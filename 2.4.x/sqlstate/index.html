§sqlstate

SQLSTATE codes with PostgreSQL, MySQL and MariaDB constants.

let code: sqlstate::SqlState = "40001".parse().unwrap();
assert!(matches!(code, sqlstate::pg::T_R_SERIALIZATION_FAILURE));
