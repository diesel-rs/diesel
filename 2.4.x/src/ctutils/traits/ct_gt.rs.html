7/// Constant time greater than.
8pub trait CtGt {
9    /// Compute whether `self > other` in constant time.
10    #[must_use]
11    fn ct_gt(&self, other: &Self) -> Choice;
12}
13
14// Share the target-specific comparison hardening with `CtLt`.
15macro_rules! impl_unsigned_ct_gt {
16    ( $($uint:ty),+ ) => {
17        $(
18            impl CtGt for $uint {
19                #[inline]
20                fn ct_gt(&self, other: &Self) -> Choice {
21                    other.ct_lt(self)
22                }
23            }
24        )+
25    };
26}

53impl CtGt for cmp::Ordering {
54    #[inline]
55    #[allow(clippy::arithmetic_side_effects, clippy::cast_sign_loss)]
56    fn ct_gt(&self, other: &Self) -> Choice {
57        let a = (*self as i8) + 1;
58        let b = (*other as i8) + 1;
59
60        // No impl of `CtGt` for `i8`, so use `u8`
61        a.cast_unsigned().ct_gt(&b.cast_unsigned())
62    }
63}
