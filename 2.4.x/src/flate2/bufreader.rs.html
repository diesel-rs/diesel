18pub struct BufReader<R> {
19    inner: R,
20    buf: Box<[u8]>,
21    pos: usize,
22    cap: usize,
23}

40impl<R: Read> BufReader<R> {
41    pub fn new(inner: R) -> BufReader<R> {
42        BufReader::with_buf(::alloc::vec::from_elem_global(0, 32 * 1024)vec![0; 32 * 1024], inner)
43    }
44
45    pub fn with_buf(buf: Vec<u8>, inner: R) -> BufReader<R> {
46        BufReader {
47            inner,
48            buf: buf.into_boxed_slice(),
49            pos: 0,
50            cap: 0,
51        }
52    }

57        &self.inner
58    }

60    pub fn get_mut(&mut self) -> &mut R {
61        &mut self.inner
62    }

65        self.inner
66    }

68    pub fn reset(&mut self, inner: R) -> R {
69        self.pos = 0;
70        self.cap = 0;

72    }

75impl<R: Read> Read for BufReader<R> {
76    fn read(&mut self, buf: &mut [u8]) -> io::Result<usize> {
77        // If we don't have any buffered data and we're doing a massive read
78        // (larger than our internal buffer), bypass our internal buffer
79        // entirely.
80        if self.pos == self.cap && buf.len() >= self.buf.len() {
81            return self.inner.read(buf);
82        }
83        let nread = {
84            let mut rem = self.fill_buf()?;
85            rem.read(buf)?
86        };
87        self.consume(nread);
88        Ok(nread)
89    }
90}
91
92impl<R: Read> BufRead for BufReader<R> {
93    fn fill_buf(&mut self) -> io::Result<&[u8]> {
94        // If we've reached the end of our internal buffer then we need to fetch
95        // some more data from the underlying reader.
96        if self.pos == self.cap {
97            self.cap = self.inner.read(&mut self.buf)?;
98            self.pos = 0;
99        }
100        Ok(&self.buf[self.pos..self.cap])
101    }
102
103    fn consume(&mut self, amt: usize) {
104        self.pos = cmp::min(self.pos + amt, self.cap);
105    }
106}
