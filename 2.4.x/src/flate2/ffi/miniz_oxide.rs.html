64impl InflateBackend for Inflate {
65    fn make(zlib_header: bool, _window_bits: u8) -> Self {
66        let format = format_from_bool(zlib_header);
67
68        Inflate {
69            inner: InflateState::new_boxed(format),
70            total_in: 0,
71            total_out: 0,
72        }
73    }
74
75    fn decompress(
76        &mut self,
77        input: &[u8],
78        output: &mut [u8],
79        flush: FlushDecompress,
80    ) -> Result<Status, DecompressError> {
81        let mz_flush = flush.into();
82        let res = inflate::stream::inflate(&mut self.inner, input, output, mz_flush);
83        self.total_in += res.bytes_consumed as u64;
84        self.total_out += res.bytes_written as u64;
85
86        match res.status {
87            Ok(status) => match status {
88                MZStatus::Ok => Ok(Status::Ok),
89                MZStatus::StreamEnd => Ok(Status::StreamEnd),
90                MZStatus::NeedDict => {
91                    mem::decompress_need_dict(self.inner.decompressor().adler32().unwrap_or(0))
92                }
93            },
94            Err(status) => match status {
95                MZError::Buf => Ok(Status::BufError),
96                _ => mem::decompress_failed(ErrorMessage),
97            },
98        }
99    }
100
101    fn reset(&mut self, zlib_header: bool) {
102        self.inner
103            .reset_as(FullReset(format_from_bool(zlib_header)));
104        self.total_in = 0;
105        self.total_out = 0;
106    }
107}

112        self.total_in
113    }

151        // Check in case the integer value changes at some point. Unlike the other zlib

153        if true {
    if !(level.level() <= 10) {
        ::core::panicking::panic("assertion failed: level.level() <= 10")
    };
};debug_assert!(level.level() <= 10);
154
155        let mut inner: Box<CompressorOxide> = Box::default();
156        let format = format_from_bool(zlib_header);
157        inner.set_format_and_level(format, level.level().try_into().unwrap_or(1));
158
159        Deflate {
160            inner,
161            total_in: 0,
162            total_out: 0,
163        }

166    fn compress(
167        &mut self,
168        input: &[u8],
169        output: &mut [u8],
170        flush: FlushCompress,
171    ) -> Result<Status, CompressError> {
172        let mz_flush = flush.into();
173        let res = deflate::stream::deflate(&mut self.inner, input, output, mz_flush);
174        self.total_in += res.bytes_consumed as u64;
175        self.total_out += res.bytes_written as u64;
176
177        match res.status {
178            Ok(status) => match status {
179                MZStatus::Ok => Ok(Status::Ok),
180                MZStatus::StreamEnd => Ok(Status::StreamEnd),
181                MZStatus::NeedDict => mem::compress_failed(ErrorMessage),
182            },
183            Err(status) => match status {
184                MZError::Buf => Ok(Status::BufError),
185                _ => mem::compress_failed(ErrorMessage),
186            },
187        }
188    }
189
190    fn reset(&mut self) {
191        self.total_in = 0;
192        self.total_out = 0;
193        self.inner.reset();
194    }
