13/// A generic connection pool.
14pub struct Pool<M: ManageConnection> {
15    pub(crate) inner: PoolInner<M>,
16}
17
18impl<M: ManageConnection> Pool<M> {
19    /// Returns a `Builder` instance to configure a new pool.
20    pub fn builder() -> Builder<M> {
21        Builder::new()
22    }
23
24    /// Retrieves a connection from the pool.
25    pub async fn get(&self) -> Result<PooledConnection<'_, M>, RunError<M::Error>> {
26        self.inner.get().await
27    }
28
29    /// Retrieves an owned connection from the pool
30    ///
31    /// Using an owning `PooledConnection` makes it easier to leak the connection pool. Therefore, [`Pool::get`]
32    /// (which stores a lifetime-bound reference to the pool) should be preferred whenever possible.
33    pub async fn get_owned(&self) -> Result<PooledConnection<'static, M>, RunError<M::Error>> {
34        Ok(PooledConnection {
35            conn: self.get().await?.take(),
36            pool: Cow::Owned(self.inner.clone()),
37            state: ConnectionState::Present,
38        })
39    }
40
41    /// Get a new dedicated connection that will not be managed by the pool.
42    /// An application may want a persistent connection (e.g. to do a
43    /// postgres LISTEN) that will not be closed or repurposed by the pool.
44    ///
45    /// This method allows reusing the manager's configuration but otherwise
46    /// bypassing the pool
47    pub async fn dedicated_connection(&self) -> Result<M::Connection, M::Error> {
48        self.inner.connect().await
49    }
50
51    /// Adds a connection to the pool.
52    ///
53    /// If the connection is broken, or the pool is at capacity, the
54    /// connection is not added and instead returned to the caller in Err.
55    pub fn add(&self, conn: M::Connection) -> Result<(), AddError<M::Connection>> {
56        self.inner.try_put(conn)
57    }
58
59    /// Returns information about the current state of the pool.
60    pub fn state(&self) -> State {
61        self.inner.state()
62    }
63
64    /// Expose a representation of the original pool configuration
65    pub fn config(&self) -> Config {
66        Config::from(self.inner.builder())
67    }
68}

70impl<M: ManageConnection> Clone for Pool<M> {
71    fn clone(&self) -> Self {
72        Pool {
73            inner: self.inner.clone(),
74        }
75    }
76}

117    pub connections_closed_broken: u64,

119    pub connections_closed_invalid: u64,

121    /// lifetime.

123    /// Total connections that were closed because they reached the max

128impl Statistics {
129    /// Total pending gets waiting for a connection.
130    pub fn pending_gets(&self) -> u64 {
131        self.get_started - self.completed_gets()

450/// A trait which provides connection-specific functionality.
451pub trait ManageConnection: Sized + Send + Sync + 'static {
452    /// The connection type this manager deals with.
453    type Connection: Send + 'static;
454    /// The error type returned by `Connection`s.
455    type Error: fmt::Debug + Send + 'static;
456
457    /// Attempts to create a new connection.
458    fn connect(&self) -> impl Future<Output = Result<Self::Connection, Self::Error>> + Send;
459    /// Determines if the connection is still connected to the database.
460    fn is_valid(
461        &self,
462        conn: &mut Self::Connection,
463    ) -> impl Future<Output = Result<(), Self::Error>> + Send;
464    /// Synchronously determine if the connection is no longer usable, if possible.
465    fn has_broken(&self, conn: &mut Self::Connection) -> bool;
466}

468/// A trait which provides functionality to initialize a connection
469pub trait CustomizeConnection<C: Send + 'static, E: 'static>:
470    fmt::Debug + Send + Sync + 'static
471{
472    /// Called with connections immediately after they are returned from
473    /// `ManageConnection::connect`.
474    ///
475    /// The default implementation simply returns `Ok(())`. If this method returns an
476    /// error, it will be forwarded to the configured error sink.
477    fn on_acquire<'a>(
478        &'a self,
479        _connection: &'a mut C,
480    ) -> Pin<Box<dyn Future<Output = Result<(), E>> + Send + 'a>> {
481        Box::pin(async { Ok(()) })
482    }
483}
