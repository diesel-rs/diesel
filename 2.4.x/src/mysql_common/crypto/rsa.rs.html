14/// Padding operation trait.
15pub trait Padding {
16    /// Padding operation for `input` bytes, where `k` is the length of modulus in octets.
17    fn pub_pad(&mut self, input: impl AsRef<[u8]>, k: usize) -> Vec<u8>;
18}

20/// Represents a source of random bytes.
21pub trait Rng {
22    type Error: std::error::Error;
23
24    /// Implementor must fill all the bytes in the `dest`.
25    fn fill(&mut self, dest: &mut [u8]) -> Result<(), Self::Error>;
26}

32    type Error = getrandom::Error;
33

36    }

82pub struct Pkcs1OaepPadding<T> {
83    rng: T,
84}
85
86impl<T> Pkcs1OaepPadding<T> {
87    /// Length of a SHA-1 hash digest.
88    const HASH_LEN: usize = 20;
89
90    pub fn new(rng: T) -> Self {
91        Self { rng }
92    }
93
94    /// Mask Generation Function as defined in rfc2437.
95    ///
96    /// It will use SHA-1 as a hash function.
97    fn mgf1(seed: &[u8], len: usize) -> Vec<u8> {
98        if len as u64 > 2u64.pow(32) * Self::HASH_LEN as u64 {
99            { ::core::panicking::panic_fmt(format_args!("mask too long")); };panic!("mask too long");
100        }
101
102        let output = (0..len.div_ceil(Self::HASH_LEN))
103            .map(|c| {
104                let cs = &mut [0u8; 4];
105                BigEndian::write_u32(cs, c as u32);
106                Sha1::digest([seed, cs].concat()).to_vec()
107            })
108            .collect::<Vec<Vec<u8>>>()
109            .concat();
110
111        output[..len].into()
112    }
113}
114
115impl<T: Rng> Padding for Pkcs1OaepPadding<T> {
116    /// Will pad input according to PKCS #1 v2 with encoding parameters equal to `[]`.
117    fn pub_pad(&mut self, input: impl AsRef<[u8]>, k: usize) -> Vec<u8> {
118        let input = input.as_ref();
119        // 1. Skip because encoding parameters == []
120        // 2. If ||M|| > emLen-2hLen-1 then output "message too long" and stop.
121        if input.len() > k - 2 * Self::HASH_LEN - 1 {
122            { ::core::panicking::panic_fmt(format_args!("message too long")); };panic!("message too long");
123        }
124        // 3. Generate an octet string PS consisting of emLen-||M||-2hLen-1 zero
125        //    octets. The length of PS may be 0.
126        let mut ps = ::alloc::vec::from_elem_global(0, k - input.len() - 2 * Self::HASH_LEN - 2)vec![0; k - input.len() - 2 * Self::HASH_LEN - 2];
127        ps.push(0x01);
128        // 4. Let pHash = Hash(P), an octet string of length hLen.
129        let p_hash = Sha1::digest([]).to_vec();
130        // 5. Concatenate pHash, PS, the message M, and other padding to form a
131        //    data block DB as: DB = pHash || PS || 01 || M
132        let db = [&*p_hash, &*ps, input].concat();
133        // 6. Generate a random octet string seed of length hLen.
134        let mut seed = ::alloc::vec::from_elem_global(0, Self::HASH_LEN)vec![0; Self::HASH_LEN];
135        self.rng.fill(&mut seed[..]).expect("rng out of entropy");
136        // 7. Let dbMask = MGF(seed, emLen-hLen).
137        let db_mask = Self::mgf1(&seed, k - Self::HASH_LEN);
138        // 8. Let maskedDB = DB \xor dbMask.
139        let masked_db: Vec<_> = db.into_iter().zip(db_mask).map(|(a, b)| a ^ b).collect();
140        // 9. Let seedMask = MGF(maskedDB, hLen).
141        let seed_mask = Self::mgf1(&masked_db, Self::HASH_LEN);
142        // 10. Let maskedSeed = seed \xor seedMask.
143        let masked_seed: Vec<_> = seed
144            .into_iter()
145            .zip(seed_mask)
146            .map(|(a, b)| a ^ b)
147            .collect();
148        // 11. Let EM = maskedSeed || maskedDB.
149        [&*masked_seed, &*masked_db].concat()
150    }
151}

153#[derive(#[automatically_derived]impl ::core::fmt::Debug for PublicKey {
    #[inline]
    #[rustc_trivial_field_reads]
    fn fmt(&self, f: &mut ::core::fmt::Formatter) -> ::core::fmt::Result {

154pub struct PublicKey {
155    modulus: BigUint,
156    exponent: BigUint,
157}
158
159impl PublicKey {
160    /// Basic constructor.
161    pub fn new(modulus: BigUint, exponent: BigUint) -> PublicKey {
162        PublicKey { modulus, exponent }
163    }
164
165    /// Will parse public key from pem representation.
166    ///
167    /// # Panic
168    ///
169    /// Will panic in case of bad pem data.
170    pub fn from_pem(pem_data: impl AsRef<[u8]>) -> PublicKey {
171        let (der, file_type) = der::pem_to_der(pem_data);
172        let (modulus, exponent) = der::parse_pub_key(&der, file_type);
173        PublicKey::new(modulus, exponent)
174    }
175
176    /// Returns number of octets in the modulus.
177    pub fn num_octets(&self) -> usize {
178        (self.modulus.bits() as usize + 6) >> 3
179    }
180
181    /// Returns modulus of the public key.
182    pub fn modulus(&self) -> &BigUint {
183        &self.modulus
184    }
185
186    /// Returns exponent of the public key.
187    pub fn exponent(&self) -> &BigUint {
188        &self.exponent
189    }
190
191    /// Will encrypt block with public key.
192    ///
193    /// # Panic
194    ///
195    /// Will panic if block is too long for key or padding.
196    pub fn encrypt_block(&self, block: impl AsRef<[u8]>, mut pad: impl Padding) -> Vec<u8> {
197        let enc_block = pad.pub_pad(block, self.num_octets());
198        let enc_int = BigUint::from_bytes_be(&enc_block);
199        let rsa = enc_int.modpow(self.exponent(), self.modulus());
200        let mut rsa_bytes = rsa.to_bytes_be();
201        // is this needed?
202        while rsa_bytes.len() < self.num_octets() {
203            rsa_bytes.insert(0, 0);
204        }
205        rsa_bytes
206    }
207}
