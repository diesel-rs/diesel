30/// A struct indicating the portion of a buffer written to, and/or not written to, during an
31/// encryption/decryption operation.
32pub struct BufferUpdate<'a> {
33    written: &'a [u8],
34    remainder: &'a mut [u8],
35}
36
37impl<'a> BufferUpdate<'a> {
38    fn new(out_buffer: &'a mut [u8], written_len: usize) -> Self {
39        let (written, remainder) = out_buffer.split_at_mut(written_len);
40        Self { written, remainder }
41    }
42}

126    ) -> Result<Self, Unspecified> {
127        let algorithm = key.algorithm();
128        if !algorithm.supports_mode(mode) {
129            return Err(Unspecified);
130        }
131        // EVP initialization reads the algorithm's IV length without receiving the slice length.
132        if !algorithm.is_valid_encryption_context(mode, &context) {
133            return Err(Unspecified);
134        }
135        // The streaming path passes raw key bytes to the EVP API rather than
136        // going through `SymmetricCipherKey` construction.  Validate
137        // algorithm-specific key constraints (e.g. DES weak-key / K1!=K2
138        // checks) that would otherwise be missed.
139        key.validate_key_material()?;
140        let mut cipher_ctx = LcPtr::new(unsafe { EVP_CIPHER_CTX_new() })?;
141        let cipher = mode.evp_cipher(key.algorithm);
142        let key_bytes = key.key_bytes.as_ref();
143        if key_bytes.len()
144            != <usize>::try_from(unsafe { EVP_CIPHER_key_length(cipher.as_const_ptr()) }).unwrap()
145        {
146            return Err(Unspecified);
147        }
148
149        match &context {
150            ctx @ EncryptionContext::Iv128(..) => {
151                let iv = <&[u8]>::try_from(ctx)?;
152                if true {

214    /// Updates the internal state of the key with the provided plaintext `input`,
215    /// potentially writing bytes of ciphertext to `output`.
216    ///
217    /// The number of bytes written to `output` can be up to `input.len()`
218    /// plus the block length of the algorithm (e.g., [`Algorithm::block_len`]) minus one.
219    ///
220    /// # Errors
221    /// * Returns an error if the `output` buffer is smaller than the length of
222    ///   the `input` plus the algorithm's block length (e.g. [`Algorithm::block_len`]) minus one.
223    /// * May return an error if the length of `input` plus the algorithm's block length is larger than `i32::MAX`.
224    pub fn update<'a>(
225        &mut self,
226        input: &[u8],
227        output: &'a mut [u8],
228    ) -> Result<BufferUpdate<'a>, Unspecified> {
229        let min_outsize = input
230            .len()
231            .checked_add(self.algorithm().block_len())
232            .ok_or(Unspecified)?
233            .checked_sub(1)
234            .ok_or(Unspecified)?;
235        self.update_internal(input, output, min_outsize)
236    }

238    /// Updates the internal state of the key with the provided plaintext `input`,
239    /// potentially writing bytes of ciphertext to `output`.
240    ///
241    /// This function has looser output buffer size requirements than [`Self::update`],
242    /// calculating the minimum required size based on the total bytes of output generated
243    /// and the cipher's block length. This is considered "less safe" because it's
244    /// based on assumptions about the state of the underlying operations.
245    ///
246    /// The minimum output buffer size is calculated based on how many bytes are needed to
247    /// reach the next block boundary after processing the input. If `next_total` is the sum
248    /// of bytes already generated plus `input.len()`, then the minimum size is:
249    /// `input.len() + ((block_len - (next_total % block_len)) % block_len)`
250    ///
251    /// # Errors
252    /// Returns an error if the `output` buffer is smaller than the calculated minimum size,
253    /// if the total output length overflows, or if the length of `input` is larger than
254    /// `i32::MAX`.
255    ///
256    /// # Panics
257    /// Panics if the number of bytes written by the cipher operation exceeds the output
258    /// buffer length.
259    pub fn less_safe_update<'a>(
260        &mut self,
261        input: &[u8],
262        output: &'a mut [u8],
263    ) -> Result<BufferUpdate<'a>, Unspecified> {
264        let next_total = self
265            .output_generated
266            .checked_add(input.len())
267            .ok_or(Unspecified)?;
268        let extra_buffer_size = (self.algorithm().block_len
269            - next_total.rem_euclid(self.algorithm().block_len))
270        .rem_euclid(self.algorithm().block_len);
271        let min_outsize = input
272            .len()
273            .checked_add(extra_buffer_size)
274            .ok_or(Unspecified)?;
275        self.update_internal(input, output, min_outsize)
276    }

661    /// Updates the internal state of the key with the provided ciphertext `input`,
662    /// potentially also writing bytes of plaintext to `output`.
663    /// The number of bytes written to `output` can be up to `input.len()`

665    ///
666    /// # Errors
667    /// * Returns an error if the `output` buffer is smaller than the length of
668    ///   the `input` plus the algorithm's block length minus one.
669    /// * May return an error if the length of `input` plus the algorithm's block length is larger
670    ///   than `i32::MAX`.
671    pub fn update<'a>(
672        &mut self,
673        input: &[u8],
674        output: &'a mut [u8],
675    ) -> Result<BufferUpdate<'a>, Unspecified> {
676        let min_outsize = input

678            .checked_add(self.algorithm().block_len())
679            .ok_or(Unspecified)?
680            .checked_sub(1)
681            .ok_or(Unspecified)?;

685    /// Updates the internal state of the key with the provided ciphertext `input`,
686    /// potentially writing bytes of plaintext to `output`.
687    ///
688    /// This function has looser output buffer size requirements than [`Self::update`],
689    /// calculating the minimum required size based on the total bytes of output generated
690    /// and the cipher's block length. This is considered "less safe" because it's
691    /// based on assumptions about the state of the underlying operations.
692    ///
693    /// The minimum output buffer size is calculated based on how many bytes are needed to
694    /// reach the next block boundary after processing the input. If `next_total` is the sum
695    /// of bytes already generated plus `input.len()`, then the minimum size is:
696    /// `input.len() + ((block_len - (next_total % block_len)) % block_len)`
697    ///
698    /// # Errors
699    /// Returns an error if the `output` buffer is smaller than the calculated minimum size,
700    /// if the total output length overflows, or if the length of `input` is larger than
701    /// `i32::MAX`.
702    ///
703    /// # Panics
704    /// Panics if the number of bytes written by the cipher operation exceeds the output
705    /// buffer length.
706    pub fn less_safe_update<'a>(
707        &mut self,
708        input: &[u8],
709        output: &'a mut [u8],
710    ) -> Result<BufferUpdate<'a>, Unspecified> {
711        let next_total = self
712            .output_generated
713            .checked_add(input.len())
714            .ok_or(Unspecified)?;
715        let extra_buffer_size = (self.algorithm().block_len
716            - next_total.rem_euclid(self.algorithm().block_len))
717        .rem_euclid(self.algorithm().block_len);
718        let min_outsize = input
719            .len()
720            .checked_add(extra_buffer_size)
721            .ok_or(Unspecified)?;
722        self.update_internal(input, output, min_outsize)
723    }

1306    enum UpdateVariant {
1307        /// [`StreamingDecryptingKey::update`]: `in_len + block_len - 1`.
1308        Documented,
1309        /// [`StreamingDecryptingKey::less_safe_update`]: `in_len` when aligned.

1313    /// Feeds block-aligned chunks through `key`, handing each call exactly the
1314    /// minimum output slice its API documents, with canary bytes just past it.
1315    /// `update`-only: the overrun precedes any padding check, so the ciphertext
1316    /// is arbitrary and `finish` is never called.
1317    fn assert_block_aligned_updates_stay_in_slice(
1318        mut key: StreamingDecryptingKey,
1319        variant: UpdateVariant,
1320        chunks: usize,
1321        canary: u8,
1322    ) {
1323        let block_len = key.algorithm().block_len();
1324        let ciphertext = vec![0x42u8; chunks * block_len];
1325        // Widest documented minimum is `2 * block_len - 1`, leaving every slice
1326        // >= `block_len + 1` canary bytes -- enough for a full-block overrun.
1327        let mut output = vec![canary; ciphertext.len() + 3 * block_len];
1328        let mut out_idx = 0usize;
1329
1330        for chunk in ciphertext.chunks(block_len) {
1331            let min_outsize = match variant {
1332                UpdateVariant::Documented => chunk.len() + block_len - 1,
1333                UpdateVariant::LessSafe => less_safe_min_out(block_len, out_idx, chunk.len()),
1334            };
1335            let out_end = out_idx + min_outsize;
1336            out_idx += apply_update_with_canary(&mut output, out_idx, out_end, canary, |out| {
1337                match variant {
1338                    UpdateVariant::Documented => key.update(chunk, out),
1339                    UpdateVariant::LessSafe => key.less_safe_update(chunk, out),
1340                }
1341                .expect("update rejected a documented-minimum output slice")
1342                .written()
1343                .len()
1344            });
1345        }
1346    }
