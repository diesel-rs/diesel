4//! CMAC is specified in [RFC 4493] and [NIST SP 800-38B].
5//!
6//! After a `Key` is constructed, it can be used for multiple signing or
7//! verification operations. Separating the construction of the key from the
8//! rest of the CMAC operation allows the per-key precomputation to be done
9//! only once, instead of it being done in every CMAC operation.
10//!
11//! Frequently all the data to be signed in a message is available in a single
12//! contiguous piece. In that case, the module-level `sign` function can be
13//! used. Otherwise, if the input is in multiple parts, `Context` should be
14//! used.
15//!
16//! # Examples:
17//!
18//! ## Signing a value and verifying it wasn't tampered with
19//!
20//! ```
21//! use aws_lc_rs::cmac;
22//!
23//! let key = cmac::Key::generate(cmac::AES_128)?;
24//!
25//! let msg = "hello, world";
26//!
27//! let tag = cmac::sign(&key, msg.as_bytes())?;
28//!
29//! // [We give access to the message to an untrusted party, and they give it
30//! // back to us. We need to verify they didn't tamper with it.]
31//!
32//! cmac::verify(&key, msg.as_bytes(), tag.as_ref())?;
33//!
34//! # Ok::<(), aws_lc_rs::error::Unspecified>(())
35//! ```
36//!
37//! ## Using the one-shot API:
38//!
39//! ```
40//! use aws_lc_rs::{cmac, rand};
41//!
42//! let msg = "hello, world";
43//!
44//! // The sender generates a secure key value and signs the message with it.
45//! // Note that in a real protocol, a key agreement protocol would be used to
46//! // derive `key_value`.
47//! let rng = rand::SystemRandom::new();
48//! let key_value: [u8; 16] = rand::generate(&rng)?.expose();
49//!
50//! let s_key = cmac::Key::new(cmac::AES_128, key_value.as_ref())?;
51//! let tag = cmac::sign(&s_key, msg.as_bytes())?;
52//!
53//! // The receiver (somehow!) knows the key value, and uses it to verify the
54//! // integrity of the message.
55//! let v_key = cmac::Key::new(cmac::AES_128, key_value.as_ref())?;
56//! cmac::verify(&v_key, msg.as_bytes(), tag.as_ref())?;
57//!
58//! # Ok::<(), aws_lc_rs::error::Unspecified>(())
59//! ```
60//!
61//! ## Using the multi-part API:
62//! ```
63//! use aws_lc_rs::{cmac, rand};
64//!
65//! let parts = ["hello", ", ", "world"];
66//!
67//! // The sender generates a secure key value and signs the message with it.
68//! // Note that in a real protocol, a key agreement protocol would be used to
69//! // derive `key_value`.
70//! let rng = rand::SystemRandom::new();
71//! let key_value: [u8; 32] = rand::generate(&rng)?.expose();
72//!
73//! let s_key = cmac::Key::new(cmac::AES_256, key_value.as_ref())?;
74//! let mut s_ctx = cmac::Context::with_key(&s_key);
75//! for part in &parts {
76//!     s_ctx.update(part.as_bytes())?;
77//! }
78//! let tag = s_ctx.sign()?;
79//!
80//! // The receiver (somehow!) knows the key value, and uses it to verify the
81//! // integrity of the message.
82//! let v_key = cmac::Key::new(cmac::AES_256, key_value.as_ref())?;
83//! let mut msg = Vec::<u8>::new();
84//! for part in &parts {
85//!     msg.extend(part.as_bytes());
86//! }
87//! cmac::verify(&v_key, &msg.as_ref(), tag.as_ref())?;
88//!
89//! # Ok::<(), aws_lc_rs::error::Unspecified>(())
90//! ```
91//! [RFC 4493]: https://tools.ietf.org/html/rfc4493
92//! [NIST SP 800-38B]: https://nvlpubs.nist.gov/nistpubs/SpecialPublications/NIST.SP.800-38b.pdf

245impl Key {
246    /// Generate a CMAC signing key using the given algorithm with a
247    /// random value.
248    ///
249    //
250    // # FIPS
251    // Use this type with one of the following algorithms:
252    // * `AES_128`
253    // * `AES_256`
254    //
255    /// # Errors
256    /// `error::Unspecified` if random generation or key construction fails.
257    pub fn generate(algorithm: Algorithm) -> Result<Self, Unspecified> {
258        let mut key_bytes = ::alloc::vec::from_elem_global(0u8, algorithm.key_len())vec![0u8; algorithm.key_len()];
259        rand::fill(&mut key_bytes)?;
260        Self::new(algorithm, &key_bytes)
261    }
262
263    /// Construct a CMAC signing key using the given algorithm and key value.
264    ///
265    /// `key_value` should be a value generated using a secure random number
266    /// generator or derived from a random key by a key derivation function.
267    ///
268    /// # Errors
269    /// `error::Unspecified` if the key length doesn't match the algorithm or if CMAC context
270    /// initialization fails.
271    pub fn new(algorithm: Algorithm, key_value: &[u8]) -> Result<Self, Unspecified> {
272        if key_value.len() != algorithm.key_len() {
273            return Err(Unspecified);
274        }
275
276        let mut ctx = LcPtr::new(unsafe { CMAC_CTX_new() })?;
277
278        unsafe {
279            let cipher = algorithm.id.evp_cipher();
280            if 1 != CMAC_Init(
281                ctx.as_mut_ptr(),
282                key_value.as_ptr().cast(),
283                key_value.len(),
284                cipher.as_const_ptr(),
285                null_mut(),
286            ) {
287                return Err(Unspecified);
288            }
289        }
290
291        Ok(Self { algorithm, ctx })
292    }
293
294    /// The algorithm for the key.
295    #[inline]
296    #[must_use]
297    pub fn algorithm(&self) -> Algorithm {
298        self.algorithm
299    }

328    /// Constructs a new CMAC signing context using the given key.

335    /// Updates the CMAC with all the data in `data`. `update` may be called
336    /// zero or more times until `sign` is called.
337    ///
338    /// # Errors
339    /// `error::Unspecified` if the CMAC cannot be updated.
340    pub fn update(&mut self, data: &[u8]) -> Result<(), Unspecified> {
341        unsafe {
342            if 1 != CMAC_Update(self.key.ctx.as_mut_ptr(), data.as_ptr(), data.len()) {
343                return Err(Unspecified);
344            }
345        }
346        Ok(())
347    }
348
349    /// Finalizes the CMAC calculation and returns the CMAC value. `sign`
350    /// consumes the context so it cannot be (mis-)used after `sign` has been
351    /// called.
352    ///
353    /// It is generally not safe to implement CMAC verification by comparing
354    /// the return value of `sign` to a tag. Use `verify` for verification
355    /// instead.
356    ///
357    //
358    // # FIPS
359    // Use this method with one of the following algorithms:
360    // * `AES_128`
361    // * `AES_256`
362    //
363    /// # Errors
364    /// `error::Unspecified` if the CMAC calculation cannot be finalized.
365    ///
366    /// # Panics
367    /// Panics if the CMAC tag length exceeds the maximum allowed length, indicating memory corruption.
368    pub fn sign(mut self) -> Result<Tag, Unspecified> {
369        let mut output = [0u8; MAX_CMAC_TAG_LEN];
370        let output_len = {
371            let result = internal_sign(&mut self, &mut output)?;
372            result.len()
373        };
374
375        Ok(Tag {
376            bytes: output,
377            len: output_len,
378        })
379    }
380
381    /// Finalizes the CMAC calculation and verifies whether the resulting value
382    /// equals the provided `tag`.
383    ///
384    /// `verify` consumes the context so it cannot be (mis-)used after `verify`
385    /// has been called.
386    ///
387    /// The verification is done in constant time to prevent timing attacks.
388    ///
389    /// # Errors
390    /// `error::Unspecified` if the tag does not match or if CMAC calculation fails.
391    //
392    // # FIPS
393    // Use this function with one of the following algorithms:
394    // * `AES_128`
395    // * `AES_256`
396    #[inline]
397    pub fn verify(mut self, tag: &[u8]) -> Result<(), Unspecified> {
398        let mut output = [0u8; MAX_CMAC_TAG_LEN];
399        let output_len = {
400            let result = internal_sign(&mut self, &mut output)?;
401            result.len()
402        };
403
404        constant_time::verify_slices_are_equal(&output[0..output_len], tag)
405    }
406}

413
414    if 1 != {
    {

                out_len.as_mut_ptr())
        }
    }
}indicator_check!(unsafe {
415        CMAC_Final(
416            ctx.key.ctx.as_mut_ptr(),
417            output.as_mut_ptr(),
418            out_len.as_mut_ptr(),
419        )
420    }) {
421        return Err(Unspecified);
422    }
423    let actual_len = unsafe { out_len.assume_init() };
424
425    // This indicates a memory corruption.
426    if true {
    if !(actual_len <= MAX_CMAC_TAG_LEN) {
        {
            ::core::panicking::panic_fmt(format_args!("CMAC tag length {0} exceeds maximum {1}",

        }
    };
};debug_assert!(
427        actual_len <= MAX_CMAC_TAG_LEN,
428        "CMAC tag length {actual_len} exceeds maximum {MAX_CMAC_TAG_LEN}"
429    );
430    if actual_len != ctx.key.algorithm.tag_len() {
431        return Err(Unspecified);
432    }
433
434    Ok(&mut output[0..actual_len])

437/// Calculates the CMAC of `data` using the key `key` in one step.
438///
439/// Use `Context` to calculate CMACs where the input is in multiple parts.
440///
441/// It is generally not safe to implement CMAC verification by comparing the
442/// return value of `sign` to a tag. Use `verify` for verification instead.
443//
444// # FIPS
445// Use this function with one of the following algorithms:
446// * `AES_128`
447// * `AES_256`
448//
449/// # Errors
450/// `error::Unspecified` if the CMAC calculation fails.
451#[inline]
452pub fn sign(key: &Key, data: &[u8]) -> Result<Tag, Unspecified> {
453    let mut ctx = Context::with_key(key);
454    ctx.update(data)?;
455    ctx.sign()
456}

458/// Calculates the CMAC of `data` using the key `key` in one step, writing the
459/// result into the provided `output` buffer.
460///
461/// Use `Context` to calculate CMACs where the input is in multiple parts.
462///
463/// The `output` buffer must be at least as large as the algorithm's tag length
464/// (obtainable via `key.algorithm().tag_len()`). The returned slice will be a
465/// sub-slice of `output` containing exactly the tag bytes.
466///
467/// It is generally not safe to implement CMAC verification by comparing the
468/// return value of `sign_to_buffer` to a tag. Use `verify` for verification instead.
469//
470// # FIPS
471// Use this function with one of the following algorithms:
472// * `AES_128`
473// * `AES_256`
474//
475/// # Errors
476/// `error::Unspecified` if the output buffer is too small or if the CMAC calculation fails.
477#[inline]
478pub fn sign_to_buffer<'out>(
479    key: &Key,
480    data: &[u8],
481    output: &'out mut [u8],
482) -> Result<&'out mut [u8], Unspecified> {
483    if output.len() < key.algorithm().tag_len() {
484        return Err(Unspecified);
485    }
486
487    let mut ctx = Context::with_key(key);
488    ctx.update(data)?;
489
490    internal_sign(&mut ctx, output)
491}

493/// Calculates the CMAC of `data` using the signing key `key`, and verifies
494/// whether the resultant value equals `tag`, in one step.
495///
496/// The verification is done in constant time to prevent timing attacks.
497///
498/// # Errors
499/// `error::Unspecified` if the tag does not match or if CMAC calculation fails.
500//
501// # FIPS
502// Use this function with one of the following algorithms:
503// * `AES_128`
504// * `AES_256`
505#[inline]
506pub fn verify(key: &Key, data: &[u8], tag: &[u8]) -> Result<(), Unspecified> {
507    let mut output = [0u8; MAX_CMAC_TAG_LEN];
508    let output_len = {
509        let result = sign_to_buffer(key, data, &mut output)?;
510        result.len()
511    };
512
513    constant_time::verify_slices_are_equal(&output[0..output_len], tag)
514}
