107pub(crate) fn parse_ed25519_public_key(key_bytes: &[u8]) -> Result<LcPtr<EVP_PKEY>, KeyRejected> {
108    // If the length of key bytes matches the raw public key size then it has to be that
109    if key_bytes.len() == ED25519_PUBLIC_KEY_LEN {
110        LcPtr::<EVP_PKEY>::parse_raw_public_key(key_bytes, EVP_PKEY_ED25519)
111    } else {
112        // Otherwise we support X.509 SubjectPublicKeyInfo formatted keys which are inherently larger
113        LcPtr::<EVP_PKEY>::parse_rfc5280_public_key(key_bytes, EVP_PKEY_ED25519)
114    }
115}

188impl AsDer<PublicKeyX509Der<'static>> for PublicKey {
189    /// Provides the public key as a DER-encoded (X.509) `SubjectPublicKeyInfo` structure.
190    /// # Errors
191    /// Returns an error if the public key fails to marshal to X.509.
192    fn as_der(&self) -> Result<PublicKeyX509Der<'static>, crate::error::Unspecified> {
193        // Initial size of 44 based on:
194        // 0:d=0  hl=2 l=  42 cons: SEQUENCE
195        // 2:d=1  hl=2 l=   5 cons:  SEQUENCE
196        // 4:d=2  hl=2 l=   3 prim:   OBJECT            :ED25519
197        // 9:d=1  hl=2 l=  33 prim:  BIT STRING
198        let der = self.evp_pkey.as_const().marshal_rfc5280_public_key()?;
199        Ok(PublicKeyX509Der::from(Buffer::new(der)))
200    }
201}

225
226        let mut public_key = [0u8; ED25519_PUBLIC_KEY_LEN];
227        let out_len: usize = evp_pkey
228            .as_const()
229            .marshal_raw_public_to_buffer(&mut public_key)?;
230        if true {
    {
        match (&public_key.len(), &out_len) {
            (left_val, right_val) => {

                    let kind = ::core::panicking::AssertKind::Eq;
                    ::core::panicking::assert_failed(kind, &*left_val,

            }
        }

};debug_assert_eq!(public_key.len(), out_len);

241    /// Generates a new key pair and returns the key pair serialized as a
242    /// PKCS#8 document.
243    ///
244    /// The PKCS#8 document will be a v2 `OneAsymmetricKey` with the public key,
245    /// as described in [RFC 5958 Section 2]; see [RFC 8410 Section 10.3] for an
246    /// example.
247    ///
248    /// [RFC 5958 Section 2]: https://tools.ietf.org/html/rfc5958#section-2
249    /// [RFC 8410 Section 10.3]: https://tools.ietf.org/html/rfc8410#section-10.3
250    ///
251    /// # *ring* Compatibility
252    /// The ring 0.16.x API did not produce encoded v2 documents that were compliant with RFC 5958.
253    /// The aws-lc-ring implementation produces PKCS#8 v2 encoded documents that are compliant per
254    /// the RFC specification.
255    ///
256    /// Our implementation ignores the `SecureRandom` parameter.
257    ///
258    // # FIPS
259    // This function must not be used.

263    pub fn generate_pkcs8(_rng: &dyn SecureRandom) -> Result<Document, Unspecified> {
264        let evp_pkey = generate_key()?;
265        Ok(Document::new(
266            evp_pkey
267                .as_const()
268                .marshal_rfc5208_private_key(Version::V2)?,

270    }

321    /// Constructs an Ed25519 key pair from the private key seed `seed` and its
322    /// public key `public_key`.
323    ///
324    /// It is recommended to use `Ed25519KeyPair::from_pkcs8()` instead.
325    ///
326    /// The private and public keys will be verified to be consistent with each
327    /// other. This helps avoid misuse of the key (e.g. accidentally swapping
328    /// the private key and public key, or using the wrong private key for the
329    /// public key). This also detects any corruption of the public or private
330    /// key.
331    ///
332    /// # Errors
333    /// `error::KeyRejected` if parse error, or if key is otherwise unacceptable.
334    pub fn from_seed_and_public_key(seed: &[u8], public_key: &[u8]) -> Result<Self, KeyRejected> {
335        let this = Self::from_seed_unchecked(seed)?;
336
337        constant_time::verify_slices_are_equal(public_key, &this.public_key.public_key_bytes)
338            .map_err(|_| KeyRejected::inconsistent_components())?;
339        Ok(this)
340    }

342    /// Constructs an Ed25519 key pair from the private key seed `seed`.
343    ///
344    /// It is recommended to use `Ed25519KeyPair::from_pkcs8()` instead. If the public key is
345    /// available, prefer to use `Ed25519KeyPair::from_seed_and_public_key()` as it will verify
346    /// the validity of the key pair.
347    ///
348    /// CAUTION: Both an Ed25519 seed and its public key are 32-bytes. If the bytes of a public key
349    /// are provided this function will create an (effectively) invalid `Ed25519KeyPair`. This
350    /// problem is undetectable by the API.
351    ///
352    /// # Errors
353    /// `error::KeyRejected` if parse error, or if key is otherwise unacceptable.
354    pub fn from_seed_unchecked(seed: &[u8]) -> Result<Self, KeyRejected> {
355        if seed.len() < ED25519_SEED_LEN {
356            return Err(KeyRejected::inconsistent_components());
357        }
358
359        let evp_pkey = LcPtr::<EVP_PKEY>::parse_raw_private_key(seed, EVP_PKEY_ED25519)?;
360
361        let mut derived_public_key = [0u8; ED25519_PUBLIC_KEY_LEN];
362        let out_len: usize = evp_pkey
363            .as_const()
364            .marshal_raw_public_to_buffer(&mut derived_public_key)?;
365        if true {
    {
        match (&derived_public_key.len(), &out_len) {
            (left_val, right_val) => {
                if !(*left_val == *right_val) {
                    let kind = ::core::panicking::AssertKind::Eq;
                    ::core::panicking::assert_failed(kind, &*left_val,
                        &*right_val, ::core::option::Option::None);
                }
            }
        }
    };
};debug_assert_eq!(derived_public_key.len(), out_len);
366
367        Ok(Self {
368            public_key: PublicKey {
369                public_key_bytes: derived_public_key,
370                evp_pkey: evp_pkey.clone(),
371            },
372            evp_pkey,

448    ///
449    // # FIPS

456        let sig_bytes = self.evp_pkey.sign(msg, None, No_EVP_PKEY_CTX_consumer)?;

458        Ok(Signature::new(|slice| {
459            slice[0..ED25519_SIGNATURE_LEN].copy_from_slice(&sig_bytes);

531    fn test_generate_pkcs8() {
532        let rng = SystemRandom::new();
533        let document = Ed25519KeyPair::generate_pkcs8(&rng).unwrap();
534        let kp1: Ed25519KeyPair = Ed25519KeyPair::from_pkcs8(document.as_ref()).unwrap();
535        assert_eq!(
536            document.as_ref(),
537            AsDer::<Pkcs8V2Der>::as_der(&kp1).unwrap().as_ref()
538        );
539        let kp2: Ed25519KeyPair =
540            Ed25519KeyPair::from_pkcs8_maybe_unchecked(document.as_ref()).unwrap();
541        assert_eq!(
542            kp1.seed().unwrap().as_be_bytes().unwrap().as_ref(),
543            kp2.seed().unwrap().as_be_bytes().unwrap().as_ref(),
544        );
545        assert_eq!(kp1.public_key.as_ref(), kp2.public_key.as_ref());
546
547        let document = Ed25519KeyPair::generate_pkcs8v1(&rng).unwrap();
548        let kp1: Ed25519KeyPair = Ed25519KeyPair::from_pkcs8(document.as_ref()).unwrap();
549        assert_eq!(
550            document.as_ref(),
551            AsDer::<Pkcs8V1Der>::as_der(&kp1).unwrap().as_ref()
552        );
553        let kp2: Ed25519KeyPair =
554            Ed25519KeyPair::from_pkcs8_maybe_unchecked(document.as_ref()).unwrap();
555        assert_eq!(
556            kp1.seed().unwrap().as_be_bytes().unwrap().as_ref(),
557            kp2.seed().unwrap().as_be_bytes().unwrap().as_ref(),
558        );
559        assert_eq!(kp1.public_key.as_ref(), kp2.public_key.as_ref());
560        let seed = kp1.seed().unwrap();
561        assert_eq!("Ed25519Seed()", format!("{seed:?}"));
562    }
