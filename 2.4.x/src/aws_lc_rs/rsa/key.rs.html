221    /// Sign `msg`. `msg` is digested using the digest algorithm from
222    /// `padding_alg` and the digest is then padded using the padding algorithm
223    /// from `padding_alg`. The signature is written into `signature`;
224    /// `signature`'s length must be exactly the length returned by
225    /// `public_modulus_len()`.
226    ///
227    /// This function does *not* take a precomputed digest; instead, `sign`
228    /// calculates the digest itself. See `sign_digest`.
229    ///
230    /// # *ring* Compatibility
231    /// Our implementation ignores the `SecureRandom` parameter.
232    // # FIPS
233    // The following conditions must be met:
234    // * RSA Key Sizes: 2048, 3072, 4096
235    // * Digest Algorithms: SHA256, SHA384, SHA512
236    //
237    /// # Errors
238    /// `error::Unspecified` on error.
239    /// With "fips" feature enabled, errors if digest length is greater than `u32::MAX`.
240    pub fn sign(
241        &self,
242        padding_alg: &'static dyn RsaEncoding,
243        _rng: &dyn rand::SecureRandom,
244        msg: &[u8],
245        signature: &mut [u8],
246    ) -> Result<(), Unspecified> {
247        let encoding = padding_alg.encoding();
248        let padding_fn = if let RsaPadding::RSA_PKCS1_PSS_PADDING = encoding.padding() {
249            Some(|pctx: *mut EVP_PKEY_CTX| {
250                configure_rsa_pkcs1_pss_padding(
251                    pctx,
252                    encoding.pss_salt_length(),
253                    encoding.digest_algorithm(),
254                    &self.evp_pkey,
255                )
256            })
257        } else {
258            None
259        };
260
261        let sig_bytes = self
262            .evp_pkey
263            .sign(msg, Some(encoding.digest_algorithm()), padding_fn)?;
264
265        signature.copy_from_slice(&sig_bytes);
266        Ok(())
267    }
268
269    /// The `digest` is padded using the padding algorithm
270    /// from `padding_alg`. The signature is written into `signature`;
271    /// `signature`'s length must be exactly the length returned by
272    /// `public_modulus_len()`.
273    ///
274    /// # *ring* Compatibility
275    /// Our implementation ignores the `SecureRandom` parameter.
276    //
277    // # FIPS
278    // Not allowed
279    //
280    /// # Errors
281    /// `error::Unspecified` on error.
282    /// With "fips" feature enabled, errors if digest length is greater than `u32::MAX`.
283    pub fn sign_digest(
284        &self,
285        padding_alg: &'static dyn RsaEncoding,
286        digest: &Digest,
287        signature: &mut [u8],
288    ) -> Result<(), Unspecified> {
289        let encoding = padding_alg.encoding();
290        if encoding.digest_algorithm() != digest.algorithm() {
291            return Err(Unspecified);
292        }
293
294        let padding_fn = Some({
295            |pctx: *mut EVP_PKEY_CTX| {
296                let evp_md = match_digest_type(&digest.algorithm().id);
297                if 1 != unsafe { EVP_PKEY_CTX_set_signature_md(pctx, evp_md.as_const_ptr()) } {
298                    return Err(());
299                }
300                if let RsaPadding::RSA_PKCS1_PSS_PADDING = encoding.padding() {
301                    configure_rsa_pkcs1_pss_padding(
302                        pctx,
303                        encoding.pss_salt_length(),
304                        encoding.digest_algorithm(),
305                        &self.evp_pkey,
306                    )
307                } else {
308                    Ok(())
309                }
310            }
311        });
312
313        let sig_bytes = self.evp_pkey.sign_digest(digest, padding_fn)?;
314
315        signature.copy_from_slice(&sig_bytes);
316        Ok(())
317    }

334    /// Constructs an RSA private key from its big-endian-encoded components.
335    ///
336    /// All components, including the CRT parameters (`dP`, `dQ`, `qInv`), are
337    /// required and are validated for consistency with one another: the key is
338    /// rejected unless `n == p * q`, `d * e == 1 (mod p-1)`,
339    /// `d * e == 1 (mod q-1)`, `dP == d (mod p-1)`, `dQ == d (mod q-1)`, and
340    /// `qInv == q**-1 (mod p)`. No primality tests are performed on `p` and
341    /// `q`.
342    ///
343    /// Only two-prime (not multi-prime) keys are supported. The public
344    /// modulus (`n`) must be 2048 to 8192 bits. The public exponent (`e`)
345    /// must be odd, greater than 1, and no longer than 33 bits.
346    ///
347    /// The public components (`n` and `e`) must be encoded without leading
348    /// zero bytes, as documented on [`PublicKeyComponents`]. Leading zero
349    /// bytes are permitted on the private components.
350    ///
351    /// # *ring* compatibility
352    ///
353    /// *aws-lc-rs* does not impose the same limitations that *ring* does, so
354    /// keys rejected by *ring* may be accepted here. In particular:
355    /// * The public modulus may be up to 8192 bits, rather than 4096.
356    /// * The public exponent has no required minimum size, whereas *ring*
357    ///   requires it to be at least 65537.
358    ///
359    /// In two respects *aws-lc-rs* is stricter than *ring*, so a key accepted
360    /// by *ring* may be rejected here:
361    /// * *ring* never uses `d` and so does not fully validate it. We do
362    ///   validate `d`, which means a key carrying a placeholder or otherwise
363    ///   inconsistent `d` is rejected.
364    /// * *ring* defers validation of the CRT parameters until the key is used
365    ///   for signing. We validate them here, so an inconsistent key fails at
366    ///   construction rather than at first use.
367    ///
368    /// # Errors
369    /// `KeyRejected` if the components do not form a valid, supported RSA
370    /// private key.
371    // The bindings use the standard RSA component names.
372    #[allow(clippy::many_single_char_names, clippy::similar_names)]
373    pub fn from_components<Public, Private>(
374        components: &KeyPairComponents<Public, Private>,
375    ) -> Result<Self, KeyRejected>
376    where
377        Public: AsRef<[u8]>,
378        Private: AsRef<[u8]>,
379    {
380        let mut rsa = LcPtr::new(unsafe { RSA_new() })?;
381        let mut p = DetachableLcPtr::try_from(components.p.as_ref())?;
382        let mut q = DetachableLcPtr::try_from(components.q.as_ref())?;
383        if 1 != unsafe { RSA_set0_factors(rsa.as_mut_ptr(), p.as_mut_ptr(), q.as_mut_ptr()) } {
384            return Err(KeyRejected::unspecified());
385        }
386        p.detach();
387        q.detach();
388
389        let mut n = public_component_to_bn(components.public_key.n.as_ref())?;
390        let mut e = public_component_to_bn(components.public_key.e.as_ref())?;
391        let mut d = DetachableLcPtr::try_from(components.d.as_ref())?;
392        if 1 != unsafe {
393            RSA_set0_key(
394                rsa.as_mut_ptr(),
395                n.as_mut_ptr(),
396                e.as_mut_ptr(),
397                d.as_mut_ptr(),
398            )
399        } {
400            return Err(KeyRejected::unspecified());
401        }
402        n.detach();
403        e.detach();
404        d.detach();
405
406        let mut dmp1 = DetachableLcPtr::try_from(components.dP.as_ref())?;
407        let mut dmq1 = DetachableLcPtr::try_from(components.dQ.as_ref())?;
408        let mut iqmp = DetachableLcPtr::try_from(components.qInv.as_ref())?;
409        if 1 != unsafe {
410            RSA_set0_crt_params(
411                rsa.as_mut_ptr(),
412                dmp1.as_mut_ptr(),
413                dmq1.as_mut_ptr(),
414                iqmp.as_mut_ptr(),
415            )
416        } {
417            return Err(KeyRejected::unspecified());
418        }
419        dmp1.detach();
420        dmq1.detach();
421        iqmp.detach();
422
423        if 1 != unsafe { RSA_check_key(rsa.as_mut_ptr()) } {
424            return Err(KeyRejected::inconsistent_components());
425        }
426        let mut evp_pkey = LcPtr::new(unsafe { EVP_PKEY_new() })?;
427        // `EVP_PKEY_set1_RSA` takes a reference on `rsa` rather than ownership
428        // of it, so `rsa` is intentionally left attached and is released when it
429        // goes out of scope. Elsewhere we use `EVP_PKEY_assign_RSA`, which
430        // requires detaching the `RSA` on success.
431        if 1 != unsafe { EVP_PKEY_set1_RSA(evp_pkey.as_mut_ptr(), rsa.as_mut_ptr()) } {
432            return Err(KeyRejected::unspecified());
433        }
434
435        Self::new(evp_pkey)
436    }
437}

488    pub(super) fn new(evp_pkey: &LcPtr<EVP_PKEY>) -> Result<Self, KeyRejected> {
489        let key = encoding::rfc8017::encode_public_key_der(evp_pkey)?;
490        #[cfg(feature = "ring-io")]

492            let evp_pkey = evp_pkey.as_const();
493            let pubkey = evp_pkey.get_rsa()?;
494            let modulus = pubkey
495                .project_const_lifetime(unsafe { |pubkey| RSA_get0_n(pubkey.as_const_ptr()) })?;
496            let modulus = modulus.to_be_bytes().into_boxed_slice();
497            let exponent = pubkey
498                .project_const_lifetime(unsafe { |pubkey| RSA_get0_e(pubkey.as_const_ptr()) })?;
499            let exponent = exponent.to_be_bytes().into_boxed_slice();
500            Ok(PublicKey {
501                key,
502                modulus,
503                exponent,
504            })
505        }
506
507        #[cfg(not(feature = "ring-io"))]
508        Ok(PublicKey { key })
509    }

600/// Converts one of the public components (`n` or `e`) of a `PublicKeyComponents`
601/// into a `BIGNUM`, enforcing the documented "without leading zeros" encoding.
602///
603/// Every API that consumes a `PublicKeyComponents` must go through this so that
604/// a given value is accepted (or rejected) consistently by all of them.
605fn public_component_to_bn(bytes: &[u8]) -> Result<DetachableLcPtr<BIGNUM>, KeyRejected> {
606    if bytes.is_empty() || bytes[0] == 0u8 {
607        return Err(KeyRejected::invalid_encoding());
608    }
609    Ok(DetachableLcPtr::try_from(bytes)?)
610}

612impl<B> PublicKeyComponents<B>
613where
614    B: AsRef<[u8]>,
615{
616    #[inline]
617    fn build_rsa(&self) -> Result<LcPtr<EVP_PKEY>, ()> {
618        let mut n_bn = public_component_to_bn(self.n.as_ref()).map_err(|_| ())?;
619        let mut e_bn = public_component_to_bn(self.e.as_ref()).map_err(|_| ())?;
620
621        let mut rsa = DetachableLcPtr::new(unsafe { RSA_new() })?;
622        if 1 != unsafe {
623            RSA_set0_key(
624                rsa.as_mut_ptr(),
625                n_bn.as_mut_ptr(),
626                e_bn.as_mut_ptr(),
627                null_mut(),
628            )
629        } {
630            return Err(());
631        }
632        n_bn.detach();
633        e_bn.detach();
634
635        let mut pkey = LcPtr::new(unsafe { EVP_PKEY_new() })?;
636        if 1 != unsafe { EVP_PKEY_assign_RSA(pkey.as_mut_ptr(), rsa.as_mut_ptr()) } {
637            return Err(());
638        }
639        rsa.detach();
640
641        Ok(pkey)
642    }
643
644    /// Verifies that `signature` is a valid signature of `message` using `self`
645    /// as the public key. `params` determine what algorithm parameters
646    /// (padding, digest algorithm, key length range, etc.) are used in the
647    /// verification.
648    ///
649    /// # Errors
650    /// `error::Unspecified` if `message` was not verified.
651    pub fn verify(
652        &self,
653        params: &RsaParameters,
654        message: &[u8],
655        signature: &[u8],
656    ) -> Result<(), Unspecified> {
657        let rsa = self.build_rsa()?;
658        super::signature::verify_rsa_signature(
659            params.digest_algorithm(),
660            params.padding(),
661            params.pss_salt_length(),
662            &rsa,
663            message,
664            signature,
665            params.bit_size_range(),
666        )
667    }
668
669    /// Parses these components into a [`crate::signature::ParsedPublicKey`],
670    /// which can then be used to verify multiple signatures while amortizing
671    /// the cost of key parsing.
672    ///
673    /// `params` specifies the RSA verification algorithm, such as
674    /// [`crate::signature::RSA_PKCS1_2048_8192_SHA256`] or
675    /// [`crate::signature::RSA_PSS_2048_8192_SHA256`].
676    ///
677    /// Note that the algorithm's accepted key-size range is *not* enforced at
678    /// this point; that check is deferred to
679    /// [`crate::signature::ParsedPublicKey::verify_sig`], matching the
680    /// behavior of [`crate::signature::ParsedPublicKey::new`].
681    ///
682    /// # Errors
683    /// `KeyRejected` if `self` does not form a valid RSA public key.
684    ///
685    /// # Examples
686    ///
687    /// ```no_run
688    /// use aws_lc_rs::signature::{self, RsaPublicKeyComponents};
689    ///
690    /// # fn main() -> Result<(), Box<dyn std::error::Error>> {
691    /// # fn modulus_bytes() -> &'static [u8] { &[] }
692    /// # fn exponent_bytes() -> &'static [u8] { &[] }
693    /// # fn signature_bytes() -> &'static [u8] { &[] }
694    /// // Public key components (big-endian, no leading zeros) received
695    /// // out-of-band from a peer.
696    /// let components = RsaPublicKeyComponents {
697    ///     n: modulus_bytes(),
698    ///     e: exponent_bytes(),
699    /// };
700    /// let parsed =
701    ///     components.to_parsed_public_key(&signature::RSA_PKCS1_2048_8192_SHA256)?;
702    /// parsed.verify_sig(b"hello, world", signature_bytes())?;
703    /// # Ok(()) }
704    /// ```
705    pub fn to_parsed_public_key(
706        &self,
707        params: &'static RsaParameters,
708    ) -> Result<crate::signature::ParsedPublicKey, KeyRejected> {
709        let pkey = self
710            .build_rsa()
711            .map_err(|()| KeyRejected::inconsistent_components())?;
712        Ok(crate::signature::ParsedPublicKey::from_rsa_evp_pkey(
713            params, pkey,
714        )?)
715    }
716}

761pub(super) fn generate_rsa_key(size: c_int) -> Result<LcPtr<EVP_PKEY>, Unspecified> {
762    let params_fn = |ctx| {
763        if 1 == unsafe { EVP_PKEY_CTX_set_rsa_keygen_bits(ctx, size) } {
764            Ok(())
765        } else {
766            Err(())
767        }
768    };
769
770    LcPtr::<EVP_PKEY>::generate(EVP_PKEY_RSA, Some(params_fn))
771}

773#[cfg(feature = "fips")]
774#[must_use]
775pub(super) fn is_valid_fips_key(key: &LcPtr<EVP_PKEY>) -> bool {
776    // This should always be an RSA key and must-never panic.
777    let evp_pkey = key.as_const();
778    let rsa_key = evp_pkey.get_rsa().expect("RSA EVP_PKEY");
779
780    1 == unsafe { RSA_check_fips((rsa_key.as_const_ptr()).cast_mut()) }
781}

783pub(super) fn is_rsa_key(key: &LcPtr<EVP_PKEY>) -> bool {
784    let id = key.as_const().id();
785    id == EVP_PKEY_RSA || id == EVP_PKEY_RSA_PSS
786}

788pub(super) fn validate_rsa_key(key: &LcPtr<EVP_PKEY>) -> Result<(), KeyRejected> {
789    if !is_rsa_key(key) {
790        return Err(KeyRejected::unspecified());
791    }
792    match key.as_const().key_size_bits() {
793        2048..=8192 => Ok(()),
794        0..=2047 => Err(KeyRejected::too_small()),
795        _ => Err(KeyRejected::too_large()),
796    }
