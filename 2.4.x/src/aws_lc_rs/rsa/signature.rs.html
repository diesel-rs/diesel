44pub enum RsaPssSaltLength {
45    /// The salt length is set to the length of the used digest algorithm.
46    ///
47    /// Corresponds to [`RSA_PSS_SALTLEN_DIGEST`]. This is the default.
48    #[default]
49    Digest,
50    /// The salt length is set to the maximum possible length with respect to
51    /// the used digest algorithm and public key.
52    ///
53    /// This corresponds to `emLen - hLen - 2`, where
54    ///  * `emLen` is `(pub_key_size_bits + 6) / 8`, so depends on the mod size;
55    ///  * `hLen` is the digest algorithm's output length;
56    ///
57    /// A verification of a signature produced using this salt length may only
58    /// work when also using this maximum length, not any length up to it.
59    Max,
60    /// The salt length is set to the given custom value.
61    ///
62    /// The passed value is not particularly checked here, so may ultimately
63    /// result in failures.
64    Custom(u16),
65}
66
67impl RsaPssSaltLength {
68    pub(crate) fn into_value(
69        self,
70        algorithm: &'static digest::Algorithm,
71        pkey: &LcPtr<EVP_PKEY>,
72    ) -> Result<c_int, ()> {
73        match self {
74            RsaPssSaltLength::Digest => Ok(RSA_PSS_SALTLEN_DIGEST),
75            RsaPssSaltLength::Max => {
76                let digest_len = u16::try_from(algorithm.output_len()).map_err(|_| ())?;
77                let pkey_size_bits =
78                    u16::try_from(pkey.as_const().key_size_bits()).map_err(|_| ())?;
79                let em_len = pkey_size_bits.checked_add(6).ok_or(())? / 8;
80                let max_len = em_len
81                    .checked_sub(digest_len)
82                    .ok_or(())?
83                    .checked_sub(2)
84                    .ok_or(())?;
85                Ok(max_len.into())
86            }
87            RsaPssSaltLength::Custom(val) => Ok(val.into()),
88        }
89    }
90}

415pub(crate) fn configure_rsa_pkcs1_pss_padding(
416    pctx: *mut EVP_PKEY_CTX,
417    saltlen: RsaPssSaltLength,
418    algorithm: &'static digest::Algorithm,
419    pkey: &LcPtr<EVP_PKEY>,
420) -> Result<(), ()> {
421    if 1 != unsafe { EVP_PKEY_CTX_set_rsa_padding(pctx, RSA_PKCS1_PSS_PADDING) } {
422        return Err(());
423    }
424
425    let saltlen = saltlen.into_value(algorithm, pkey)?;
426
427    if 1 != unsafe { EVP_PKEY_CTX_set_rsa_pss_saltlen(pctx, saltlen) } {
428        return Err(());
429    }
430
431    Ok(())
432}

434#[inline]
435pub(crate) fn verify_rsa_signature(
436    algorithm: &'static digest::Algorithm,
437    padding: &'static RsaPadding,
438    pss_salt_length: RsaPssSaltLength,
439    public_key: &LcPtr<EVP_PKEY>,
440    msg: &[u8],
441    signature: &[u8],
442    allowed_bit_size: &RangeInclusive<u32>,
443) -> Result<(), Unspecified> {
444    if !allowed_bit_size.contains(&public_key.as_const().key_size_bits().try_into()?) {
445        return Err(Unspecified);
446    }
447
448    let padding_fn = if let RsaPadding::RSA_PKCS1_PSS_PADDING = padding {
449        Some(|pctx: *mut EVP_PKEY_CTX| {
450            configure_rsa_pkcs1_pss_padding(pctx, pss_salt_length, algorithm, public_key)
451        })
452    } else {
453        None
454    };
455
456    public_key.verify(msg, Some(algorithm), padding_fn, signature)
457}
458
459#[inline]
460pub(crate) fn verify_rsa_digest_signature(
461    algorithm: &'static digest::Algorithm,
462    padding: &'static RsaPadding,
463    pss_salt_length: RsaPssSaltLength,
464    public_key: &LcPtr<EVP_PKEY>,
465    digest: &Digest,
466    signature: &[u8],
467    allowed_bit_size: &RangeInclusive<u32>,
468) -> Result<(), Unspecified> {
469    // Enforced here so no caller can omit it; the ctx below is configured from `digest`.
470    if algorithm != digest.algorithm() {
471        return Err(Unspecified);
472    }
473
474    if !allowed_bit_size.contains(&public_key.as_const().key_size_bits().try_into()?) {
475        return Err(Unspecified);
476    }
477
478    let padding_fn = Some({
479        |pctx: *mut EVP_PKEY_CTX| {
480            let evp_md = match_digest_type(&digest.algorithm().id);
481            if 1 != unsafe { EVP_PKEY_CTX_set_signature_md(pctx, evp_md.as_const_ptr()) } {
482                return Err(());
483            }
484            if let RsaPadding::RSA_PKCS1_PSS_PADDING = padding {
485                configure_rsa_pkcs1_pss_padding(pctx, pss_salt_length, algorithm, public_key)
486            } else {
487                Ok(())
488            }
489        }
490    });
491
492    public_key.verify_digest_sig(digest, padding_fn, signature)
493}
