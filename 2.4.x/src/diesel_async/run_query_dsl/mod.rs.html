36    /// The `execute` method
37    ///
38    /// This trait should not be relied on directly by most apps. Its behavior is

74    /// The `load` method
75    ///
76    /// This trait should not be relied on directly by most apps. Its behavior is
77    /// provided by [`RunQueryDsl`]. However, you may need a where clause on this trait
78    /// to call `load` from generic code.
79    ///
80    /// [`RunQueryDsl`]: super::RunQueryDsl
81    pub trait LoadQuery<'query, Conn: AsyncConnectionCore, U> {
82        /// The future returned by [`LoadQuery::internal_load`]
83        type LoadFuture<'conn>: Future<Output = QueryResult<Self::Stream<'conn>>> + Send
84        where
85            Conn: 'conn;
86
87        /// The inner stream returned by [`LoadQuery::internal_load`]
88        type Stream<'conn>: Stream<Item = QueryResult<U>> + Send
89        where
90            Conn: 'conn;
91
92        /// Load this query
93        fn internal_load(self, conn: &mut Conn) -> Self::LoadFuture<'_>;
94    }
95
96    impl<'query, Conn, DB, T, U, ST> LoadQuery<'query, Conn, U> for T
97    where
98        Conn: AsyncConnectionCore<Backend = DB>,
99        U: Send,
100        DB: Backend + 'static,
101        T: AsQuery + Send + 'query,
102        T::Query: QueryFragment<DB> + QueryId + Send + 'query,
103        T::SqlType: CompatibleType<U, DB, SqlType = ST>,
104        U: FromSqlRow<ST, DB> + Send + 'static,
105        DB: QueryMetadata<T::SqlType>,
106        ST: 'static,
107    {
108        type LoadFuture<'conn>
109            = utils::MapOk<Conn::LoadFuture<'conn, 'query>, Self::Stream<'conn>>
110        where
111            Conn: 'conn;
112
113        type Stream<'conn>
114            = stream::Map<
115            Conn::Stream<'conn, 'query>,
116            fn(QueryResult<Conn::Row<'conn, 'query>>) -> QueryResult<U>,
117        >
118        where
119            Conn: 'conn;
120
121        fn internal_load(self, conn: &mut Conn) -> Self::LoadFuture<'_> {
122            utils::MapOk::new(conn.load(self), |stream| {
123                stream.map(|row| {
124                    U::build_from_row(&row?).map_err(diesel::result::Error::DeserializationError)
125                })
126            })
127        }
128    }
129}

161    /// Executes the given command, returning the number of rows affected.
162    ///
163    /// `execute` is usually used in conjunction with [`insert_into`](diesel::insert_into()),
164    /// [`update`](diesel::update()) and [`delete`](diesel::delete()) where the number of
165    /// affected rows is often enough information.
166    ///
167    /// When asking the database to return data from a query, [`load`](crate::run_query_dsl::RunQueryDsl::load()) should
168    /// probably be used instead.
169    ///
170    /// # Example
171    ///
172    /// ```rust
173    /// # include!("../doctest_setup.rs");
174    /// #
175    /// use diesel_async::RunQueryDsl;
176    ///
177    /// # #[tokio::main(flavor = "current_thread")]
178    /// # async fn main() {
179    /// #     run_test().await;
180    /// # }
181    /// #
182    /// # async fn run_test() -> QueryResult<()> {
183    /// #     use diesel::insert_into;
184    /// #     use schema::users::dsl::*;
185    /// #     let connection = &mut establish_connection().await;
186    /// let inserted_rows = insert_into(users)
187    ///     .values(name.eq("Ruby"))
188    ///     .execute(connection)
189    ///     .await?;
190    /// assert_eq!(1, inserted_rows);
191    ///
192    /// # #[cfg(not(feature = "sqlite"))]
193    /// let inserted_rows = insert_into(users)
194    ///     .values(&vec![name.eq("Jim"), name.eq("James")])
195    ///     .execute(connection)
196    ///     .await?;
197    /// # #[cfg(not(feature = "sqlite"))]
198    /// assert_eq!(2, inserted_rows);
199    /// #     Ok(())
200    /// # }
201    /// ```
202    fn execute<'conn, 'query>(self, conn: &'conn mut Conn) -> Conn::ExecuteFuture<'conn, 'query>
203    where
204        Conn: AsyncConnectionCore + Send,
205        Self: methods::ExecuteDsl<Conn> + 'query,
206    {
207        methods::ExecuteDsl::execute(self, conn)
208    }

459    /// Runs the command, and returns the affected row.
460    ///
461    /// `Err(NotFound)` will be returned if the query affected 0 rows. You can

467    /// unless a returning clause was already specified.
468    ///
469    /// This method only returns the first row that was affected, even if more
470    /// rows are affected.
471    ///
472    /// # Example
473    ///
474    /// ```rust
475    /// # include!("../doctest_setup.rs");
476    /// use diesel_async::RunQueryDsl;
477    ///
478    /// #
479    /// # #[tokio::main(flavor = "current_thread")]

482    /// # }

485    /// # async fn run_test() -> QueryResult<()> {
486    /// #     use diesel::{insert_into, update};
487    /// #     use schema::users::dsl::*;
488    /// #     let connection = &mut establish_connection().await;
489    /// let inserted_row = insert_into(users)
490    ///     .values(name.eq("Ruby"))

492    ///     .await?;

602/// Sugar for types which implement both `AsChangeset` and `Identifiable`
603///
604/// On backends which support the `RETURNING` keyword,
605/// `foo.save_changes(&conn)` is equivalent to
606/// `update(&foo).set(&foo).get_result(&conn)`.
607/// On other backends, two queries will be executed.
608///
609/// # Example
610///
611/// ```rust
612/// # include!("../doctest_setup.rs");
613/// # use schema::animals;
614/// #
615/// use diesel_async::{SaveChangesDsl, AsyncConnection};
616///
617/// #[derive(Queryable, Debug, PartialEq)]
618/// struct Animal {
619///    id: i32,
620///    species: String,
621///    legs: i32,
622///    name: Option<String>,
623/// }
624///
625/// #[derive(AsChangeset, Identifiable)]
626/// #[diesel(table_name = animals)]
627/// struct AnimalForm<'a> {
628///     id: i32,
629///     name: &'a str,
630/// }
631///
632/// # #[tokio::main(flavor = "current_thread")]
633/// # async fn main() {
634/// #     run_test().await.unwrap();
635/// # }
636/// #
637/// # async fn run_test() -> QueryResult<()> {
638/// #     use self::animals::dsl::*;
639/// #     let connection = &mut establish_connection().await;
640/// let form = AnimalForm { id: 2, name: "Super scary" };
641/// # #[cfg(not(feature = "sqlite"))]
642/// let changed_animal = form.save_changes(connection).await?;
643/// let expected_animal = Animal {
644///     id: 2,
645///     species: String::from("spider"),
646///     legs: 8,
647///     name: Some(String::from("Super scary")),
648/// };
649/// # #[cfg(not(feature = "sqlite"))]
650/// assert_eq!(expected_animal, changed_animal);
651/// #     Ok(())
652/// # }
653/// ```
654pub trait SaveChangesDsl<Conn> {
655    /// See the trait documentation
656    fn save_changes<'life0, 'async_trait, T>(
657        self,
658        connection: &'life0 mut Conn,
659    ) -> impl Future<Output = QueryResult<T>> + Send + 'async_trait
660    where
661        Self: Sized + diesel::prelude::Identifiable,
662        Conn: UpdateAndFetchResults<Self, T>,
663        T: 'async_trait,
664        'life0: 'async_trait,
665        Self: ::core::marker::Send + 'async_trait,
666    {
667        connection.update_and_fetch(self)
668    }
669}

671impl<T, Conn> SaveChangesDsl<Conn> for T where
672    T: Copy + AsChangeset<Target = <T as HasTable>::Table> + IntoUpdateTarget
673{
674}

676/// A trait defining how to update a record and fetch the updated entry
677/// on a certain backend.
678///
679/// The only case where it is required to work with this trait is while
680/// implementing a new connection type.
681/// Otherwise use [`SaveChangesDsl`]
682///
683/// For implementing this trait for a custom backend:
684/// * The `Changes` generic parameter represents the changeset that should be stored
685/// * The `Output` generic parameter represents the type of the response.
686pub trait UpdateAndFetchResults<Changes, Output>: AsyncConnectionCore
687where
688    Changes: diesel::prelude::Identifiable + HasTable,
689{
690    /// See the traits documentation.
691    fn update_and_fetch<'conn, 'changes>(
692        &'conn mut self,
693        changeset: Changes,
694    ) -> BoxFuture<'changes, QueryResult<Output>>
695    // cannot use impl future due to rustc bugs
696    // https://github.com/rust-lang/rust/issues/135619
697    //impl Future<Output = QueryResult<Output>> + Send + 'changes
698    where
699        Changes: 'changes,
700        'conn: 'changes,
701        Self: 'changes;
702}
703
704#[cfg(any(feature = "mysql", feature = "mariadb"))]
705impl<'b, Changes, Output, Tab, V, DB> UpdateAndFetchResults<Changes, Output>
706    for crate::mysql_like::AsyncMysqlLikeConnection<DB>
707where
708    Output: Send + 'static,
709    Changes:
710        Copy + AsChangeset<Target = Tab> + Send + diesel::associations::Identifiable<Table = Tab>,
711    Tab: diesel::Table + diesel::query_dsl::methods::FindDsl<Changes::Id> + 'b,
712    diesel::dsl::Find<Tab, Changes::Id>: IntoUpdateTarget<Table = Tab, WhereClause = V>,
713    diesel::query_builder::UpdateStatement<Tab, V, Changes::Changeset>:
714        diesel::query_builder::AsQuery,
715    diesel::dsl::Update<Changes, Changes>: methods::ExecuteDsl<Self>,
716    V: Send + 'b,
717    DB: MysqlLikeAsyncBackend,
718    Self: AsyncConnectionCore,
719    Changes::Changeset: Send + 'b,
720    Changes::Id: 'b,
721    Tab::FromClause: Send,
722    diesel::dsl::Find<Changes::Table, Changes::Id>: methods::LoadQuery<'b, crate::mysql_like::AsyncMysqlLikeConnection<DB>, Output>
723        + RunQueryDsl<crate::mysql_like::AsyncMysqlLikeConnection<DB>>
724        + Send,
725{
726    fn update_and_fetch<'conn, 'changes>(
727        &'conn mut self,
728        changeset: Changes,
729    ) -> BoxFuture<'changes, QueryResult<Output>>
730    where
731        Changes: 'changes,
732        Changes::Changeset: 'changes,
733        'conn: 'changes,
734        Self: 'changes,
735    {
736        async move {
737            diesel::update(changeset)
738                .set(changeset)
739                .execute(self)
740                .await?;
741            Changes::table().find(changeset.id()).get_result(self).await
742        }
743        .boxed()
744    }
745}
