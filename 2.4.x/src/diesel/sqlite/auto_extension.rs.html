43/// Registers an auto-extension that runs for every SQLite connection opened in
44/// this process, including non-Diesel ones.
45///
46/// This is a safe wrapper around [`sqlite3_auto_extension`][docs]. The callback
47/// receives the [`SqliteConnection`] being opened and returns `Ok(())` to
48/// continue or an error to fail the open. Use it to register SQL functions,
49/// collations, or aggregates through the usual connection API, or to initialize
50/// a statically linked C extension such as Spatialite or sqlite-vec via
51/// [`SqliteConnection::with_raw_connection`].
52///
53/// Call this before opening any connection. Extensions run in registration
54/// order, and the first error aborts the open. The callback must be a `fn` item
55/// or a closure that captures only zero-sized values (enforced at compile
56/// time), and registering the same `fn` twice is a no-op. It may run on several
57/// threads at once and must not open another connection (which would re-enter
58/// the auto-extensions and recurse) or call [`register_auto_extension`],
59/// [`cancel_auto_extension`], or [`reset_auto_extension`]. Panics are caught and
60/// turned into a failed open.
61///
62/// [docs]: https://www.sqlite.org/c3ref/auto_extension.html
63///
64/// # Example
65///
66/// ```rust
67/// # // This test registers Rust callbacks that sqlite invokes through the
68/// # // FFI boundary, which is not supported by miri `-Zmiri-native-lib`.
69/// # #[cfg(not(miri))]
70/// # fn main() {
71/// use diesel::dsl::sql;
72/// use diesel::prelude::*;
73/// use diesel::sql_types::Integer;
74/// use diesel::sqlite::{register_auto_extension, reset_auto_extension, SqliteConnection, BorrowedSqliteConnection};
75/// use std::pin::Pin;
76///
77/// // Registers a case-insensitive collation on every new connection.
78/// fn my_ext(conn: &mut BorrowedSqliteConnection) -> QueryResult<()> {
79///     unsafe {
80///         // SAFETY: This is safe as long as we don't move or replace out the
81///         // inner SqliteConnection. None of the functions defined on SqliteConnection
82///         // do that
83///         conn.inner_mut().register_collation("RUSTNOCASE", |a, b| a.to_lowercase().cmp(&b.to_lowercase()))
84///     }
85/// }
86///
87/// register_auto_extension(my_ext).unwrap();
88///
89/// // Every future connection now has the collation.
90/// let mut conn = SqliteConnection::establish(":memory:").unwrap();
91/// let equal: i32 = sql::<Integer>("SELECT 'a' = 'A' COLLATE RUSTNOCASE")
92///     .get_result(&mut conn)
93///     .unwrap();
94/// assert_eq!(equal, 1);
95/// # reset_auto_extension();
96/// # }
97/// # #[cfg(miri)]
98/// # fn main() {}
99/// ```
100#[allow(unsafe_code)]
101pub fn register_auto_extension<F>(extension: F) -> QueryResult<()>
102where
103    F: Fn(&'_ mut BorrowedSqliteConnection) -> QueryResult<()> + Sync + 'static,
104{
105    // SAFETY: `entry_point` returns a stable function pointer with SQLite's callback ABI.
106    let result =
107        unsafe { auto_extension_ffi::sqlite3_auto_extension(Some(entry_point(extension))) };
108    if result == ffi::SQLITE_OK {
109        Ok(())
110    } else {
111        Err(DatabaseError(
112            DatabaseErrorKind::Unknown,
113            Box::new(ffi::code_to_str(result).to_string()),
114        ))
115    }
116}

118/// Removes a previously registered auto-extension, returning `true` if it was
119/// found ([docs][cancel_docs]).
120///
121/// Pass the same `fn` item given to [`register_auto_extension`]. A closure
122/// cannot be cancelled this way, because its type cannot be named again. Use
123/// [`reset_auto_extension`] to clear everything instead.
124///
125/// [cancel_docs]: https://www.sqlite.org/c3ref/cancel_auto_extension.html
126#[allow(unsafe_code)]
127pub fn cancel_auto_extension<F>(extension: F) -> bool
128where
129    F: Fn(&'_ mut BorrowedSqliteConnection) -> QueryResult<()> + Sync + 'static,
130{
131    // SAFETY: `entry_point` returns the same stable pointer used for registration.
132    unsafe { auto_extension_ffi::sqlite3_cancel_auto_extension(Some(entry_point(extension))) != 0 }
133}

136///
137/// After this call, no auto-extensions will run for newly opened connections.

140#[allow(unsafe_code)]

145/// Returns the trampoline for `F`. `extension` is taken by value to infer `F`,
146/// then `forget`-ten so the zero-sized callback stays conceptually alive for the
147/// process and its destructor never runs.
148fn entry_point<F>(extension: F) -> RawAutoExtension
149where
150    F: Fn(&'_ mut BorrowedSqliteConnection) -> QueryResult<()> + Sync + 'static,
151{
152    core::mem::forget(extension);
153    trampoline::<F>
154}

156/// The C entry point handed to SQLite, monomorphized per callback type `F` so
157/// each distinct callback maps to a distinct, stable address. SQLite's
158/// pointer-based deduplication and [`cancel_auto_extension`] rely on that.
159#[allow(unsafe_code)]
160unsafe extern "C" fn trampoline<F>(
161    db: *mut ffi::sqlite3,
162    pz_err_msg: *mut *mut c_char,
163    _p_api: RawApiPointer,
164) -> c_int
165where
166    F: Fn(&'_ mut BorrowedSqliteConnection) -> QueryResult<()> + Sync + 'static,
167{
168    const {
169        if !(core::mem::size_of::<F>() == 0) {
    {
        ::core::panicking::panic_fmt(format_args!("an auto-extension callback must not capture non-zero-sized state. Use a `fn` item or a closure that captures only zero-sized values"));
    }
};assert!(
170            core::mem::size_of::<F>() == 0,
171            "an auto-extension callback must not capture non-zero-sized state. \
172             Use a `fn` item or a closure that captures only zero-sized values"
173        );
174    }
175
176    // `_p_api` matters only for runtime-loaded shared libraries. Statically
177    // linked extensions link the SQLite symbols directly, so we ignore it.
178    let result: Result<(), String> =
179        crate::util::std_compat::catch_unwind(core::panic::AssertUnwindSafe(|| {
180            let Some(db) = core::ptr::NonNull::new(db) else {
181                return Err(String::from(
182                    "auto-extension received a null database handle",
183                ));
184            };
185            // Reconstruct a *reference* to the zero-sized callback, never an
186            // owned value, so its destructor never runs. `&F: Fn` because `F: Fn`.
187            // SAFETY: `F` is zero-sized (asserted above), so a dangling, aligned,
188            // non-null pointer is a valid `&F`, which `NonNull::dangling` provides.
189            let extension: &F = unsafe { core::ptr::NonNull::<F>::dangling().as_ref() };
190            // SAFETY: `db` is a valid handle for the duration of this call, and
191            // the borrowed connection does not take ownership of it.
192            unsafe { SqliteConnection::with_borrowed_connection(db, extension) }
193                .map_err(|e| e.to_string())
194        }))
195        .unwrap_or_else(|panic| {
196            Err(match panic_detail(panic) {
197                Some(message) => ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("auto-extension panicked: {0}",
                message))
    })alloc::format!("auto-extension panicked: {message}"),
198                None => String::from("auto-extension panicked"),
199            })
200        });
201
202    match result {
203        Ok(()) => ffi::SQLITE_OK,
204        Err(message) => {
205            set_error_message(pz_err_msg, &message);
206            ffi::SQLITE_ERROR
207        }
208    }
209}

211/// Best-effort message from a caught panic payload. The no_std `catch_unwind`
212/// carries no payload, so only the `std` variant can recover the text.
213#[cfg(feature = "std")]
214fn panic_detail(panic: alloc::boxed::Box<dyn core::any::Any + Send>) -> Option<String> {
215    panic
216        .downcast_ref::<&str>()
217        .map(|s| (*s).to_owned())
218        .or_else(|| panic.downcast_ref::<String>().cloned())
219}

221#[cfg(not(feature = "std"))]
222fn panic_detail(_panic: ()) -> Option<String> {
223    None

226/// Writes `message` into `*pz_err_msg` with `sqlite3_malloc`, which is the
227/// allocator SQLite later frees it with. The message is truncated at the first
228/// NUL byte to form a valid C string, and allocation failure is ignored.
229#[allow(unsafe_code)]
230fn set_error_message(pz_err_msg: *mut *mut c_char, message: &str) {
231    if pz_err_msg.is_null() {
232        return;
233    }
234
235    let bytes = message.as_bytes();
236    let len = bytes.iter().position(|&b| b == 0).unwrap_or(bytes.len());
237
238    // SQLite sizes allocations with a C `int`. A message that does not fit is
239    // dropped rather than truncated to a bogus length.
240    let Ok(size) = c_int::try_from(len + 1) else {
241        return;
242    };
243    let buffer = unsafe { ffi::sqlite3_malloc(size) } as *mut u8;
244    if buffer.is_null() {
245        return;
246    }
247
248    unsafe {
249        core::ptr::copy_nonoverlapping(bytes.as_ptr(), buffer, len);
250        *buffer.add(len) = 0;
251        *pz_err_msg = buffer as *mut c_char;
252    }
253}

306    fn auto_extension_lifecycle() {
307        let _lock = AUTO_EXT_TEST_LOCK.lock().unwrap_or_else(|e| e.into_inner());
308        let _guard = TestResetGuard;
309        reset_auto_extension();
310
311        // -- 1. register + new connection has the collation --
312        register_auto_extension(test_ext_init).unwrap();
313        assert_eq!(probe_collation().unwrap(), 1);
314
315        // -- 2. cancel + new connection does NOT have the collation --
316        let removed = cancel_auto_extension(test_ext_init);
317        assert!(

321        assert!(

326        // -- 3. cancel returns false for unregistered --
327        let removed = cancel_auto_extension(test_ext_init);

335        reset_auto_extension();

342        register_auto_extension(test_ext_init).unwrap();
343        register_auto_extension(test_ext_init).unwrap();

387                let mut err: *mut c_char = core::ptr::null_mut();

512    // Counts calls and destructor runs of the hook closures registered by
513    // `register_all_hooks`. Tests using them hold `AUTO_EXT_TEST_LOCK`.
514    mod hook_tokens {
515        use std::sync::atomic::{AtomicUsize, Ordering};
516        pub(super) static FIRED: AtomicUsize = AtomicUsize::new(0);
517        pub(super) static DROPPED: AtomicUsize = AtomicUsize::new(0);
518        /// Dropped together with its hook closure; panics on drop if the flag is set.
519        pub(super) struct Token(pub(super) bool);
520        impl Token {
521            pub(super) fn fire(&self) {
522                FIRED.fetch_add(1, Ordering::SeqCst);
523            }
524        }
525        impl Drop for Token {
526            fn drop(&mut self) {
527                DROPPED.fetch_add(1, Ordering::SeqCst);
528                if self.0 {
529                    panic!("hook closure dropped");
530                }
531            }
532        }
533    }

535    // Registers all nine hooks through the borrowed connection. The closure of hook
536    // number `panicking` (in `RawConnection` field order) panics when dropped.
537    #[allow(unsafe_code)]
538    fn register_all_hooks(conn: &mut BorrowedSqliteConnection, panicking: usize) {
539        use crate::sqlite::{
540            AuthorizerDecision, BusyDecision, CommitDecision, ProgressDecision, SqliteChangeOps,
541            SqliteTraceFlags, SqliteUpdateRouter,
542        };
543        use core::num::NonZeroU32;
544        use hook_tokens::Token;
545        let conn = unsafe { conn.inner_mut() };
546        let t = Token(panicking == 0);
547        conn.on_update(SqliteUpdateRouter::new().on_any(SqliteChangeOps::ALL, move |_| t.fire()));
548        let t = Token(panicking == 1);
549        conn.on_commit(move || {
550            t.fire();
551            CommitDecision::Proceed
552        });
553        let t = Token(panicking == 2);
554        conn.on_rollback(move || t.fire());
555        let t = Token(panicking == 3);
556        conn.on_progress(NonZeroU32::new(1).unwrap(), move || {
557            t.fire();
558            ProgressDecision::Continue
559        });
560        let t = Token(panicking == 4);
561        conn.on_wal(move |_, _, _| t.fire());
562        let t = Token(panicking == 5);
563        conn.on_busy(move |_| {
564            t.fire();
565            BusyDecision::GiveUp
566        });
567        let t = Token(panicking == 6);
568        conn.on_authorize(move |_| {
569            t.fire();
570            AuthorizerDecision::Allow
571        });
572        let t = Token(panicking == 7);
573        conn.on_trace(SqliteTraceFlags::all(), move |_| t.fire());
574        let t = Token(panicking == 8);
575        conn.on_collation_needed(move |_| {
576            t.fire();
577            None
578        });

658    // A panicking callback becomes `SQLITE_ERROR` with the payload recovered
659    // into the message (`&str` and `String` payloads, generic fallback
660    // otherwise), and drives the panic-unwind path through the drop guard. Gated
661    // off WASM, where `catch_unwind` aborts because `panic = "abort"`.
662    #[test]
663    #[allow(unsafe_code)]
664    #[cfg(not(all(target_family = "wasm", target_os = "unknown")))]
665    fn trampoline_reports_panic_message() {
666        fn panic_str(_conn: &mut BorrowedSqliteConnection) -> QueryResult<()> {
667            panic!("boom-str");
668        }
669        fn panic_string(_conn: &mut BorrowedSqliteConnection) -> QueryResult<()> {
670            panic!("boom-{}", 7);
671        }
672        fn panic_other(_conn: &mut BorrowedSqliteConnection) -> QueryResult<()> {
673            std::panic::panic_any(7_u8);
674        }
675
676        let cases: [(RawAutoExtension, &str); 3] = [
677            (entry_point(panic_str), "auto-extension panicked: boom-str"),
678            (entry_point(panic_string), "auto-extension panicked: boom-7"),
679            (entry_point(panic_other), "auto-extension panicked"),
680        ];
681
682        let mut conn = open_memory_connection();
683        // SAFETY: the pointer is only used while `conn` is alive.
684        unsafe {
685            conn.with_raw_connection(|db| {
686                for (tramp, expected) in cases {
687                    let mut err: *mut c_char = core::ptr::null_mut();
688                    let rc = tramp(db, &mut err, core::ptr::null());
689                    assert_eq!(rc, ffi::SQLITE_ERROR);
690                    assert!(!err.is_null());
691                    let message = core::ffi::CStr::from_ptr(err)
692                        .to_string_lossy()
693                        .into_owned();
694                    assert_eq!(message, expected);
695                    ffi::sqlite3_free(err as *mut core::ffi::c_void);
696                }
697            })
698        }
699    }
