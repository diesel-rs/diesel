45    ///
46    /// # Cancel safety
47    ///
48    /// This method is cancel safe.
49    ///
50    /// # Examples
51    ///
52    /// ```
53    /// # use tokio::io::AsyncBufRead;
54    /// use tokio::io::AsyncBufReadExt;
55    ///
56    /// # async fn dox(my_buf_read: impl AsyncBufRead + Unpin) -> std::io::Result<()> {
57    /// let mut lines = my_buf_read.lines();
58    ///
59    /// while let Some(line) = lines.next_line().await? {
60    ///     println!("length = {}", line.len())
61    /// }

63    /// # }
64    /// ```
65    pub async fn next_line(&mut self) -> io::Result<Option<String>> {
66        use std::future::poll_fn;
67
68        poll_fn(|cx| Pin::new(&mut *self).poll_next_line(cx)).await
69    }

94    /// Polls for the next line in the stream.
95    ///
96    /// This method returns:
97    ///
98    ///  * `Poll::Pending` if the next line is not yet available.
99    ///  * `Poll::Ready(Ok(Some(line)))` if the next line is available.
100    ///  * `Poll::Ready(Ok(None))` if there are no more lines in this stream.
101    ///  * `Poll::Ready(Err(err))` if an IO error occurred while reading the next line,
102    ///    or if the line was not valid UTF-8.
103    ///
104    /// A line that is not valid UTF-8 is returned as the [`FromUtf8Error`]
105    /// inside the error, and the next call reads the following line.
106    ///
107    /// [`FromUtf8Error`]: std::string::FromUtf8Error
108    ///
109    /// When the method returns `Poll::Pending`, the `Waker` in the provided
110    /// `Context` is scheduled to receive a wakeup when more bytes become
111    /// available on the underlying IO resource.  Note that on multiple calls to
112    /// `poll_next_line`, only the `Waker` from the `Context` passed to the most
113    /// recent call is scheduled to receive a wakeup.
114    pub fn poll_next_line(
115        self: Pin<&mut Self>,
116        cx: &mut Context<'_>,
117    ) -> Poll<io::Result<Option<String>>> {
118        let me = self.project();
119
120        let mut read = 0;
121        let n = match read_until_internal(me.reader, cx, b'\n', me.buf, &mut read) {
    ::core::task::Poll::Ready(t) => t,
    ::core::task::Poll::Pending => { return ::core::task::Poll::Pending; }
}ready!(read_until_internal(me.reader, cx, b'\n', me.buf, &mut read))?;
122
123        if n == 0 && me.buf.is_empty() {
124            return Poll::Ready(Ok(None));
125        }
126
127        let mut bytes = mem::take(me.buf);
128
129        if bytes.last() == Some(&b'\n') {
130            bytes.pop();
131
132            if bytes.last() == Some(&b'\r') {
133                bytes.pop();
134            }
135        }
136
137        match String::from_utf8(bytes) {
138            Ok(line) => Poll::Ready(Ok(Some(line))),
139            Err(err) => Poll::Ready(Err(io::Error::new(io::ErrorKind::InvalidData, err))),
140        }
141    }
142}
