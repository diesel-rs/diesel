10    /// Wraps a type that is [`AsyncWrite`] and [`AsyncRead`], and buffers its input and output.
11    ///
12    /// It can be excessively inefficient to work directly with something that implements [`AsyncWrite`]
13    /// and [`AsyncRead`]. For example, every `write`, however small, has to traverse the syscall
14    /// interface, and similarly, every read has to do the same. The [`BufWriter`] and [`BufReader`]
15    /// types aid with these problems respectively, but do so in only one direction. `BufStream` wraps
16    /// one in the other so that both directions are buffered. See their documentation for details.
17    #[derive(Debug)]
18    #[cfg_attr(docsrs, doc(cfg(feature = "io-util")))]
19    pub struct BufStream<RW> {
20        #[pin]
21        inner: BufReader<BufWriter<RW>>,
22    }
23}

25impl<RW: AsyncRead + AsyncWrite> BufStream<RW> {
26    /// Wraps a type in both [`BufWriter`] and [`BufReader`].
27    ///
28    /// See the documentation for those types and [`BufStream`] for details.
29    pub fn new(stream: RW) -> BufStream<RW> {
30        BufStream {
31            inner: BufReader::new(BufWriter::new(stream)),
32        }
33    }

35    /// Creates a `BufStream` with the specified [`BufReader`] capacity and [`BufWriter`]
36    /// capacity.
37    ///
38    /// See the documentation for those types and [`BufStream`] for details.
39    ///
40    /// # Panics
41    ///
42    /// Panics if `reader_capacity` is zero.
43    #[track_caller]
44    pub fn with_capacity(
45        reader_capacity: usize,
46        writer_capacity: usize,
47        stream: RW,
48    ) -> BufStream<RW> {
49        BufStream {
50            inner: BufReader::with_capacity(
51                reader_capacity,
52                BufWriter::with_capacity(writer_capacity, stream),
53            ),
54        }
55    }

57    /// Gets a reference to the underlying I/O object.

62    }

64    /// Gets a mutable reference to the underlying I/O object.
65    ///
66    /// It is inadvisable to directly read from the underlying I/O object.
67    pub fn get_mut(&mut self) -> &mut RW {
68        self.inner.get_mut().get_mut()
69    }
70
71    /// Gets a pinned mutable reference to the underlying I/O object.
72    ///
73    /// It is inadvisable to directly read from the underlying I/O object.
74    pub fn get_pin_mut(self: Pin<&mut Self>) -> Pin<&mut RW> {
75        self.project().inner.get_pin_mut().get_pin_mut()
76    }
77
78    /// Consumes this `BufStream`, returning the underlying I/O object.
79    ///
80    /// Note that any leftover data in the internal buffer is lost.
81    pub fn into_inner(self) -> RW {
82        self.inner.into_inner().into_inner()
83    }

166/// Seek to an offset, in bytes, in the underlying stream.
167///
168/// The position used for seeking with `SeekFrom::Current(_)` is the
169/// position the underlying stream would be at if the `BufStream` had no
170/// internal buffer.
171///
172/// Seeking always discards the internal buffer, even if the seek position
173/// would otherwise fall within it. This guarantees that calling
174/// `.into_inner()` immediately after a seek yields the underlying reader
175/// at the same position.
176///
177/// See [`AsyncSeek`] for more details.
178///
179/// Note: In the edge case where you're seeking with `SeekFrom::Current(n)`
180/// where `n` minus the internal buffer length overflows an `i64`, two
181/// seeks will be performed instead of one. If the second seek returns
182/// `Err`, the underlying reader will be left at the same position it would
183/// have if you called `seek` with `SeekFrom::Current(0)`.
184impl<RW: AsyncRead + AsyncWrite + AsyncSeek> AsyncSeek for BufStream<RW> {
185    fn start_seek(self: Pin<&mut Self>, position: SeekFrom) -> io::Result<()> {
186        self.project().inner.start_seek(position)
187    }
188
189    fn poll_complete(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<u64>> {
190        self.project().inner.poll_complete(cx)
191    }
192}
