42    type Output = io::Result<()>;

44    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<()>> {
45        const MAX_VECTOR_ELEMENTS: usize = 64;
46
47        let me = self.project();
48        while me.buf.has_remaining() {
49            let n = loop {
50                let result = if me.writer.is_write_vectored() {
51                    let mut slices = [IoSlice::new(&[]); MAX_VECTOR_ELEMENTS];
52                    let cnt = me.buf.chunks_vectored(&mut slices);
53                    match Pin::new(&mut *me.writer).poll_write_vectored(cx, &slices[..cnt]) {
    ::core::task::Poll::Ready(t) => t,
    ::core::task::Poll::Pending => { return ::core::task::Poll::Pending; }
}ready!(Pin::new(&mut *me.writer).poll_write_vectored(cx, &slices[..cnt]))
54                } else {
55                    match Pin::new(&mut *me.writer).poll_write(cx, me.buf.chunk()) {
    ::core::task::Poll::Ready(t) => t,
    ::core::task::Poll::Pending => { return ::core::task::Poll::Pending; }
}ready!(Pin::new(&mut *me.writer).poll_write(cx, me.buf.chunk()))
56                };
57
58                match result {
59                    Ok(n) => break n,
60                    Err(e) if e.kind() == io::ErrorKind::Interrupted => continue,
61                    Err(e) => return Poll::Ready(Err(e)),
62                }
63            };
64            me.buf.advance(n);
65            if n == 0 {
66                return Poll::Ready(Err(io::ErrorKind::WriteZero.into()));
67            }
68        }
69
70        Poll::Ready(Ok(()))
71    }
