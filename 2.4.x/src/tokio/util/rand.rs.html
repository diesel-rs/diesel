34impl RngSeed {
35    /// Creates a random seed using loom internally.
36    pub(crate) fn new() -> Self {
37        Self::from_u64(crate::loom::rand::seed())
38    }
39
40    fn from_u64(seed: u64) -> Self {
41        let one = (seed >> 32) as u32;
42        let two = seed as u32;
43
44        Self::from_pair(one, two)
45    }
46
47    fn from_pair(s: u32, r: u32) -> Self {
48        if s == 0 && r == 0 {
49            Self { s: 0, r: 1 }
50        } else {
51            Self { s, r }
52        }
53    }
54}

56impl FastRand {
57    /// Initialize a new fast random number generator using the default source of entropy.
58    pub(crate) fn new() -> FastRand {
59        FastRand::from_seed(RngSeed::new())
60    }

62    /// Initializes a new, thread-local, fast random number generator.
63    pub(crate) fn from_seed(seed: RngSeed) -> FastRand {
64        FastRand {
65            one: seed.s,
66            two: seed.r,
67        }
68    }

70    #[cfg(any(feature = "macros", feature = "sync", feature = "rt"))]
71    pub(crate) fn fastrand_n(&mut self, n: u32) -> u32 {
72        // This is similar to fastrand() % n, but faster.
73        // See https://lemire.me/blog/2016/06/27/a-fast-alternative-to-the-modulo-reduction/
74        let mul = (self.fastrand() as u64).wrapping_mul(n as u64);
75        (mul >> 32) as u32
76    }
77
78    fn fastrand(&mut self) -> u32 {
79        let mut s1 = self.one;
80        let s0 = self.two;
81
82        s1 ^= s1 << 17;
83        s1 = s1 ^ s0 ^ s1 >> 7 ^ s0 >> 16;
84
85        self.one = s0;
86        self.two = s1;
87
88        s0.wrapping_add(s1)
89    }
90}
