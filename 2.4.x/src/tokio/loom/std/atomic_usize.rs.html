6/// `AtomicUsize` providing an additional `unsync_load` function.
7pub(crate) struct AtomicUsize {
8    inner: UnsafeCell<std::sync::atomic::AtomicUsize>,
9}
10
11unsafe impl Send for AtomicUsize {}
12unsafe impl Sync for AtomicUsize {}
13impl panic::RefUnwindSafe for AtomicUsize {}
14impl panic::UnwindSafe for AtomicUsize {}
15
16impl AtomicUsize {
17    pub(crate) const fn new(val: usize) -> AtomicUsize {
18        let inner = UnsafeCell::new(std::sync::atomic::AtomicUsize::new(val));
19        AtomicUsize { inner }
20    }
21
22    /// Performs an unsynchronized load.
23    ///
24    /// # Safety
25    ///
26    /// All mutations must have happened before the unsynchronized load.
27    /// Additionally, there must be no concurrent mutations.
28    pub(crate) unsafe fn unsync_load(&self) -> usize {
29        unsafe { core::ptr::read(self.inner.get() as *const usize) }
30    }
31
32    pub(crate) fn with_mut<R>(&mut self, f: impl FnOnce(&mut usize) -> R) -> R {
33        // safety: we have mutable access
34        f(unsafe { (*self.inner.get()).get_mut() })
35    }
36
37    // TODO(MSRV 1.95): When bumping MSRV, switch to `try_update`.
38    pub(crate) fn fetch_update<F>(
39        &self,
40        set_order: std::sync::atomic::Ordering,
41        fetch_order: std::sync::atomic::Ordering,
42        f: F,
43    ) -> Result<usize, usize>
44    where
45        F: FnMut(usize) -> Option<usize>,
46    {
47        #[allow(deprecated)]
48        ops::Deref::deref(self).fetch_update(set_order, fetch_order, f)
49    }
50}
51
52impl ops::Deref for AtomicUsize {
53    type Target = std::sync::atomic::AtomicUsize;
54
55    fn deref(&self) -> &Self::Target {
56        // safety: it is always safe to access `&self` fns on the inner value as
57        // we never perform unsafe mutations.
58        unsafe { &*self.inner.get() }
59    }
60}
61
62impl ops::DerefMut for AtomicUsize {
63    fn deref_mut(&mut self) -> &mut Self::Target {
64        // safety: we hold `&mut self`
65        unsafe { &mut *self.inner.get() }
66    }
67}

69impl fmt::Debug for AtomicUsize {
70    fn fmt(&self, fmt: &mut fmt::Formatter<'_>) -> fmt::Result {
71        (**self).fmt(fmt)
72    }
