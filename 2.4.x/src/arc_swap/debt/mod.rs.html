1//! Debt handling.
2//!
3//! A debt is a reference count of a smart pointer that is owed. This module provides a lock-free
4//! storage for debts.
5//!
6//! Each thread has its own node with bunch of slots. Only that thread can allocate debts in there,
7//! but others are allowed to inspect and pay them. The nodes form a linked list for the reason of
8//! inspection. The nodes are never removed (even after the thread terminates), but if the thread
9//! gives it up, another (new) thread can claim it.
10//!
11//! The writers walk the whole chain and pay the debts (by bumping the ref counts) of the just
12//! removed pointer ‒ the confirmed ones and the unconfirmed ones taken on their own storage.
13//!
14//! Each node has some fast (but fallible) nodes and a fallback node, with different algorithms to
15//! claim them (see the relevant submodules).
16

27/// One debt slot.
28///
29/// It may contain an „owed“ reference count.
30#[derive(#[automatically_derived]impl ::core::fmt::Debug for Debt {
    #[inline]
    #[rustc_trivial_field_reads]
    fn fmt(&self, f: &mut ::core::fmt::Formatter) -> ::core::fmt::Result {
        ::core::fmt::Formatter::debug_struct_field2_finish(f,
            "Debt ptr storage", &self.ptr, &&self.storage)
    }
}Debug)]
31pub(crate) struct Debt {
32    /// The owed pointer (encoded), or `NONE`.
33    pub(crate) ptr: AtomicUsize,
34    /// Address of the storage an unconfirmed debt was taken on, or `CONFIRMED`.
35    ///
36    /// An unconfirmed debt may name a reused address, so writers skip other storages' unconfirmed
37    /// debts. Confirmed debts carry no key, because the `ArcSwap` may move while a guard lives.
38    storage: AtomicUsize,
39}
40
41impl Debt {
42    /// The `storage` of a confirmed debt (no `AtomicPtr` lives at address 0).
43    const CONFIRMED: usize = 0;
44
45    /// The value of pointer `3` should be pretty safe, for two reasons:
46    ///
47    /// * It's an odd number, but the pointers we have are likely aligned at least to the word size,
48    ///   because the data at the end of the `Arc` has the counters.
49    /// * It's in the very first page where NULL lives, so it's not mapped.
50    ///
51    /// Encoded values can't be `3` either (that needs an address 1..=3, in the same page).
52    pub(crate) const NONE: usize = 0b11;
53
54    /// The value stored in a debt slot for the given pointer.
55    ///
56    /// NULL stays untagged, because it carries no reference count of any kind.
57    #[inline]
58    pub(crate) fn encode<T: RefCnt>(ptr: *const T::Base) -> usize {
59        if ptr.is_null() {
60            0
61        } else {
62            ptr as usize | T::TAG
63        }
64    }
65}
66
67impl Default for Debt {
68    fn default() -> Self {
69        Debt {
70            ptr: AtomicUsize::new(Self::NONE),
71            storage: AtomicUsize::new(Self::CONFIRMED),
72        }
73    }
74}
75
76impl Debt {
77    /// Marks the debt confirmed: writers of any storage may pay it from now on.
78    #[inline]
79    pub(crate) fn mark_confirmed(&self) {
80        // Relaxed: a writer of this storage matches the address anyway and a writer of a moved
81        // one runs after the move, which needs exclusive access, so after the whole `load`.
82        self.storage.store(Self::CONFIRMED, Relaxed);
83    }
84
85    /// Tries to pay the given debt.
86    ///
87    /// If the debt is still there, for the given pointer, it is paid and `true` is returned. If it
88    /// is empty or if there's some other pointer, it is not paid and `false` is returned, meaning
89    /// the debt was paid previously by someone else.
90    ///
91    /// # Notes
92    ///
93    /// * It is possible that someone paid the debt and then someone else put a debt for the same
94    ///   pointer in there. This is fine, as we'll just pay the debt for that someone else.
95    /// * Only the reader that put the debt there pays it this way, so it knows the type.
96    /// * The tag lets `Arc` and `Weak` of the same object coexist. It also relies on the fact the
97    ///   same thing is not stuffed both inside an `Arc` and `Rc` or something like that, but that
98    ///   sounds like a reasonable assumption. Someone storing it through `ArcSwap<T>` and someone
99    ///   else with `ArcSwapOption<T>` will work.
100    #[inline]
101    pub(crate) fn pay<T: RefCnt>(&self, ptr: *const T::Base) -> bool {
102        self.ptr
103            // On failure, we have observed that the debt has been paid, but we need to establish a
104            // happens-before relationship with that debt being paid before we do anything that
105            // relies on the Arc's strong counter being incremented, so we need to Acquire.
106            // Arc does not sufficiently take care of this, as the memory model permits its Drop's
107            // `fetch_sub` to observe itself to be last remaining instance before the `fetch_add`s
108            // from the debt being repaid, and only after that observation does Arc have an Acquire
109            // fence.
110            //
111            // Unfortunately, we need SeqCst on the success
112            .compare_exchange(Self::encode::<T>(ptr), Self::NONE, SeqCst, SeqCst)
113            .is_ok()
114    }
115
116    /// Writer-side payment: pays a debt on `ptr` if it is confirmed or was taken on `storage_addr`.
117    fn pay_for_storage<T: RefCnt>(&self, ptr: *const T::Base, storage_addr: usize) -> bool {
118        let encoded = Self::encode::<T>(ptr);
119        // SeqCst: keeps the load in the same total order as the writer's pointer swap and the
120        // reader's confirm load (see fast.rs); it also acquires the `storage` store.
121        if self.ptr.load(SeqCst) != encoded {
122            return false;
123        }
124        let owner = self.storage.load(Relaxed);
125        if owner != storage_addr && owner != Self::CONFIRMED {
126            return false;
127        }
128        self.ptr
129            .compare_exchange(encoded, Self::NONE, SeqCst, SeqCst)
130            .is_ok()
131    }
132
133    /// Pays all the debts on the given pointer and the storage.
134    pub(crate) fn pay_all<T, R>(ptr: *const T::Base, storage_addr: usize, replacement: R)
135    where
136        T: RefCnt,
137        R: Fn() -> T,
138    {
139        LocalNode::with(|local| {
140            let val = unsafe { T::from_ptr(ptr) };
141            // Pre-pay one ref count that can be safely put into a debt slot to pay it.
142            T::inc(&val);
143
144            Node::traverse::<(), _>(|node| {
145                // Make the cooldown trick know we are poking into this node.
146                let _reservation = node.reserve_writer();
147
148                local.help(node, storage_addr, &replacement);
149
150                let all_slots = node
151                    .fast_slots()
152                    .chain(core::iter::once(node.helping_slot()));
153                for slot in all_slots {
154                    // Skips unconfirmed debts of other storages (they may name a reused address).
155                    if slot.pay_for_storage::<T>(ptr, storage_addr) {
156                        // Pre-pay one more, for another future slot
157                        T::inc(&val);
158                    }
159                }
160
161                None
162            });
163            // Implicit dec by dropping val in here, pair for the above
164        })
165    }
166}
