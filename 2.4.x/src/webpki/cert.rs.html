29/// A parsed X509 certificate.
30pub struct Cert<'a> {
31    pub(crate) serial: untrusted::Input<'a>,
32    pub(crate) signed_data: SignedData<'a>,
33    pub(crate) issuer: untrusted::Input<'a>,
34    pub(crate) validity: untrusted::Input<'a>,
35    pub(crate) subject: untrusted::Input<'a>,
36    pub(crate) spki: untrusted::Input<'a>,
37
38    pub(crate) basic_constraints: Option<untrusted::Input<'a>>,
39    // key usage (KU) extension (if any). When validating certificate revocation lists (CRLs) this
40    // field will be consulted to determine if the cert is allowed to sign CRLs. For cert validation
41    // this field is ignored (for more detail see in `verify_cert.rs` and
42    // `check_issuer_independent_properties`).
43    pub(crate) key_usage: Option<untrusted::Input<'a>>,
44    pub(crate) eku: Option<untrusted::Input<'a>>,
45    pub(crate) name_constraints: Option<untrusted::Input<'a>>,
46    pub(crate) subject_alt_name: Option<untrusted::Input<'a>>,
47    pub(crate) crl_distribution_points: Option<untrusted::Input<'a>>,
48    pub(crate) scts: Option<untrusted::Input<'a>>,
49
50    der: CertificateDer<'a>,
51}

53impl<'a> Cert<'a> {
54    pub(crate) fn for_trust_anchor(cert_der: untrusted::Input<'a>) -> Result<Self, Error> {
55        Self::from_input(cert_der, UnknownExtensionPolicy::IgnoreCritical)
56    }
57
58    pub(crate) fn from_der(cert_der: untrusted::Input<'a>) -> Result<Self, Error> {
59        Self::from_input(cert_der, UnknownExtensionPolicy::default())
60    }
61
62    fn from_input(
63        cert_der: untrusted::Input<'a>,
64        ext_policy: UnknownExtensionPolicy,
65    ) -> Result<Self, Error> {
66        let (tbs, signed_data) =
67            cert_der.read_all(Error::TrailingData(DerTypeId::Certificate), |cert_der| {
68                der::nested(
69                    cert_der,
70                    der::Tag::Sequence,
71                    Error::TrailingData(DerTypeId::SignedData),
72                    |der| {
73                        // limited to SEQUENCEs of size 2^16 or less.
74                        SignedData::from_der(der, der::TWO_BYTE_DER_SIZE)
75                    },
76                )
77            })?;
78
79        tbs.read_all(
80            Error::TrailingData(DerTypeId::CertificateTbsCertificate),
81            |tbs| {
82                version3(tbs)?;
83
84                let serial = lenient_certificate_serial_number(tbs)?;
85
86                let signature = der::expect_tag(tbs, der::Tag::Sequence)?;
87                // TODO: In mozilla::pkix, the comparison is done based on the
88                // normalized value (ignoring whether or not there is an optional NULL
89                // parameter for RSA-based algorithms), so this may be too strict.
90                if !public_values_eq(signature, signed_data.algorithm) {
91                    return Err(Error::SignatureAlgorithmMismatch);
92                }
93
94                let issuer = der::expect_tag(tbs, der::Tag::Sequence)?;
95                let validity = der::expect_tag(tbs, der::Tag::Sequence)?;
96                let subject = der::expect_tag(tbs, der::Tag::Sequence)?;
97                let spki = der::expect_tag(tbs, der::Tag::Sequence)?;
98
99                // In theory there could be fields [1] issuerUniqueID and [2]
100                // subjectUniqueID, but in practice there never are, and to keep the
101                // code small and simple we don't accept any certificates that do
102                // contain them.
103
104                let mut cert = Cert {
105                    signed_data,
106                    serial,
107                    issuer,
108                    validity,
109                    subject,
110                    spki,
111
112                    basic_constraints: None,
113                    key_usage: None,
114                    eku: None,
115                    name_constraints: None,
116                    subject_alt_name: None,
117                    crl_distribution_points: None,
118                    scts: None,
119
120                    der: CertificateDer::from(cert_der.as_slice_less_safe()),
121                };
122
123                // When used to read X509v3 Certificate.tbsCertificate.extensions, we allow
124                // the extensions to be empty.  This is in spite of RFC5280:
125                //
126                //   "If present, this field is a SEQUENCE of one or more certificate extensions."
127                //
128                // Unfortunately other implementations don't get this right, eg:
129                // - https://github.com/golang/go/issues/52319
130                // - https://github.com/openssl/openssl/issues/20877
131                const ALLOW_EMPTY: bool = true;
132
133                if !tbs.at_end() {
134                    der::nested(
135                        tbs,
136                        der::Tag::ContextSpecificConstructed3,
137                        Error::TrailingData(DerTypeId::CertificateExtensions),
138                        |tagged| {
139                            der::nested_of_mut(
140                                tagged,
141                                der::Tag::Sequence,
142                                der::Tag::Sequence,
143                                Error::TrailingData(DerTypeId::Extension),
144                                ALLOW_EMPTY,
145                                |extension| {
146                                    remember_cert_extension(
147                                        &mut cert,
148                                        &Extension::from_der(extension)?,
149                                        ext_policy,
150                                    )
151                                },
152                            )
153                        },
154                    )?;
155                }
156
157                Ok(cert)
158            },
159        )
160    }
161
162    /// Returns a list of valid DNS names provided in the subject alternative names extension
163    ///
164    /// This function must not be used to implement custom DNS name verification.
165    /// Checking that a certificate is valid for a given subject name should always be done with
166    /// [EndEntityCert::verify_is_valid_for_subject_name].
167    ///
168    /// [EndEntityCert::verify_is_valid_for_subject_name]: crate::EndEntityCert::verify_is_valid_for_subject_name
169    pub fn valid_dns_names(&self) -> impl Iterator<Item = &'a str> {
170        NameIterator::new(self.subject_alt_name).filter_map(|result| {
171            let presented_id = match result.ok()? {
172                GeneralName::DnsName(presented) => presented,
173                _ => return None,
174            };
175
176            // if the name could be converted to a DNS name, return it; otherwise,
177            // keep going.
178            let dns_str = core::str::from_utf8(presented_id.as_slice_less_safe()).ok()?;
179            match DnsName::try_from(dns_str) {
180                Ok(_) => Some(dns_str),
181                Err(_) => {
182                    match WildcardDnsNameRef::try_from_ascii(presented_id.as_slice_less_safe()) {
183                        Ok(wildcard_dns_name) => Some(wildcard_dns_name.as_str()),
184                        Err(_) => None,
185                    }
186                }
187            }
188        })
189    }
190
191    /// Returns a list of valid URI names provided in the subject alternative names extension
192    ///
193    /// This function returns URIs as strings without performing validation beyond checking that
194    /// they are valid UTF-8.
195    pub fn valid_uri_names(&self) -> impl Iterator<Item = &'a str> {
196        NameIterator::new(self.subject_alt_name).filter_map(|result| {
197            let presented_id = match result.ok()? {
198                GeneralName::UniformResourceIdentifier(presented) => presented,
199                _ => return None,
200            };
201
202            // if the URI can be converted to a valid UTF-8 string, return it; otherwise,
203            // keep going.
204            core::str::from_utf8(presented_id.as_slice_less_safe()).ok()
205        })
206    }

208    /// Raw certificate serial number.
209    ///
210    /// This is in big-endian byte order, in twos-complement encoding.
211    ///
212    /// If the caller were to add an `INTEGER` tag and suitable length, this
213    /// would become a valid DER encoding.
214    pub fn serial(&self) -> &[u8] {
215        self.serial.as_slice_less_safe()
216    }
217
218    /// Raw DER-encoded certificate issuer.
219    ///
220    /// This does not include the outer `SEQUENCE` tag or length.
221    pub fn issuer(&self) -> &[u8] {
222        self.issuer.as_slice_less_safe()
223    }
224
225    /// Raw DER encoded certificate subject.
226    ///
227    /// This does not include the outer `SEQUENCE` tag or length.
228    pub fn subject(&self) -> &[u8] {
229        self.subject.as_slice_less_safe()
230    }
231
232    /// Get the RFC 5280-compliant [`SubjectPublicKeyInfoDer`] (SPKI) of this [`Cert`].
233    ///
234    /// This **does** include the outer `SEQUENCE` tag and length.
235    #[cfg(feature = "alloc")]
236    pub fn subject_public_key_info(&self) -> SubjectPublicKeyInfoDer<'static> {
237        // Our SPKI representation contains only the content of the RFC 5280 SEQUENCE
238        // So we wrap the SPKI contents back into a properly-encoded ASN.1 SEQUENCE
239        SubjectPublicKeyInfoDer::from(der::asn1_wrap(
240            Tag::Sequence,
241            self.spki.as_slice_less_safe(),
242        ))
243    }
244
245    /// Returns an iterator over the certificate's cRLDistributionPoints extension values, if any.
246    pub(crate) fn crl_distribution_points(
247        &self,
248    ) -> Option<impl Iterator<Item = Result<CrlDistributionPoint<'a>, Error>>> {
249        self.crl_distribution_points.map(DerIterator::new)
250    }
251
252    /// Raw DER-encoded representation of the certificate.
253    pub fn der(&self) -> CertificateDer<'a> {
254        self.der.clone() // This is cheap, just cloning a reference.
255    }

258// mozilla::pkix supports v1, v2, v3, and v4, including both the implicit
259// (correct) and explicit (incorrect) encoding of v1. We allow only v3.
260fn version3(input: &mut untrusted::Reader<'_>) -> Result<(), Error> {
261    der::nested(
262        input,
263        der::Tag::ContextSpecificConstructed0,
264        Error::UnsupportedCertVersion,
265        |input| {
266            let version = u8::from_der(input)?;
267            if version != 2 {
268                // v3
269                return Err(Error::UnsupportedCertVersion);
270            }
271            Ok(())
272        },
273    )
274}

276pub(crate) fn lenient_certificate_serial_number<'a>(
277    input: &mut untrusted::Reader<'a>,
278) -> Result<untrusted::Input<'a>, Error> {
279    // https://tools.ietf.org/html/rfc5280#section-4.1.2.2:
280    // * Conforming CAs MUST NOT use serialNumber values longer than 20 octets."
281    // * "The serial number MUST be a positive integer [...]"
282    //
283    // However, we don't enforce these constraints, as there are widely-deployed trust anchors
284    // and many X.509 implementations in common use that violate these constraints. This is called
285    // out by the same section of RFC 5280 as cited above:
286    //   Note: Non-conforming CAs may issue certificates with serial numbers
287    //   that are negative or zero.  Certificate users SHOULD be prepared to
288    //   gracefully handle such certificates.
289    der::expect_tag(input, Tag::Integer)
290}

292fn remember_cert_extension<'a>(
293    cert: &mut Cert<'a>,
294    extension: &Extension<'a>,
295    ext_policy: UnknownExtensionPolicy,
296) -> Result<(), Error> {
297    // We don't do anything with certificate policies so we can safely ignore
298    // all policy-related stuff. We assume that the policy-related extensions
299    // are not marked critical.
300
301    if extension.id.as_slice_less_safe() == SCT_LIST_OID {
302        return set_extension_once(&mut cert.scts, || {
303            extension.value.read_all(Error::BadDer, |value| {
304                der::expect_tag(value, Tag::OctetString)
305            })
306        });
307    }
308
309    remember_extension(extension, ext_policy, |id| {
310        let out = match id {
311            // id-ce-keyUsage 2.5.29.15.
312            15 => &mut cert.key_usage,
313
314            // id-ce-subjectAltName 2.5.29.17
315            17 => &mut cert.subject_alt_name,
316
317            // id-ce-basicConstraints 2.5.29.19
318            19 => &mut cert.basic_constraints,
319
320            // id-ce-nameConstraints 2.5.29.30
321            30 => &mut cert.name_constraints,
322
323            // id-ce-cRLDistributionPoints 2.5.29.31
324            31 => &mut cert.crl_distribution_points,
325
326            // id-ce-extKeyUsage 2.5.29.37
327            37 => &mut cert.eku,
328
329            // Unsupported extension
330            _ => return extension.unsupported(ext_policy),
331        };
332
333        set_extension_once(out, || {
334            extension.value.read_all(Error::BadDer, |value| match id {
335                // Unlike the other extensions we remember KU is a BitString and not a Sequence. We
336                // read the raw bytes here and parse at the time of use.
337                15 => Ok(value.read_bytes_to_end()),
338                // All other remembered certificate extensions are wrapped in a Sequence.
339                _ => der::expect_tag(value, Tag::Sequence),
340            })
341        })
342    })
343}

345/// A certificate revocation list (CRL) distribution point, describing a source of
346/// CRL information for a given certificate as described in RFC 5280 section 4.2.3.13[^1].
347///
348/// [^1]: <https://datatracker.ietf.org/doc/html/rfc5280#section-4.2.1.13>
349pub(crate) struct CrlDistributionPoint<'a> {
350    /// distributionPoint describes the location of CRL information.
351    distribution_point: Option<untrusted::Input<'a>>,
352
353    /// reasons holds a bit flag set of certificate revocation reasons associated with the
354    /// CRL distribution point.
355    pub(crate) reasons: Option<der::BitStringFlags<'a>>,
356
357    /// when the CRL issuer is not the certificate issuer, crl_issuer identifies the issuer of the
358    /// CRL.
359    pub(crate) crl_issuer: Option<untrusted::Input<'a>>,
360}
361
362impl<'a> CrlDistributionPoint<'a> {
363    /// Return the distribution point names (if any).
364    pub(crate) fn names(&self) -> Result<Option<DistributionPointName<'a>>, Error> {
365        self.distribution_point
366            .map(|input| DistributionPointName::from_der(&mut untrusted::Reader::new(input)))
367            .transpose()
368    }
369}
370
371impl<'a> FromDer<'a> for CrlDistributionPoint<'a> {
372    fn from_der(reader: &mut untrusted::Reader<'a>) -> Result<Self, Error> {
373        // RFC 5280 section §4.2.1.13:
374        //   A DistributionPoint consists of three fields, each of which is optional:
375        //   distributionPoint, reasons, and cRLIssuer.
376        let mut result = CrlDistributionPoint {
377            distribution_point: None,
378            reasons: None,
379            crl_issuer: None,
380        };
381
382        der::nested(
383            reader,
384            Tag::Sequence,
385            Error::TrailingData(Self::TYPE_ID),
386            |der| {
387                const DISTRIBUTION_POINT_TAG: u8 = CONTEXT_SPECIFIC | CONSTRUCTED;
388                const REASONS_TAG: u8 = CONTEXT_SPECIFIC | 1;
389                const CRL_ISSUER_TAG: u8 = CONTEXT_SPECIFIC | CONSTRUCTED | 2;
390
391                while !der.at_end() {
392                    let (tag, value) = der::read_tag_and_get_value(der)?;
393                    match tag {
394                        DISTRIBUTION_POINT_TAG => {
395                            set_extension_once(&mut result.distribution_point, || Ok(value))?
396                        }
397                        REASONS_TAG => set_extension_once(&mut result.reasons, || {
398                            der::bit_string_flags(value)
399                        })?,
400                        CRL_ISSUER_TAG => set_extension_once(&mut result.crl_issuer, || Ok(value))?,
401                        _ => return Err(Error::BadDer),
402                    }
403                }
404
405                // RFC 5280 section §4.2.1.13:
406                //   a DistributionPoint MUST NOT consist of only the reasons field; either distributionPoint or
407                //   cRLIssuer MUST be present.
408                match (result.distribution_point, result.crl_issuer) {
409                    (None, None) => Err(Error::MalformedExtensions),
410                    _ => Ok(result),
411                }
412            },
413        )
414    }
415
416    const TYPE_ID: DerTypeId = DerTypeId::CrlDistributionPoint;
417}

697    #[test]
698    #[cfg(feature = "alloc")]
699    fn test_crl_distribution_point_multiple() {
700        let der = include_bytes!("../tests/crl_distrib_point/multiple_distribution_points.der");
701        let cert =
702            Cert::from_der(untrusted::Input::from(der)).expect("failed to parse certificate");
703
704        // We expect to be able to parse the intermediate certificate's CRL distribution points.
705        let crl_distribution_points = cert
706            .crl_distribution_points()
707            .expect("missing distribution points extension")
708            .collect::<Result<Vec<_>, Error>>()
709            .expect("failed to parse distribution points");
710
711        // There should be two distribution points present.
712        let (point_a, point_b) = (
713            crl_distribution_points
714                .first()
715                .expect("missing first distribution point"),
716            crl_distribution_points
717                .get(1)
718                .expect("missing second distribution point"),
719        );
720
721        fn get_names<'a>(
722            point: &'a CrlDistributionPoint<'a>,
723        ) -> impl Iterator<Item = Result<GeneralName<'a>, Error>> {
724            match point
725                .names()
726                .expect("failed to parse distribution point names")
727                .expect("missing distribution point name")
728            {
729                DistributionPointName::NameRelativeToCrlIssuer => {
730                    panic!("unexpected relative name")
731                }
732                DistributionPointName::FullName(names) => names,
733            }
734        }
735
736        fn uri_bytes<'a>(name: &'a GeneralName<'a>) -> &'a [u8] {
737            match name {
738                GeneralName::UniformResourceIdentifier(uri) => uri.as_slice_less_safe(),
739                _ => panic!("unexpected name type"),
740            }
741        }
742
743        // We expect to find three URIs across the two distribution points.
744        let expected_names = [
745            "http://example.com/crl.1.der".as_bytes(),
746            "http://example.com/crl.2.der".as_bytes(),
747            "http://example.com/crl.3.der".as_bytes(),
748        ];
749        let all_names = get_names(point_a)
750            .chain(get_names(point_b))
751            .collect::<Result<Vec<_>, Error>>()
752            .expect("failed to parse names");
753
754        assert_eq!(
755            all_names.iter().map(uri_bytes).collect::<Vec<_>>(),
756            expected_names
757        );
758    }
