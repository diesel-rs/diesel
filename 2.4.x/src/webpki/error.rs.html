281impl Error {
282    // Compare the Error with the new error by rank, returning the higher rank of the two as
283    // the most specific error.
284    pub(crate) fn most_specific(self, new: Self) -> Self {
285        // Assign an error a numeric value ranking it by specificity.
286        if self.rank() >= new.rank() { self } else { new }
287    }
288
289    // Return a numeric indication of how specific the error is, where an error with a higher rank
290    // is considered more useful to an end user than an error with a lower rank. This is used by
291    // Error::most_specific to compare two errors in order to return which is more specific.
292    #[allow(clippy::as_conversions)] // We won't exceed u32 errors.
293    pub(crate) fn rank(&self) -> u32 {
294        match &self {
295            // Errors related to certificate validity

373impl From<Error> for ControlFlow<Error, Error> {
374    fn from(value: Error) -> Self {
375        match value {
376            // If an error is fatal, we've exhausted the potential for continued search.
377            err if err.is_fatal() => Self::Break(err),
378            // Otherwise we've rejected one candidate chain, but may continue to search for others.
379            err => Self::Continue(err),
380        }
381    }
