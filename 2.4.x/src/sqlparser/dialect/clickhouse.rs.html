27impl Dialect for ClickHouseDialect {
28    fn is_identifier_start(&self, ch: char) -> bool {
29        // See https://clickhouse.com/docs/en/sql-reference/syntax/#syntax-identifiers
30        ch.is_ascii_lowercase() || ch.is_ascii_uppercase() || ch == '_'
31    }
32
33    /// Aliases are identifiers, quoted with `"` or a backtick, see
34    /// <https://clickhouse.com/docs/reference/syntax#identifiers>
35    fn supports_string_literal_column_alias(&self, _explicit: bool) -> bool {
36        false
37    }
38
39    /// See <https://clickhouse.com/docs/reference/syntax#identifiers>
40    fn supports_string_literal_table_alias(&self, _explicit: bool) -> bool {
41        false
42    }
43
44    fn is_identifier_part(&self, ch: char) -> bool {
45        // Undocumented, but the server accepts `$` in identifiers for MySQL compatibility
46        // since https://github.com/ClickHouse/ClickHouse/pull/14232
47        self.is_identifier_start(ch) || ch.is_ascii_digit() || ch == '$'
48    }
49
50    /// ClickHouse keywords are not reserved, so any of them can be a table name.
51    /// See <https://clickhouse.com/docs/en/sql-reference/syntax>
52    fn is_table_factor(&self, _kw: &Keyword, _parser: &mut Parser) -> bool {
53        true
54    }
55
56    /// ClickHouse accepts both backticks and double quotes, but its own
57    /// formatter emits backticks by default.
58    /// See <https://clickhouse.com/docs/en/sql-reference/syntax#identifiers>
59    fn identifier_quote_style(&self, _identifier: &str) -> Option<char> {
60        Some('`')
61    }
62
63    fn supports_string_literal_backslash_escape(&self) -> bool {
64        true
65    }
66
67    fn supports_select_wildcard_except(&self) -> bool {
68        true
69    }

76        true

80        true
