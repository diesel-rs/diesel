307    fn get_next_precedence(&self, parser: &Parser) -> Option<Result<u8, ParserError>> {
308        let token = parser.peek_token_ref();
309        {
    {
        let lvl = ::log::Level::Debug;
        if lvl <= ::log::STATIC_MAX_LEVEL && lvl <= ::log::max_level() {
            ::log::__private_api::log({ ::log::__private_api::GlobalLogger },
                format_args!("get_next_precedence() {0:?}", token), lvl,
                &("sqlparser::dialect::postgresql",
                        "sqlparser::dialect::postgresql",
                        ::log::__private_api::loc()), ());
        }
    }
};debug!("get_next_precedence() {token:?}");
310
311        // we only return some custom value here when the behaviour (not merely the numeric value) differs
312        // from the default implementation
313        match &token.token {
314            Token::Word(w)
315                if w.keyword == Keyword::COLLATE && !parser.in_column_definition_state() =>
316            {
317                Some(Ok(COLLATE_PREC))
318            }
319            Token::LBracket => Some(Ok(BRACKET_PREC)),
320            Token::Arrow
321            | Token::LongArrow
322            | Token::HashArrow
323            | Token::HashLongArrow
324            | Token::AtArrow
325            | Token::ArrowAt
326            | Token::HashMinus
327            | Token::AtQuestion
328            | Token::AtAt
329            | Token::Question
330            | Token::QuestionAnd
331            | Token::QuestionPipe
332            | Token::ExclamationMark
333            | Token::Overlap
334            | Token::CaretAt
335            | Token::StringConcat
336            | Token::Sharp
337            | Token::ShiftRight
338            | Token::ShiftLeft
339            | Token::CustomBinaryOperator(_) => Some(Ok(PG_OTHER_PREC)),
340            // lowest prec to prevent it from turning into a binary op
341            Token::Colon | Token::Assignment => Some(Ok(self.prec_unknown())),
342            _ => None,
343        }
344    }

370    fn prec_value(&self, prec: Precedence) -> u8 {
371        match prec {
372            Precedence::Period => PERIOD_PREC,
373            Precedence::DoubleColon => DOUBLE_COLON_PREC,
374            Precedence::AtTz => AT_TZ_PREC,
375            Precedence::MulDivModOp => MUL_DIV_MOD_OP_PREC,
376            Precedence::PlusMinus => PLUS_MINUS_PREC,
377            Precedence::Xor => XOR_PREC,
378            Precedence::Ampersand => PG_OTHER_PREC,
379            Precedence::Caret => CARET_PREC,
380            Precedence::Pipe => PG_OTHER_PREC,
381            Precedence::Colon => PG_OTHER_PREC,
382            Precedence::Between => BETWEEN_LIKE_PREC,
383            Precedence::Eq => EQ_PREC,
384            Precedence::Like => BETWEEN_LIKE_PREC,
385            Precedence::Is => IS_PREC,
386            Precedence::PgOther => PG_OTHER_PREC,
387            Precedence::UnaryNot => NOT_PREC,
388            Precedence::And => AND_PREC,
389            Precedence::Or => OR_PREC,
390        }
391    }
