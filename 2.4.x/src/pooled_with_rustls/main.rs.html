14async fn main() -> Result<(), Box<dyn std::error::Error>> {
15    let db_url = std::env::var("DATABASE_URL").expect("Env var `DATABASE_URL` not set");
16
17    let mut config = ManagerConfig::default();
18    config.custom_setup = Box::new(establish_connection);
19
20    // First we have to construct a connection manager with our custom `establish_connection`
21    // function
22    let mgr = AsyncDieselConnectionManager::<AsyncPgConnection>::new_with_config(db_url, config);
23    // From that connection we can then create a pool, here given with some example settings.
24    //
25    // This creates a TLS configuration that's equivalent to `libpq'` `sslmode=verify-full`, which
26    // means this will check whether the provided certificate is valid for the given database host.
27    //
28    // `libpq` does not perform these checks by default (https://www.postgresql.org/docs/current/libpq-connect.html)
29    // If you hit a TLS error while connecting to the database double check your certificates
30    let pool = Pool::builder()
31        .max_size(10)
32        .min_idle(Some(5))
33        .max_lifetime(Some(Duration::from_secs(60 * 60 * 24)))
34        .idle_timeout(Some(Duration::from_secs(60 * 2)))
35        .build(mgr)
36        .await?;
37
38    // Now we can use our pool to run queries over a TLS-secured connection:
39    let conn = pool.get().await?;
40    let _ = conn;
41
42    Ok(())
43}

45fn establish_connection(config: &str) -> BoxFuture<'_, ConnectionResult<AsyncPgConnection>> {
46    let fut = async {
47        // We first set up the way we want rustls to work.
48        let rustls_config =
49            ClientConfig::with_platform_verifier().expect("Failed to initialize rustls");
50        let tls = tokio_postgres_rustls::MakeRustlsConnect::new(rustls_config);
51        let (client, conn) = tokio_postgres::connect(config, tls)
52            .await
53            .map_err(|e| ConnectionError::BadConnection(e.to_string()))?;
54
55        AsyncPgConnection::try_from_client_and_connection(client, conn).await
56    };
57    fut.boxed()
58}
