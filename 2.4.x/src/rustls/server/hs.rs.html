39pub(super) fn can_resume(
40    suite: SupportedCipherSuite,
41    sni: &Option<DnsName<'_>>,
42    using_ems: bool,
43    resumedata: &persist::ServerSessionValue,
44) -> bool {
45    // The RFCs underspecify what happens if we try to resume to
46    // an unoffered/varying suite.  We merely don't resume in weird cases.
47    //
48    // RFC 6066 says "A server that implements this extension MUST NOT accept
49    // the request to resume the session if the server_name extension contains
50    // a different name. Instead, it proceeds with a full handshake to
51    // establish a new session."
52    //
53    // RFC 8446: "The server MUST ensure that it selects
54    // a compatible PSK (if any) and cipher suite."
55    resumedata.cipher_suite == suite.suite()
56        && (resumedata.extended_ms == using_ems || (resumedata.extended_ms && !using_ems))
57        && &resumedata.sni == sni
58}

102                .iter()

104                    their_protocols
105                        .iter()
106                        .any(|theirs| theirs.as_ref() == ours.as_slice())
107                })
108                .map(|bytes| ProtocolName::from(bytes.clone()));
109            if let Some(selected_protocol) = &cx.common.alpn_protocol {
110                {

                lvl,

                        ::log::__private_api::loc()), ());
        }
    }

113                    Some(SingleProtocolName::new(selected_protocol.clone()));
114            } else if !our_protocols.is_empty() {
115                return Err(cx.common.send_fatal_alert(
116                    AlertDescription::NoApplicationProtocol,
117                    Error::NoApplicationProtocol,
118                ));
119            }
120        }
121

123            // QUIC has strict ALPN, unlike TLS's more backwards-compatible behavior. RFC 9001
124            // says: "The server MUST treat the inability to select a compatible application
125            // protocol as a connection error of type 0x0178". We judge that ALPN was desired
126            // (rather than some out-of-band protocol negotiation mechanism) if and only if any ALPN

128            // successful establishment of connections between peers that can't understand

132            {
133                return Err(cx.common.send_fatal_alert(
134                    AlertDescription::NoApplicationProtocol,
135                    Error::NoApplicationProtocol,
136                ));
137            }
138
139            let transport_params = hello
140                .transport_parameters
141                .as_ref()
142                .or(hello
143                    .transport_parameters_draft
144                    .as_ref());
145            match transport_params {
146                Some(params) => cx.common.quic.params = Some(params.to_owned().into_vec()),
147                None => {
148                    return Err(cx
149                        .common
150                        .missing_extension(PeerMisbehaved::MissingQuicTransportParameters));
151                }
152            }
153        }
154
155        let for_resume = resumedata.is_some();
156        // SNI
157        if let (false, Some(ServerNamePayload::SingleDnsName(_))) = (for_resume, &hello.server_name)
158        {

160        }
161

163        // if we're resuming, and is only triggered if we have an OCSP response
164        // to send.
165        if !for_resume

167                .certificate_status_request
168                .is_some()

170            if ocsp_response.is_some() && !cx.common.is_tls13() {
171                // Only TLS1.2 sends confirmation in ServerHello
172                self.extensions
173                    .certificate_status_request_ack = Some(());
174            }
175        } else {
176            // Throw away any OCSP response so we don't try to send it later.
177            ocsp_response.take();
178        }
179
180        self.validate_server_cert_type_extension(hello, config, cx)?;
181        self.validate_client_cert_type_extension(hello, config, cx)?;
182
183        Ok(())
184    }

187    pub(super) fn process_tls12(
188        &mut self,
189        config: &ServerConfig,
190        hello: &ClientHelloPayload,
191        using_ems: bool,

193        // Renegotiation.
194        // (We don't do reneg at all, but would support the secure version if we did.)

207        // If we get any SessionTicket extension and have tickets enabled,

215        if using_ems {
216            self.extensions
217                .extended_master_secret_ack = Some(());

352        let tls13_enabled = self
353            .config
354            .supports_version(ProtocolVersion::TLSv1_3, cx.common.protocol);
355        let tls12_enabled = self
356            .config
357            .supports_version(ProtocolVersion::TLSv1_2, cx.common.protocol)
358            && !self.previous_hello.is_some();

360        // Are we doing TLS1.3?
361        let version = if let Some(versions) = &client_hello.supported_versions {
362            if versions.tls13 && tls13_enabled {
363                ProtocolVersion::TLSv1_3

366                    AlertDescription::ProtocolVersion,
367                    PeerIncompatible::Tls12NotOfferedOrEnabled,

370                return Err(cx.common.send_fatal_alert(
371                    AlertDescription::ProtocolVersion,
372                    PeerIncompatible::Tls13RequiredForQuic,
373                ));
374            } else {
375                ProtocolVersion::TLSv1_2
376            }
377        } else if u16::from(client_hello.client_version) < u16::from(ProtocolVersion::TLSv1_2) {
378            return Err(cx.common.send_fatal_alert(
379                AlertDescription::ProtocolVersion,
380                PeerIncompatible::Tls12NotOffered,
381            ));
382        } else if cx.common.is_quic() {
383            return Err(cx.common.send_fatal_alert(
384                AlertDescription::ProtocolVersion,
385                PeerIncompatible::Tls13RequiredForQuic,
386            ));
387        } else if !tls12_enabled && tls13_enabled {
388            return Err(cx.common.send_fatal_alert(
389                AlertDescription::ProtocolVersion,
390                PeerIncompatible::SupportedVersionsExtensionRequired,

549    fn choose_suite_and_kx_group(
550        &self,
551        selected_version: ProtocolVersion,
552        sig_key_algorithm: SignatureAlgorithm,
553        protocol: Protocol,
554        client_groups: &[NamedGroup],
555        client_suites: &[CipherSuite],
556    ) -> Result<(SupportedCipherSuite, &'static dyn SupportedKxGroup), PeerIncompatible> {
557        // Determine which `KeyExchangeAlgorithm`s are theoretically possible, based
558        // on the offered and supported groups.
559        let mut ecdhe_possible = false;
560        let mut ffdhe_possible = false;
561        let mut ffdhe_offered = false;
562        let mut supported_groups = Vec::with_capacity(client_groups.len());
563
564        for offered_group in client_groups {
565            let supported = self
566                .config
567                .provider
568                .kx_groups
569                .iter()
570                .find(|skxg| {
571                    skxg.usable_for_version(selected_version) && skxg.name() == *offered_group
572                });
573
574            match offered_group.key_exchange_algorithm() {
575                KeyExchangeAlgorithm::DHE => {
576                    ffdhe_possible |= supported.is_some();
577                    ffdhe_offered = true;
578                }
579                KeyExchangeAlgorithm::ECDHE => {
580                    ecdhe_possible |= supported.is_some();
581                }
582            }
583
584            supported_groups.push(supported);
585        }
586
587        let first_supported_dhe_kxg = if selected_version == ProtocolVersion::TLSv1_2 {
588            // https://datatracker.ietf.org/doc/html/rfc7919#section-4 (paragraph 2)
589            let first_supported_dhe_kxg = self
590                .config
591                .provider
592                .kx_groups
593                .iter()
594                .find(|skxg| skxg.name().key_exchange_algorithm() == KeyExchangeAlgorithm::DHE);
595            ffdhe_possible |= !ffdhe_offered && first_supported_dhe_kxg.is_some();
596            first_supported_dhe_kxg
597        } else {
598            // In TLS1.3, the server may only directly negotiate a group.
599            None
600        };
601
602        if !ecdhe_possible && !ffdhe_possible {
603            return Err(PeerIncompatible::NoKxGroupsInCommon);
604        }
605
606        let mut suitable_suites_iter = self
607            .config
608            .provider
609            .cipher_suites
610            .iter()
611            .filter(|suite| {
612                // Reduce our supported ciphersuites by the certified key's algorithm.
613                suite.usable_for_signature_algorithm(sig_key_algorithm)
614                // And version
615                && suite.version().version == selected_version
616                // And protocol
617                && suite.usable_for_protocol(protocol)
618                // And support one of key exchange groups
619                && (ecdhe_possible && suite.usable_for_kx_algorithm(KeyExchangeAlgorithm::ECDHE)
620                || ffdhe_possible && suite.usable_for_kx_algorithm(KeyExchangeAlgorithm::DHE))
621            });
622
623        // RFC 7919 (https://datatracker.ietf.org/doc/html/rfc7919#section-4) requires us to send
624        // the InsufficientSecurity alert in case we don't recognize client's FFDHE groups (i.e.,
625        // `suitable_suites` becomes empty). But that does not make a lot of sense (e.g., client
626        // proposes FFDHE4096 and we only support FFDHE2048), so we ignore that requirement here,
627        // and continue to send HandshakeFailure.
628
629        let suite = if self.config.ignore_client_order {
630            suitable_suites_iter.find(|suite| client_suites.contains(&suite.suite()))
631        } else {
632            let suitable_suites = suitable_suites_iter.collect::<Vec<_>>();
633            client_suites
634                .iter()
635                .find_map(|client_suite| {
636                    suitable_suites
637                        .iter()
638                        .find(|x| *client_suite == x.suite())
639                })
640                .copied()
641        }
642        .ok_or(PeerIncompatible::NoCipherSuitesInCommon)?;
643
644        // Finally, choose a key exchange group that is compatible with the selected cipher
645        // suite.
646        let maybe_skxg = supported_groups
647            .iter()
648            .find_map(|maybe_skxg| match maybe_skxg {
649                Some(skxg) => suite
650                    .usable_for_kx_algorithm(skxg.name().key_exchange_algorithm())
651                    .then_some(*skxg),
652                None => None,
653            });
654
655        if selected_version == ProtocolVersion::TLSv1_3 {
656            // This unwrap is structurally guaranteed by the early return for `!ffdhe_possible && !ecdhe_possible`
657            return Ok((*suite, *maybe_skxg.unwrap()));
658        }
659
660        // For TLS1.2, the server can unilaterally choose a DHE group if it has one and
661        // there was no better option.
662        match maybe_skxg {
663            Some(skxg) => Ok((*suite, *skxg)),
664            None if suite.usable_for_kx_algorithm(KeyExchangeAlgorithm::DHE) => {
665                // If kx for the selected cipher suite is DHE and no DHE groups are specified in the extension,
666                // the server is free to choose DHE params, we choose the first DHE kx group of the provider.
667                if let Some(server_selected_ffdhe_skxg) = first_supported_dhe_kxg {
668                    Ok((*suite, *server_selected_ffdhe_skxg))
669                } else {
670                    Err(PeerIncompatible::NoKxGroupsInCommon)
671                }
672            }
673            None => Err(PeerIncompatible::NoKxGroupsInCommon),
674        }
675    }
676}

702/// Configuration-independent validation of a `ClientHello` message.
703///
704/// This represents the first part of the `ClientHello` handling, where we do all validation that
705/// doesn't depend on a `ServerConfig` being available and extract everything needed to build a
706/// [`ClientHello`] value for a [`ResolvesServerCert`].
707///
708/// Note that this will modify `data.sni` even if config or certificate resolution fail.
709///
710/// [`ResolvesServerCert`]: crate::server::ResolvesServerCert
711pub(super) fn process_client_hello<'m>(
712    m: &'m Message<'m>,
713    done_retry: bool,
714    cx: &mut ServerContext<'_>,
715) -> Result<(&'m ClientHelloPayload, Vec<SignatureScheme>), Error> {
716    let client_hello =
717        match &m.payload {
    MessagePayload::Handshake {
        parsed: crate::msgs::handshake::HandshakeMessagePayload(HandshakePayload::ClientHello(hm)),
        .. } => Ok(hm),
    payload =>
        Err(crate::check::inappropriate_handshake_message(payload,
                &[crate::ContentType::Handshake],
                &[HandshakeType::ClientHello])),
}require_handshake_msg!(m, HandshakeType::ClientHello, HandshakePayload::ClientHello)?;
718    {
    {
        let lvl = ::log::Level::Trace;
        if lvl <= ::log::STATIC_MAX_LEVEL && lvl <= ::log::max_level() {
            ::log::__private_api::log({ ::log::__private_api::GlobalLogger },
                format_args!("we got a clienthello {0:?}", client_hello), lvl,
                &("rustls::server::hs", "rustls::server::hs",
                        ::log::__private_api::loc()), ());
        }
    }
};trace!("we got a clienthello {client_hello:?}");
719
720    if !client_hello
721        .compression_methods
722        .contains(&Compression::Null)
723    {
724        return Err(cx.common.send_fatal_alert(
725            AlertDescription::IllegalParameter,
726            PeerIncompatible::NullCompressionRequired,
727        ));
728    }
729
730    // No handshake messages should follow this one in this flight.
731    cx.common.check_aligned_handshake()?;
732
733    // Extract and validate the SNI DNS name, if any, before giving it to
734    // the cert resolver. In particular, if it is invalid then we should
735    // send an Illegal Parameter alert instead of the Internal Error alert
736    // (or whatever) that we'd send if this were checked later or in a
737    // different way.
738    //
739    // [RFC6066][] specifies that literal IP addresses are illegal in
740    // `ServerName`s with a `name_type` of `host_name`.
741    //
742    // Some clients incorrectly send such extensions: we choose to
743    // successfully parse these (into `ServerNamePayload::IpAddress`)
744    // but then act like the client sent no `server_name` extension.
745    //
746    // [RFC6066]: https://datatracker.ietf.org/doc/html/rfc6066#section-3
747    let sni = match &client_hello.server_name {
748        Some(ServerNamePayload::SingleDnsName(dns_name)) => Some(dns_name.to_lowercase_owned()),
749        Some(ServerNamePayload::IpAddress) => None,
750        Some(ServerNamePayload::Invalid) => {
751            return Err(cx.common.send_fatal_alert(
752                AlertDescription::IllegalParameter,
753                PeerMisbehaved::ServerNameMustContainOneHostName,
754            ));
755        }
756        None => None,
757    };
758
759    // save only the first SNI
760    if let (Some(sni), false) = (&sni, done_retry) {
761        // Save the SNI into the session.
762        // The SNI hostname is immutable once set.
763        if !cx.data.sni.is_none() {
    ::core::panicking::panic("assertion failed: cx.data.sni.is_none()")
};assert!(cx.data.sni.is_none());
764        cx.data.sni = Some(sni.clone());
765    } else if cx.data.sni != sni {
766        return Err(PeerMisbehaved::ServerNameDifferedOnRetry.into());
767    }
768
769    let sig_schemes = client_hello
770        .signature_schemes
771        .as_ref()
772        .ok_or_else(|| {
773            cx.common.send_fatal_alert(
774                AlertDescription::HandshakeFailure,
775                PeerIncompatible::SignatureAlgorithmsExtensionRequired,
776            )
777        })?;
778
779    Ok((client_hello, sig_schemes.to_owned()))
780}
