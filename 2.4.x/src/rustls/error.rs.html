623// * [OpenSSL Implementation](https://github.com/openssl/openssl/blob/45bb98bfa223efd3258f445ad443f878011450f0/ssl/statem/statem_lib.c#L1434)

625impl From<CertificateError> for AlertDescription {
626    fn from(e: CertificateError) -> Self {
627        use CertificateError::*;
628        match e {
629            BadEncoding
630            | UnhandledCriticalExtension
631            | NotValidForName
632            | NotValidForNameContext { .. } => Self::BadCertificate,
633            // RFC 5246/RFC 8446
634            // certificate_expired
635            //  A certificate has expired or **is not currently valid**.
636            Expired | ExpiredContext { .. } | NotValidYet | NotValidYetContext { .. } => {
637                Self::CertificateExpired
638            }
639            Revoked => Self::CertificateRevoked,
640            // OpenSSL, BoringSSL and AWS-LC all generate an Unknown CA alert for
641            // the case where revocation status can not be determined, so we do the same here.
642            UnknownIssuer
643            | UnknownRevocationStatus
644            | ExpiredRevocationList
645            | ExpiredRevocationListContext { .. } => Self::UnknownCA,
646            InvalidOcspResponse => Self::BadCertificateStatusResponse,
647            #[allow(deprecated)]
648            BadSignature
649            | UnsupportedSignatureAlgorithm
650            | UnsupportedSignatureAlgorithmContext { .. }
651            | UnsupportedSignatureAlgorithmForPublicKeyContext { .. } => Self::DecryptError,
652            InvalidPurpose | InvalidPurposeContext { .. } => Self::UnsupportedCertificate,
653            ApplicationVerificationFailure => Self::AccessDenied,
654            // RFC 5246/RFC 8446
655            // certificate_unknown
656            //  Some other (unspecified) issue arose in processing the
657            //  certificate, rendering it unacceptable.
658            Other(..) => Self::CertificateUnknown,
659        }
660    }
