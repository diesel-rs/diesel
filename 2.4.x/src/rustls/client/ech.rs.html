120                        _i + 1,
121                        _version
122                    );
123                    continue; // Unsupported version.
124                }
125            };
126
127            if contents.has_unknown_mandatory_extension() || contents.has_duplicate_extension() {
128                {
    {

129                continue; // Unsupported, or malformed extensions.
130            }

213    /// Build a GREASE ECH extension based on the placeholder configuration.
214    ///
215    /// See <https://datatracker.ietf.org/doc/html/draft-ietf-tls-esni-18#name-grease-ech> for
216    /// more information.
217    pub(crate) fn grease_ext(
218        &self,
219        secure_random: &'static dyn SecureRandom,
220        inner_name: ServerName<'static>,
221        outer_hello: &ClientHelloPayload,
222    ) -> Result<EncryptedClientHello, Error> {
223        {
    {
        let lvl = ::log::Level::Trace;
        if lvl <= ::log::STATIC_MAX_LEVEL && lvl <= ::log::max_level() {
            ::log::__private_api::log({ ::log::__private_api::GlobalLogger },
                format_args!("Preparing GREASE ECH extension"), lvl,
                &("rustls::client::ech", "rustls::client::ech",
                        ::log::__private_api::loc()), ());
        }
    }
};trace!("Preparing GREASE ECH extension");
224
225        // Pick a random config id.
226        let mut config_id: [u8; 1] = [0; 1];
227        secure_random.fill(&mut config_id[..])?;
228
229        let suite = self.suite.suite();
230
231        // Construct a dummy ECH state - we don't have a real ECH config from a server since
232        // this is for GREASE.
233        let mut grease_state = EchState::new(
234            &EchConfig {
235                config: EchConfigPayload::V18(EchConfigContents {
236                    key_config: HpkeKeyConfig {
237                        config_id: config_id[0],
238                        kem_id: HpkeKem::DHKEM_P256_HKDF_SHA256,
239                        public_key: PayloadU16::new(self.placeholder_key.0.clone()),
240                        symmetric_cipher_suites: ::alloc::boxed::box_assume_init_into_vec_unsafe(::alloc::intrinsics::write_box_via_move(::alloc::boxed::Box::new_uninit(),
        [suite.sym]))vec![suite.sym],
241                    },
242                    maximum_name_length: 0,
243                    public_name: DnsName::try_from("filler").unwrap(),
244                    extensions: Vec::default(),
245                }),
246                suite: self.suite,
247            },
248            inner_name,
249            false,
250            secure_random,
251            false, // Does not matter if we enable/disable SNI here. Inner hello is not used.
252        )?;
253
254        // Construct an inner hello using the outer hello - this allows us to know the size of
255        // dummy payload we should use for the GREASE extension.
256        let encoded_inner_hello = grease_state.encode_inner_hello(outer_hello, None, &None);
257
258        // Generate a payload of random data equivalent in length to a real inner hello.
259        let payload_len = encoded_inner_hello.len()
260            + suite
261                .sym
262                .aead_id
263                .tag_len()
264                // Safety: we have confirmed the AEAD is supported when building the config. All
265                //  supported AEADs have a tag length.
266                .unwrap();
267        let mut payload = ::alloc::vec::from_elem_global(0, payload_len)vec![0; payload_len];
268        secure_random.fill(&mut payload)?;
269
270        // Return the GREASE extension.
271        Ok(EncryptedClientHello::Outer(EncryptedClientHelloOuter {
272            cipher_suite: suite.sym,
273            config_id: config_id[0],
274            enc: PayloadU16::new(grease_state.enc.0),
275            payload: PayloadU16::new(payload),
276        }))
277    }

380    /// Construct a ClientHelloPayload offering ECH.
381    ///
382    /// An outer hello, with a protected inner hello for the `inner_name` will be returned, and the
383    /// ECH context will be updated to reflect the inner hello that was offered.
384    ///
385    /// If `retry_req` is `Some`, then the outer hello will be constructed for a hello retry request.
386    ///
387    /// If `resuming` is `Some`, then the inner hello will be constructed for a resumption handshake.
388    pub(crate) fn ech_hello(
389        &mut self,
390        mut outer_hello: ClientHelloPayload,
391        retry_req: Option<&HelloRetryRequest>,
392        resuming: &Option<Retrieved<&persist::Tls13ClientSessionValue>>,
393    ) -> Result<ClientHelloPayload, Error> {
394        {
    {
        let lvl = ::log::Level::Trace;
        if lvl <= ::log::STATIC_MAX_LEVEL && lvl <= ::log::max_level() {
            ::log::__private_api::log({ ::log::__private_api::GlobalLogger },
                format_args!("Preparing ECH offer {0}",
                    if retry_req.is_some() { "for retry" } else { "" }), lvl,
                &("rustls::client::ech", "rustls::client::ech",
                        ::log::__private_api::loc()), ());
        }
    }
};trace!(
395            "Preparing ECH offer {}",
396            if retry_req.is_some() { "for retry" } else { "" }
397        );
398
399        // Construct the encoded inner hello and update the transcript.
400        let encoded_inner_hello = self.encode_inner_hello(&outer_hello, retry_req, resuming);
401
402        // Complete the ClientHelloOuterAAD with an ech extension, the payload should be a placeholder
403        // of size L, all zeroes. L == length of encrypting encoded client hello inner w/ the selected
404        // HPKE AEAD. (sum of plaintext + tag length, typically).
405        let payload_len = encoded_inner_hello.len()
406            + self
407                .cipher_suite
408                .aead_id
409                .tag_len()
410                // Safety: we've already verified this AEAD is supported when loading the config
411                // that was used to create the ECH context. All supported AEADs have a tag length.
412                .unwrap();
413
414        // Outer hello's created in response to a hello retry request omit the enc value.
415        let enc = match retry_req.is_some() {
416            true => Vec::default(),
417            false => self.enc.0.clone(),
418        };
419
420        fn outer_hello_ext(ctx: &EchState, enc: Vec<u8>, payload: Vec<u8>) -> EncryptedClientHello {
421            EncryptedClientHello::Outer(EncryptedClientHelloOuter {
422                cipher_suite: ctx.cipher_suite,
423                config_id: ctx.config_id,
424                enc: PayloadU16::new(enc),
425                payload: PayloadU16::new(payload),
426            })
427        }
428
429        // The outer handshake is not permitted to resume a session. If we're resuming in the
430        // inner handshake we remove the PSK extension from the outer hello, replacing it
431        // with a GREASE PSK to implement the "ClientHello Malleability Mitigation" mentioned
432        // in 10.12.3.
433        if let Some(psk_offer) = outer_hello.preshared_key_offer.as_mut() {
434            self.grease_psk(psk_offer)?;
435        }
436
437        // To compute the encoded AAD we add a placeholder extension with an empty payload.
438        outer_hello.encrypted_client_hello =
439            Some(outer_hello_ext(self, enc.clone(), ::alloc::vec::from_elem_global(0, payload_len)vec![0; payload_len]));
440
441        // Next we compute the proper extension payload.
442        let payload = self
443            .sender
444            .seal(&outer_hello.get_encoding(), &encoded_inner_hello)?;
445
446        // And then we replace the placeholder extension with the real one.
447        outer_hello.encrypted_client_hello = Some(outer_hello_ext(self, enc, payload));
448
449        Ok(outer_hello)
450    }

463        // without consulting the server hello, see RFC 9849 section 6.1.6.

473            .start_hash(hash);

476        // We need to preserve the original inner_transcript to use if this confirmation succeeds.

480        // encoding rather than reencoding it.
481        confirmation_transcript

491        // server random. This match signals that the server accepted the ECH offer.
492        // Indexing safety: Random is [0; 32] by construction.
493

499                    random: self.inner_hello_random,

504            // -- <https://www.rfc-editor.org/info/rfc9849/#section-6.1.5>
505            false if self.accepted_by_hello_retry_request == Some(true) => Err(common
506                .send_fatal_alert(

510            false => {

514                // for ECHConfig.contents.public_name"
515                // -- <https://www.rfc-editor.org/info/rfc9849/#section-6.1.6>

594    // 5.1 "Encoding the ClientHelloInner"
595    fn encode_inner_hello(
596        &mut self,
597        outer_hello: &ClientHelloPayload,
598        retryreq: Option<&HelloRetryRequest>,
599        resuming: &Option<Retrieved<&persist::Tls13ClientSessionValue>>,
600    ) -> Vec<u8> {
601        // Start building an inner hello using the outer_hello as a template.
602        let mut inner_hello = ClientHelloPayload {
603            // Some information is copied over as-is.
604            client_version: outer_hello.client_version,
605            session_id: outer_hello.session_id,
606            compression_methods: outer_hello.compression_methods.clone(),
607
608            // We will build up the included extensions ourselves.
609            extensions: Box::new(ClientExtensions::default()),
610
611            // Set the inner hello random to the one we generated when creating the ECH state.
612            // We hold on to the inner_hello_random in the ECH state to use later for confirming
613            // whether ECH was accepted or not.
614            random: self.inner_hello_random,
615
616            // We remove the empty renegotiation info SCSV from the outer hello's ciphersuite.
617            // Similar to the TLS 1.2 specific extensions we will filter out, this is seen as a
618            // TLS 1.2 only feature by bogo.
619            cipher_suites: outer_hello
620                .cipher_suites
621                .iter()
622                .filter(|cs| **cs != TLS_EMPTY_RENEGOTIATION_INFO_SCSV)
623                .cloned()
624                .collect(),
625        };
626
627        inner_hello.order_seed = outer_hello.order_seed;
628
629        // The inner hello will always have an inner variant of the ECH extension added.
630        // See Section 6.1 rule 4.
631        inner_hello.encrypted_client_hello = Some(EncryptedClientHello::Inner);
632
633        let inner_sni = match &self.inner_name {
634            // The inner hello only gets a SNI value if enable_sni is true and the inner name
635            // is a domain name (not an IP address).
636            ServerName::DnsName(dns_name) if self.enable_sni => Some(dns_name),
637            _ => None,
638        };
639
640        // Now we consider each of the outer hello's extensions - we can either:
641        // 1. Omit the extension if it isn't appropriate (e.g. is a TLS 1.2 extension).
642        // 2. Add the extension to the inner hello as-is.
643        // 3. Compress the extension, by collecting it into a list of to-be-compressed
644        //    extensions we'll handle separately.
645        let outer_extensions = outer_hello.used_extensions_in_encoding_order();
646        let mut compressed_exts = Vec::with_capacity(outer_extensions.len());
647        for ext in outer_extensions {
648            // Some outer hello extensions are only useful in the context where a TLS 1.3
649            // connection allows TLS 1.2. This isn't the case for ECH so we skip adding them
650            // to the inner hello.
651            if #[allow(non_exhaustive_omitted_patterns)] match ext {
    ExtensionType::ExtendedMasterSecret | ExtensionType::SessionTicket |
        ExtensionType::ECPointFormats => true,
    _ => false,
}matches!(
652                ext,
653                ExtensionType::ExtendedMasterSecret
654                    | ExtensionType::SessionTicket
655                    | ExtensionType::ECPointFormats
656            ) {
657                continue;
658            }
659
660            if ext == ExtensionType::ServerName {
661                // We may want to replace the outer hello SNI with our own inner hello specific SNI.
662                if let Some(sni_value) = inner_sni {
663                    inner_hello.server_name = Some(ServerNamePayload::from(sni_value));
664                }
665                // We don't want to add, or compress, the SNI from the outer hello.
666                continue;
667            }
668
669            // Compressed extensions need to be put aside to include in one contiguous block.
670            // Uncompressed extensions get added directly to the inner hello.
671            if ext.ech_compress() {
672                compressed_exts.push(ext);
673            }
674
675            inner_hello.clone_one(outer_hello, ext);
676        }
677
678        // We've added all the uncompressed extensions. Now we need to add the contiguous
679        // block of to-be-compressed extensions.
680        inner_hello.contiguous_extensions = compressed_exts.clone();
681
682        // Note which extensions we're sending in the inner hello. This may differ from
683        // the outer hello (e.g. the inner hello may omit SNI while the outer hello will
684        // always have the ECH cover name in SNI).
685        self.sent_extensions = inner_hello.collect_used();
686
687        // If we're resuming, we need to update the PSK binder in the inner hello.
688        if let Some(resuming) = resuming.as_ref() {
689            let mut chp = HandshakeMessagePayload(HandshakePayload::ClientHello(inner_hello));
690
691            // Retain the early key schedule we get from processing the binder.
692            self.early_data_key_schedule = Some(tls13::fill_in_psk_binder(
693                resuming,
694                &self.inner_hello_transcript,
695                &mut chp,
696            ));
697
698            // fill_in_psk_binder works on an owned HandshakeMessagePayload, so we need to
699            // extract our inner hello back out of it to retain ownership.
700            inner_hello = match chp.0 {
701                HandshakePayload::ClientHello(chp) => chp,
702                // Safety: we construct the HMP above and know its type unconditionally.
703                _ => ::core::panicking::unreachable_default_msg()unreachable!(),
704            };
705        }
706
707        {
    {
        let lvl = ::log::Level::Trace;
        if lvl <= ::log::STATIC_MAX_LEVEL && lvl <= ::log::max_level() {
            ::log::__private_api::log({ ::log::__private_api::GlobalLogger },
                format_args!("ECH Inner Hello: {0:#?}", inner_hello), lvl,
                &("rustls::client::ech", "rustls::client::ech",
                        ::log::__private_api::loc()), ());
        }
    }
};trace!("ECH Inner Hello: {inner_hello:#?}");
708
709        // Encode the inner hello according to the rules required for ECH. This differs
710        // from the standard encoding in several ways. Notably this is where we will
711        // replace the block of contiguous to-be-compressed extensions with a marker.
712        let mut encoded_hello = inner_hello.ech_inner_encoding(compressed_exts);
713
714        // Calculate padding
715        // max_name_len = L
716        let max_name_len = usize::from(self.maximum_name_length);
717        let max_name_len = if max_name_len > 0 { max_name_len } else { 255 };
718
719        let name_padding_len = match &inner_hello.server_name {
720            Some(ServerNamePayload::SingleDnsName(name)) => {
721                // name.len() = D
722                // max(0, L - D)
723                Ord::max(0, max_name_len.saturating_sub(name.as_ref().len()))
724            }
725            // L + 9
726            // "This is the length of a "server_name" extension with an L-byte name."
727            _ => max_name_len + 9,
728        };
729        encoded_hello.extend(iter::repeat(0).take(name_padding_len));
730
731        // Let L be the length of the EncodedClientHelloInner with all the padding computed so far
732        // Let N = 31 - ((L - 1) % 32) and add N bytes of padding.
733        let padding_len = 31 - ((encoded_hello.len() - 1) % 32);
734        encoded_hello.extend(iter::repeat(0).take(padding_len));
735
736        // Construct the inner hello message that will be used for the transcript.
737        let inner_hello_msg = Message {
738            version: match retryreq {
739                // <https://datatracker.ietf.org/doc/html/rfc8446#section-5.1>:
740                // "This value MUST be set to 0x0303 for all records generated
741                //  by a TLS 1.3 implementation ..."
742                Some(_) => ProtocolVersion::TLSv1_2,
743                // "... other than an initial ClientHello (i.e., one not
744                // generated after a HelloRetryRequest), where it MAY also be
745                // 0x0301 for compatibility purposes"
746                //
747                // (retryreq == None means we're in the "initial ClientHello" case)
748                None => ProtocolVersion::TLSv1_0,
749            },
750            payload: MessagePayload::handshake(HandshakeMessagePayload(
751                HandshakePayload::ClientHello(inner_hello),
752            )),
753        };
754
755        // Update the inner transcript buffer with the inner hello message.
756        self.inner_hello_transcript
757            .add_message(&inner_hello_msg);
758
759        encoded_hello
760    }

762    // See https://datatracker.ietf.org/doc/html/draft-ietf-tls-esni-18#name-grease-psk
763    fn grease_psk(&self, psk_offer: &mut PresharedKeyOffer) -> Result<(), Error> {
764        for ident in psk_offer.identities.iter_mut() {
765            // "For each PSK identity advertised in the ClientHelloInner, the
766            // client generates a random PSK identity with the same length."
767            self.secure_random
768                .fill(&mut ident.identity.0)?;
769            // "It also generates a random, 32-bit, unsigned integer to use as
770            // the obfuscated_ticket_age."
771            let mut ticket_age = [0_u8; 4];
772            self.secure_random
773                .fill(&mut ticket_age)?;
774            ident.obfuscated_ticket_age = u32::from_be_bytes(ticket_age);
775        }
776
777        // "Likewise, for each inner PSK binder, the client generates a random string
778        // of the same length."
779        psk_offer.binders = psk_offer
780            .binders
781            .iter()
782            .map(|old_binder| {
783                // We can't access the wrapped binder PresharedKeyBinder's PayloadU8 mutably,
784                // so we construct new PresharedKeyBinder's from scratch with the same length.
785                let mut new_binder = ::alloc::vec::from_elem_global(0, old_binder.as_ref().len())vec![0; old_binder.as_ref().len()];
786                self.secure_random
787                    .fill(&mut new_binder)?;
788                Ok::<PresharedKeyBinder, Error>(PresharedKeyBinder::from(new_binder))
789            })
790            .collect::<Result<_, _>>()?;
791        Ok(())
792    }

794    fn server_hello_conf(
795        server_hello: &ServerHelloPayload,
796        server_hello_encoded: &Payload<'_>,
797    ) -> Message<'static> {
798        // The confirmation is computed over the server hello, which has had
799        // its `random` field altered to zero the final 8 bytes.
800        //
801        // nb. we don't require that we can round-trip a `ServerHelloPayload`, to
802        // allow for efficiency in its in-memory representation.  That means
803        // we operate here on the received encoding, as the confirmation needs
804        // to be computed on that.
805        let mut encoded = server_hello_encoded.clone().into_vec();
806        encoded[SERVER_HELLO_ECH_CONFIRMATION_SPAN].fill(0x00);
807
808        Message {
809            version: ProtocolVersion::TLSv1_3,
810            payload: MessagePayload::Handshake {
811                encoded: Payload::Owned(encoded),
812                parsed: HandshakeMessagePayload(HandshakePayload::ServerHello(
813                    server_hello.clone(),
814                )),
815            },
816        }
817    }

1022            let suite = tls13_suite();
1023            let hash = suite.common.hash_provider;
1024            let mut key_schedule = handshake_key_schedule(suite);
1025
1026            // without a hello retry request, a confirming server hello accepts ECH
1027            let mut state = ech_state();
1028            state
1029                .ech_hello(outer_hello(), None, &None)
1030                .unwrap();
1031            let server_hello = confirming_server_hello(&state, &mut key_schedule, suite);
1032            let mut server_name = inner_name();

1034                state
1035                    .confirm_acceptance(
1036                        &mut key_schedule,
1037                        &server_hello,
1038                        &encode(&server_hello),
1039                        hash,

1041                        &mut CommonState::new(Side::Client),
1042                    )
1043                    .unwrap()
1044                    .is_some()
1045            );
1046            assert_eq!(server_name, inner_name());

1053            let hrr = hello_retry_request(suite, None);

1058            );

1061            let mut server_name = inner_name();
1062            assert!(
1063                state
1064                    .confirm_acceptance(
1065                        &mut key_schedule,
1066                        &server_hello,
1067                        &encode(&server_hello),
1068                        hash,
1069                        &mut server_name,
1070                        &mut CommonState::new(Side::Client),

1073                    .is_none()
1074            );

1146        /// A hello retry request accepting the offer in `state`, see RFC 9849 section 7.2.1.
1147        fn accepting_hello_retry_request(
1148            state: &EchState,
1149            suite: &'static Tls13CipherSuite,
1150        ) -> HelloRetryRequest {
1151            let mut hrr = hello_retry_request(suite, Some(Payload::new(vec![0u8; 8])));
1152            let mut transcript = state
1153                .inner_hello_transcript
1154                .clone()
1155                .start_hash(suite.common.hash_provider);
1156            transcript.rollup_for_hrr();
1157            transcript.add_message(&EchState::hello_retry_request_conf(&hrr));
1158            let confirmation = server_ech_hrr_confirmation_secret(
1159                suite.hkdf_provider,
1160                &state.inner_hello_random.0,
1161                transcript.current_hash(),
1162            );
1163            hrr.encrypted_client_hello = Some(Payload::new(confirmation.to_vec()));
1164            hrr
1165        }

1187        /// A server hello accepting the offer in `state`, see RFC 9849 section 7.2.
1188        fn confirming_server_hello(
1189            state: &EchState,
1190            key_schedule: &mut KeyScheduleHandshakeStart,
1191            suite: &'static Tls13CipherSuite,
1192        ) -> ServerHelloPayload {
1193            let mut server_hello = server_hello(suite);
1194            let mut transcript = state
1195                .inner_hello_transcript
1196                .clone()
1197                .start_hash(suite.common.hash_provider);
1198            transcript.add_message(&EchState::server_hello_conf(
1199                &server_hello,
1200                &encode(&server_hello),
1201            ));
1202            let confirmation = key_schedule.server_ech_confirmation_secret(
1203                &state.inner_hello_random.0,
1204                transcript.current_hash(),
1205            );
1206            server_hello.random.0[24..].copy_from_slice(&confirmation);
1207            server_hello
1208        }

1263    fn dns_name_of_len(mut len: usize) -> DnsName<'static> {
1264        let mut s = String::new();
1265        let labels = len.div_ceil(63);
1266        for _ in 0..labels {
1267            let chars = Ord::min(len, 63);
1268            len -= chars;
1269            for _ in 0..chars {
1270                s.push('a');
1271            }
1272            if len != 0 {
1273                s.push('.');
1274            }
1275        }
1276        DnsName::try_from(s).unwrap()
1277    }
