1//! Certificate compression and decompression support
2//!
3//! This crate supports compression and decompression everywhere
4//! certificates are used, in accordance with [RFC8879][rfc8879].
5//!
6//! Note that this is only supported for TLS1.3 connections.
7//!
8//! # Getting started
9//!
10//! Build this crate with the `brotli` and/or `zlib` crate features.  This
11//! adds dependencies on these crates.  They are used by default if enabled.
12//!
13//! We especially recommend `brotli` as it has the widest deployment so far.
14//!
15//! # Custom compression/decompression implementations
16//!
17//! 1. Implement the [`CertCompressor`] and/or [`CertDecompressor`] traits
18//! 2. Provide those to:
19//!   - [`ClientConfig::cert_compressors`][cc_cc] or [`ServerConfig::cert_compressors`][sc_cc].
20//!   - [`ClientConfig::cert_decompressors`][cc_cd] or [`ServerConfig::cert_decompressors`][sc_cd].
21//!
22//! These are used in these circumstances:
23//!
24//! | Peer | Client authentication | Server authentication |
25//! | ---- | --------------------- | --------------------- |
26//! | *Client* | [`ClientConfig::cert_compressors`][cc_cc] | [`ClientConfig::cert_decompressors`][cc_cd] |
27//! | *Server* | [`ServerConfig::cert_decompressors`][sc_cd] | [`ServerConfig::cert_compressors`][sc_cc] |
28//!
29//! [rfc8879]: https://datatracker.ietf.org/doc/html/rfc8879
30//! [cc_cc]: crate::ClientConfig::cert_compressors
31//! [sc_cc]: crate::ServerConfig::cert_compressors
32//! [cc_cd]: crate::ClientConfig::cert_decompressors
33//! [sc_cd]: crate::ServerConfig::cert_decompressors
34

59/// An available certificate decompression algorithm.
60pub trait CertDecompressor: Debug + Send + Sync {
61    /// Decompress `input`, writing the result to `output`.
62    ///
63    /// `output` is sized to match the declared length of the decompressed data.
64    ///
65    /// `Err(DecompressionFailed)` should be returned if decompression produces more, or fewer
66    /// bytes than fit in `output`, or if the `input` is in any way malformed.
67    fn decompress(&self, input: &[u8], output: &mut [u8]) -> Result<(), DecompressionFailed>;
68
69    /// Which algorithm this decompressor handles.
70    fn algorithm(&self) -> CertificateCompressionAlgorithm;
71}

84/// An available certificate compression algorithm.
85pub trait CertCompressor: Debug + Send + Sync {
86    /// Compress `input`, returning the result.
87    ///
88    /// `input` is consumed by this function so (if the underlying implementation
89    /// supports it) the compression can be performed in-place.
90    ///
91    /// `level` is a hint as to how much effort to expend on the compression.
92    ///
93    /// `Err(CompressionFailed)` may be returned for any reason.
94    fn compress(
95        &self,
96        input: Vec<u8>,
97        level: CompressionLevel,
98    ) -> Result<Vec<u8>, CompressionFailed>;
99
100    /// Which algorithm this compressor handles.
101    fn algorithm(&self) -> CertificateCompressionAlgorithm;
102}

137    #[derive(Debug)]
138    struct ZlibRsDecompressor;
139
140    impl CertDecompressor for ZlibRsDecompressor {
141        fn decompress(&self, input: &[u8], output: &mut [u8]) -> Result<(), DecompressionFailed> {
142            // `Inflate` rather than `decompress_slice()`, because the latter does not
143            // report how much of `input` the stream used: see `total_in()` below.
144            let mut state = Inflate::new(true, ZLIB_WINDOW_BITS);
145            match state.decompress(input, output, InflateFlush::Finish) {
146                Ok(Status::StreamEnd)
147                    if state.total_in() == input.len() as u64
148                        && state.total_out() == output.len() as u64 =>
149                {
150                    Ok(())
151                }
152                _ => Err(DecompressionFailed),
153            }
154        }
155
156        fn algorithm(&self) -> CertificateCompressionAlgorithm {
157            CertificateCompressionAlgorithm::Zlib
158        }
159    }

162    pub const ZLIB_COMPRESSOR: &dyn CertCompressor = &ZlibRsCompressor;

164    #[derive(Debug)]
165    struct ZlibRsCompressor;
166
167    impl CertCompressor for ZlibRsCompressor {
168        fn compress(
169            &self,
170            input: Vec<u8>,
171            level: CompressionLevel,
172        ) -> Result<Vec<u8>, CompressionFailed> {
173            let mut output = alloc::vec![0u8; compress_bound(input.len())];
174            let config = match level {
175                CompressionLevel::Interactive => DeflateConfig::default(),
176                CompressionLevel::Amortized => DeflateConfig::best_compression(),
177            };
178            let (output_filled, rc) = compress_slice(&mut output, &input, config);
179            if rc != ReturnCode::Ok {
180                return Err(CompressionFailed);
181            }
182
183            let used = output_filled.len();
184            output.truncate(used);
185            Ok(output)
186        }
187
188        fn algorithm(&self) -> CertificateCompressionAlgorithm {
189            CertificateCompressionAlgorithm::Zlib
190        }
191    }
192
193    /// Largest LZ77 window, matching `InflateConfig::default()`.
194    const ZLIB_WINDOW_BITS: u8 = 15;
195}

216        fn decompress(&self, input: &[u8], output: &mut [u8]) -> Result<(), DecompressionFailed> {
217            let mut state = BrotliState::new_strict(
218                StandardAlloc::default(),
219                StandardAlloc::default(),
220                StandardAlloc::default(),
221            );
222            let (mut available_in, mut input_offset) = (input.len(), 0);
223            let (mut available_out, mut output_offset, mut total_out) = (output.len(), 0, 0);
224
225            match BrotliDecompressStream(
226                &mut available_in,
227                &mut input_offset,
228                input,
229                &mut available_out,
230                &mut output_offset,
231                output,
232                &mut total_out,
233                &mut state,
234            ) {
235                BrotliResult::ResultSuccess if available_out == 0 && available_in == 0 => Ok(()),
236                _ => Err(DecompressionFailed),
237            }
238        }

241            CertificateCompressionAlgorithm::Brotli
242        }

251    impl CertCompressor for BrotliCompressor {
252        fn compress(
253            &self,
254            input: Vec<u8>,
255            level: CompressionLevel,
256        ) -> Result<Vec<u8>, CompressionFailed> {
257            let quality = match level {
258                CompressionLevel::Interactive => QUALITY_FAST,
259                CompressionLevel::Amortized => QUALITY_SLOW,
260            };
261            let output = Cursor::new(Vec::with_capacity(input.len() / 2));
262            let mut compressor = brotli::CompressorWriter::new(output, BUFFER_SIZE, quality, LGWIN);
263            compressor
264                .write_all(&input)
265                .map_err(|_| CompressionFailed)?;
266            Ok(compressor.into_inner().into_inner())
267        }
268
269        fn algorithm(&self) -> CertificateCompressionAlgorithm {
270            CertificateCompressionAlgorithm::Brotli
271        }
272    }
273
274    /// Brotli buffer size.
275    ///
276    /// Chosen based on brotli `examples/compress.rs`.
277    const BUFFER_SIZE: usize = 4096;
278
279    /// This is the default lgwin parameter, see `BrotliEncoderInitParams()`
280    const LGWIN: u32 = 22;
281
282    /// Compression quality we use for interactive compressions.
283    /// See <https://blog.cloudflare.com/results-experimenting-brotli> for data.
284    const QUALITY_FAST: u32 = 4;
285
286    /// Compression quality we use for offline compressions (the maximum).
287    const QUALITY_SLOW: u32 = 11;
288}

294///
295/// The prospect of being able to reuse a given compression for many connections
296/// means we can afford to spend more time on that compression (by passing
297/// `CompressionLevel::Amortized` to the compressor).
298#[derive(#[automatically_derived]impl ::core::fmt::Debug for CompressionCache {

}Debug)]

314pub struct CompressionCacheInner {
315    /// Maximum size of underlying storage.
316    size: usize,
317
318    /// LRU-order entries.
319    ///
320    /// First is least-used, last is most-used.
321    entries: Mutex<VecDeque<Arc<CompressionCacheEntry>>>,
322}
323
324impl CompressionCache {
325    /// Make a `CompressionCache` that stores up to `size` compressed
326    /// certificate messages.
327    #[cfg(feature = "std")]
328    pub fn new(size: usize) -> Self {
329        if size == 0 {
330            return Self::Disabled;
331        }
332
333        Self::Enabled(CompressionCacheInner {
334            size,
335            entries: Mutex::new(VecDeque::with_capacity(size)),
336        })
337    }
338
339    /// Return a `CompressionCacheEntry`, which is an owning
340    /// wrapper for a `CompressedCertificatePayload`.
341    ///
342    /// `compressor` is the compression function we have negotiated.
343    /// `original` is the uncompressed certificate message.
344    pub(crate) fn compression_for(
345        &self,
346        compressor: &dyn CertCompressor,
347        original: &CertificatePayloadTls13<'_>,
348    ) -> Result<Arc<CompressionCacheEntry>, CompressionFailed> {
349        match self {
350            Self::Disabled => Self::uncached_compression(compressor, original),
351
352            #[cfg(feature = "std")]
353            Self::Enabled(_) => self.compression_for_impl(compressor, original),
354        }
355    }
356
357    #[cfg(feature = "std")]
358    fn compression_for_impl(
359        &self,
360        compressor: &dyn CertCompressor,
361        original: &CertificatePayloadTls13<'_>,
362    ) -> Result<Arc<CompressionCacheEntry>, CompressionFailed> {
363        let (max_size, entries) = match self {
364            Self::Enabled(CompressionCacheInner { size, entries }) => (*size, entries),
365            _ => ::core::panicking::unreachable_default_msg()unreachable!(),
366        };
367
368        // context is a per-connection quantity, and included in the compressed data.
369        // it is not suitable for inclusion in the cache.
370        if !original.context.0.is_empty() {
371            return Self::uncached_compression(compressor, original);
372        }
373
374        // cache probe:
375        let encoding = original.get_encoding();
376        let algorithm = compressor.algorithm();
377
378        let mut cache = entries
379            .lock()
380            .map_err(|_| CompressionFailed)?;
381        for (i, item) in cache.iter().enumerate() {
382            if item.algorithm == algorithm && item.original == encoding {
383                // this item is now MRU
384                let item = cache.remove(i).unwrap();
385                cache.push_back(item.clone());
386                return Ok(item);
387            }
388        }
389        drop(cache);
390
391        // do compression:
392        let uncompressed_len = encoding.len() as u32;
393        let compressed = compressor.compress(encoding.clone(), CompressionLevel::Amortized)?;
394        let new_entry = Arc::new(CompressionCacheEntry {
395            algorithm,
396            original: encoding,
397            compressed: CompressedCertificatePayload {
398                alg: algorithm,
399                uncompressed_len,
400                compressed: PayloadU24::from(Payload::new(compressed)),
401            },
402        });
403
404        // insert into cache
405        let mut cache = entries
406            .lock()
407            .map_err(|_| CompressionFailed)?;
408        if cache.len() == max_size {
409            cache.pop_front();
410        }
411        cache.push_back(new_entry.clone());
412        Ok(new_entry)
413    }
414
415    /// Compress `original` using `compressor` at `Interactive` level.
416    fn uncached_compression(
417        compressor: &dyn CertCompressor,
418        original: &CertificatePayloadTls13<'_>,
419    ) -> Result<Arc<CompressionCacheEntry>, CompressionFailed> {
420        let algorithm = compressor.algorithm();
421        let encoding = original.get_encoding();
422        let uncompressed_len = encoding.len() as u32;
423        let compressed = compressor.compress(encoding, CompressionLevel::Interactive)?;
424
425        // this `CompressionCacheEntry` in fact never makes it into the cache, so
426        // `original` is left empty
427        Ok(Arc::new(CompressionCacheEntry {
428            algorithm,
429            original: Vec::new(),
430            compressed: CompressedCertificatePayload {
431                alg: algorithm,
432                uncompressed_len,
433                compressed: PayloadU24::from(Payload::new(compressed)),
434            },
435        }))
436    }
437}

701        #[derive(Debug)]
702        struct RequireCompress(&'static dyn CertCompressor, AtomicBool, bool);
703
704        impl CertCompressor for RequireCompress {
705            fn compress(
706                &self,
707                input: Vec<u8>,
708                level: CompressionLevel,
709            ) -> Result<Vec<u8>, CompressionFailed> {
710                self.1.store(true, Ordering::SeqCst);
711                self.0.compress(input, level)
712            }
713
714            fn algorithm(&self) -> CertificateCompressionAlgorithm {
715                self.0.algorithm()
716            }
717        }
718
719        impl Drop for RequireCompress {
720            fn drop(&mut self) {
721                assert_eq!(self.1.load(Ordering::SeqCst), self.2);
722            }
723        }
724    }
