54    /// Decrypt a TLS message.
55    ///
56    /// `encr` is a decoded message allegedly received from the peer.
57    /// If it can be decrypted, its decryption is returned.  Otherwise,
58    /// an error is returned.
59    pub(crate) fn decrypt_incoming<'a>(
60        &mut self,
61        encr: InboundOpaqueMessage<'a>,
62    ) -> Result<Option<Decrypted<'a>>, Error> {
63        if self.decrypt_state != DirectionState::Active {
64            return Ok(Some(Decrypted {
65                want_close_before_decrypt: false,
66                plaintext: encr.into_plain_message(),
67            }));
68        }
69
70        // Set to `true` if the peer appears to getting close to encrypting
71        // too many messages with this key.
72        //
73        // Perhaps if we send an alert well before their counter wraps, a
74        // buggy peer won't make a terrible mistake here?
75        //
76        // Note that there's no reason to refuse to decrypt: the security
77        // failure has already happened.
78        let want_close_before_decrypt = self.read_seq == SEQ_SOFT_LIMIT;
79
80        let encrypted_len = encr.payload.len();
81        match self
82            .message_decrypter
83            .decrypt(encr, self.read_seq)
84        {
85            Ok(plaintext) => {
86                self.read_seq += 1;
87                if !self.has_decrypted {
88                    self.has_decrypted = true;
89                }
90                Ok(Some(Decrypted {
91                    want_close_before_decrypt,
92                    plaintext,
93                }))
94            }
95            Err(Error::DecryptError) if self.doing_trial_decryption(encrypted_len) => {
96                {
    {
        let lvl = ::log::Level::Trace;
        if lvl <= ::log::STATIC_MAX_LEVEL && lvl <= ::log::max_level() {
            ::log::__private_api::log({ ::log::__private_api::GlobalLogger },
                format_args!("Dropping undecryptable message after aborted early_data"),
                lvl,
                &("rustls::record_layer", "rustls::record_layer",
                        ::log::__private_api::loc()), ());
        }
    }
};trace!("Dropping undecryptable message after aborted early_data");
97                Ok(None)
98            }
99            Err(err) => Err(err),
100        }
101    }

195    /// Return a remedial action when we are near to encrypting too many messages.
196    ///
197    /// `add` is added to the current sequence number.  `add` as `0` means
198    /// "the next message processed by `encrypt_outgoing`"
199    pub(crate) fn pre_encrypt_action(&self, add: u64) -> PreEncryptAction {
200        match self.write_seq.saturating_add(add) {
201            SEQ_HARD_LIMIT.. => PreEncryptAction::Refuse,
202            v if v >= self.write_seq_max => PreEncryptAction::RefreshOrClose,
203            _ => PreEncryptAction::Nothing,
204        }
205    }

207    pub(crate) fn is_encrypting(&self) -> bool {
208        self.encrypt_state == DirectionState::Active
209    }

230    fn doing_trial_decryption(&mut self, requested: usize) -> bool {
231        match self
232            .trial_decryption_len
233            .and_then(|value| value.checked_sub(requested))
234        {
235            Some(remaining) => {
236                self.trial_decryption_len = Some(remaining);
237                true
238            }
239            _ => false,
240        }
241    }

254pub(crate) enum PreEncryptAction {
255    /// No action is needed before calling `encrypt_outgoing`
256    Nothing,
257
258    /// A `key_update` request should be sent ASAP.
259    ///
260    /// If that is not possible (for example, the connection is TLS1.2), a `close_notify`
261    /// alert should be sent instead.
262    RefreshOrClose,
263
264    /// Do not call `encrypt_outgoing` further, it will panic rather than
265    /// over-use the key.
266    Refuse,
267}

277    fn pre_encrypt_action_checks_limits() {
278        let action = |write_seq_max, write_seq, add| {
279            let mut record_layer = RecordLayer::new();
280            record_layer.write_seq_max = write_seq_max;
281            record_layer.write_seq = write_seq;
282            record_layer.pre_encrypt_action(add)
283        };
284
285        assert_eq!(action(10, 8, 1), PreEncryptAction::Nothing);
286        assert_eq!(action(10, 8, 2), PreEncryptAction::RefreshOrClose);
287        assert_eq!(action(10, 11, 0), PreEncryptAction::RefreshOrClose);
288        assert_eq!(
289            action(SEQ_SOFT_LIMIT, SEQ_HARD_LIMIT - 1, 0),
290            PreEncryptAction::RefreshOrClose
291        );
292        assert_eq!(
293            action(SEQ_SOFT_LIMIT, SEQ_HARD_LIMIT - 1, 1),
294            PreEncryptAction::Refuse
295        );
296    }

298    #[test]
299    fn test_has_decrypted() {
300        use crate::{ContentType, ProtocolVersion};
301
302        struct PassThroughDecrypter;
303        impl MessageDecrypter for PassThroughDecrypter {
304            fn decrypt<'a>(
305                &mut self,
306                m: InboundOpaqueMessage<'a>,
307                _: u64,
308            ) -> Result<InboundPlainMessage<'a>, Error> {
309                Ok(m.into_plain_message())
310            }
311        }
312
313        // A record layer starts out invalid, having never decrypted.
314        let mut record_layer = RecordLayer::new();
315        assert!(matches!(
316            record_layer.decrypt_state,
317            DirectionState::Invalid
318        ));
319        assert_eq!(record_layer.read_seq, 0);
320        assert!(!record_layer.has_decrypted());
321
322        // Preparing the record layer should update the decrypt state, but shouldn't affect whether it
323        // has decrypted.
324        record_layer.prepare_message_decrypter(Box::new(PassThroughDecrypter));
325        assert!(matches!(
326            record_layer.decrypt_state,
327            DirectionState::Prepared
328        ));
329        assert_eq!(record_layer.read_seq, 0);
330        assert!(!record_layer.has_decrypted());
331
332        // Starting decryption should update the decrypt state, but not affect whether it has decrypted.
333        record_layer.start_decrypting();
334        assert!(matches!(record_layer.decrypt_state, DirectionState::Active));
335        assert_eq!(record_layer.read_seq, 0);
336        assert!(!record_layer.has_decrypted());
337
338        // Decrypting a message should update the read_seq and track that we have now performed
339        // a decryption.
340        record_layer
341            .decrypt_incoming(InboundOpaqueMessage::new(
342                ContentType::Handshake,
343                ProtocolVersion::TLSv1_2,
344                &mut [0xC0, 0xFF, 0xEE],
345            ))
346            .unwrap();
347        assert!(matches!(record_layer.decrypt_state, DirectionState::Active));
348        assert_eq!(record_layer.read_seq, 1);
349        assert!(record_layer.has_decrypted());
350
351        // Resetting the record layer message decrypter (as if a key update occurred) should reset
352        // the read_seq number, but not our knowledge of whether we have decrypted previously.
353        record_layer.set_message_decrypter(Box::new(PassThroughDecrypter));
354        assert!(matches!(record_layer.decrypt_state, DirectionState::Active));
355        assert_eq!(record_layer.read_seq, 0);
356        assert!(record_layer.has_decrypted());
357    }
