impl ::digest::common::hazmat::SerializableState for Md5 {
    type SerializedStateSize =
        ::digest::typenum::Sum<<block_api::Md5Core as
        ::digest::common::hazmat::SerializableState>::SerializedStateSize,
        ::digest::block_buffer::SerializedBufferSize<<block_api::Md5Core as
        ::digest::block_api::BlockSizeUser>::BlockSize,
        <block_api::Md5Core as
        ::digest::block_api::BufferKindUser>::BufferKind>>;
    #[inline]
    fn serialize(&self) -> ::digest::common::hazmat::SerializedState<Self> {
        let serialized_core = self.core.serialize();
        let serialized_buf = self.buffer.serialize();
        serialized_core.concat(serialized_buf)
    }
    #[inline]
    fn deserialize(serialized_state:
            &::digest::common::hazmat::SerializedState<Self>)
        -> Result<Self, ::digest::common::hazmat::DeserializeStateError> {
        use ::digest::common::hazmat::{
            SerializableState, DeserializeStateError,
        };
        let (serialized_core, serialized_buf) =
            serialized_state.split_ref::<<block_api::Md5Core as
                SerializableState>::SerializedStateSize>();
        let core = SerializableState::deserialize(serialized_core)?;
        let buffer =
            ::digest::block_buffer::BlockBuffer::deserialize(serialized_buf).map_err(|_|
                        DeserializeStateError)?;
        Ok(Self { core, buffer })
    }
}
