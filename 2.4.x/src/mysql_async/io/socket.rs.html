21/// Unix domain socket connection on unix, or named pipe connection on windows.
22#[derive(#[automatically_derived]impl ::core::fmt::Debug for Socket {
    #[inline]
    #[rustc_trivial_field_reads]
    fn fmt(&self, f: &mut ::core::fmt::Formatter) -> ::core::fmt::Result {
        ::core::fmt::Formatter::debug_struct_field1_finish(f, "Socket inner",
            &&self.inner)
    }
}Debug)]
23pub(crate) struct Socket {
24    #[cfg(unix)]
25    inner: tokio::net::UnixStream,
26}
27
28impl Socket {
29    /// Connects a new socket.
30    #[cfg(unix)]
31    pub async fn new<P: AsRef<Path>>(path: P) -> Result<Socket, io::Error> {
32        Ok(Socket {
33            inner: tokio::net::UnixStream::connect(path).await?,
34        })
35    }
36}
37
38impl AsyncRead for Socket {
39    fn poll_read(
40        mut self: Pin<&mut Self>,
41        cx: &mut Context<'_>,
42        buf: &mut ReadBuf<'_>,
43    ) -> Poll<Result<(), Error>> {
44        loop {
    match Pin::new(&mut self.inner).poll_read(cx, buf) {
        Poll::Ready(Err(err)) if err.kind() == Interrupted => continue,
        x => break x,
    }
}with_interrupted!(Pin::new(&mut self.inner).poll_read(cx, buf))
45    }
46}
