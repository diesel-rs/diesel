26pub(crate) struct Recycler {
27    inner: Arc<Inner>,
28    discard: FuturesUnordered<BoxFuture<'static, ()>>,
29    discarded: usize,
30    cleaning: FuturesUnordered<BoxFuture<'static, Conn>>,
31    reset: FuturesUnordered<BoxFuture<'static, Conn>>,
32
33    // Option<Conn> so that we have a way to send a "I didn't make a Conn after all" signal
34    dropped: mpsc::UnboundedReceiver<Option<Conn>>,
35    /// Pool options.
36    pool_opts: PoolOpts,
37    eof: bool,
38}
39
40impl Recycler {
41    pub fn new(
42        pool_opts: PoolOpts,
43        inner: Arc<Inner>,
44        dropped: UnboundedReceiver<Option<Conn>>,
45    ) -> Self {
46        Self {
47            inner,
48            discard: FuturesUnordered::new(),
49            discarded: 0,
50            cleaning: FuturesUnordered::new(),
51            reset: FuturesUnordered::new(),
52            dropped,
53            pool_opts,
54            eof: false,
55        }
56    }
57
58    fn conn_return(&mut self, conn: Conn, pool_is_closed: bool) {
59        let mut exchange = self.inner.exchange.lock().unwrap();
60        if pool_is_closed || exchange.available.len() >= self.pool_opts.active_bound() {
61            drop(exchange);
62            self.inner
63                .metrics
64                .discarded_superfluous_connection
65                .fetch_add(1, Ordering::Relaxed);
66            self.discard.push(conn.close_conn().boxed());
67        } else {
68            self.inner
69                .metrics
70                .connection_returned_to_pool
71                .fetch_add(1, Ordering::Relaxed);
72            #[cfg(feature = "hdrhistogram")]
73            self.inner
74                .metrics
75                .connection_active_duration
76                .lock()
77                .unwrap()
78                .record(conn.inner.active_since.elapsed().as_micros() as u64)
79                .ok();
80            exchange.available.push_back(conn.into());
81            self.inner
82                .metrics
83                .connections_in_pool
84                .store(exchange.available.len(), Ordering::Relaxed);
85            exchange.waiting.wake();
86        }
87    }
88
89    fn conn_decision(&mut self, conn: Conn, close: bool) {
90        if conn.inner.stream.is_none() || conn.inner.disconnected {
91            // drop unestablished connection
92            self.inner
93                .metrics
94                .discarded_unestablished_connection
95                .fetch_add(1, Ordering::Relaxed);
96            self.discard.push(futures_util::future::ok(()).boxed());
97        } else if conn.inner.tx_status != TxStatus::None || conn.has_pending_result() {
98            self.inner
99                .metrics
100                .dirty_connection_return
101                .fetch_add(1, Ordering::Relaxed);
102            self.cleaning.push(conn.cleanup_for_pool().boxed());
103        } else if conn.expired() || close {
104            self.inner
105                .metrics
106                .discarded_expired_connection
107                .fetch_add(1, Ordering::Relaxed);
108            self.discard.push(conn.close_conn().boxed());
109        } else if conn.inner.reset_upon_returning_to_a_pool {
110            self.inner
111                .metrics
112                .resetting_connection
113                .fetch_add(1, Ordering::Relaxed);
114            self.reset.push(conn.reset_for_pool().boxed());
115        } else {
116            self.conn_return(conn, false);
117        }
118    }
119}

121impl Future for Recycler {
122    type Output = ();
123
124    fn poll(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
125        let mut close = self.inner.close.load(Ordering::Acquire);
126
127        while !self.eof {
128            // see if there are more connections for us to recycle
129            match Pin::new(&mut self.dropped).poll_recv(cx) {
130                Poll::Ready(Some(Some(conn))) => {
131                    if !conn.inner.pool.is_none() {
    ::core::panicking::panic("assertion failed: conn.inner.pool.is_none()")
};assert!(conn.inner.pool.is_none());
132                    self.conn_decision(conn, close);
133                }
134                Poll::Ready(Some(None)) => {
135                    // someone signaled us that it's exit time
136                    close = self.inner.close.load(Ordering::Acquire);
137                    if !close { ::core::panicking::panic("assertion failed: close") };assert!(close);
138                    continue;
139                }
140                Poll::Ready(None) => {
141                    // no more connections are coming -- time to exit!
142                    self.inner.close.store(true, Ordering::Release);
143                    self.eof = true;
144                    close = true;
145                }
146                Poll::Pending => {
147                    // nope -- but let's still make progress on the ones we have
148                    break;
149                }
150            }
151        }
152
153        // if we've been asked to close, reclaim any idle connections
154        if close || self.eof {
155            loop {
156                let Some(IdlingConn { conn, .. }) =
157                    self.inner.exchange.lock().unwrap().available.pop_front()
158                else {
159                    break;
160                };
161                if !conn.inner.pool.is_none() {
    ::core::panicking::panic("assertion failed: conn.inner.pool.is_none()")
};assert!(conn.inner.pool.is_none());
162                self.conn_decision(conn, close);
163            }
164        }
165
166        // are any dirty connections ready for us to reclaim?
167        loop {
168            match Pin::new(&mut self.cleaning).poll_next(cx) {
169                Poll::Pending | Poll::Ready(None) => break,
170                Poll::Ready(Some(Ok(conn))) => self.conn_decision(conn, close),
171                Poll::Ready(Some(Err(e))) => {
172                    // an error occurred while cleaning a connection.
173                    // what do we do? replace it with a new connection?
174                    // for a conn to end up in cleaning, it must have come through .dropped.
175                    // anything that comes through .dropped we know has .pool.is_none().
176                    // therefore, dropping the conn won't decrement .exist, so we need to do that.
177                    self.discarded += 1;
178                    self.inner
179                        .metrics
180                        .discarded_error_during_cleanup
181                        .fetch_add(1, Ordering::Relaxed);
182                    // NOTE: we're discarding the error here
183                    let _ = e;
184                }
185            }
186        }
187
188        // let's iterate through connections being successfully reset
189        loop {
190            match Pin::new(&mut self.reset).poll_next(cx) {
191                Poll::Pending | Poll::Ready(None) => break,
192                Poll::Ready(Some(Ok(conn))) => self.conn_return(conn, close),
193                Poll::Ready(Some(Err(e))) => {
194                    // an error during reset.
195                    // replace with a new connection
196                    self.discarded += 1;
197                    self.inner
198                        .metrics
199                        .discarded_error_during_cleanup
200                        .fetch_add(1, Ordering::Relaxed);
201                    // NOTE: we're discarding the error here
202                    let _ = e;
203                }
204            }
205        }
206
207        // are there any torn-down connections for us to deal with?
208        loop {
209            match Pin::new(&mut self.discard).poll_next(cx) {
210                Poll::Pending | Poll::Ready(None) => break,
211                Poll::Ready(Some(Ok(()))) => {
212                    // yes! count it.
213                    // note that we must decrement .exist since the connection does not have a
214                    // .pool, and therefore won't do anything useful when it is dropped.
215                    self.discarded += 1
216                }
217                Poll::Ready(Some(Err(e))) => {
218                    // an error occurred while closing a connection.
219                    // what do we do? we still replace it with a new connection..
220                    self.discarded += 1;
221                    self.inner
222                        .metrics
223                        .discarded_error_during_cleanup
224                        .fetch_add(1, Ordering::Relaxed);
225                    // NOTE: we're discarding the error here
226                    let _ = e;
227                }
228            }
229        }
230
231        if self.discarded != 0 {
232            // we need to open up slots for new connctions to be established!
233            let mut exchange = self.inner.exchange.lock().unwrap();
234            exchange.exist -= self.discarded;
235            self.inner
236                .metrics
237                .connection_count
238                .store(exchange.exist, Ordering::Relaxed);
239            for _ in 0..self.discarded {
240                exchange.waiting.wake();
241            }
242            drop(exchange);
243            self.discarded = 0;
244        }
245
246        // NOTE: we are asserting here that no more connections will ever be returned to
247        // us. see the explanation in Pool::poll_new_conn for why this is okay, even during
248        // races on .exist
249        let effectively_eof = close && self.inner.exchange.lock().unwrap().exist == 0;
250
251        if (self.eof || effectively_eof)
252            && self.cleaning.is_empty()
253            && self.discard.is_empty()
254            && self.reset.is_empty()
255        {
256            // we know that all Pool handles have been dropped (self.dropped.poll returned None).
257
258            // if this assertion fails, where are the remaining connections?
259            {
    match (&self.inner.exchange.lock().unwrap().available.len(), &0) {
        (left_val, right_val) => {
            if !(*left_val == *right_val) {
                let kind = ::core::panicking::AssertKind::Eq;
                ::core::panicking::assert_failed(kind, &*left_val,
                    &*right_val, ::core::option::Option::None);
            }
        }
    }
};assert_eq!(self.inner.exchange.lock().unwrap().available.len(), 0);
260
261            // NOTE: it is _necessary_ that we set this _before_ we call .wake
262            // otherwise, the following may happen to the DisconnectPool future:
263            //
264            //  - We wake all in .wake
265            //  - DisconnectPool::poll adds to .wake
266            //  - DisconnectPool::poll reads .closed == false
267            //  - We set .closed = true
268            //
269            // At this point, DisconnectPool::poll will never be notified again.
270            self.inner.closed.store(true, Ordering::Release);
271        }
272
273        if self.inner.closed.load(Ordering::Acquire) {
274            // `DisconnectPool` might still wait to be woken up.
275            let mut exchange = self.inner.exchange.lock().unwrap();
276            while exchange.waiting.wake() {}
277            // we're about to exit, so there better be no outstanding connections
278            {
    match (&exchange.exist, &0) {
        (left_val, right_val) => {
            if !(*left_val == *right_val) {
                let kind = ::core::panicking::AssertKind::Eq;
                ::core::panicking::assert_failed(kind, &*left_val,
                    &*right_val, ::core::option::Option::None);
            }
        }
    }
};assert_eq!(exchange.exist, 0);
279            {
    match (&exchange.available.len(), &0) {
        (left_val, right_val) => {
            if !(*left_val == *right_val) {
                let kind = ::core::panicking::AssertKind::Eq;
                ::core::panicking::assert_failed(kind, &*left_val,
                    &*right_val, ::core::option::Option::None);
            }
        }
    }
};assert_eq!(exchange.available.len(), 0);
280            drop(exchange);
281
282            Poll::Ready(())
283        } else {
284            Poll::Pending
285        }
286    }
287}

289impl Drop for Recycler {
290    fn drop(&mut self) {
291        if !self.inner.closed.load(Ordering::Acquire) {
292            // user did not wait for outstanding connections to finish!
293            // this is not good -- we won't be able to shut down our connections cleanly
294            // all we can do is try to ensure a clean shutdown
295            self.inner.close.store(true, Ordering::SeqCst);
296        }
297    }
