23            queue: Default::default(),
24            metrics,
25        }

28    pub(crate) fn push(&mut self, waker: Waker, queue_id: QueueId) {
29        // The documentation of Future::poll says:
30        //   Note that on multiple calls to poll, only the Waker from
31        //   the Context passed to the most recent call should be
32        //   scheduled to receive a wakeup.
33        //
34        // But the the documentation of KeyedPriorityQueue::push says:
35        //   Adds new element to queue if missing key or replace its
36        //   priority if key exists. In second case doesn’t replace key.
37        //
38        // This means we have to remove first to have the most recent
39        // waker in the queue.
40        self.remove(queue_id);
41        self.queue.push(QueuedWaker { queue_id, waker }, queue_id);
42
43        self.metrics
44            .active_wait_requests
45            .fetch_add(1, atomic::Ordering::Relaxed);
46    }
47
48    /// Returns `true` if anyone was awaken
49    pub(crate) fn wake(&mut self) -> bool {
50        match self.queue.pop() {
51            Some((qw, _)) => {
52                self.metrics
53                    .active_wait_requests
54                    .fetch_sub(1, atomic::Ordering::Relaxed);
55                qw.waker.wake();
56                true
57            }
58            None => false,
59        }
60    }
61
62    pub(crate) fn remove(&mut self, id: QueueId) {
63        if self.queue.remove(&id).is_some() {
64            self.metrics
65                .active_wait_requests
66                .fetch_sub(1, atomic::Ordering::Relaxed);
67        }
68    }
69
70    pub(crate) fn peek_id(&mut self) -> Option<QueueId> {
71        self.queue.peek().map(|(qw, _)| qw.queue_id)
72    }
73
74    // only used in tests for now
75    #[allow(dead_code)]
76    pub(crate) fn len(&self) -> usize {
77        self.queue.len()
78    }
79}

120mod test {
121    use std::cmp::Reverse;
122    use std::task::RawWaker;
123    use std::task::RawWakerVTable;
124    use std::task::Waker;
125
126    use super::*;
127
128    #[test]
129    fn waitlist_integrity() {
130        const DATA: *const () = &();
131        const NOOP_CLONE_FN: unsafe fn(*const ()) -> RawWaker = |_| RawWaker::new(DATA, &RW_VTABLE);
132        const NOOP_FN: unsafe fn(*const ()) = |_| {};
133        static RW_VTABLE: RawWakerVTable =
134            RawWakerVTable::new(NOOP_CLONE_FN, NOOP_FN, NOOP_FN, NOOP_FN);
135        let w = unsafe { Waker::from_raw(RawWaker::new(DATA, &RW_VTABLE)) };
136
137        let mut waitlist = Waitlist::new(Default::default());
138        assert_eq!(0, waitlist.queue.len());
139
140        waitlist.push(w.clone(), QueueId(Reverse(4)));
141        waitlist.push(w.clone(), QueueId(Reverse(2)));
142        waitlist.push(w.clone(), QueueId(Reverse(8)));
143        waitlist.push(w.clone(), QUEUE_END_ID);
144        waitlist.push(w.clone(), QueueId(Reverse(10)));
145
146        waitlist.remove(QueueId(Reverse(8)));
147
148        assert_eq!(4, waitlist.queue.len());
149
150        let (_, id) = waitlist.queue.pop().unwrap();
151        assert_eq!(2, id.0 .0);
152        let (_, id) = waitlist.queue.pop().unwrap();
153        assert_eq!(4, id.0 .0);
154        let (_, id) = waitlist.queue.pop().unwrap();
155        assert_eq!(10, id.0 .0);
156        let (_, id) = waitlist.queue.pop().unwrap();
157        assert_eq!(QUEUE_END_ID, id);
158
159        assert_eq!(0, waitlist.queue.len());
160    }
