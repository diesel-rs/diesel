19/// States of the GetConn future.
20pub(crate) enum GetConnInner {
21    New,
22    Done,
23    // TODO: one day this should be an existential
24    Connecting(crate::BoxFuture<'static, Conn>),
25    /// This future will check, that idling connection is alive.
26    Checking(crate::BoxFuture<'static, Conn>),
27}
28
29impl fmt::Debug for GetConnInner {
30    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
31        match self {
32            GetConnInner::New => f.debug_tuple("GetConnInner::New").finish(),
33            GetConnInner::Done => f.debug_tuple("GetConnInner::Done").finish(),
34            GetConnInner::Connecting(_) => f
35                .debug_tuple("GetConnInner::Connecting")
36                .field(&"<future>")
37                .finish(),
38            GetConnInner::Checking(_) => f
39                .debug_tuple("GetConnInner::Checking")
40                .field(&"<future>")
41                .finish(),
42        }
43    }
44}

46#[derive(#[automatically_derived]impl ::core::fmt::Debug for GetConnState {
    #[inline]
    #[rustc_trivial_field_reads]
    fn fmt(&self, f: &mut ::core::fmt::Formatter) -> ::core::fmt::Result {
        ::core::fmt::Formatter::debug_struct_field3_finish(f,
            "GetConnState queue_id pool inner", &self.queue_id, &self.pool,
            &&self.inner)
    }
}Debug)]
47struct GetConnState {
48    queue_id: QueueId,
49    pool: Option<Pool>,
50    inner: GetConnInner,
51}
52
53impl GetConnState {
54    fn pool_mut(&mut self) -> &mut Pool {
55        self.pool
56            .as_mut()
57            .expect("GetConn::poll polled after returning Async::Ready")
58    }
59
60    fn pool_take(&mut self) -> Pool {
61        self.pool
62            .take()
63            .expect("GetConn::poll polled after returning Async::Ready")
64    }
65
66    fn ready_connection(&mut self, mut c: Conn) -> Result<Conn> {
67        let pool = self.pool_take();
68        pool.inner
69            .metrics
70            .connections_in_use
71            .fetch_add(1, Ordering::Relaxed);
72
73        c.inner.reset_upon_returning_to_a_pool = pool.opts.pool_opts().reset_connection();
74        c.inner.pool = Some(pool);
75
76        Ok(c)
77    }
78}

80/// This future will take connection from a pool and resolve to [`Conn`].
81#[cfg_attr(feature = "tracing", tracing::instrument(level = "debug", skip_all))]
82pub(crate) async fn get_conn(pool: Pool) -> Result<Conn> {
83    let queue_id = QueueId::next();
84    let mut state = GetConnState {
85        queue_id,
86        pool: Some(pool),
87        inner: GetConnInner::New,
88    };
89
90    loop {
91        match state.inner {
92            GetConnInner::New => {
93                let pool = state.pool_mut();
94                let poll_new = |cx: &mut Context<'_>| pool.poll_new_conn(cx, queue_id);
95                let next = poll_fn(poll_new).await?;
96                match next {
97                    GetConnInner::Connecting(conn_fut) => {
98                        state.inner = GetConnInner::Connecting(conn_fut);
99                    }
100                    GetConnInner::Checking(conn_fut) => {
101                        state.inner = GetConnInner::Checking(conn_fut);
102                    }
103                    GetConnInner::Done => {
    ::core::panicking::panic_fmt(format_args!("internal error: entered unreachable code: {0}",
            format_args!("Pool::poll_new_conn never gives out already-consumed GetConns")));
}unreachable!(
104                        "Pool::poll_new_conn never gives out already-consumed GetConns"
105                    ),
106                    GetConnInner::New => {
107                        {
    ::core::panicking::panic_fmt(format_args!("internal error: entered unreachable code: {0}",
            format_args!("Pool::poll_new_conn never gives out GetConnInner::New")));
}unreachable!("Pool::poll_new_conn never gives out GetConnInner::New")
108                    }
109                }
110            }
111            GetConnInner::Done => {
112                {
    ::core::panicking::panic_fmt(format_args!("internal error: entered unreachable code: {0}",
            format_args!("GetConn::poll polled after returning Async::Ready")));
};unreachable!("GetConn::poll polled after returning Async::Ready");
113            }
114            GetConnInner::Connecting(ref mut f) => {
115                let result = f.await;
116                state.inner = GetConnInner::Done;
117
118                return match result {
119                    Ok(c) => state.ready_connection(c),
120                    Err(e) => {
121                        let pool = state.pool_take();
122                        pool.cancel_connection();
123                        Err(e)
124                    }
125                };
126            }
127            GetConnInner::Checking(ref mut f) => {
128                let result = f.await;
129                match result {
130                    Ok(c) => {
131                        state.inner = GetConnInner::Done;
132
133                        return state.ready_connection(c);
134                    }
135                    Err(_) => {
136                        // Idling connection is broken. We'll drop it and try again.
137                        state.inner = GetConnInner::New;
138
139                        let pool = state.pool_mut();
140                        pool.cancel_connection();
141                        continue;
142                    }
143                }
144            }
145        }
146    }
147}
148
149impl Drop for GetConnState {
150    fn drop(&mut self) {
151        // We drop a connection before it can be resolved, a.k.a. cancelling it.
152        // Make sure we maintain the necessary invariants towards the pool.
153        if let Some(pool) = self.pool.take() {
154            // Remove the waker from the pool's waitlist in case this task was
155            // woken by another waker, like from tokio::time::timeout.
156            pool.unqueue(self.queue_id);
157            if let GetConnInner::Connecting(..) = self.inner {
158                pool.cancel_connection();
159            }
160        }
161    }
162}
