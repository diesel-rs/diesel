22impl<P> Routine<()> for NextSetRoutine<P>
23where
24    P: Protocol,
25{
26    fn call<'a>(self, conn: &'a mut Conn) -> BoxFuture<'a, crate::Result<()>>
27    where
28        Self: 'a,
29    {
30        #[cfg(feature = "tracing")]
31        let span = debug_span!(
32            "mysql_async::next_set",
33            mysql_async.connection.id = conn.id()
34        );
35        conn.sync_seq_id();
36        let fut = async move {
37            // Cached metadata can be for binary protocol only. With binary we can't have a batch prepared. Multiple results
38            // are only possible with SP call. But with them we won't have metadata skipped(on 2nd or more result).
39            conn.read_result_set::<P>(false, None).await?;
40            Ok(())
41        };
42
43        #[cfg(feature = "tracing")]
44        let fut = instrument_result!(fut, span);
45
46        fut.boxed()
47    }
