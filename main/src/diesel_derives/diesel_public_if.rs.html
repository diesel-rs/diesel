75impl syn::parse::Parse for EntryWithVisibility {
76    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
77        let meta = syn::Attribute::parse_outer(input)?;
78        if input.peek(::syn::token::FnToken![fn]) || input.peek(::syn::token::TypeToken![type]) {
79            let tail = input.parse()?;
80            Ok(Self::TraitFunction { meta, tail })
81        } else {
82            let vis = input.parse()?;
83            if input.peek(::syn::token::StructToken![struct]) {
84                let s = DeriveInput::parse(input)?;
85                if let syn::Data::Struct(def) = s.data {
86                    Ok(Self::Struct {
87                        meta,
88                        vis,
89                        def,
90                        generics: s.generics,
91                        ident: s.ident,
92                    })
93                } else {
94                    ::core::panicking::panic("internal error: entered unreachable code")unreachable!()
95                }
96            } else {
97                let tail = input.parse()?;
98                Ok(Self::Item { meta, vis, tail })
99            }
100        }
101    }
