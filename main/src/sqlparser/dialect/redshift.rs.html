31// In most cases the redshift dialect is identical to [`PostgresSqlDialect`].
32//
33// Notable differences:
34// 1. Redshift treats brackets `[` and `]` differently. For example, `SQL SELECT a[1][2] FROM b`
35// in the Postgres dialect, the query will be parsed as an array, while in the Redshift dialect it will
36// be a json path
37impl Dialect for RedshiftSqlDialect {
38    /// Determine if a character starts a potential nested quoted identifier.
39    /// Example: RedShift supports the following quote styles to all mean the same thing:
40    /// ```sql
41    /// SELECT 1 AS foo;
42    /// SELECT 1 AS "foo";
43    /// SELECT 1 AS [foo];
44    /// SELECT 1 AS ["foo"];
45    /// ```
46    fn is_nested_delimited_identifier_start(&self, ch: char) -> bool {
47        ch == '['
48    }
49
50    /// Only applicable whenever [`Self::is_nested_delimited_identifier_start`] returns true
51    /// If the next sequence of tokens potentially represent a nested identifier, then this method
52    /// returns a tuple containing the outer quote style, and if present, the inner (nested) quote style.
53    ///
54    /// Example (Redshift):
55    /// ```text
56    /// `["foo"]` => Some(`[`, Some(`"`))
57    /// `[foo]` => Some(`[`, None)
58    /// `[0]` => None
59    /// `"foo"` => None
60    /// ```
61    fn peek_nested_delimited_identifier_quotes(
62        &self,
63        mut chars: Peekable<Chars<'_>>,
64    ) -> Option<(char, Option<char>)> {
65        if chars.peek() != Some(&'[') {
66            return None;
67        }
68
69        chars.next();
70
71        let mut not_white_chars = chars.skip_while(|ch| ch.is_whitespace()).peekable();
72
73        if let Some(&ch) = not_white_chars.peek() {
74            if ch == '"' {
75                return Some(('[', Some('"')));
76            }
77            if self.is_identifier_start(ch) {
78                return Some(('[', None));
79            }
80        }
81
82        None
83    }
84
85    fn is_identifier_start(&self, ch: char) -> bool {
86        // UTF-8 multibyte characters are supported in identifiers via the PostgreSqlDialect.
87        // https://docs.aws.amazon.com/redshift/latest/dg/r_names.html
88        PostgreSqlDialect {}.is_identifier_start(ch) || ch == '#'
89    }
90
91    fn is_identifier_part(&self, ch: char) -> bool {
92        // UTF-8 multibyte characters are supported in identifiers via the PostgreSqlDialect.
93        // https://docs.aws.amazon.com/redshift/latest/dg/r_names.html
94        PostgreSqlDialect {}.is_identifier_part(ch) || ch == '#'
95    }
96
97    /// See <https://docs.aws.amazon.com/redshift/latest/dg/r_names.html>
98    fn identifier_quote_style(&self, _identifier: &str) -> Option<char> {
99        Some('"')
100    }
101
102    /// redshift has `CONVERT(type, value)` instead of `CONVERT(value, type)`
103    /// <https://docs.aws.amazon.com/redshift/latest/dg/r_CONVERT_function.html>
104    fn convert_type_before_value(&self) -> bool {
105        true
106    }
107
108    fn supports_connect_by(&self) -> bool {
109        true
110    }
111
112    /// Redshift expects the `TOP` option before the `ALL/DISTINCT` option:
113    /// <https://docs.aws.amazon.com/redshift/latest/dg/r_SELECT_list.html#r_SELECT_list-parameters>
114    fn supports_top_before_distinct(&self) -> bool {
115        true
116    }
117
118    /// Redshift supports PartiQL: <https://docs.aws.amazon.com/redshift/latest/dg/super-overview.html>
119    fn supports_partiql(&self) -> bool {
120        true
121    }
122

124        true
125    }

153    }

159    }

179    }

181    /// `INTERVAL` is listed as reserved in Redshift docs but the actual parser allows it as an
182    /// identifier, consistent with pgsql.
183    fn is_reserved_for_identifier(&self, kw: Keyword) -> bool {
184        if #[allow(non_exhaustive_omitted_patterns)] match kw {
    Keyword::INTERVAL => true,
    _ => false,
}matches!(kw, Keyword::INTERVAL) {
185            false
186        } else {
187            RESERVED_FOR_IDENTIFIER.contains(&kw)
188        }
189    }
