31impl Comments {
32    /// Accepts `comment` if its the first or is located strictly after the
33    /// last accepted comment.  In other words, this method will skip the
34    /// comment if its comming out of order (as encountered in the parsed
35    /// source code.)
36    pub fn offer(&mut self, comment: CommentWithSpan) {
37        if self
38            .0
39            .last()
40            .map(|last| last.span < comment.span)
41            .unwrap_or(true)
42        {
43            self.0.push(comment);
44        }
45    }

47    /// Finds comments starting within the given location range. The order of
48    /// iterator reflects the order of the comments as encountered in the parsed
49    /// source code.
50    ///
51    /// # Example
52    /// ```rust
53    /// use sqlparser::{dialect::GenericDialect, parser::Parser, tokenizer::Location};
54    ///
55    /// let sql = r#"/*
56    ///  header comment ...
57    ///  ... spanning multiple lines
58    /// */

60    ///  -- first statement

62    ///

72    /// assert_eq!(

88    /// information for certain AST nodes.

90        let (start, end) = (
91            self.start_index(range.start_bound()),
92            self.end_index(range.end_bound()),
93        );
94        if true {

        ::core::panicking::panic("assertion failed: (0..=self.0.len()).contains(&start)")
    };
};debug_assert!((0..=self.0.len()).contains(&start));
95        if true {
    if !(0..=self.0.len()).contains(&end) {
        ::core::panicking::panic("assertion failed: (0..=self.0.len()).contains(&end)")
    };
};debug_assert!((0..=self.0.len()).contains(&end));

104    /// Finds comments immediately before the token starting at `location`, in source order.
105    ///
106    /// # Example
107    /// ```rust
108    /// use sqlparser::{dialect::GenericDialect, parser::Parser, tokenizer::Location};
109    ///
110    /// let sql = r#"SELECT 1; -- one
111    /// -- two
112    /// SELECT 2"#;
113    /// let (_, comments) = Parser::parse_sql_with_comments(&GenericDialect, sql).unwrap();
114    ///
115    /// let second = Location::new(3, 1);
116    /// assert_eq!(
117    ///    &comments.find_before(second).map(|c| c.as_str()).collect::<Vec<_>>(),
118    ///    &[" one", " two"]);
119    /// ```
120    pub fn find_before(&self, location: Location) -> Iter<'_> {
121        let end = self.end_index(Bound::Excluded(&location));
122        let start = self.0[..end].partition_point(|c| c.next_token_start != Some(location));
123        Iter(self.0[start..end].iter())
124    }
125
126    pub(crate) fn set_next_token_start(&mut self, next_token_start: Location) {
127        for comment in self
128            .0
129            .iter_mut()
130            .rev()
131            .take_while(|c| c.next_token_start.is_none())
132        {
133            comment.next_token_start = Some(next_token_start);
134        }
135    }
136
137    /// Find the index of the first comment starting "before" the given location.
138    ///
139    /// The returned index is _inclusive_ and within the range of `0..=self.0.len()`.
140    fn start_index(&self, location: Bound<&Location>) -> usize {
141        match location {
142            Bound::Included(location) => {
143                match self.0.binary_search_by(|c| c.span.start.cmp(location)) {
144                    Ok(i) => i,
145                    Err(i) => i,
146                }
147            }
148            Bound::Excluded(location) => {
149                match self.0.binary_search_by(|c| c.span.start.cmp(location)) {
150                    Ok(i) => i + 1,
151                    Err(i) => i,
152                }
153            }
154            Bound::Unbounded => 0,
155        }
156    }
157
158    /// Find the index of the first comment starting "after" the given location.
159    ///
160    /// The returned index is _exclusive_ and within the range of `0..=self.0.len()`.
161    fn end_index(&self, location: Bound<&Location>) -> usize {
162        match location {
163            Bound::Included(location) => {
164                match self.0.binary_search_by(|c| c.span.start.cmp(location)) {
165                    Ok(i) => i + 1,
166                    Err(i) => i,
167                }
168            }
169            Bound::Excluded(location) => {
170                match self.0.binary_search_by(|c| c.span.start.cmp(location)) {
171                    Ok(i) => i,
172                    Err(i) => i,
173                }
174            }
175            Bound::Unbounded => self.0.len(),
176        }
177    }

209pub enum Comment {
210    /// A single line comment, typically introduced with a prefix and spanning
211    /// until end-of-line or end-of-file in the source code.
212    ///
213    /// Note: `content` will include the terminating new-line character, if any.
214    /// A single-line comment, typically introduced with a prefix and spanning
215    /// until end-of-line or end-of-file in the source code.
216    ///
217    /// Note: `content` will include the terminating new-line character, if any.
218    SingleLine {
219        /// The content of the comment (including trailing newline, if any).
220        content: String,
221        /// The prefix introducing the comment (e.g. `--`, `#`).
222        prefix: String,
223    },
224
225    /// A multi-line comment, typically enclosed in `/* .. */` markers. The
226    /// string represents the content excluding the markers.
227    MultiLine(String),
228}
229
230impl Comment {
231    /// Retrieves the content of the comment as string slice.
232    pub fn as_str(&self) -> &str {
233        match self {
234            Comment::SingleLine { content, prefix: _ } => content.as_str(),
235            Comment::MultiLine(content) => content.as_str(),
236        }
237    }
238}

385    fn test_find_before() {
386        let first_token = Location::new(2, 1);
387        let second_token = Location::new(4, 1);
388        let mut comments = Comments::default();
389        for comment in [
390            CommentWithSpan {
391                comment: Comment::SingleLine {
392                    content: " first".into(),
393                    prefix: "--".into(),
394                },
395                span: Span::new((1, 1).into(), (1, 9).into()),
396                prev_token_end: None,
397                next_token_start: Some(first_token),
398            },
399            CommentWithSpan {
400                comment: Comment::SingleLine {
401                    content: " second".into(),
402                    prefix: "--".into(),
403                },
404                span: Span::new((2, 11).into(), (2, 20).into()),
405                prev_token_end: Some(Location::new(2, 10)),
406                next_token_start: Some(second_token),
407            },
408            CommentWithSpan {
409                comment: Comment::SingleLine {
410                    content: " third".into(),
411                    prefix: "--".into(),
412                },
413                span: Span::new((3, 1).into(), (3, 9).into()),
414                prev_token_end: Some(Location::new(2, 10)),
415                next_token_start: Some(second_token),
416            },
417            CommentWithSpan {
418                comment: Comment::SingleLine {
419                    content: " tail".into(),
420                    prefix: "--".into(),
421                },
422                span: Span::new((4, 11).into(), (4, 18).into()),
423                prev_token_end: Some(Location::new(4, 10)),
424                next_token_start: None,
425            },
426        ] {
427            comments.offer(comment);
428        }
429
430        fn find_before(comments: &Comments, location: Location) -> Vec<&str> {
431            comments.find_before(location).map(|c| c.as_str()).collect()
432        }
433
434        assert_eq!(find_before(&comments, first_token), vec![" first"]);
435        assert_eq!(
436            find_before(&comments, second_token),
437            vec![" second", " third"]
438        );
439        for location in [
440            Location::new(0, 0),
441            Location::new(1, 3),
442            Location::new(2, 8),
443            Location::new(3, 12),
444            Location::new(4, 3),
445            Location::new(5, 1),
446        ] {
447            assert_eq!(
448                find_before(&comments, location),
449                Vec::<&str>::new(),
450                "{location:?}"
451            );
452        }
453        assert_eq!(
454            find_before(&Comments::default(), first_token),
455            Vec::<&str>::new()
456        );
