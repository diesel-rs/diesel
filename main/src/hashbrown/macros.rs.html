4/// Fallback `cfg_select!` on older Rust.
5///
6/// Requires braces around output, but works for items and expressions.
7#[cfg(not(feature = "nightly"))]
8#[macro_export]
9macro_rules! cfg_select {
10    ({ $($tt:tt)* }) => {{
11        $crate::cfg_select! { $($tt)* }
12    }};
13    (_ => { $($output:tt)* }) => {
14        $($output)*
15    };
16    (
17        $cfg:meta => $output:tt
18        $($( $rest:tt )+)?
19    ) => {
20        #[cfg($cfg)]
21        $crate::cfg_select! { _ => $output }
22        $(
23            #[cfg(not($cfg))]
24            $crate::cfg_select! { $($rest)+ }
25        )?
26    }
27}
