7/// Receiver for derived `FromAttributes` impls.
8pub struct FromAttributesOptions {
9    // Note: FromAttributes has no behaviors beyond those common
10    // to all the `OuterFrom` traits.
11    pub base: OuterFrom,
12}
13
14impl FromAttributesOptions {
15    pub fn new(di: &syn::DeriveInput) -> Result<Self> {
16        let opts = (Self {
17            base: OuterFrom::start(di)?,
18        })
19        .parse_attributes(&di.attrs)?
20        .parse_body(&di.data)?;
21
22        let is_transparent = opts
23            .base
24            .container
25            .data
26            .as_struct()
27            .map(|fields| {
28                (fields.len() == 1 && fields.style.is_tuple())
29                    || fields.style.is_struct() && opts.base.container.transparent.is_present()
30            })
31            .unwrap_or(false);
32
33        if !is_transparent && opts.base.attr_names.is_empty() {
34            Err(Error::custom(
35                "FromAttributes without attributes collects nothing",
36            ))
37        } else {
38            Ok(opts)
39        }
40    }
41}
