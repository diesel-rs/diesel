56pub fn expand(input: &mut Item, is_local: bool) {
57    match input {
58        Item::Trait(input) => {
59            let context = Context::Trait {
60                generics: &input.generics,
61                supertraits: &input.supertraits,
62            };
63            for inner in &mut input.items {
64                if let TraitItem::Fn(method) = inner {
65                    let sig = &mut method.sig;
66                    if sig.asyncness.is_some() {
67                        let block = &mut method.default;
68                        let mut has_self = has_self_in_sig(sig);
69                        if let Some(block) = block {
70                            has_self |= has_self_in_block(block);
71                            transform_block(context, sig, block);
72                            method.attrs.push(lint_suppress_with_body());
73                        } else {
74                            method.attrs.push(lint_suppress_without_body());
75                        }
76                        let has_default = method.default.is_some();
77                        transform_sig(context, sig, has_self, has_default, is_local);
78                    }
79                }
80            }
81        }
82        Item::Impl(input) => {
83            let mut associated_type_impl_traits = Set::new();
84            for inner in &input.items {
85                if let ImplItem::Type(assoc) = inner {
86                    if let Type::ImplTrait(_) = assoc.ty {
87                        associated_type_impl_traits.insert(assoc.ident.clone());
88                    }
89                }
90            }
91
92            let context = Context::Impl {
93                impl_generics: &input.generics,
94                associated_type_impl_traits: &associated_type_impl_traits,
95            };
96            for inner in &mut input.items {
97                match inner {
98                    ImplItem::Fn(method) if method.sig.asyncness.is_some() => {
99                        let sig = &mut method.sig;
100                        let block = &mut method.block;
101                        let has_self = has_self_in_sig(sig);
102                        transform_block(context, sig, block);
103                        transform_sig(context, sig, has_self, false, is_local);
104                        method.attrs.push(lint_suppress_with_body());
105                    }
106                    ImplItem::Verbatim(tokens) => {
107                        let mut method = match syn::parse2::<VerbatimFn>(tokens.clone()) {
108                            Ok(method) if method.sig.asyncness.is_some() => method,
109                            _ => continue,
110                        };
111                        let sig = &mut method.sig;
112                        let has_self = has_self_in_sig(sig);
113                        transform_sig(context, sig, has_self, false, is_local);
114                        method.attrs.push(lint_suppress_with_body());
115                        *tokens = {
    let mut _s = ::quote::__private::TokenStream::new();
    ::quote::ToTokens::to_tokens(&method, &mut _s);
    _s
}quote!(#method);
116                    }
117                    _ => {}
118                }
119            }
120        }
121    }

450fn positional_arg(i: usize, pat: &Pat) -> Ident {
451    let span = syn::spanned::Spanned::span(pat).resolved_at(Span::mixed_site());
452    match ::quote::__private::IdentFragmentAdapter(&i) {
    arg =>
        ::quote::__private::mk_ident(&::alloc::__export::must_use({
                        ::alloc::fmt::format(format_args!("__arg{0}", arg))
                    }),
            ::quote::__private::Option::Some::<::quote::__private::Span>(span)),
}format_ident!("__arg{}", i, span = span)
453}

455fn contains_associated_type_impl_trait(context: Context, ret: &mut Type) -> bool {
456    struct AssociatedTypeImplTraits<'a> {
457        set: &'a Set<Ident>,
458        contains: bool,
459    }
460
461    impl<'a> VisitMut for AssociatedTypeImplTraits<'a> {
462        fn visit_type_path_mut(&mut self, ty: &mut TypePath) {
463            if ty.qself.is_none()
464                && ty.path.segments.len() == 2
465                && ty.path.segments[0].ident == "Self"
466                && self.set.contains(&ty.path.segments[1].ident)
467            {
468                self.contains = true;
469            }
470            visit_mut::visit_type_path_mut(self, ty);
471        }
472    }
473
474    match context {
475        Context::Trait { .. } => false,
476        Context::Impl {
477            associated_type_impl_traits,
478            ..
479        } => {
480            let mut visit = AssociatedTypeImplTraits {
481                set: associated_type_impl_traits,
482                contains: false,
483            };
484            visit.visit_type_mut(ret);
485            visit.contains
486        }
487    }
488}

497fn replace_impl_trait_with_infer(ty: &mut Type) {
498    struct ReplaceImplTraitWithInfer;
