26pub enum IsNull {
27    /// The expression might produce a `NULL` value
28    IsNullable,
29    /// The expression cannot produce a `NULL` values
30    NotNullable,
31    /// It's unknown whether or not the expression
32    /// can produce a `NULL` value
33    Unknown,

36impl IsNull {
37    fn or(self, other: Self) -> Self {

40            (Self::IsNullable, _) | (_, Self::IsNullable) => Self::IsNullable,
41            (Self::NotNullable, Self::NotNullable) => Self::NotNullable,
