14pub(crate) fn infer_expr(
15    expr: &sqlparser::ast::Expr,
16    query_source_lookup: &HashMap<Option<&str>, QuerySource>,
17    context: &mut ParseContext,
18) -> Result<Expression> {
19    match expr {
20        Expr::Value(v) => Ok(Expression::Literal {
21            value: v.value.clone().into_string(),
22            is_null: #[allow(non_exhaustive_omitted_patterns)] match v.value {
    Value::Null => true,
    _ => false,
}matches!(v.value, Value::Null),
23        }),
24        Expr::Identifier(id) => {
25            if query_source_lookup.len() == 1 {
26                let table = query_source_lookup
27                    .values()
28                    .next()
29                    .expect("We checked there is only one value");
30                Ok(Expression::Field {
31                    schema: table.schema.map(|s| s.to_owned()),
32                    query_source: table.name.map(|s| s.to_owned()),
33                    field_name: id.value.clone(),
34                    source_id: table.source_id,
35                    // no joins here so we should be fine
36                    via_left_join: false,
37                })
38            } else {
39                Err(Error::UnsupportedSql {
40                    msg: ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Queries without more than one table are not supported yet: `{0}`",
                id))
    })format!(
41                        "Queries without more than one table are not supported yet: `{id}`"
42                    ),
43                })
44            }
45        }
46        s @ Expr::CompoundIdentifier(ids) => match ids.as_slice() {
47            [table, field] => {
48                let table = query_source_lookup
49                    .get(&Some(table.value.as_str()))
50                    .ok_or_else(|| Error::InvalidQuerySource {
51                        query_source: table.value.clone(),
52                    })?;
53                // lookup if this field comes in via a `LEFT JOIN` somewhere in the chain
54                let via_left_join = table.contains_left_join(query_source_lookup)?;
55                Ok(Expression::Field {
56                    schema: table.schema.map(|s| s.to_owned()),
57                    query_source: table.name.map(|s| s.to_owned()),
58                    field_name: field.value.clone(),
59                    source_id: table.source_id,
60                    via_left_join,
61                })
62            }
63            _ => Err(Error::UnsupportedSql {
64                msg: ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Queries with more than 2 identifier segments are not supported yet: `{0}`",
                s))
    })format!(
65                    "Queries with more than 2 identifier segments are not supported yet: `{s}`"
66                ),
67            }),
68        },
69        Expr::Cast {
70            expr, data_type, ..
71        } => {
72            let inner = infer_expr(expr, query_source_lookup, context)?;
73            Ok(Expression::Cast {
74                inner: Box::new(inner),
75                tpe: data_type.to_string(),
76            })
77        }
78        Expr::BinaryOp { left, op, right } => {
79            let left = Box::new(infer_expr(left, query_source_lookup, context)?);
80            let right = Box::new(infer_expr(right, query_source_lookup, context)?);
81            Ok(Expression::BinaryOp {
82                left,
83                right,
84                op: op.to_string(),
85                statically_not_null: false,
86            })
87        }
88        Expr::IsNull(e) => {
89            let inner = Box::new(infer_expr(e, query_source_lookup, context)?);
90            Ok(Expression::PostfixOp {
91                expr: inner,
92                op: String::from("IS NULL"),
93                statically_not_null: true,
94            })
95        }
96        Expr::IsNotNull(e) => {
97            let inner = Box::new(infer_expr(e, query_source_lookup, context)?);
98            Ok(Expression::PostfixOp {
99                expr: inner,
100                op: String::from("IS NOT NULL"),
101                statically_not_null: true,
102            })
103        }
104        Expr::Function(f) => infer_functions(f, query_source_lookup, context),
105        Expr::Like {
106            negated,
107            any,
108            expr,
109            pattern,
110            escape_char,
111        } if !*any && escape_char.is_none() => {
112            let op = if *negated { "NOT LIKE" } else { "LIKE" };
113            Ok(Expression::BinaryOp {
114                left: Box::new(infer_expr(expr, query_source_lookup, context)?),
115                right: Box::new(infer_expr(pattern, query_source_lookup, context)?),
116                op: String::from(op),
117                statically_not_null: false,
118            })
119        }
120        Expr::ILike {
121            negated,
122            any,
123            expr,
124            pattern,
125            escape_char,
126        } if !*any && escape_char.is_none() => {
127            let op = if *negated { "NOT ILIKE" } else { "ILIKE" };
128            Ok(Expression::BinaryOp {
129                left: Box::new(infer_expr(expr, query_source_lookup, context)?),
130                right: Box::new(infer_expr(pattern, query_source_lookup, context)?),
131                op: String::from(op),
132                statically_not_null: false,
133            })
134        }
135        Expr::IsDistinctFrom(a, b) => Ok(Expression::BinaryOp {
136            left: Box::new(infer_expr(a, query_source_lookup, context)?),
137            right: Box::new(infer_expr(b, query_source_lookup, context)?),
138            op: String::from("IS DISTINCT FROM"),
139            statically_not_null: true,
140        }),
141        Expr::IsNotDistinctFrom(a, b) => Ok(Expression::BinaryOp {
142            left: Box::new(infer_expr(a, query_source_lookup, context)?),
143            right: Box::new(infer_expr(b, query_source_lookup, context)?),
144            op: String::from("IS NOT DISTINCT FROM"),
145            statically_not_null: true,
146        }),
147        Expr::Between {
148            expr,
149            negated,
150            low,
151            high,
152        } => Ok(Expression::Between {
153            left: Box::new(infer_expr(expr, query_source_lookup, context)?),
154            negated: *negated,
155            low: Box::new(infer_expr(low, query_source_lookup, context)?),
156            high: Box::new(infer_expr(high, query_source_lookup, context)?),
157        }),
158        Expr::SimilarTo {
159            negated,
160            expr,
161            pattern,
162            escape_char,
163        } if escape_char.is_none() => {
164            let op = if *negated {
165                "NOT SIMILAR TO"
166            } else {
167                "SIMILAR TO"
168            };
169            Ok(Expression::BinaryOp {
170                left: Box::new(infer_expr(expr, query_source_lookup, context)?),
171                right: Box::new(infer_expr(pattern, query_source_lookup, context)?),
172                op: String::from(op),
173                statically_not_null: false,
174            })
175        }
176        Expr::RLike {
177            negated,
178            expr,
179            pattern,
180            regexp,
181        } => {
182            let op = if *regexp {
183                if *negated { "NOT REGEXP" } else { "REGEXP" }
184            } else if *negated {
185                "NOT RLIKE"
186            } else {
187                "RLIKE"
188            };
189            Ok(Expression::BinaryOp {
190                left: Box::new(infer_expr(expr, query_source_lookup, context)?),
191                right: Box::new(infer_expr(pattern, query_source_lookup, context)?),
192                op: String::from(op),
193                statically_not_null: false,
194            })
195        }
196        Expr::Case {
197            operand,
198            conditions,
199            else_result,
200            ..
201        } => {
202            let operand = operand
203                .as_ref()
204                .map(|o| infer_expr(o, query_source_lookup, context))
205                .transpose()?
206                .map(Box::new);
207            let else_clause = else_result
208                .as_ref()
209                .map(|e| infer_expr(e, query_source_lookup, context))
210                .transpose()?
211                .map(Box::new);
212            let conditions = conditions
213                .iter()
214                .map(|c| -> std::result::Result<_, _> {
215                    let condition = infer_expr(&c.condition, query_source_lookup, context)?;
216                    let result = infer_expr(&c.result, query_source_lookup, context)?;
217                    Ok(CaseCondition { condition, result })
218                })
219                .collect::<Result<Vec<_>>>()?;
220            Ok(Expression::Case {
221                operand,
222                conditions,
223                else_clause,
224            })
225        }
226        Expr::InList {
227            expr,
228            list,
229            negated,
230        } => Ok(Expression::In {
231            left: Box::new(infer_expr(expr, query_source_lookup, context)?),
232            negated: *negated,
233            list: list
234                .iter()
235                .map(|e| infer_expr(e, query_source_lookup, context))
236                .collect::<Result<Vec<_>, _>>()?,
237        }),
238        Expr::Nested(n) => infer_expr(n, query_source_lookup, context)
239            .map(Box::new)
240            .map(Expression::Grouped),
241        Expr::Subquery(query) => {
242            let results = crate::select::parse_query(query, Some(query_source_lookup), context)?;
243            Ok(Expression::Subquery { selection: results })
244        }
245        Expr::InSubquery {
246            expr,
247            subquery,
248            negated,
249        } => {
250            let results = crate::select::parse_query(subquery, Some(query_source_lookup), context)?;
251            Ok(Expression::InSubQuery {
252                left: Box::new(infer_expr(expr, query_source_lookup, context)?),
253                negated: *negated,
254                subquery: results,
255            })
256        }
257        // other kinds of expressions still need to be supported
258        Expr::CompoundFieldAccess { .. }
259        | Expr::JsonAccess { .. }
260        | Expr::IsFalse(..)
261        | Expr::IsNotFalse(..)
262        | Expr::IsTrue(..)
263        | Expr::IsNotTrue(..)
264        | Expr::IsUnknown(..)
265        | Expr::IsNotUnknown(..)
266        | Expr::IsNormalized { .. }
267        | Expr::InUnnest { .. }
268        | Expr::Like { .. }
269        | Expr::ILike { .. }
270        | Expr::SimilarTo { .. }
271        | Expr::AnyOp { .. }
272        | Expr::AllOp { .. }
273        | Expr::UnaryOp { .. }
274        | Expr::Convert { .. }
275        | Expr::AtTimeZone { .. }
276        | Expr::Extract { .. }
277        | Expr::Ceil { .. }
278        | Expr::Floor { .. }
279        | Expr::Position { .. }
280        | Expr::Substring { .. }
281        | Expr::Trim { .. }
282        | Expr::Overlay { .. }
283        | Expr::Collate { .. }
284        | Expr::Prefixed { .. }
285        | Expr::TypedString(..)
286        | Expr::Exists { .. }
287        | Expr::GroupingSets(..)
288        | Expr::Cube(..)
289        | Expr::Rollup(..)
290        | Expr::Tuple(..)
291        | Expr::Struct { .. }
292        | Expr::Named { .. }
293        | Expr::Dictionary(..)
294        | Expr::Map(..)
295        | Expr::Array(..)
296        | Expr::Interval(..)
297        | Expr::MatchAgainst { .. }
298        | Expr::Wildcard(..)
299        | Expr::QualifiedWildcard(..)
300        | Expr::OuterJoin(..)
301        | Expr::Prior(..)
302        | Expr::Lambda(..)
303        | Expr::MemberOf(..)
304        | Expr::IsJson { .. } => Ok(Expression::Unknown),
305    }
306}

308fn infer_functions(
309    f: &sqlparser::ast::Function,
310    query_source_lookup: &HashMap<Option<&str>, QuerySource<'_>>,
311    context: &mut ParseContext,
312) -> Result<Expression> {
313    let (name, schema) = match f.name.0.as_slice() {
314        [ObjectNamePart::Identifier(name)] => (&name.value, None),
315        [
316            ObjectNamePart::Identifier(schema),
317            ObjectNamePart::Identifier(name),
318        ] => (&name.value, Some(&schema.value)),
319        _ => return Ok(Expression::Unknown),
320    };
321    let args = match &f.args {
322        FunctionArguments::List(l) if l.duplicate_treatment.is_none() && l.clauses.is_empty() => l
323            .args
324            .iter()
325            .map(|arg| match arg {
326                FunctionArg::Named { arg, .. }
327                | FunctionArg::ExprNamed { arg, .. }
328                | FunctionArg::Unnamed(arg) => match arg {
329                    FunctionArgExpr::Expr(expr) => infer_expr(expr, query_source_lookup, context),
330                    FunctionArgExpr::QualifiedWildcard(object_name) => {
331                        if let Some(item) = object_name
332                            .0
333                            .last()
334                            .and_then(|a| a.as_ident())
335                            .map(|a| a.value.as_str())
336                            .and_then(|k| query_source_lookup.get(&Some(k)))
337                        {
338                            let is_left_joined = item.contains_left_join(query_source_lookup)?;
339                            Ok(Expression::Wildcard {
340                                is_left_joined,
341                                relation: item.name.map(|s| s.to_owned()),
342                                source_id: item.source_id,
343                                schema: item.schema.map(|t| t.to_owned()),
344                            })
345                        } else {
346                            Ok(Expression::Unknown)
347                        }
348                    }
349                    FunctionArgExpr::Wildcard if query_source_lookup.len() == 1 => {
350                        let query_source_lookup = query_source_lookup
351                            .values()
352                            .next()
353                            .expect("We have exactly one element");
354                        Ok(Expression::Wildcard {
355                            is_left_joined: false,
356                            relation: query_source_lookup.name.map(|s| s.to_owned()),
357                            schema: query_source_lookup.schema.map(|t| t.to_owned()),
358                            source_id: query_source_lookup.source_id,
359                        })
360                    }
361                    FunctionArgExpr::WildcardWithOptions(_) | FunctionArgExpr::Wildcard => {
362                        Ok(Expression::Unknown)
363                    }
364                },
365            })
366            .collect::<Result<Vec<_>, _>>()?,
367        _ => return Ok(Expression::Unknown),
368    };
369    Ok(Expression::Function {
370        name: name.clone(),
371        schema: schema.cloned(),
372        arguments: args,
373    })
