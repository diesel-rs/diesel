9/// A generic interface that allows this crate
10/// to request more information about certain database
11/// relations
12pub trait SchemaResolver {
13    /// Resolve a specific database field
14    fn resolve_field<'s>(
15        &'s mut self,
16        relation_schema: Option<&str>,
17        query_relation: Option<&str>,
18        field_name: &str,
19    ) -> Result<&'s dyn SchemaField, Box<dyn std::error::Error + Send + Sync + 'static>>;
20
21    /// Get a list of fields for a specific database relation
22    /// in the order returned by the database
23    fn list_fields<'s>(
24        &'s mut self,
25        relation_schema: Option<&str>,
26        query_relation: Option<&str>,
27    ) -> Result<Vec<&'s dyn SchemaField>, Box<dyn std::error::Error + Send + Sync + 'static>>;
28}
29
30/// A generic representation of a database field
31pub trait SchemaField {
32    /// Is this field nullable
33    fn is_nullable(&self) -> IsNull;
34    /// The database side name of the field
35    fn name(&self) -> Option<&str>;
36}
37
38pub(crate) struct CombinedResolver<'b> {
39    subqueries: Vec<Vec<ResolvedField>>,
40    fallback: &'b mut dyn SchemaResolver,
41}
42
43impl<'b> CombinedResolver<'b> {
44    pub(crate) fn empty(capacity: usize, fallback: &'b mut dyn SchemaResolver) -> Self {
45        Self {
46            fallback,
47            subqueries: Vec::with_capacity(capacity),
48        }
49    }
50
51    pub(crate) fn new(
52        subqueries: &[SubQuery],
53        fallback: &'b mut dyn SchemaResolver,
54    ) -> Result<Self> {
55        let mut resolver = Self::empty(subqueries.len(), fallback);
56        for subquery in subqueries {
57            resolver.insert(&subquery.fields)?;
58        }
59        Ok(resolver)
60    }
61
62    pub(crate) fn insert(&mut self, fields: &[SelectField]) -> Result<()> {
63        let fields = fields
64            .iter()
65            .map(|field| {
66                Ok(ResolvedField {
67                    is_null: field.infer_nullability(self)?,
68                    ident: field.ident.clone(),
69                })
70            })
71            .collect::<Result<Vec<_>>>()?;
72        self.subqueries.push(fields);
73        Ok(())
74    }
75
76    pub(crate) fn resolve_derived_field(
77        &self,
78        source_id: usize,
79        relation: Option<&str>,
80        field_name: &str,
81    ) -> Result<&dyn SchemaField> {
82        self.subqueries
83            .get(source_id)
84            .and_then(|fields| {
85                fields
86                    .iter()
87                    .find(|field| field.ident.as_deref() == Some(field_name))
88            })
89            .map(|field| field as &dyn SchemaField)
90            .ok_or_else(|| Error::UnknownField {
91                relation_schema: None,
92                query_relation: relation.map(str::to_owned),
93                field_name: field_name.to_owned(),
94            })
95    }
96
97    pub(crate) fn list_derived_fields(&self, source_id: usize) -> Result<Vec<&dyn SchemaField>> {
98        self.subqueries
99            .get(source_id)
100            .map(|fields| {
101                fields
102                    .iter()
103                    .map(|field| field as &dyn SchemaField)
104                    .collect()
105            })
106            .ok_or_else(|| Error::UnsupportedSql {
107                msg: "Unresolved subquery".to_owned(),
108            })
109    }
110}
111
112impl SchemaResolver for CombinedResolver<'_> {
113    fn resolve_field<'s>(
114        &'s mut self,
115        relation_schema: Option<&str>,
116        query_relation: Option<&str>,
117        field_name: &str,
118    ) -> Result<&'s dyn SchemaField, Box<dyn std::error::Error + Send + Sync + 'static>> {
119        self.fallback
120            .resolve_field(relation_schema, query_relation, field_name)
121    }
122
123    fn list_fields<'s>(
124        &'s mut self,
125        relation_schema: Option<&str>,
126        query_relation: Option<&str>,
127    ) -> Result<Vec<&'s dyn SchemaField>, Box<dyn std::error::Error + Send + Sync + 'static>> {
128        self.fallback.list_fields(relation_schema, query_relation)
129    }
130}
