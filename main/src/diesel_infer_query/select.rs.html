36pub enum Expression {
37    /// A literal value like `1` or `'foo'`
38    Literal {
39        /// The actual value
40        value: Option<String>,
41        /// is this value `NULL`?
42        is_null: bool,
43    },
44    /// A field of a query source
45    Field {
46        /// the schema of the query source
47        schema: Option<String>,
48        /// the name of the query source
49        query_source: Option<String>,
50        /// the derived table or CTE this field refers to, `None` for a database relation
51        source_id: Option<usize>,
52        /// the name of the field
53        field_name: String,
54        /// is this field coming from a query source joined via a `LEFT JOIN`
55        via_left_join: bool,
56    },
57    /// A cast expression
58    Cast {
59        /// inner expression
60        inner: Box<Expression>,
61        /// target cast type
62        tpe: String,
63    },
64    /// A binary operation
65    BinaryOp {
66        /// left side of the operation
67        left: Box<Expression>,
68        /// right side of the operation
69        right: Box<Expression>,
70        /// operator
71        op: String,
72        /// is this operation statically known not to produce null values
73        statically_not_null: bool,
74    },
75    // A postfix operation like `IS NULL`
76    PostfixOp {
77        expr: Box<Expression>,
78        op: String,
79        statically_not_null: bool,
80    },
81    /// A function call
82    Function {
83        name: String,
84        schema: Option<String>,
85        arguments: Vec<Expression>,
86    },
87    /// A wild card `*` expression
88    Wildcard {
89        schema: Option<String>,
90        relation: Option<String>,
91        is_left_joined: bool,
92        source_id: Option<usize>,
93    },
94    /// A `left BETWEEN low AND high` expression
95    Between {
96        left: Box<Expression>,
97        negated: bool,
98        low: Box<Expression>,
99        high: Box<Expression>,
100    },
101    // A `CASE [operand] [conditions] ELSE [else]` expression
102    Case {
103        operand: Option<Box<Expression>>,
104        conditions: Vec<CaseCondition>,
105        else_clause: Option<Box<Expression>>,
106    },
107    /// A `[left] (NOT) IN ([LIST]) expression
108    In {
109        left: Box<Expression>,
110        negated: bool,
111        list: Vec<Expression>,
112    },
113    /// A `[left] (NOT) IN ([QUERY]) expression
114    InSubQuery {
115        left: Box<Expression>,
116        negated: bool,
117        subquery: Vec<SelectField>,
118    },
119    /// An expression grouping by `(inner)`
120    Grouped(Box<Expression>),
121    /// SubQuery,
122    Subquery { selection: Vec<SelectField> },
123    /// A field based on a combined set
124    Combined {
125        left: Box<Expression>,
126        right: Box<Expression>,
127        set_operator: SetOperator,
128    },
129    /// A unknown select expression
130    Unknown,
131}

153impl Expression {
154    pub(crate) fn infer_nullability(&self, resolver: &mut CombinedResolver<'_>) -> Result<IsNull> {
155        match self {
156            Self::Literal { is_null, .. } => Ok(if *is_null {
157                IsNull::IsNullable
158            } else {
159                IsNull::NotNullable
160            }),
161            Self::Field {
162                via_left_join: true,
163                ..
164            } => Ok(IsNull::IsNullable),
165            Self::Field {
166                schema,
167                query_source: table,
168                source_id,
169                field_name,
170                ..
171            } => {
172                let field = match source_id {
173                    Some(id) => {
174                        resolver.resolve_derived_field(*id, table.as_deref(), field_name)?
175                    }
176                    None => resolver
177                        .resolve_field(schema.as_deref(), table.as_deref(), field_name)
178                        .map_err(|inner| Error::ResolverFailure { inner })?,
179                };
180                Ok(field.is_nullable())
181            }
182            Self::Cast { inner, .. } => inner.infer_nullability(resolver),
183            Self::BinaryOp {
184                left,
185                right,
186                statically_not_null,
187                ..
188            } => {
189                if *statically_not_null {
190                    Ok(IsNull::NotNullable)
191                } else {
192                    Ok(left
193                        .infer_nullability(resolver)?
194                        .or(right.infer_nullability(resolver)?))
195                }
196            }
197            Expression::PostfixOp {
198                expr,
199                statically_not_null,
200                ..
201            } => {
202                if *statically_not_null {
203                    Ok(IsNull::NotNullable)
204                } else {
205                    expr.infer_nullability(resolver)
206                }
207            }
208            Expression::Function {
209                name,
210                schema,
211                arguments,
212            } => {
213                match name.to_lowercase().as_str() {
214                    // we consider count as only not nullable function for now
215                    "count" if schema.is_none() => Ok(IsNull::NotNullable),
216                    // coalesce is not nullable if any argument cannot contain
217                    // null values as it returns the first non-null value
218                    "coalesce" if schema.is_none() => {
219                        for arg in arguments {
220                            match arg.infer_nullability(resolver)? {
221                                IsNull::NotNullable => return Ok(IsNull::NotNullable),
222                                IsNull::IsNullable | IsNull::Unknown => {}
223                            }
224                        }
225                        Ok(IsNull::IsNullable)
226                    }
227                    _ => Ok(IsNull::IsNullable),
228                }
229            }
230            Expression::Between {
231                left, low, high, ..
232            } => {
233                let nullability = [
234                    left.infer_nullability(resolver)?,
235                    low.infer_nullability(resolver)?,
236                    high.infer_nullability(resolver)?,
237                ];
238                Ok(nullability[0].or(nullability[1]).or(nullability[2]))
239            }
240            Expression::Case {
241                conditions,
242                else_clause,
243                ..
244            } => conditions
245                .iter()
246                .map(|c| &c.result)
247                .chain(else_clause.iter().map(|c| &**c))
248                .map(|c| c.infer_nullability(resolver))
249                .try_fold(IsNull::NotNullable, |agg, v| Ok(agg.or(v?))),
250            Expression::In { left, list, .. } => {
251                let left = left.infer_nullability(resolver);
252                list.iter()
253                    .map(|l| l.infer_nullability(resolver))
254                    .chain([left])
255                    .try_fold(IsNull::NotNullable, |agg, v| Ok(agg.or(v?)))
256            }
257            Expression::Grouped(inner) => inner.infer_nullability(resolver),
258            Expression::Subquery { selection } => match selection.as_slice() {
259                [_s] => {
260                    // always nullable as the subquery might return an empty result
261                    Ok(IsNull::IsNullable)
262                }
263                _ => Ok(IsNull::Unknown),
264            },
265            Expression::InSubQuery { left, subquery, .. } => {
266                match subquery.as_slice() {
267                    [s] => {
268                        // `x IN (SELECT y …)` yields NULL when x matches no row and the
269                        // subquery contains a NULL, so the result is nullable whenever
270                        // either side is nullable
271                        Ok(left
272                            .infer_nullability(resolver)?
273                            .or(s.infer_nullability(resolver)?))
274                    }
275                    // A subquery should return only one column
276                    // if that's not the case, give up
277                    _ => Ok(IsNull::Unknown),
278                }
279            }
280            // for except set operations only the left
281            // side is actually returned
282            Expression::Combined {
283                left,
284                set_operator: SetOperator::Except,
285                ..
286            } => left.infer_nullability(resolver),
287            // for `UNION` set operations both sides are
288            // returned, so if any side can contain
289            // null values we need to assume this for the
290            // whole expression
291            Expression::Combined {
292                left,
293                right,
294                set_operator: SetOperator::Union,
295            } => {
296                let left = left.infer_nullability(resolver)?;
297                let right = right.infer_nullability(resolver)?;
298                Ok(left.or(right))
299            }
300            // for `INTERSECT` we only return values
301            // that are in both result sets, so
302            // if any side is not nullable we only return
303            // not nullable values for this expression
304            Expression::Combined {
305                left,
306                right,
307                set_operator: SetOperator::Intersect,
308            } => {
309                let left = left.infer_nullability(resolver)?;
310                let right = right.infer_nullability(resolver)?;
311                let ret = match (left, right) {
312                    (IsNull::Unknown, _) | (_, IsNull::Unknown) => IsNull::Unknown,
313                    (IsNull::NotNullable, _) | (_, IsNull::NotNullable) => IsNull::NotNullable,
314                    (IsNull::IsNullable, IsNull::IsNullable) => IsNull::IsNullable,
315                };
316                Ok(ret)
317            }
318            Expression::Wildcard { .. } => Err(Error::UnresolvedWildcard),
319            Expression::Unknown => Ok(IsNull::Unknown),
320        }
321    }
322}

324pub(crate) fn infer_from_select(
325    select: &sqlparser::ast::Select,
326    outer_lookup: Option<&HashMap<Option<&str>, QuerySource>>,
327    context: &mut ParseContext,
328) -> Result<Vec<SelectField>> {
329    let mut query_source_lookup = collect_query_sources(&select.from)?;
330    for source in &select.from {
331        register_derived(&source.relation, &mut query_source_lookup, context)?;
332        for join in &source.joins {
333            register_derived(&join.relation, &mut query_source_lookup, context)?;
334        }
335    }
336    for source in query_source_lookup.values_mut() {
337        if source.source_id.is_none() && source.schema.is_none() {
338            source.source_id = source.name.and_then(|name| context.cte(name));
339        }
340    }
341    if let Some(outer) = outer_lookup {
342        for (k, v) in outer {
343            query_source_lookup.entry(*k).or_insert_with(|| v.clone());
344        }
345    }
346    select
347        .projection
348        .iter()
349        .map(|p| infer_projection(p, &query_source_lookup, context))
350        .collect()
351}

353fn register_derived<'a>(
354    source: &'a sqlparser::ast::TableFactor,
355    lookup: &mut HashMap<Option<&'a str>, QuerySource<'a>>,
356    context: &mut ParseContext,
357) -> Result<()> {
358    match source {
359        sqlparser::ast::TableFactor::Derived {
360            subquery, alias, ..
361        } => {
362            let fields = parse_query(subquery, None, context)?;
363            let id = context.register(fields);
364            let key = alias.as_ref().map(|alias| alias.name.value.as_str());
365            if let Some(source) = lookup.get_mut(&key) {
366                source.source_id = Some(id);
367            }
368        }
369        sqlparser::ast::TableFactor::NestedJoin {
370            table_with_joins, ..
371        } => {
372            register_derived(&table_with_joins.relation, lookup, context)?;
373            for join in &table_with_joins.joins {
374                register_derived(&join.relation, lookup, context)?;
375            }
376        }
377        _ => {}
378    }
379    Ok(())
380}

384) -> Result<HashMap<Option<&'a str>, QuerySource<'a>>> {
385    let mut out = HashMap::with_capacity(sources.len());
386    for s in sources {
387        QuerySource::fill_from_table_with_joins(&mut out, s)?;
388    }
389    Ok(out)
390}

392pub(crate) fn infer_projection(
393    item: &SelectItem,
394    query_source_lookup: &HashMap<Option<&str>, QuerySource>,
395    context: &mut ParseContext,
396) -> Result<SelectField> {
397    match item {
398        SelectItem::UnnamedExpr(expr) => {
399            let kind = crate::expression::infer_expr(expr, query_source_lookup, context)?;
400            let ident = if let Expression::Field { field_name, .. } = &kind {
401                Some(field_name.clone())
402            } else {
403                None
404            };
405            Ok(SelectField { ident, kind })
406        }
407        SelectItem::ExprWithAlias { expr, alias } => Ok(SelectField {
408            ident: Some(alias.value.clone()),
409            kind: crate::expression::infer_expr(expr, query_source_lookup, context)?,
410        }),
411        SelectItem::QualifiedWildcard(
412            SelectItemQualifiedWildcardKind::ObjectName(name),
413            _wildcard_additional_options,
414        ) => {
415            if let Some(item) = name
416                .0
417                .last()
418                .and_then(|a| a.as_ident())
419                .map(|a| a.value.as_str())
420                .and_then(|k| query_source_lookup.get(&Some(k)))
421            {
422                let is_left_joined = item.contains_left_join(query_source_lookup)?;
423                Ok(SelectField {
424                    ident: None,
425                    kind: Expression::Wildcard {
426                        schema: item.schema.map(|s| s.to_owned()),
427                        relation: item.name.map(|c| c.to_owned()),
428                        is_left_joined,
429                        source_id: item.source_id,
430                    },
431                })
432            } else {
433                ::core::panicking::todo_default_msg()todo!()
434            }
435        }
436        SelectItem::Wildcard(_) if query_source_lookup.len() == 1 => {
437            let wildcard = query_source_lookup.values().next().expect("Is exactly one");
438            Ok(SelectField {
439                ident: None,
440                kind: Expression::Wildcard {
441                    schema: wildcard.schema.map(|c| c.to_owned()),
442                    relation: wildcard.name.map(|c| c.to_owned()),
443                    is_left_joined: false,
444                    source_id: wildcard.source_id,
445                },
446            })
447        }
448        s @ SelectItem::Wildcard(_wildcard_additional_options) => Err(Error::UnsupportedSql {
449            msg: ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Unsupported `SELECT` expression: `{0}`",
                s))
    })format!("Unsupported `SELECT` expression: `{s}`"),
450        }),
451        s @ SelectItem::QualifiedWildcard(
452            _select_item_qualified_wildcard_kind,
453            _wildcard_additional_options,
454        ) => Err(Error::UnsupportedSql {
455            msg: ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Unsupported `SELECT` expression: `{0}`",
                s))
    })format!("Unsupported `SELECT` expression: `{s}`"),
456        }),
457        s @ SelectItem::ExprWithAliases { .. } => Err(Error::UnsupportedSql {
458            msg: ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Unsupported `SELECT` expression: `{0}`",
                s))
    })format!("Unsupported `SELECT` expression: `{s}`"),
459        }),
460    }
461}

463pub(crate) fn parse_query(
464    select: &sqlparser::ast::Query,
465    outer_lookup: Option<&HashMap<Option<&str>, QuerySource>>,
466    context: &mut ParseContext,
467) -> Result<Vec<SelectField>> {
468    let outer_ctes = context.ctes.len();
469    if let Some(with) = &select.with {
470        for cte in &with.cte_tables {
471            let mut fields = parse_query(&cte.query, None, context)?;
472            if !cte.alias.columns.is_empty() {
473                if fields.len() != cte.alias.columns.len() {
474                    return Err(Error::UnsupportedSql {
475                        msg: ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Not matching field count for a CTE. Got {0} fields, but expected {1} fields",
                fields.len(), cte.alias.columns.len()))
    })format!(
476                            "Not matching field count for a CTE. Got {} fields, but expected {} fields",
477                            fields.len(),
478                            cte.alias.columns.len()
479                        ),
480                    });
481                }
482                for (field, alias) in fields.iter_mut().zip(&cte.alias.columns) {
483                    field.ident = Some(alias.name.value.clone());
484                }
485            }
486            let id = context.register(fields);
487            context.ctes.push((cte.alias.name.value.clone(), id));
488        }
489    }
490    let result = parse_from_set_expr(outer_lookup, &select.body, context);
491    context.ctes.truncate(outer_ctes);
492    result
493}
494
495fn parse_from_set_expr(
496    outer_lookup: Option<&HashMap<Option<&str>, QuerySource<'_>>>,
497    expr: &sqlparser::ast::SetExpr,
498    context: &mut ParseContext,
499) -> Result<Vec<SelectField>> {
500    match expr {
501        sqlparser::ast::SetExpr::Select(select_expr) => {
502            infer_from_select(select_expr, outer_lookup, context)
503        }
504        // Set expressions like `UNION`, `INTERSECT` and `EXCEPT`
505        sqlparser::ast::SetExpr::SetOperation {
506            left, op, right, ..
507        } => {
508            let op = SetOperator::from_sql_parser_ast(op)?;
509            let left = parse_from_set_expr(outer_lookup, left, context)?;
510            let right = parse_from_set_expr(outer_lookup, right, context)?;
511            if left.len() == right.len() {
512                Ok(left
513                    .into_iter()
514                    .zip(right)
515                    .map(|(l, r)| SelectField {
516                        ident: l.ident,
517                        kind: Expression::Combined {
518                            left: Box::new(l.kind),
519                            right: Box::new(r.kind),
520                            set_operator: op,
521                        },
522                    })
523                    .collect())
524            } else {
525                Err(Error::UnsupportedSql {
526                    msg: ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("We expect set operations to have the same number of fields on both sides. We got {0} fields on the left side and {1} fields on the right side",
                left.len(), right.len()))
    })format!(
527                        "We expect set operations to have the same number of fields on both sides. \
528                         We got {} fields on the left side and {} fields on the right side",
529                        left.len(),
530                        right.len()
531                    ),
532                })
533            }
534        }
535        s => Err(Error::UnsupportedSql {
536            msg: ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Unsupported query kind: `{0}`", s))
    })format!("Unsupported query kind: `{s}`"),
537        }),
538    }
539}
