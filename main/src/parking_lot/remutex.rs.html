5/// Implementation of the `GetThreadId` trait for `lock_api::ReentrantMutex`.
6pub struct RawThreadId;

8unsafe impl GetThreadId for RawThreadId {
9    const INIT: RawThreadId = RawThreadId;
10
11    fn nonzero_thread_id(&self) -> NonZeroUsize {
12        // The address of a thread-local variable is guaranteed to be unique to the
13        // current thread, and is also guaranteed to be non-zero. The variable has to have a
14        // non-zero size to guarantee it has a unique address for each thread.
15        const KEY: ::std::thread::LocalKey<u8> =
    {
        const __RUST_STD_INTERNAL_INIT: u8 = { 0 };
        unsafe {
            ::std::thread::LocalKey::new(const {
                        if ::std::mem::needs_drop::<u8>() {
                            |__rust_std_internal_init|
                                {
                                    #[thread_local]
                                    static __RUST_STD_INTERNAL_VAL:
                                        ::std::thread::local_impl::EagerStorage<u8> =
                                        ::std::thread::local_impl::EagerStorage::new(__RUST_STD_INTERNAL_INIT);
                                    __RUST_STD_INTERNAL_VAL.get()
                                }
                        } else {
                            |__rust_std_internal_init|
                                {
                                    #[thread_local]
                                    static __RUST_STD_INTERNAL_VAL: u8 =
                                        __RUST_STD_INTERNAL_INIT;
                                    &__RUST_STD_INTERNAL_VAL
                                }
                        }
                    })
        }
    };thread_local!(static KEY: u8 = const { 0 });
16        KEY.with(|x| {
17            NonZeroUsize::new(core::ptr::from_ref(x).addr())
18                .expect("thread-local variable address is null")
19        })
20    }
21}

23/// A reentrant mutual exclusion lock.
24///
25/// This lock blocks other threads waiting for it to become available. A thread
26/// which already holds the lock can acquire it additional times without
27/// blocking.
28///
29/// Unlike [`Mutex`](crate::Mutex), [`ReentrantMutexGuard`] does not provide
30/// mutable references to the locked data, because multiple guards can coexist
31/// on the same thread. Use interior mutability, such as [`RefCell`](core::cell::RefCell),
32/// to mutate the guarded data.
33///
34/// See [`Mutex`](crate::Mutex) for more details about the underlying mutex
35/// primitive.
36///
37/// # Examples
38///
39/// ```
40/// use parking_lot::ReentrantMutex;
41/// use std::cell::RefCell;
42///
43/// let lock = ReentrantMutex::new(RefCell::new(0));
44/// let first = lock.lock();
45/// let second = lock.lock();
46/// *first.borrow_mut() += 1;
47/// *second.borrow_mut() += 1;
48/// assert_eq!(*lock.lock().borrow(), 2);
49/// ```
50pub type ReentrantMutex<T> = lock_api::ReentrantMutex<RawMutex, RawThreadId, T>;
51
52/// An RAII guard which releases one level of recursive locking when dropped.
53///
54/// The data protected by the mutex can be accessed through this guard via its
55/// [`Deref`](core::ops::Deref) implementation.
56pub type ReentrantMutexGuard<'a, T> = lock_api::ReentrantMutexGuard<'a, RawMutex, RawThreadId, T>;
57
58/// An RAII mutex guard returned by [`ReentrantMutexGuard::map`], which can point to a
59/// subfield of the protected data.
60///
61/// The main difference between `MappedReentrantMutexGuard` and `ReentrantMutexGuard` is that the
62/// former doesn't support temporarily unlocking and re-locking, since that
63/// could introduce soundness issues if the locked object is modified by another
64/// thread.
65pub type MappedReentrantMutexGuard<'a, T> =
66    lock_api::MappedReentrantMutexGuard<'a, RawMutex, RawThreadId, T>;

99        let m = Arc::new(ReentrantMutex::new(RefCell::new(0)));
100        let m2 = m.clone();
101        let lock = m.lock();
102        let child = thread::spawn(move || {
103            let lock = m2.lock();
104            assert_eq!(*lock.borrow(), 4950);
105        });
106        for i in 0..100 {
107            let lock = m.lock();
108            *lock.borrow_mut() += i;
109        }
