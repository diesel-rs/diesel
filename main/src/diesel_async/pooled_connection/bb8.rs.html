60use super::{AsyncDieselConnectionManager, PoolError, PoolableConnection};
61use bb8::ManageConnection;
62use diesel::query_builder::QueryFragment;

64/// Type alias for using [`bb8::Pool`] with [`diesel-async`]
65///
66/// This is **not** equal to [`bb8::Pool`]. It already uses the correct
67/// connection manager and expects only the connection type as generic argument
68pub type Pool<C> = bb8::Pool<AsyncDieselConnectionManager<C>>;
69/// Type alias for using [`bb8::PooledConnection`] with [`diesel-async`]
70pub type PooledConnection<'a, C> = bb8::PooledConnection<'a, AsyncDieselConnectionManager<C>>;
71/// Type alias for using [`bb8::RunError`] with [`diesel-async`]
72pub type RunError = bb8::RunError<super::PoolError>;
73
74impl<C> ManageConnection for AsyncDieselConnectionManager<C>
75where
76    C: PoolableConnection + 'static,
77    for<'a> diesel::dsl::select<diesel::dsl::AsExprOf<i32, diesel::sql_types::Integer>>:
78        crate::methods::LoadQuery<'a, C, i32>,
79    diesel::query_builder::SqlQuery: QueryFragment<C::Backend>,
80{
81    type Connection = C;
82
83    type Error = PoolError;
84
85    async fn connect(&self) -> Result<Self::Connection, Self::Error> {
86        (self.manager_config.custom_setup)(&self.connection_url)
87            .await
88            .map_err(PoolError::ConnectionError)
89    }
90
91    async fn is_valid(&self, conn: &mut Self::Connection) -> Result<(), Self::Error> {
92        conn.ping(&self.manager_config.recycling_method)
93            .await
94            .map_err(PoolError::QueryError)
95    }
96
97    fn has_broken(&self, conn: &mut Self::Connection) -> bool {
98        std::thread::panicking() || conn.is_broken()
99    }
