22pub(crate) struct Wheel {
23    /// The number of milliseconds elapsed since the wheel started.
24    elapsed: u64,
25
26    /// Timer wheel.
27    ///
28    /// Levels:
29    ///
30    /// * 1 ms slots / 64 ms range
31    /// * 64 ms slots / ~ 4 sec range
32    /// * ~ 4 sec slots / ~ 4 min range
33    /// * ~ 4 min slots / ~ 4 hr range
34    /// * ~ 4 hr slots / ~ 12 day range
35    /// * ~ 12 day slots / ~ 2 yr range
36    levels: Box<[Level; NUM_LEVELS]>,
37
38    /// Entries queued for firing
39    pending: LinkedList<TimerShared>,
40}
41
42/// Number of levels. Each level has 64 slots. By using 6 levels with 64 slots
43/// each, the timer is able to track time up to 2 years into the future with a
44/// precision of 1 millisecond.
45const NUM_LEVELS: usize = 6;
46
47const BITS_PER_LEVEL: usize = 6;
48
49/// The maximum duration of a `Sleep`.
50const MAX_DURATION: u64 = 1 << (BITS_PER_LEVEL * NUM_LEVELS);

52impl Wheel {

54    pub(crate) fn new() -> Wheel {
55        let levels = (0..NUM_LEVELS).map(Level::new).collect::<Box<_>>();
56        Wheel {
57            elapsed: 0,
58            levels: levels.try_into().unwrap(),
59            pending: LinkedList::new(),
60        }
61    }

69    /// Inserts an entry into the timing wheel.
70    ///
71    /// # Arguments
72    ///
73    /// * `item`: The item to insert into the wheel.
74    ///
75    /// # Return
76    ///
77    /// Returns `Ok` when the item is successfully inserted, `Err` otherwise.
78    ///
79    /// `Err(Elapsed)` indicates that `when` represents an instant that has
80    /// already passed. In this case, the caller should fire the timeout
81    /// immediately.
82    ///
83    /// # Safety
84    ///
85    /// This function registers item into an intrusive linked list. The caller
86    /// must ensure that `item` is pinned and will not be dropped without first
87    /// being deregistered.
88    pub(crate) unsafe fn insert(
89        &mut self,
90        item: TimerHandle,
91    ) -> Result<u64, (TimerHandle, InsertError)> {
92        let when = unsafe { item.sync_when() };
93
94        if when <= self.elapsed {
95            return Err((item, InsertError::Elapsed));
96        }
97
98        // Get the level at which the entry should be stored
99        let level = self.level_for(when);
100
101        unsafe {
102            self.levels[level].add_entry(item);
103        }
104
105        if true {
    if !{
                self.levels[level].next_expiration(self.elapsed).is_none_or(|e|
                        e.deadline >= self.elapsed)
            } {
        ::core::panicking::panic("assertion failed: {\n    self.levels[level].next_expiration(self.elapsed).is_none_or(|e|\n            e.deadline >= self.elapsed)\n}")
    };
};debug_assert!({
106            self.levels[level]
107                .next_expiration(self.elapsed)
108                .is_none_or(|e| e.deadline >= self.elapsed)
109        });
110
111        Ok(when)
112    }

114    /// Removes `item` from the timing wheel.
115    pub(crate) unsafe fn remove(&mut self, item: NonNull<TimerShared>) {
116        unsafe {
117            let when = item.as_ref().registered_when();
118            if when == STATE_DEREGISTERED {
119                self.pending.remove(item);
120            } else {
121                if true {
    if !(self.elapsed <= when) {
        {
            ::core::panicking::panic_fmt(format_args!("elapsed={0}; when={1}",
                    self.elapsed, when));
        }
    };
};debug_assert!(
122                    self.elapsed <= when,
123                    "elapsed={}; when={}",
124                    self.elapsed,
125                    when
126                );
127
128                let level = self.level_for(when);
129                self.levels[level].remove_entry(item);
130            }
131        }
132    }

134    /// Instant at which to poll.
135    pub(crate) fn poll_at(&self) -> Option<u64> {
136        self.next_expiration().map(|expiration| expiration.deadline)
137    }

139    /// Advances the timer up to the instant represented by `now`.
140    pub(crate) fn poll(&mut self, now: u64) -> Option<TimerHandle> {
141        loop {
142            if let Some(handle) = self.pending.pop_back() {
143                return Some(handle);
144            }
145
146            match self.next_expiration() {
147                Some(ref expiration) if expiration.deadline <= now => {
148                    self.process_expiration(expiration, now);
149
150                    // During shutdown (`now == u64::MAX`), slots are drained
151                    // directly without cascading entries across levels. Keep
152                    // `self.elapsed` unchanged until all slots are empty so
153                    // `level_for` remains valid if the lock is temporarily
154                    // dropped to wake a batch of wakers.
155                    if now != u64::MAX {
156                        self.set_elapsed(expiration.deadline);
157                    }
158                }
159                _ => {
160                    // in this case the poll did not indicate an expiration
161                    // _and_ we were not able to find a next expiration in
162                    // the current list of timers.  advance to the poll's
163                    // current time and do nothing else.
164                    self.set_elapsed(now);
165                    break;
166                }
167            }
168        }
169
170        self.pending.pop_back()
171    }

173    /// Returns the instant at which the next timeout expires.
174    fn next_expiration(&self) -> Option<Expiration> {
175        if !self.pending.is_empty() {
176            // Expire immediately as we have things pending firing
177            return Some(Expiration {
178                level: 0,
179                slot: 0,
180                deadline: self.elapsed,
181            });
182        }
183
184        // Check all levels
185        for (level_num, level) in self.levels.iter().enumerate() {
186            if let Some(expiration) = level.next_expiration(self.elapsed) {
187                // There cannot be any expirations at a higher level that happen
188                // before this one.
189                if true {
    if !self.no_expirations_before(level_num + 1, expiration.deadline) {
        ::core::panicking::panic("assertion failed: self.no_expirations_before(level_num + 1, expiration.deadline)")
    };
};debug_assert!(self.no_expirations_before(level_num + 1, expiration.deadline));
190
191                return Some(expiration);
192            }
193        }
194
195        None
196    }

198    /// Returns the tick at which this timer wheel next needs to perform some
199    /// processing, or None if there are no timers registered.
200    pub(super) fn next_expiration_time(&self) -> Option<u64> {
201        self.next_expiration().map(|ex| ex.deadline)
202    }

204    /// Used for debug assertions
205    fn no_expirations_before(&self, start_level: usize, before: u64) -> bool {
206        let mut res = true;
207
208        for level in &self.levels[start_level..] {
209            if let Some(e2) = level.next_expiration(self.elapsed) {
210                if e2.deadline < before {
211                    res = false;
212                }
213            }
214        }
215
216        res
217    }

219    /// iteratively find entries that are between the wheel's current
220    /// time and the expiration time.  for each in that population either
221    /// queue it for notification (in the case of the last level) or tier
222    /// it down to the next level (in all other cases).
223    pub(crate) fn process_expiration(&mut self, expiration: &Expiration, now: u64) {
224        // Note that we need to take _all_ of the entries off the list before
225        // processing any of them. This is important because it's possible that
226        // those entries might need to be reinserted into the same slot.
227        //
228        // This happens only on the highest level, when an entry is inserted
229        // more than MAX_DURATION into the future. When this happens, we wrap
230        // around, and process some entries a multiple of MAX_DURATION before
231        // they actually need to be dropped down a level. We then reinsert them
232        // back into the same position; we must make sure we don't then process
233        // those entries again or we'll end up in an infinite loop.
234        let mut entries = self.take_entries(expiration);
235        let deadline = if now == u64::MAX {
236            now
237        } else {
238            expiration.deadline
239        };
240
241        while let Some(item) = entries.pop_back() {
242            if expiration.level == 0 && now != u64::MAX {
243                if true {
    {
        match (&unsafe { item.registered_when() }, &expiration.deadline) {
            (left_val, right_val) => {
                if !(*left_val == *right_val) {
                    let kind = ::core::panicking::AssertKind::Eq;
                    ::core::panicking::assert_failed(kind, &*left_val,
                        &*right_val, ::core::option::Option::None);
                }
            }
        }
    };
};debug_assert_eq!(unsafe { item.registered_when() }, expiration.deadline);
244            }
245
246            // Try to expire the entry; this is cheap (doesn't synchronize) if
247            // the timer is not expired, and updates registered_when.
248            match unsafe { item.mark_pending(deadline) } {
249                Ok(()) => {
250                    // Item was expired
251                    self.pending.push_front(item);
252                }
253                Err(expiration_tick) => {
254                    let level = level_for(expiration.deadline, expiration_tick);
255                    unsafe {
256                        self.levels[level].add_entry(item);
257                    }
258                }
259            }
260        }
261    }

263    fn set_elapsed(&mut self, when: u64) {
264        if !(self.elapsed <= when) {
    {
        ::core::panicking::panic_fmt(format_args!("elapsed={0:?}; when={1:?}",
                self.elapsed, when));
    }
};assert!(
265            self.elapsed <= when,
266            "elapsed={:?}; when={:?}",
267            self.elapsed,
268            when
269        );
270
271        if when > self.elapsed {
272            self.elapsed = when;
273        }
274    }

276    /// Obtains the list of entries that need processing for the given expiration.
277    fn take_entries(&mut self, expiration: &Expiration) -> LinkedList<TimerShared> {
278        self.levels[expiration.level].take_slot(expiration.slot)
279    }

281    fn level_for(&self, when: u64) -> usize {
282        level_for(self.elapsed, when)
283    }

286fn level_for(elapsed: u64, when: u64) -> usize {
287    const SLOT_MASK: u64 = (1 << BITS_PER_LEVEL) - 1;
288
289    // Mask in the trailing bits ignored by the level calculation in order to cap
290    // the possible leading zeros
291    let masked = elapsed ^ when | SLOT_MASK;
292
293    if masked >= MAX_DURATION {
294        // Fudge the timer into the top level
295        return NUM_LEVELS - 1;
296    }
297
298    masked.ilog2() as usize / BITS_PER_LEVEL
299}
