49/// Work-stealing based thread pool for executing futures.
50pub(crate) struct MultiThread;
51
52// ===== impl MultiThread =====
53
54impl MultiThread {
55    #[allow(clippy::too_many_arguments)]
56    pub(crate) fn new(
57        size: usize,
58        driver: Driver,
59        driver_handle: driver::Handle,
60        blocking_spawner: blocking::Spawner,
61        seed_generator: RngSeedGenerator,
62        config: Config,
63        timer_flavor: TimerFlavor,
64        name: Option<String>,
65    ) -> (MultiThread, Arc<Handle>, Launch) {
66        let parker = Parker::new(driver);
67        let (handle, launch) = worker::create(
68            size,
69            parker,
70            driver_handle,
71            blocking_spawner,
72            seed_generator,
73            config,
74            timer_flavor,
75            name,
76        );
77
78        (MultiThread, handle, launch)
79    }
80
81    /// Blocks the current thread waiting for the future to complete.
82    ///
83    /// The future will execute on the current thread, but all spawned tasks
84    /// will be executed on the thread pool.
85    pub(crate) fn block_on<F>(&self, handle: &scheduler::Handle, future: F) -> F::Output
86    where
87        F: Future,
88    {
89        crate::runtime::context::enter_runtime(handle, true, |blocking| {
90            blocking.block_on(future).expect("failed to park thread")
91        })
92    }
93
94    pub(crate) fn shutdown(&mut self, handle: &scheduler::Handle) {
95        match handle {
96            scheduler::Handle::MultiThread(handle) => handle.shutdown(),
97            _ => {
    ::core::panicking::panic_fmt(format_args!("expected MultiThread scheduler"));
}panic!("expected MultiThread scheduler"),
98        }
99    }
100}

105    }
