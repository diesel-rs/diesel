224#[cfg(target_os = "freebsd")]
225pub(crate) mod impl_freebsd {
226    use crate::net::unix::{self, UnixStream};
227
228    use libc::{c_void, getsockopt, socklen_t, xucred, LOCAL_PEERCRED, XUCRED_VERSION};
229    use std::io;
230    use std::mem::{size_of, MaybeUninit};
231    use std::os::unix::io::AsRawFd;
232
233    pub(crate) fn get_peer_cred(sock: &UnixStream) -> io::Result<super::UCred> {
234        // `SOL_LOCAL` is not re-exported by `libc` for FreeBSD; it is defined
235        // as 0 in `<sys/un.h>`.
236        const SOL_LOCAL: libc::c_int = 0;
237
238        unsafe {
239            let raw_fd = sock.as_raw_fd();
240
241            let mut xucred = MaybeUninit::<xucred>::zeroed();
242            let mut len = size_of::<xucred>() as socklen_t;
243
244            let ret = getsockopt(
245                raw_fd,
246                SOL_LOCAL,
247                LOCAL_PEERCRED,
248                xucred.as_mut_ptr() as *mut c_void,
249                &mut len,
250            );
251
252            if ret != 0 {
253                return Err(io::Error::last_os_error());
254            }
255            if len as usize != size_of::<xucred>() {
256                return Err(io::Error::new(
257                    io::ErrorKind::InvalidData,
258                    "unexpected xucred size from LOCAL_PEERCRED",
259                ));
260            }
261
262            let xucred = xucred.assume_init();
263
264            // Match `getpeereid(3)` and reject any `xucred` whose version we
265            // don't know how to interpret.
266            if xucred.cr_version != XUCRED_VERSION {
267                return Err(io::Error::new(
268                    io::ErrorKind::InvalidData,
269                    "unexpected xucred version from LOCAL_PEERCRED",
270                ));
271            }
272
273            // `cr_pid` is populated by the kernel since FreeBSD 13. PID 0 is
274            // the kernel scheduler and never a real userland peer, so we
275            // surface it as `None` rather than a misleading `Some(0)`.
276            let pid = match xucred.cr_pid__c_anonymous_union.cr_pid {
277                0 => None,
278                p => Some(p as unix::pid_t),
279            };
280
281            // `xucred` carries the effective uid in `cr_uid` and the effective
282            // gid in `cr_groups[0]`, matching what `getpeereid(2)` returns.
283            Ok(super::UCred {
284                uid: xucred.cr_uid as unix::uid_t,
285                gid: xucred.cr_groups[0] as unix::gid_t,
286                pid,
287            })
288        }
289    }
290}

299pub(crate) mod impl_macos {
300    use crate::net::unix::{self, UnixStream};
301
302    use libc::{c_void, getpeereid, getsockopt, pid_t, LOCAL_PEEREPID, SOL_LOCAL};
303    use std::io;
304    use std::mem::size_of;
305    use std::mem::MaybeUninit;
306    use std::os::unix::io::AsRawFd;
307
308    pub(crate) fn get_peer_cred(sock: &UnixStream) -> io::Result<super::UCred> {
309        unsafe {
310            let raw_fd = sock.as_raw_fd();
311
312            let mut uid = MaybeUninit::uninit();
313            let mut gid = MaybeUninit::uninit();
314            let mut pid: MaybeUninit<pid_t> = MaybeUninit::uninit();
315            let mut pid_size: MaybeUninit<u32> = MaybeUninit::new(size_of::<pid_t>() as u32);
316
317            if getsockopt(
318                raw_fd,
319                SOL_LOCAL,
320                LOCAL_PEEREPID,
321                pid.as_mut_ptr() as *mut c_void,
322                pid_size.as_mut_ptr(),
323            ) != 0
324            {
325                return Err(io::Error::last_os_error());
326            }
327
328            assert!(pid_size.assume_init() == (size_of::<pid_t>() as u32));
329
330            let ret = getpeereid(raw_fd, uid.as_mut_ptr(), gid.as_mut_ptr());
331
332            if ret == 0 {
333                Ok(super::UCred {
334                    uid: uid.assume_init() as unix::uid_t,
335                    gid: gid.assume_init() as unix::gid_t,
336                    pid: Some(pid.assume_init() as unix::pid_t),
337                })
338            } else {
339                Err(io::Error::last_os_error())
340            }
341        }
342    }
343}

348    use std::io;
349    use std::os::unix::io::AsRawFd;
350    use std::ptr;
351
352    pub(crate) fn get_peer_cred(sock: &UnixStream) -> io::Result<super::UCred> {
353        unsafe {
354            let raw_fd = sock.as_raw_fd();
355
356            let mut cred = ptr::null_mut();
357            let ret = libc::getpeerucred(raw_fd, &mut cred);
358
359            if ret == 0 {
360                let uid = libc::ucred_geteuid(cred);
361                let gid = libc::ucred_getegid(cred);
362                let pid = libc::ucred_getpid(cred);
363
364                libc::ucred_free(cred);
365
366                Ok(super::UCred {
367                    uid: uid as unix::uid_t,
368                    gid: gid as unix::gid_t,
369                    pid: Some(pid as unix::pid_t),
370                })
371            } else {
372                Err(io::Error::last_os_error())
373            }
374        }
375    }
