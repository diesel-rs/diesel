13/// A bidirectional pipe to read and write bytes in memory.
14///
15/// A pair of `DuplexStream`s are created together, and they act as a "channel"
16/// that can be used as in-memory IO types. Writing to one of the pairs will
17/// allow that data to be read from the other, and vice versa.
18///
19/// # Closing a `DuplexStream`
20///
21/// If one end of the `DuplexStream` channel is dropped, any pending reads on
22/// the other side will continue to read data until the buffer is drained, then
23/// they will signal EOF by returning 0 bytes. Any writes to the other side,
24/// including pending ones (that are waiting for free space in the buffer) will
25/// return `Err(BrokenPipe)` immediately.
26///
27/// # Example
28///
29/// ```
30/// # async fn ex() -> std::io::Result<()> {
31/// # use tokio::io::{AsyncReadExt, AsyncWriteExt};
32/// let (mut client, mut server) = tokio::io::duplex(64);
33///
34/// client.write_all(b"ping").await?;
35///
36/// let mut buf = [0u8; 4];
37/// server.read_exact(&mut buf).await?;
38/// assert_eq!(&buf, b"ping");
39///
40/// server.write_all(b"pong").await?;
41///
42/// client.read_exact(&mut buf).await?;
43/// assert_eq!(&buf, b"pong");
44/// # Ok(())
45/// # }
46/// ```

        ::core::fmt::Formatter::debug_struct_field2_finish(f,
            "DuplexStream read write", &self.read, &&self.write)
    }
}Debug)]

79    ///
80    /// Using a `BytesMut` because it has efficient `Buf` and `BufMut`
81    /// functionality already. Additionally, it can try to copy data in the
82    /// same buffer if there read index has advanced far enough.

99/// Create a new pair of `DuplexStream`s that act like a pair of connected sockets.
100///
101/// The `max_buf_size` argument is the maximum amount of bytes that can be
102/// written to a side before the write returns `Poll::Pending`.
103///
104/// # Panics
105///
106/// This function panics if `max_buf_size` is 0.
107#[track_caller]
108#[cfg_attr(docsrs, doc(cfg(feature = "io-util")))]
109pub fn duplex(max_buf_size: usize) -> (DuplexStream, DuplexStream) {
110    let one = Arc::new(Mutex::new(SimplexStream::new_unsplit(max_buf_size)));
111    let two = Arc::new(Mutex::new(SimplexStream::new_unsplit(max_buf_size)));
112
113    (
114        DuplexStream {
115            read: one.clone(),
116            write: two.clone(),
117        },
118        DuplexStream {
119            read: two,
120            write: one,
121        },
122    )
123}

125impl AsyncRead for DuplexStream {
126    // Previous rustc required this `self` to be `mut`, even though newer
127    // versions recognize it isn't needed to call `lock()`. So for
128    // compatibility, we include the `mut` and `allow` the lint.
129    //
130    // See https://github.com/rust-lang/rust/issues/73592
131    #[allow(unused_mut)]
132    fn poll_read(
133        mut self: Pin<&mut Self>,
134        cx: &mut task::Context<'_>,
135        buf: &mut ReadBuf<'_>,
136    ) -> Poll<std::io::Result<()>> {
137        let mut wakers = Wakers::default();
138        let ret = Pin::new(&mut *self.read.lock()).poll_read_deferred(cx, buf, &mut wakers);
139        wakers.wake();
140        ret
141    }
142}

194impl Drop for DuplexStream {
195    fn drop(&mut self) {
196        // notify the other side of the closure
197        let read_waker = self.write.lock().close_write();
198        let write_waker = self.read.lock().close_read();
199        // wake after releasing the locks, as the wakers may access the pipe
200        if let Some(waker) = read_waker {
201            waker.wake();
202        }
203        if let Some(waker) = write_waker {
204            waker.wake();
205        }
206    }
207}

209/// Wakers taken out of a `SimplexStream` by a read or write.
210///
211/// A `DuplexStream` keeps its `SimplexStream`s behind a mutex, and wakes and
212/// drops these wakers only after releasing it, as they may access the pipe.
213/// The halves returned by `simplex()` are still polled under the mutex in
214/// `split`.
215#[derive(#[automatically_derived]impl ::core::default::Default for Wakers {
    #[inline]
    fn default() -> Self {
        Wakers {
            wake: ::core::default::Default::default(),
            replaced: ::core::default::Default::default(),
        }
    }
}Default)]
216struct Wakers {
217    /// The waker of the other side of the pipe, which needs to be woken.
218    wake: Option<Waker>,
219    /// The previously registered waker, which was replaced by a new one.
220    replaced: Option<Waker>,
221}
222
223impl Wakers {
224    fn wake(self) {
225        if let Some(waker) = self.wake {
226            waker.wake();
227        }
228        drop(self.replaced);
229    }
230}

234/// Creates unidirectional buffer that acts like in memory pipe.
235///
236/// The `max_buf_size` argument is the maximum amount of bytes that can be
237/// written to a buffer before it returns `Poll::Pending`.
238///
239/// # Unify reader and writer
240///
241/// The reader and writer half can be unified into a single structure
242/// of `SimplexStream` that supports both reading and writing or
243/// the `SimplexStream` can be already created as unified structure
244/// using [`SimplexStream::new_unsplit()`].
245///
246/// ```
247/// # async fn ex() -> std::io::Result<()> {
248/// # use tokio::io::{AsyncReadExt, AsyncWriteExt};
249/// let (reader, writer) = tokio::io::simplex(64);
250/// let mut simplex_stream = reader.unsplit(writer);
251/// simplex_stream.write_all(b"hello").await?;
252///
253/// let mut buf = [0u8; 5];
254/// simplex_stream.read_exact(&mut buf).await?;
255/// assert_eq!(&buf, b"hello");
256/// # Ok(())
257/// # }
258/// ```
259///
260/// # Panics
261///
262/// This function panics if `max_buf_size` is 0.
263#[track_caller]
264#[cfg_attr(docsrs, doc(cfg(feature = "io-util")))]
265pub fn simplex(max_buf_size: usize) -> (ReadHalf<SimplexStream>, WriteHalf<SimplexStream>) {
266    split(SimplexStream::new_unsplit(max_buf_size))
267}

308    fn poll_read_internal(
309        mut self: Pin<&mut Self>,
310        cx: &mut task::Context<'_>,
311        buf: &mut ReadBuf<'_>,
312        wakers: &mut Wakers,
313    ) -> Poll<std::io::Result<()>> {
314        if buf.remaining() == 0 {
315            return Poll::Ready(Ok(()));
316        }
317
318        if self.buffer.has_remaining() {
319            let max = self.buffer.remaining().min(buf.remaining());
320            buf.put_slice(&self.buffer[..max]);
321            self.buffer.advance(max);
322            if max > 0 {
323                // The passed `buf` might have been empty, don't wake up if
324                // no bytes have been moved.
325                wakers.wake = self.write_waker.take();
326            }
327            Poll::Ready(Ok(()))
328        } else if self.is_closed {
329            Poll::Ready(Ok(()))
330        } else {
331            wakers.replaced = self.read_waker.replace(cx.waker().clone());
332            Poll::Pending
333        }
334    }
335
336    fn poll_write_internal(
337        mut self: Pin<&mut Self>,
338        cx: &mut task::Context<'_>,
339        buf: &[u8],
340        wakers: &mut Wakers,
341    ) -> Poll<std::io::Result<usize>> {
342        if self.is_closed {
343            return Poll::Ready(Err(std::io::ErrorKind::BrokenPipe.into()));
344        }
345        if buf.is_empty() {
346            return Poll::Ready(Ok(0));
347        }
348
349        let avail = self.max_buf_size - self.buffer.len();
350        if avail == 0 {
351            wakers.replaced = self.write_waker.replace(cx.waker().clone());
352            return Poll::Pending;
353        }
354
355        let len = buf.len().min(avail);
356        self.buffer.extend_from_slice(&buf[..len]);
357        wakers.wake = self.read_waker.take();
358        Poll::Ready(Ok(len))
359    }
360
361    fn poll_write_vectored_internal(
362        mut self: Pin<&mut Self>,
363        cx: &mut task::Context<'_>,
364        bufs: &[std::io::IoSlice<'_>],
365        wakers: &mut Wakers,
366    ) -> Poll<Result<usize, std::io::Error>> {
367        if self.is_closed {
368            return Poll::Ready(Err(std::io::ErrorKind::BrokenPipe.into()));
369        }
370        if bufs.iter().all(|buf| buf.is_empty()) {
371            return Poll::Ready(Ok(0));
372        }
373
374        let avail = self.max_buf_size - self.buffer.len();
375        if avail == 0 {
376            wakers.replaced = self.write_waker.replace(cx.waker().clone());
377            return Poll::Pending;
378        }
379
380        let mut rem = avail;
381        for buf in bufs {
382            if rem == 0 {
383                break;
384            }
385
386            let len = buf.len().min(rem);
387            self.buffer.extend_from_slice(&buf[..len]);
388            rem -= len;
389        }
390
391        wakers.wake = self.read_waker.take();
392        Poll::Ready(Ok(avail - rem))
393    }
