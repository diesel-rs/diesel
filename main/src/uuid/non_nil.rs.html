11/// A UUID that is guaranteed not to be the [nil UUID](https://www.ietf.org/rfc/rfc9562.html#name-nil-uuid).
12///
13/// This is useful for representing optional UUIDs more efficiently, as `Option<NonNilUuid>`
14/// takes up the same space as `Uuid`.
15///
16/// Note that `Uuid`s created by the following methods are guaranteed to be non-nil:
17///
18/// - [`Uuid::new_v1`]
19/// - [`Uuid::now_v1`]
20/// - [`Uuid::new_v3`]
21/// - [`Uuid::new_v4`]
22/// - [`Uuid::new_v5`]
23/// - [`Uuid::new_v6`]
24/// - [`Uuid::now_v6`]
25/// - [`Uuid::new_v7`]
26/// - [`Uuid::now_v7`]
27/// - [`Uuid::new_v8`]
28///
29/// # ABI
30///
31/// The `NonNilUuid` type does not yet have a stable ABI. Its representation or alignment
32/// may change. It is currently only guaranteed that `NonNilUuid` and `Option<NonNilUuid>`
33/// are the same size as `Uuid`.

74impl NonNilUuid {
75    /// Creates a non-nil UUID if the value is non-nil.
76    pub const fn new(uuid: Uuid) -> Option<Self> {
77        match NonZeroU128::new(uuid.as_u128()) {
78            Some(non_nil) => Some(NonNilUuid(non_nil)),
79            None => None,
80        }
81    }
82
83    /// Creates a non-nil without checking whether the value is non-nil. This results in undefined behavior if the value is nil.
84    ///
85    /// # Safety
86    ///
87    /// The value must not be nil.
88    pub const unsafe fn new_unchecked(uuid: Uuid) -> Self {
89        NonNilUuid(unsafe { NonZeroU128::new_unchecked(uuid.as_u128()) })
90    }
91
92    /// Get the underlying [`Uuid`] value.
93    #[inline]
94    pub const fn get(self) -> Uuid {
95        Uuid::from_u128(self.0.get())
96    }
97
98    /// Creates a non-nil version 1 UUID.
99    ///
100    /// See [`Uuid::new_v1`] for details.
101    #[cfg(feature = "v1")]
102    pub fn new_v1(ts: crate::Timestamp, node_id: &[u8; 6]) -> Self {
103        // SAFETY: `Uuid::new_v1` sets the version bits, so its result cannot be nil.
104        unsafe { Self::new_unchecked(Uuid::new_v1(ts, node_id)) }
105    }

189impl From<NonNilUuid> for Uuid {
190    /// Converts a [`NonNilUuid`] back into a [`Uuid`].
191    ///
192    /// # Examples
193    /// ```
194    /// # use uuid::{NonNilUuid, Uuid};
195    /// let uuid = Uuid::from_u128(0x0123456789abcdef0123456789abcdef);
196    /// let non_nil = NonNilUuid::try_from(uuid).unwrap();
197    /// let uuid_again = Uuid::from(non_nil);
198    ///
199    /// assert_eq!(uuid, uuid_again);
200    /// ```
201    fn from(non_nil: NonNilUuid) -> Self {
202        Uuid::from_u128(non_nil.0.get())
203    }
204}

206impl TryFrom<Uuid> for NonNilUuid {
207    type Error = Error;
208
209    /// Attempts to convert a [`Uuid`] into a [`NonNilUuid`].
210    ///
211    /// # Examples
212    /// ```
213    /// # use uuid::{NonNilUuid, Uuid};
214    /// let uuid = Uuid::from_u128(0x0123456789abcdef0123456789abcdef);
215    /// let non_nil = NonNilUuid::try_from(uuid).unwrap();
216    /// ```
217    fn try_from(uuid: Uuid) -> Result<Self, Self::Error> {
218        NonZeroU128::new(uuid.as_u128())
219            .map(Self)
220            .ok_or(Error(ErrorKind::Nil))
221    }
222}

228    #[test]
229    fn test_non_nil_with_option_size() {
230        assert_eq!(
231            std::mem::size_of::<Option<NonNilUuid>>(),
232            std::mem::size_of::<Uuid>()

236    #[test]
237    fn test_non_nil() {
238        let uuid = Uuid::from_u128(0x0123456789abcdef0123456789abcdef);
239
240        assert_eq!(Uuid::from(NonNilUuid::try_from(uuid).unwrap()), uuid);
241        assert_eq!(NonNilUuid::new(uuid).unwrap(), uuid);
242        assert_eq!(unsafe { NonNilUuid::new_unchecked(uuid) }, uuid);
243
244        assert!(NonNilUuid::try_from(Uuid::nil()).is_err());
245        assert!(NonNilUuid::new(Uuid::nil()).is_none());
246    }
247
248    #[test]
249    fn test_non_nil_formatting() {
250        let uuid = Uuid::from_u128(0x0123456789abcdef0123456789abcdef);
251        let non_nil = NonNilUuid::try_from(uuid).unwrap();
252
253        assert_eq!(format!("{uuid}"), format!("{non_nil}"));
254        assert_eq!(format!("{uuid:?}"), format!("{non_nil:?}"));
255    }
256
257    #[test]
258    fn test_non_nil_ord() {
259        let uuid1 = Uuid::from_u128(0x0123456789abcdef0123456789abcdef);
260        let uuid2 = Uuid::from_u128(0x0123456789abcdef0123456789abcdf0);
261
262        let non_nil1 = NonNilUuid::try_from(uuid1).unwrap();
263        let non_nil2 = NonNilUuid::try_from(uuid2).unwrap();
264
265        // Ordering between NonNilUuid instances
266        assert!(non_nil1 < non_nil2);
267        assert!(non_nil2 > non_nil1);
268
269        // Ordering between NonNilUuid and Uuid
270        assert!(non_nil1 < uuid2);
271        assert!(non_nil2 > uuid1);
272
273        // Ordering between Uuid and NonNilUuid
274        assert!(uuid1 < non_nil2);
275        assert!(uuid2 > non_nil1);
276
277        // Equality
278        let non_nil1_copy = NonNilUuid::try_from(uuid1).unwrap();
279        assert_eq!(non_nil1, non_nil1_copy);
280        assert!(non_nil1 <= non_nil1_copy);
281        assert!(non_nil1 >= non_nil1_copy);
282
283        // Equality between NonNilUuid and Uuid
284        assert_eq!(non_nil1, uuid1);
285        assert_eq!(uuid1, non_nil1);
286        assert!(non_nil1 >= uuid1);
287        assert!(non_nil1 <= uuid1);
288        assert!(uuid1 >= non_nil1);
289        assert!(uuid1 <= non_nil1);
290    }
