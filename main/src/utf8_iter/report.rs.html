48impl Utf8Handler for ErrorReportingHandler {
49    type Output = Result<char, Utf8CharsError>;
50
51    /// Map a single-byte UTF-8 sequence to `Output`.
52    ///
53    /// When `Output` is `char`, `ascii.to_char()`
54    /// is the appropriate implementation.
55    ///
56    /// The implementation is expected to be marked
57    /// `#[inline(always)]`.
58    #[inline(always)]
59    fn single_byte(&self, ascii: Ascii) -> Self::Output {
60        Ok(ascii.to_char())
61    }
62
63    /// Map a two-byte UTF-8 sequence to `Output`.
64    ///
65    /// When `Output` is `char`, `sequence.to_char()`
66    /// is the appropriate implementation.
67    ///
68    /// The implementation is expected to be marked
69    /// `#[inline(always)]`.
70    #[inline(always)]
71    fn two_byte(&self, sequence: TwoByteSequence) -> Self::Output {
72        Ok(sequence.to_char())
73    }
74
75    /// Map a three-byte UTF-8 sequence to `Output`.
76    ///
77    /// When `Output` is `char`, `sequence.to_char()`
78    /// is the appropriate implementation.
79    ///
80    /// The implementation is expected to be marked
81    /// `#[inline(always)]`.
82    #[inline(always)]
83    fn three_byte(&self, sequence: ThreeByteSequence) -> Self::Output {
84        Ok(sequence.to_char())
85    }
86
87    /// Map a four-byte UTF-8 sequence to `Output`.
88    ///
89    /// When `Output` is `char`, `sequence.to_char()`
90    /// is the appropriate implementation.
91    ///
92    /// The implementation is expected to be marked
93    /// `#[inline(always)]`.
94    #[inline(always)]
95    fn four_byte(&self, sequence: FourByteSequence) -> Self::Output {
96        Ok(sequence.to_char())
97    }
98
99    /// Map a single UTF-8 error to `Output`.
100    ///
101    /// What constitutes a single error is defined by the
102    /// WHATWG Encoding Standard.
103    ///
104    /// When `Output` is `char`,
105    /// `char::REPLACEMENT_CHARACTER`
106    /// is the appropriate implementation.
107    ///
108    /// The implementation of this method is expected to
109    /// be declared `#[inline(always)]`.
110    #[inline(always)]
111    fn error(&self) -> Self::Output {
112        Err(Utf8CharsError)
113    }
114}
