21//! Provides iteration by `char` over `&[u8]` containing potentially-invalid
22//! UTF-8 such that errors are handled according to the [WHATWG Encoding
23//! Standard](https://encoding.spec.whatwg.org/#utf-8-decoder) (i.e. the same
24//! way as in `String::from_utf8_lossy`).
25//!
26//! The trait `Utf8CharsEx` provides the convenience method `chars()` on
27//! byte slices themselves instead of having to use the more verbose
28//! `Utf8Chars::new(slice)`.
29//!
30//! ```rust
31//! use utf8_iter::Utf8CharsEx;
32//! let data = b"\xFF\xC2\xE2\xE2\x98\xF0\xF0\x9F\xF0\x9F\x92\xE2\x98\x83";
33//! let from_iter: String = data.chars().collect();
34//! let from_std = String::from_utf8_lossy(data);
35//! assert_eq!(from_iter, from_std);
36//! ```
37
