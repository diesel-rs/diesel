44impl Utf8Handler for DefaultHandler {
45    type Output = char;
46
47    /// Map a single-byte UTF-8 sequence to `Output`.
48    #[inline(always)]
49    fn single_byte(&self, ascii: Ascii) -> Self::Output {
50        ascii.to_char()
51    }
52
53    /// Map a two-byte UTF-8 sequence to `Output`.
54    #[inline(always)]
55    fn two_byte(&self, sequence: TwoByteSequence) -> Self::Output {
56        sequence.to_char()
57    }
58
59    /// Map a three-byte UTF-8 sequence to `Output`.
60    #[inline(always)]
61    fn three_byte(&self, sequence: ThreeByteSequence) -> Self::Output {
62        sequence.to_char()
63    }
64
65    /// Map a four-byte UTF-8 sequence to `Output`.
66    #[inline(always)]
67    fn four_byte(&self, sequence: FourByteSequence) -> Self::Output {
68        sequence.to_char()
69    }
70
71    /// Map a single UTF-8 error to `Output`.
72    #[inline(always)]
73    fn error(&self) -> Self::Output {
74        char::REPLACEMENT_CHARACTER
75    }
76}

impl<'a> Utf8Chars<'a> {
    #[inline(always)]
    /// Creates the iterator from a byte slice.
    pub fn new(bytes: &'a [u8]) -> Self {
        Self {
            inner: crate::Utf8CharsWithHandler::new(bytes,
                DefaultHandler::new()),
        }
    }
    /// Views the current remaining data in the iterator as a subslice
    /// of the original slice.
    #[must_use]
    #[inline(always)]
    pub fn as_slice(&self) -> &'a [u8] { self.inner.as_slice() }
}

    fn size_hint(&self) -> (usize, Option<usize>) { self.inner.size_hint() }

    fn last(mut self) -> Option<Self::Item> { self.next_back() }

91/// Convenience trait that adds `chars()` and `char_indices()` methods
92/// similar to the ones on string slices to byte slices.
93pub trait Utf8CharsEx {
94    /// Iterator over `char`s
95    fn chars(&self) -> Utf8Chars<'_>;
96    /// Iterator over `char`s and their indices
97    fn char_indices(&self) -> Utf8CharIndices<'_>;
98}

100impl Utf8CharsEx for [u8] {
101    /// Convenience method for creating an UTF-8 iterator
102    /// for the slice.
103    #[inline(always)]
104    fn chars(&self) -> Utf8Chars<'_> {
105        Utf8Chars::new(self)
106    }
107    /// Convenience method for creating a byte index and
108    /// UTF-8 iterator for the slice.
109    #[inline(always)]
110    fn char_indices(&self) -> Utf8CharIndices<'_> {
111        Utf8CharIndices::new(self)
112    }
