14/// Returns a copy of the global [`TimeZoneDatabase`].
15///
16/// This is the same database used for convenience routines like
17/// [`Timestamp::in_tz`](crate::Timestamp::in_tz) and parsing routines for
18/// [`ZonedDateTime`](crate::ZonedDateTime) that need to do IANA time zone
19/// identifier lookups. Basically, whenever an implicit time zone database is
20/// needed, it is *this* copy of the time zone database that is used.
21///
22/// In feature configurations where a time zone database cannot interact with
23/// the file system (like when `std` is not enabled), this returns a database
24/// where every lookup will fail.
25///
26/// # Example
27///
28/// ```
29/// use jiff::tz;
30///
31/// assert!(tz::db().get("Antarctica/Troll").is_ok());
32/// assert!(tz::db().get("does-not-exist").is_err());
33/// ```
34pub fn db() -> &'static TimeZoneDatabase {
35    // #[cfg(any(not(feature = "std"), miri))]
36    #[cfg(not(feature = "std"))]
37    {
38        // Without `std` there's no lazily initialized global state. But when
39        // the tzdb is bundled into the binary, we can still hand out a usable
40        // database: the bundled database needs no allocation, so it can be
41        // constructed in a `const`. (Without `std`, the zoneinfo and
42        // concatenated databases are both unavailable, so there's nothing to
43        // prefer over the bundle.)
44        //
45        // Ref: https://github.com/BurntSushi/jiff/issues/533
46        #[cfg(any(
47            feature = "tzdb-bundle-always",
48            all(
49                feature = "tzdb-bundle-platform",
50                any(windows, target_family = "wasm"),
51            ),
52        ))]
53        static DB: TimeZoneDatabase = TimeZoneDatabase {
54            inner: Repr::Bundled(bundled::Database::new()),
55        };
56        #[cfg(not(any(
57            feature = "tzdb-bundle-always",
58            all(
59                feature = "tzdb-bundle-platform",
60                any(windows, target_family = "wasm"),
61            ),
62        )))]
63        static DB: TimeZoneDatabase = TimeZoneDatabase::none();
64        &DB
65    }
66    // #[cfg(all(feature = "std", not(miri)))]
67    #[cfg(feature = "std")]
68    {
69        use std::sync::OnceLock;
70
71        static DB: OnceLock<TimeZoneDatabase> = OnceLock::new();
72        DB.get_or_init(|| {
73            let db = TimeZoneDatabase::from_env();
74            debug!("initialized global time zone database: {db:?}");
75            db
76        })
77    }
78}

80/// A handle to a [IANA Time Zone Database].
81///
82/// A `TimeZoneDatabase` provides a way to lookup [`TimeZone`]s by their
83/// human readable identifiers, such as `America/Los_Angeles` and
84/// `Europe/Warsaw`.
85///
86/// It is rare to need to create or use this type directly. Routines like zoned
87/// datetime parsing and time zone conversion provide convenience routines for
88/// using an implicit global time zone database by default. This global time
89/// zone database is available via [`jiff::tz::db`](crate::tz::db()`). But
90/// lower level parsing routines such as
91/// [`fmt::temporal::DateTimeParser::parse_zoned_datetime_with`](crate::fmt::temporal::DateTimeParser::parse_zoned_datetime_with)
92/// and
93/// [`civil::DateTime::to_zoned_datetime`](crate::civil::DateTime::to_zoned_datetime)
94/// provide a means to use a custom copy of a `TimeZoneDatabase`.
95///
96/// # Platform behavior
97///
98/// This behavior is subject to change.
99///
100/// On Unix systems, and when the `tzdb-zoneinfo` crate feature is enabled
101/// (which it is by default), Jiff will read the `/usr/share/zoneinfo`
102/// directory for time zone data.
103///
104/// On Windows systems and when the `tzdb-bundle-platform` crate feature is
105/// enabled (which it is by default), _or_ when the `tzdb-bundle-always` crate
106/// feature is enabled, then the `jiff-tzdb` crate will be used to embed the
107/// entire Time Zone Database into the compiled artifact.
108///
109/// On Android systems, and when the `tzdb-concatenated` crate feature is
110/// enabled (which it is by default), Jiff will attempt to read a concatenated
111/// zoneinfo database using the `ANDROID_DATA` or `ANDROID_ROOT` environment
112/// variables.
113///
114/// In general, using `/usr/share/zoneinfo` (or an equivalent) is heavily
115/// preferred in lieu of embedding the database into your compiled artifact.
116/// The reason is because your system copy of the Time Zone Database may be
117/// updated, perhaps a few times a year, and it is better to get seamless
118/// updates through your system rather than needing to wait on a Rust crate
119/// to update and then rebuild your software. The bundling approach should
120/// only be used when there is no plausible alternative. For example, Windows
121/// has no canonical location for a copy of the Time Zone Database. Indeed,
122/// this is why the Cargo configuration of Jiff specifically does not enabled
123/// bundling by default on Unix systems, but does enable it by default on
124/// Windows systems. Of course, if you really do need a copy of the database
125/// bundled, then you can enable the `tzdb-bundle-always` crate feature.
126///
127/// # Cloning
128///
129/// A `TimeZoneDatabase` can be cheaply cloned. It will share a thread safe
130/// cache with other copies of the same `TimeZoneDatabase`.
131///
132/// # Caching
133///
134/// Because looking up a time zone on disk, reading the file into memory
135/// and parsing the time zone transitions out of that file requires
136/// a fair amount of work, a `TimeZoneDatabase` does a fair bit of
137/// caching. This means that the vast majority of calls to, for example,
138/// [`Timestamp::in_tz`](crate::Timestamp::in_tz) don't actually need to hit
139/// disk. It will just find a cached copy of a [`TimeZone`] and return that.
140///
141/// Of course, with caching comes problems of cache invalidation. Invariably,
142/// there are parameters that Jiff uses to manage when the cache should be
143/// invalidated. Jiff tries to emit log messages about this when it happens. If
144/// you find the caching behavior of Jiff to be sub-optimal for your use case,
145/// please create an issue. (The plan is likely to expose some options for
146/// configuring the behavior of a `TimeZoneDatabase`, but I wanted to collect
147/// user feedback first.)
148///
149/// [IANA Time Zone Database]: https://en.wikipedia.org/wiki/Tz_database
150///
151/// # Example: list all available time zones
152///
153/// ```no_run
154/// use jiff::tz;
155///
156/// for tzid in tz::db().available() {
157///     println!("{tzid}");
158/// }
159/// ```
160///
161/// # Example: using multiple time zone databases
162///
163/// Jiff supports opening and using multiple time zone databases by default.
164/// All you need to do is point [`TimeZoneDatabase::from_dir`] to your own
165/// copy of the Time Zone Database, and it will handle the rest.
166///
167/// This example shows how to utilize multiple databases by parsing a datetime
168/// using an older copy of the IANA Time Zone Database. This example leverages
169/// the fact that the 2018 copy of the database preceded Brazil's announcement
170/// that daylight saving time would be abolished. This meant that datetimes
171/// in the future, when parsed with the older copy of the Time Zone Database,
172/// would still follow the old daylight saving time rules. But a mere update of
173/// the database would otherwise change the meaning of the datetime.
174///
175/// This scenario can come up if one stores datetimes in the future. This is
176/// also why the default offset conflict resolution strategy when parsing zoned
177/// datetimes is [`OffsetConflict::Reject`](crate::tz::OffsetConflict::Reject),
178/// which prevents one from silently re-interpreting datetimes to a different
179/// timestamp.
180///
181/// ```no_run
182/// use jiff::{fmt::temporal::DateTimeParser, tz::{self, TimeZoneDatabase}};
183///
184/// static PARSER: DateTimeParser = DateTimeParser::new();
185///
186/// // Open a version of tzdb from before Brazil announced its abolition
187/// // of daylight saving time.
188/// let tzdb2018 = TimeZoneDatabase::from_dir("path/to/tzdb-2018b")?;
189/// // Open the system tzdb.
190/// let tzdb = tz::db();
191///
192/// // Parse the same datetime string with the same parser, but using two
193/// // different versions of tzdb.
194/// let dt = "2020-01-15T12:00[America/Sao_Paulo]";
195/// let zdt2018 = PARSER.parse_zoned_datetime_with(&tzdb2018, dt)?;
196/// let zdt = PARSER.parse_zoned_datetime_with(tzdb, dt)?;
197///
198/// // Before DST was abolished, 2020-01-15 was in DST, which corresponded
199/// // to UTC offset -02. Since DST rules applied to datetimes in the
200/// // future, the 2018 version of tzdb would lead one to interpret
201/// // 2020-01-15 as being in DST.
202/// assert_eq!(zdt2018.offset(), tz::offset(-2));
203/// // But DST was abolished in 2019, which means that 2020-01-15 was no
204/// // no longer in DST. So after a tzdb update, the same datetime as above
205/// // now has a different offset.
206/// assert_eq!(zdt.offset(), tz::offset(-3));
207///
208/// // So if you try to parse a datetime serialized from an older copy of
209/// // tzdb, you'll get an error under the default configuration because
210/// // of `OffsetConflict::Reject`. This would succeed if you parsed it
211/// // using tzdb2018!
212/// assert!(
213///     PARSER.parse_zoned_datetime_with(tzdb, zdt2018.to_string()).is_err()
214/// );
215///
216/// # Ok::<(), Box<dyn std::error::Error>>(())
217/// ```

223/// The internal representation of a `TimeZoneDatabase`.
224///
225/// The bundled database is kept out of the `Arc` because it carries no data
226/// of its own (the tzdb is compiled into the binary and any parsed zones are
227/// cached in a global). This lets it be constructed in a `const`, which is
228/// what makes it possible for `db()` to return the bundled database even when
229/// `std` is unavailable (and so there is no lazily initialized global state).
230///
231/// Ref: https://github.com/BurntSushi/jiff/issues/533
232#[derive(#[automatically_derived]impl ::core::clone::Clone for Repr {
    #[inline]
    #[rustc_trivial_field_reads]
    fn clone(&self) -> Self {
        match self {
            Self::Empty => Repr::Empty,
            Self::Bundled(__self_0) =>
                Repr::Bundled(::core::clone::Clone::clone(__self_0)),
            Self::Arc(__self_0) =>
                Repr::Arc(::core::clone::Clone::clone(__self_0)),
        }
    }
}Clone)]
233enum Repr {
234    /// A database for which all lookups fail.
235    Empty,
236    /// The bundled database. Needs no allocation, so no `Arc`.
237    Bundled(bundled::Database),
238    /// A database backed by an `Arc` so that clones share its cache.
239    Arc(Arc<Kind>),
240}

265    /// Returns a time zone database initialized from the current environment.
266    ///
267    /// This routine never fails, but it may not be able to find a copy of
268    /// your Time Zone Database. When this happens, log messages (with some
269    /// at least at the `WARN` level) will be emitted. They can be viewed by
270    /// installing a [`log`] compatible logger such as [`env_logger`].
271    ///
272    /// Typically, one does not need to call this routine directly. Instead,
273    /// it's done for you as part of [`jiff::tz::db`](crate::tz::db()).
274    /// This does require Jiff's `std` feature to be enabled though. So for
275    /// example, you might use this constructor when the features `alloc`
276    /// and `tzdb-bundle-always` are enabled to get access to a bundled
277    /// copy of the IANA time zone database. (Accessing the system copy at
278    /// `/usr/share/zoneinfo` requires `std`.)
279    ///
280    /// Beware that calling this constructor will create a new _distinct_
281    /// handle from the one returned by `jiff::tz::db` with its own cache.
282    ///
283    /// [`log`]: https://docs.rs/log
284    /// [`env_logger`]: https://docs.rs/env_logger
285    ///
286    /// # Platform behavior
287    ///
288    /// When the `TZDIR` environment variable is set, this will attempt to
289    /// open the Time Zone Database at the directory specified. Otherwise,
290    /// this will search a list of predefined directories for a system
291    /// installation of the Time Zone Database. Typically, it's found at
292    /// `/usr/share/zoneinfo`.
293    ///
294    /// On Windows systems, under the default crate configuration, this will
295    /// return an embedded copy of the Time Zone Database since Windows does
296    /// not have a canonical installation of the Time Zone Database.
297    pub fn from_env() -> TimeZoneDatabase {
298        // On Android, try the concatenated database first, since that's
299        // typically what is used.
300        //
301        // Overall this logic might be sub-optimal. Like, does it really make
302        // sense to check for the zoneinfo or concatenated database on non-Unix
303        // platforms? Probably not to be honest. But these should only be
304        // executed ~once generally, so it doesn't seem like a big deal to try.
305        // And trying makes things a little more flexible I think.
306        #[cfg(not(miri))]
307        {
308            if falsecfg!(target_os = "android") {
309                let db = concatenated::Database::from_env();
310                if !db.is_definitively_empty() {
311                    return TimeZoneDatabase::new(Kind::Concatenated(db));
312                }
313
314                let db = zoneinfo::Database::from_env();
315                if !db.is_definitively_empty() {
316                    return TimeZoneDatabase::new(Kind::ZoneInfo(db));
317                }
318            } else {
319                let db = zoneinfo::Database::from_env();
320                if !db.is_definitively_empty() {
321                    return TimeZoneDatabase::new(Kind::ZoneInfo(db));
322                }
323
324                let db = concatenated::Database::from_env();
325                if !db.is_definitively_empty() {
326                    return TimeZoneDatabase::new(Kind::Concatenated(db));
327                }
328            }
329        }
330
331        let db = bundled::Database::new();
332        if !db.is_definitively_empty() {
333            return TimeZoneDatabase { inner: Repr::Bundled(db) };
334        }
335
336        warn!(
337            "could not find zoneinfo, concatenated tzdata or \
338             bundled time zone database",
339        );
340        TimeZoneDatabase::none()
341    }

411    /// Returns a time zone database initialized from the bundled copy of
412    /// the [IANA Time Zone Database].
413    ///
414    /// While this API is always available, in order to get a non-empty
415    /// database back, this requires that one of the crate features
416    /// `tzdb-bundle-always` or `tzdb-bundle-platform` is enabled. In the
417    /// latter case, the bundled database is only available on platforms known
418    /// to lack a system copy of the IANA Time Zone Database (i.e., non-Unix
419    /// systems).
420    ///
421    /// This routine is infallible, but it may return a database
422    /// that is definitively empty if the bundled data is not
423    /// available. To query whether the data is empty or not, use
424    /// [`TimeZoneDatabase::is_definitively_empty`].
425    ///
426    /// # Data generation
427    ///
428    /// The data in this crate comes from the [IANA Time Zone Database] "data
429    /// only" distribution. [`jiff-cli`] is used to first compile the release
430    /// into binary TZif data using the `zic` compiler, and secondly, converts
431    /// the binary data into a flattened and de-duplicated representation that
432    /// is embedded into this crate's source code.
433    ///
434    /// The conversion into the TZif binary data uses the following settings:
435    ///
436    /// * The "rearguard" data is used (see below).
437    /// * The binary data itself is compiled using the "slim" format. Which
438    ///   effectively means that the TZif data primarily only uses explicit
439    ///   time zone transitions for historical data and POSIX time zones for
440    ///   current time zone transition rules. This doesn't have any impact
441    ///   on the actual results. The reason that there are "slim" and "fat"
442    ///   formats is to support legacy applications that can't deal with
443    ///   POSIX time zones. For example, `/usr/share/zoneinfo` on my modern
444    ///   Archlinux installation (2025-02-27) is in the "fat" format.
445    ///
446    /// The reason that rearguard data is used is a bit more subtle and has
447    /// to do with a difference in how the IANA Time Zone Database treats its
448    /// internal "daylight saving time" flag and what people in the "real
449    /// world" consider "daylight saving time." For example, in the standard
450    /// distribution of the IANA Time Zone Database, `Europe/Dublin` has its
451    /// daylight saving time flag set to _true_ during Winter and set to
452    /// _false_ during Summer. The actual time shifts are the same as, e.g.,
453    /// `Europe/London`, but which one is actually labeled "daylight saving
454    /// time" is not.
455    ///
456    /// The IANA Time Zone Database does this for `Europe/Dublin`, presumably,
457    /// because _legally_, time during the Summer in Ireland is called `Irish
458    /// Standard Time`, and time during the Winter is called `Greenwich Mean
459    /// Time`. These legal names are reversed from what is typically the case,
460    /// where "standard" time is during the Winter and daylight saving time is
461    /// during the Summer. The IANA Time Zone Database implements this tweak in
462    /// legal language via a "negative daylight saving time offset." This is
463    /// somewhat odd, and some consumers of the IANA Time Zone Database cannot
464    /// handle it. Thus, the rearguard format was born for, seemingly, legacy
465    /// programs.
466    ///
467    /// Jiff can handle negative daylight saving time offsets just fine,
468    /// but we use the rearguard format anyway so that the underlying data
469    /// more accurately reflects on-the-ground reality for humans living in
470    /// `Europe/Dublin`. In particular, using the rearguard data enables
471    /// [localization of time zone names] to be done correctly.
472    ///
473    /// [IANA Time Zone Database]: https://en.wikipedia.org/wiki/Tz_database
474    /// [`jiff-cli`]: https://github.com/BurntSushi/jiff/tree/master/crates/jiff-cli
475    /// [localization of time zone names]: https://github.com/BurntSushi/jiff/issues/258
476    pub fn bundled() -> TimeZoneDatabase {
477        let db = bundled::Database::new();
478        if db.is_definitively_empty() {
479            warn!("could not find embedded/bundled zoneinfo");
480        }
481        TimeZoneDatabase { inner: Repr::Bundled(db) }
482    }

489    /// Returns a [`TimeZone`] corresponding to the IANA time zone identifier
490    /// given.
491    ///
492    /// The lookup is performed without regard to ASCII case.
493    ///
494    /// To see a list of all available time zone identifiers for this database,
495    /// use [`TimeZoneDatabase::available`].
496    ///
497    /// It is guaranteed that if the given time zone name is case insensitively
498    /// equivalent to `UTC`, then the time zone returned will be equivalent to
499    /// `TimeZone::UTC`. Similarly for `Etc/Unknown` and `TimeZone::unknown()`.
500    ///
501    /// # Example
502    ///
503    /// ```
504    /// use jiff::tz;
505    ///
506    /// let tz = tz::db().get("america/NEW_YORK")?;
507    /// assert_eq!(tz.iana_name(), Some("America/New_York"));
508    ///
509    /// # Ok::<(), Box<dyn std::error::Error>>(())
510    /// ```
511    pub fn get(&self, name: &str) -> Result<TimeZone, Error> {
512        let found = match self.inner {
513            Repr::Empty => {
514                return Err(Error::from(
515                    E::failed_time_zone_no_database_configured(name),
516                ));
517            }
518            Repr::Bundled(ref db) => db.get(name),
519            Repr::Arc(ref kind) => match **kind {
520                Kind::ZoneInfo(ref db) => db.get(name),
521                Kind::Concatenated(ref db) => db.get(name),
522            },
523        };
524        if let Some(tz) = found {
525            trace!("found time zone `{name}` in {db:?}", db = self);
526            return Ok(tz);
527        }
528        Err(Error::from(E::failed_time_zone(name)))
529    }

561    ///
562    /// Subsequent interactions with this database will need to re-read time
563    /// zone data from disk.
564    ///
565    /// It might be useful to call this if you know the time zone database
566    /// has changed on disk and want to force Jiff to re-load it immediately
567    /// without spawning a new process or waiting for Jiff's internal cache
568    /// invalidation heuristics to kick in.
569    pub fn reset(&self) {
570        match self.inner {
571            Repr::Empty => {}
572            Repr::Bundled(ref db) => db.reset(),
573            Repr::Arc(ref kind) => match **kind {
574                Kind::ZoneInfo(ref db) => db.reset(),
575                Kind::Concatenated(ref db) => db.reset(),
576            },
577        }
578    }
579
580    /// Returns true if it is known that this time zone database is empty.
581    ///
582    /// When this returns true, it is guaranteed that all
583    /// [`TimeZoneDatabase::get`] calls will fail, and that
584    /// [`TimeZoneDatabase::available`] will always return an empty iterator.
585    ///
586    /// Note that if this returns false, it is still possible for this database
587    /// to be empty.
588    ///
589    /// # Example
590    ///
591    /// ```
592    /// use jiff::tz::TimeZoneDatabase;
593    ///
594    /// let db = TimeZoneDatabase::none();
595    /// assert!(db.is_definitively_empty());
596    /// ```
597    pub fn is_definitively_empty(&self) -> bool {
598        match self.inner {
599            Repr::Empty => true,
600            Repr::Bundled(ref db) => db.is_definitively_empty(),
601            Repr::Arc(ref kind) => match **kind {
602                Kind::ZoneInfo(ref db) => db.is_definitively_empty(),
603                Kind::Concatenated(ref db) => db.is_definitively_empty(),
604            },
605        }
606    }
607}

845        .ok()?;

854/// Checks if `name` is a "special" time zone and returns one if so.
855///
856/// This is limited to special constants that should have consistent values
857/// across time zone database implementations. For example, `UTC`.
858fn special_time_zone(name: &str) -> Option<TimeZone> {
859    if utf8::cmp_ignore_ascii_case("utc", name).is_eq() {
860        return Some(TimeZone::UTC);
861    }
862    if utf8::cmp_ignore_ascii_case("etc/unknown", name).is_eq() {
863        return Some(TimeZone::unknown());
864    }
