18/// A representation of a civil date in the Gregorian calendar.
19///
20/// A `Date` value corresponds to a triple of year, month and day. Every `Date`
21/// value is guaranteed to be a valid Gregorian calendar date. For example,
22/// both `2023-02-29` and `2023-11-31` are invalid and cannot be represented by
23/// a `Date`.
24///
25/// # Civil dates
26///
27/// A `Date` value behaves without regard to daylight saving time or time
28/// zones in general. When doing arithmetic on dates with spans defined in
29/// units of time (such as with [`Date::checked_add`]), days are considered to
30/// always be precisely `86,400` seconds long.
31///
32/// # Parsing and printing
33///
34/// The `Date` type provides convenient trait implementations of
35/// [`std::str::FromStr`] and [`std::fmt::Display`]:
36///
37/// ```
38/// use jiff::civil::Date;
39///
40/// let date: Date = "2024-06-19".parse()?;
41/// assert_eq!(date.to_string(), "2024-06-19");
42///
43/// # Ok::<(), Box<dyn std::error::Error>>(())
44/// ```
45///
46/// A civil `Date` can also be parsed from something that _contains_ a date,
47/// but with perhaps other data (such as an offset or time zone):
48///
49/// ```
50/// use jiff::civil::Date;
51///
52/// let date: Date = "2024-06-19T15:22:45-04[America/New_York]".parse()?;
53/// assert_eq!(date.to_string(), "2024-06-19");
54///
55/// # Ok::<(), Box<dyn std::error::Error>>(())
56/// ```
57///
58/// For more information on the specific format supported, see the
59/// [`fmt::temporal`](crate::fmt::temporal) module documentation.
60///
61/// # Default value
62///
63/// For convenience, this type implements the `Default` trait. Its default
64/// value corresponds to `0000-01-01`. One can also access this value via the
65/// `Date::ZERO` constant.
66///
67/// # Comparisons
68///
69/// The `Date` type provides both `Eq` and `Ord` trait implementations to
70/// facilitate easy comparisons. When a date `d1` occurs before a date `d2`,
71/// then `d1 < d2`. For example:
72///
73/// ```
74/// use jiff::civil::date;
75///
76/// let d1 = date(2024, 3, 11);
77/// let d2 = date(2025, 1, 31);
78/// assert!(d1 < d2);
79/// ```
80///
81/// # Arithmetic
82///
83/// This type provides routines for adding and subtracting spans of time, as
84/// well as computing the span of time between two `Date` values.
85///
86/// For adding or subtracting spans of time, one can use any of the following
87/// routines:
88///
89/// * [`Date::checked_add`] or [`Date::checked_sub`] for checked arithmetic.
90/// * [`Date::saturating_add`] or [`Date::saturating_sub`] for saturating
91/// arithmetic.
92///
93/// Additionally, checked arithmetic is available via the `Add` and `Sub`
94/// trait implementations. When the result overflows, a panic occurs.
95///
96/// ```
97/// use jiff::{civil::date, ToSpan};
98///
99/// let start = date(2024, 2, 25);
100/// let one_week_later = start + 1.weeks();
101/// assert_eq!(one_week_later, date(2024, 3, 3));
102/// ```
103///
104/// One can compute the span of time between two dates using either
105/// [`Date::until`] or [`Date::since`]. It's also possible to subtract two
106/// `Date` values directly via a `Sub` trait implementation:
107///
108/// ```
109/// use jiff::{civil::date, ToSpan};
110///
111/// let date1 = date(2024, 3, 3);
112/// let date2 = date(2024, 2, 25);
113/// assert_eq!(date1 - date2, 7.days().fieldwise());
114/// ```
115///
116/// The `until` and `since` APIs are polymorphic and allow re-balancing and
117/// rounding the span returned. For example, the default largest unit is days
118/// (as exemplified above), but we can ask for bigger units:
119///
120/// ```
121/// use jiff::{civil::date, ToSpan, Unit};
122///
123/// let date1 = date(2024, 5, 3);
124/// let date2 = date(2024, 2, 25);
125/// assert_eq!(
126///     date1.since((Unit::Year, date2))?,
127///     2.months().days(7).fieldwise(),
128/// );
129///
130/// # Ok::<(), Box<dyn std::error::Error>>(())
131/// ```
132///
133/// Or even round the span returned:
134///
135/// ```
136/// use jiff::{civil::{DateDifference, date}, RoundMode, ToSpan, Unit};
137///
138/// let date1 = date(2024, 5, 15);
139/// let date2 = date(2024, 2, 25);
140/// assert_eq!(
141///     date1.since(
142///         DateDifference::new(date2)
143///             .smallest(Unit::Month)
144///             .largest(Unit::Year),
145///     )?,
146///     2.months().fieldwise(),
147/// );
148/// // `DateDifference` uses truncation as a rounding mode by default,
149/// // but you can set the rounding mode to break ties away from zero:
150/// assert_eq!(
151///     date1.since(
152///         DateDifference::new(date2)
153///             .smallest(Unit::Month)
154///             .largest(Unit::Year)
155///             .mode(RoundMode::HalfExpand),
156///     )?,
157///     // Rounds up to 8 days.
158///     3.months().fieldwise(),
159/// );
160///
161/// # Ok::<(), Box<dyn std::error::Error>>(())
162/// ```
163///
164/// # Rounding
165///
166/// Rounding dates is currently not supported. If you want this functionality,
167/// please participate in the [issue tracking its support][add-date-rounding].
168///
169/// [add-date-rounding]: https://github.com/BurntSushi/jiff/issues/1

422    /// Returns the year and its era.
423    ///
424    /// This crate specifically allows years to be negative or `0`, where as
425    /// years written for the Gregorian calendar are always positive and
426    /// greater than `0`. In the Gregorian calendar, the era labels `BCE` and
427    /// `CE` are used to disambiguate between years less than or equal to `0`
428    /// and years greater than `0`, respectively.
429    ///
430    /// The crate is designed this way so that years in the latest era (that
431    /// is, `CE`) are aligned with years in this crate.
432    ///
433    /// The year returned is guaranteed to be in the range `1..=10000`.
434    ///
435    /// # Example
436    ///
437    /// ```
438    /// use jiff::civil::{Era, date};
439    ///
440    /// let d = date(2024, 10, 3);
441    /// assert_eq!(d.era_year(), (2024, Era::CE));
442    ///
443    /// let d = date(1, 10, 3);
444    /// assert_eq!(d.era_year(), (1, Era::CE));
445    ///
446    /// let d = date(0, 10, 3);
447    /// assert_eq!(d.era_year(), (1, Era::BCE));
448    ///
449    /// let d = date(-1, 10, 3);
450    /// assert_eq!(d.era_year(), (2, Era::BCE));
451    ///
452    /// let d = date(-10, 10, 3);
453    /// assert_eq!(d.era_year(), (11, Era::BCE));
454    ///
455    /// let d = date(-9_999, 10, 3);
456    /// assert_eq!(d.era_year(), (10_000, Era::BCE));
457    /// ```
458    #[inline]
459    pub fn era_year(self) -> (i16, Era) {
460        let year = self.year();
461        if year >= 1 {
462            (year, Era::CE)
463        } else {
464            // We specifically ensure our min/max bounds on `Year` always leave
465            // room in its representation to add or subtract 1, so this will
466            // never fail.
467            let era_year = -year + 1;
468            (era_year, Era::BCE)
469        }
470    }

1119    /// ```

1239    /// Add the given span of time to this date. If the sum would overflow the
1240    /// minimum or maximum date values, then an error is returned.
1241    ///
1242    /// This operation accepts three different duration types: [`Span`],
1243    /// [`SignedDuration`] or [`std::time::Duration`]. This is achieved via
1244    /// `From` trait implementations for the [`DateArithmetic`] type.
1245    ///
1246    /// # Properties
1247    ///
1248    /// When adding a [`Span`] duration, this routine is _not_ reversible
1249    /// because some additions may be ambiguous. For example, adding `1 month`
1250    /// to the date `2024-03-31` will produce `2024-04-30` since April has only
1251    /// 30 days in a month. Conversely, subtracting `1 month` from `2024-04-30`
1252    /// will produce `2024-03-30`, which is not the date we started with.
1253    ///
1254    /// If spans of time are limited to units of days (or less), then this
1255    /// routine _is_ reversible. This also implies that all operations with
1256    /// a [`SignedDuration`] or a [`std::time::Duration`] are reversible.
1257    ///
1258    /// # Errors
1259    ///
1260    /// If the span added to this date would result in a date that exceeds the
1261    /// range of a `Date`, then this will return an error.
1262    ///
1263    /// # Examples
1264    ///
1265    /// This shows a few examples of adding spans of time to various dates.
1266    /// We make use of the [`ToSpan`](crate::ToSpan) trait for convenient
1267    /// creation of spans.
1268    ///
1269    /// ```
1270    /// use jiff::{civil::date, ToSpan};
1271    ///
1272    /// let d = date(2024, 3, 31);
1273    /// assert_eq!(d.checked_add(1.months())?, date(2024, 4, 30));
1274    /// // Adding two months gives us May 31, not May 30.
1275    /// let d = date(2024, 3, 31);
1276    /// assert_eq!(d.checked_add(2.months())?, date(2024, 5, 31));
1277    /// // Any time in the span that does not exceed a day is ignored.
1278    /// let d = date(2024, 3, 31);
1279    /// assert_eq!(d.checked_add(23.hours())?, date(2024, 3, 31));
1280    /// // But if the time exceeds a day, that is accounted for!
1281    /// let d = date(2024, 3, 31);
1282    /// assert_eq!(d.checked_add(28.hours())?, date(2024, 4, 1));
1283    ///
1284    /// # Ok::<(), Box<dyn std::error::Error>>(())

1355    #[inline]
1356    fn checked_add_span(self, span: &Span) -> Result<Date, Error> {
1357        if span.is_zero() {
1358            return Ok(self);
1359        }
1360        if span.units().contains_only(Unit::Day) {
1361            return self
1362                .inner
1363                .checked_add(span.get_days())
1364                .map(Date::from_jcore)
1365                .map_err(Error::jcore_range);
1366        }
1367
1368        let (month, years) =
1369            month_add_overflowing(self.month(), span.get_months());
1370        let year = b::Year::checked_add(self.year(), years)
1371            .and_then(|years| b::Year::checked_add(years, span.get_years()))?;
1372        let date = Date::new_constrain(year, month, self.day())?;
1373        let mut days = date
1374            .to_unix_epoch_day()
1375            .checked_add(7 * span.get_weeks())
1376            .and_then(|days| days.checked_add(span.get_days()))
1377            .map_err(Error::jcore_range)?;
1378        if !span.units().only_time().is_empty() {
1379            let time_days = b::UnixEpochDays::check(
1380                span.to_invariant_duration_time_only().as_civil_days(),
1381            )?;
1382            days = days.checked_add(time_days).map_err(Error::jcore_range)?;
1383        }
1384        Ok(Date::from_unix_epoch_day(days))
1385    }

1481    /// Returns a span representing the elapsed time from this date until
1482    /// the given `other` date.
1483    ///
1484    /// When `other` occurs before this date, then the span returned will be
1485    /// negative.
1486    ///
1487    /// Depending on the input provided, the span returned is rounded. It may
1488    /// also be balanced up to bigger units than the default. By default, the
1489    /// span returned is balanced such that the biggest and smallest possible
1490    /// unit is days.
1491    ///
1492    /// This operation is configured by providing a [`DateDifference`]
1493    /// value. Since this routine accepts anything that implements
1494    /// `Into<DateDifference>`, once can pass a `Date` directly. One
1495    /// can also pass a `(Unit, Date)`, where `Unit` is treated as
1496    /// [`DateDifference::largest`].
1497    ///
1498    /// # Properties
1499    ///
1500    /// It is guaranteed that if the returned span is subtracted from `other`,
1501    /// and if no rounding is requested, and if the largest unit request is at
1502    /// most `Unit::Day`, then the original date will be returned.
1503    ///
1504    /// This routine is equivalent to `self.since(other).map(|span| -span)`
1505    /// if no rounding options are set. If rounding options are set, then
1506    /// it's equivalent to
1507    /// `self.since(other_without_rounding_options).map(|span| -span)`,
1508    /// followed by a call to [`Span::round`] with the appropriate rounding
1509    /// options set. This is because the negation of a span can result in
1510    /// different rounding results depending on the rounding mode.
1511    ///
1512    /// # Errors
1513    ///
1514    /// An error can occur if `DateDifference` is misconfigured. For example,
1515    /// if the smallest unit provided is bigger than the largest unit.
1516    ///
1517    /// It is guaranteed that if one provides a date with the default
1518    /// [`DateDifference`] configuration, then this routine will never fail.
1519    ///
1520    /// # Examples
1521    ///
1522    /// ```
1523    /// use jiff::{civil::date, ToSpan};
1524    ///
1525    /// let earlier = date(2006, 8, 24);
1526    /// let later = date(2019, 1, 31);
1527    /// assert_eq!(earlier.until(later)?, 4543.days().fieldwise());
1528    ///
1529    /// // Flipping the dates is fine, but you'll get a negative span.
1530    /// let earlier = date(2006, 8, 24);
1531    /// let later = date(2019, 1, 31);
1532    /// assert_eq!(later.until(earlier)?, -4543.days().fieldwise());
1533    ///
1534    /// # Ok::<(), Box<dyn std::error::Error>>(())
1535    /// ```
1536    ///
1537    /// # Example: using bigger units
1538    ///
1539    /// This example shows how to expand the span returned to bigger units.
1540    /// This makes use of a `From<(Unit, Date)> for DateDifference` trait
1541    /// implementation.
1542    ///
1543    /// ```
1544    /// use jiff::{civil::date, Unit, ToSpan};
1545    ///
1546    /// let d1 = date(1995, 12, 07);
1547    /// let d2 = date(2019, 01, 31);
1548    ///
1549    /// // The default limits durations to using "days" as the biggest unit.
1550    /// let span = d1.until(d2)?;
1551    /// assert_eq!(span.to_string(), "P8456D");
1552    ///
1553    /// // But we can ask for units all the way up to years.
1554    /// let span = d1.until((Unit::Year, d2))?;
1555    /// assert_eq!(span.to_string(), "P23Y1M24D");
1556    ///
1557    /// # Ok::<(), Box<dyn std::error::Error>>(())
1558    /// ```
1559    ///
1560    /// # Example: rounding the result
1561    ///
1562    /// This shows how one might find the difference between two dates and
1563    /// have the result rounded to the nearest month.
1564    ///
1565    /// In this case, we need to hand-construct a [`DateDifference`]
1566    /// in order to gain full configurability.
1567    ///
1568    /// ```
1569    /// use jiff::{civil::{date, DateDifference}, Unit, ToSpan};
1570    ///
1571    /// let d1 = date(1995, 12, 07);
1572    /// let d2 = date(2019, 02, 06);
1573    ///
1574    /// let span = d1.until(DateDifference::from(d2).smallest(Unit::Month))?;
1575    /// assert_eq!(span, 277.months().fieldwise());
1576    ///
1577    /// // Or even include years to make the span a bit more comprehensible.
1578    /// let span = d1.until(
1579    ///     DateDifference::from(d2)
1580    ///         .smallest(Unit::Month)
1581    ///         .largest(Unit::Year),
1582    /// )?;
1583    /// // Notice that we are one day shy of 23y2m. Rounding spans computed
1584    /// // between dates uses truncation by default.
1585    /// assert_eq!(span, 23.years().months(1).fieldwise());
1586    ///
1587    /// # Ok::<(), Box<dyn std::error::Error>>(())
1588    /// ```
1589    ///
1590    /// # Example: units biggers than days inhibit reversibility
1591    ///
1592    /// If you ask for units bigger than days, then adding the span
1593    /// returned to the `other` date is not guaranteed to result in the
1594    /// original date. For example:
1595    ///
1596    /// ```
1597    /// use jiff::{civil::date, Unit, ToSpan};
1598    ///
1599    /// let d1 = date(2024, 3, 2);
1600    /// let d2 = date(2024, 5, 1);
1601    ///
1602    /// let span = d1.until((Unit::Month, d2))?;
1603    /// assert_eq!(span, 1.month().days(29).fieldwise());
1604    /// let maybe_original = d2.checked_sub(span)?;
1605    /// // Not the same as the original datetime!
1606    /// assert_eq!(maybe_original, date(2024, 3, 3));
1607    ///
1608    /// // But in the default configuration, days are always the biggest unit
1609    /// // and reversibility is guaranteed.
1610    /// let span = d1.until(d2)?;
1611    /// assert_eq!(span, 60.days().fieldwise());
1612    /// let is_original = d2.checked_sub(span)?;
1613    /// assert_eq!(is_original, d1);
1614    ///
1615    /// # Ok::<(), Box<dyn std::error::Error>>(())
1616    /// ```
1617    ///
1618    /// This occurs because spans are added as if by adding the biggest units
1619    /// first, and then the smaller units. Because months vary in length,
1620    /// their meaning can change depending on how the span is added. In this
1621    /// case, adding one month to `2024-03-02` corresponds to 31 days, but
1622    /// subtracting one month from `2024-05-01` corresponds to 30 days.

1625        self,
1626        other: A,
1627    ) -> Result<Span, Error> {
1628        let args: DateDifference = other.into();
1629        let span = args.since_with_largest_unit(self)?;
1630        if args.rounding_may_change_span() {
1631            span.round(args.round.relative(self))
1632        } else {
1633            Ok(span)
1634        }
1635    }

1673    /// Returns an absolute duration representing the elapsed time from this
1674    /// date until the given `other` date.
1675    ///
1676    /// When `other` occurs before this date, then the duration returned will
1677    /// be negative.
1678    ///
1679    /// Unlike [`Date::until`], this returns a duration corresponding to a
1680    /// 96-bit integer of nanoseconds between two dates. In this case of
1681    /// computing durations between civil dates where all days are assumed to
1682    /// be 24 hours long, the duration returned will always be divisible by
1683    /// 24 hours. (That is, `24 * 60 * 60 * 1_000_000_000` nanoseconds.)
1684    ///
1685    /// # Fallibility
1686    ///
1687    /// This routine never panics or returns an error. Since there are no
1688    /// configuration options that can be incorrectly provided, no error is
1689    /// possible when calling this routine. In contrast, [`Date::until`] can
1690    /// return an error in some cases due to misconfiguration. But like this
1691    /// routine, [`Date::until`] never panics or returns an error in its
1692    /// default configuration.
1693    ///
1694    /// # When should I use this versus [`Date::until`]?
1695    ///
1696    /// See the type documentation for [`SignedDuration`] for the section on
1697    /// when one should use [`Span`] and when one should use `SignedDuration`.
1698    /// In short, use `Span` (and therefore `Date::until`) unless you have a
1699    /// specific reason to do otherwise.
1700    ///
1701    /// # Example
1702    ///
1703    /// ```
1704    /// use jiff::{civil::date, SignedDuration};
1705    ///
1706    /// let earlier = date(2006, 8, 24);
1707    /// let later = date(2019, 1, 31);
1708    /// assert_eq!(
1709    ///     earlier.duration_until(later),
1710    ///     SignedDuration::from_hours(4543 * 24),
1711    /// );
1712    /// ```
1713    ///
1714    /// # Example: difference with [`Date::until`]
1715    ///
1716    /// The main difference between this routine and `Date::until` is that the
1717    /// latter can return units other than a 96-bit integer of nanoseconds.
1718    /// While a 96-bit integer of nanoseconds can be converted into other
1719    /// units like hours, this can only be done for uniform units. (Uniform
1720    /// units are units for which each individual unit always corresponds to
1721    /// the same elapsed time regardless of the datetime it is relative to.)
1722    /// This can't be done for units like years, months or days without a
1723    /// relative date.
1724    ///
1725    /// ```
1726    /// use jiff::{civil::date, SignedDuration, Span, SpanRound, ToSpan, Unit};
1727    ///
1728    /// let d1 = date(2024, 1, 1);
1729    /// let d2 = date(2025, 4, 1);
1730    ///
1731    /// let span = d1.until((Unit::Year, d2))?;
1732    /// assert_eq!(span, 1.year().months(3).fieldwise());
1733    ///
1734    /// let duration = d1.duration_until(d2);
1735    /// assert_eq!(duration, SignedDuration::from_hours(456 * 24));
1736    /// // There's no way to extract years or months from the signed
1737    /// // duration like one might extract hours (because every hour
1738    /// // is the same length). Instead, you actually have to convert
1739    /// // it to a span and then balance it by providing a relative date!
1740    /// let options = SpanRound::new().largest(Unit::Year).relative(d1);
1741    /// let span = Span::try_from(duration)?.round(options)?;
1742    /// assert_eq!(span, 1.year().months(3).fieldwise());
1743    ///
1744    /// # Ok::<(), Box<dyn std::error::Error>>(())
1745    /// ```
1746    ///
1747    /// # Example: getting an unsigned duration
1748    ///
1749    /// If you're looking to find the duration between two dates as a
1750    /// [`std::time::Duration`], you'll need to use this method to get a
1751    /// [`SignedDuration`] and then convert it to a `std::time::Duration`:
1752    ///
1753    /// ```
1754    /// use std::time::Duration;
1755    ///
1756    /// use jiff::{civil::date, SignedDuration};
1757    ///
1758    /// let d1 = date(2024, 7, 1);
1759    /// let d2 = date(2024, 8, 1);
1760    /// let duration = Duration::try_from(d1.duration_until(d2))?;
1761    /// assert_eq!(duration, Duration::from_secs(31 * 24 * 60 * 60));
1762    ///
1763    /// // Note that unsigned durations cannot represent all
1764    /// // possible differences! If the duration would be negative,
1765    /// // then the conversion fails:

2723    /// The biggest units allowed are determined by the `smallest` and
2724    /// `largest` settings, but defaults to `Unit::Day`.
2725    #[inline]
2726    fn since_with_largest_unit(&self, d1: Date) -> Result<Span, Error> {
2727        // N.B. A logical (but inefficient yet easier to understand)
2728        // description of the behavior here can be found in the Temporal spec:
2729        // https://tc39.es/proposal-temporal/#sec-temporal-calendardateuntil
2730        //
2731        // Immense discussion discussing and justifying this behavior:
2732        // https://github.com/tc39/proposal-temporal/issues/2535

2735        if smallest < Unit::Day {
2736            return Err(Error::from(UnitConfigError::CivilDate {
2737                given: smallest,

2739        }
2740

2742        if largest < Unit::Day {
2743            return Err(Error::from(UnitConfigError::CivilDate {
2744                given: largest,
2745            }));
2746        }
2747
2748        let d2 = self.date;
2749        if d1 == d2 {
2750            return Ok(Span::new());
2751        }
2752        if largest <= Unit::Week {
2753            let mut weeks: i32 = 0;
2754            let mut days = d1.until_days(d2);
2755            if largest == Unit::Week {
2756                weeks = days / 7;
2757                days %= 7;
2758            }
2759            return Ok(Span::new().weeks(weeks).days(days));
2760        }
2761
2762        let year1 = d1.year();
2763        let month1 = d1.month();
2764        let day1 = d1.day();
2765        let mut year2 = d2.year();
2766        let mut month2 = d2.month();
2767        let day2 = d2.day();
2768
2769        let mut years = year2 - year1;
2770        let mut months = i32::from(month2 - month1);
2771        let mut days = i32::from(day2 - day1);
2772        if years != 0 || months != 0 {
2773            let sign = if years != 0 {
2774                Sign::from(years)
2775            } else {
2776                Sign::from(months)
2777            };
2778            let mut days_in_month2 = d2.days_in_month();
2779            let mut day_correct = 0;
2780            if Sign::from(days) == -sign {
2781                // Justifying this operation as infallible is quite tricky.
2782                // It can only fail in two cases: year2=9999, month2=12 and
2783                // sign=Negative or year2=-9999, month2=1 and sign=Positive. In
2784                // both of those cases, year will overflow its boundaries.
2785                //
2786                // However, neither case is possible. Namely, whenever
2787                // sign=Negative, it follows that either `year1 > year2` or
2788                // `month1 > month2`. But there are no such values of `year1`
2789                // or `month1` that satisfy `year1 > 9999 or month1 > 12`
2790                // because `9999` is the maximum legal year value.
2791                //
2792                // A similar argument follows for sign=Positive and
2793                // `year2=-9999` and `month2=1`. Therefore, this `unwrap()` is
2794                // okay.
2795                let (y, m) = month_add_one(year2, month2, -sign).unwrap();
2796                year2 = y;
2797                month2 = m;
2798
2799                years = year2 - year1;
2800                months = i32::from(month2 - month1);
2801                let original_days_in_month1 = days_in_month2;
2802                days_in_month2 = jcore::civil::days_in_month(year2, month2);
2803                day_correct = if sign.is_negative() {
2804                    -original_days_in_month1
2805                } else {
2806                    days_in_month2
2807                };
2808            }
2809
2810            let day0_trunc = i32::from(day1.min(days_in_month2));
2811            days = i32::from(day2) - day0_trunc + i32::from(day_correct);
2812
2813            if years != 0 {
2814                months = i32::from(month2 - month1);
2815                if Sign::from(months) == -sign {
2816                    let month_correct = sign * 12;
2817                    year2 -= sign.as_i16();
2818                    years = year2 - year1;
2819
2820                    months = i32::from(month2 - month1) + month_correct;
2821                }
2822            }
2823        }
2824        if largest == Unit::Month && years != 0 {
2825            months =
2826                b::SpanMonths::checked_add(months, i32::from(years) * 12)?;
2827            years = 0;
2828        }
2829        Ok(Span::new().years(years).months(months).days(days))
2830    }
