82/// Target platform support
83pub(crate) trait Target: Sized {
84    /// Return a semicolon-delimited string of language/COUNTRY codes.
85    fn lang_prefs(self) -> Result<LanguagePreferences>;
86    /// Return the user's "real" / "full" name.
87    fn realname(self) -> Result<OsString>;
88    /// Return the user's username.
89    fn username(self) -> Result<OsString>;
90    /// Return the computer's "fancy" / "pretty" name.
91    fn devicename(self) -> Result<OsString>;
92    /// Return the computer's hostname.
93    fn hostname(self) -> Result<String>;
94    /// Return the OS distribution's name.
95    fn distro(self) -> Result<String>;
96    /// Return the desktop environment.
97    fn desktop_env(self) -> Option<DesktopEnvironment>;
98    /// Return the target platform.
99    fn platform(self) -> Platform;
100    /// Return the computer's CPU architecture.
101    fn arch(self) -> Result<CpuArchitecture>;
102
103    /// Return the user's account name (usually just the username, but may
104    /// include an account server hostname).
105    fn account(self) -> Result<OsString> {
106        self.username()
107    }
108}

110// This is only used on some platforms
111#[cfg(feature = "std")]
112#[allow(dead_code)]
113fn unix_lang() -> Result<LanguagePreferences> {
114    use std::{
115        env::{self, VarError},
116        str::FromStr,
117        vec::Vec,
118    };
119
120    use crate::{Error, Language};
121
122    let env_var = |var: &str| match env::var(var) {
123        Ok(value) => Ok(if value.is_empty() { None } else { Some(value) }),
124        Err(VarError::NotPresent) => Ok(None),
125        Err(VarError::NotUnicode(_)) => {
126            Err(Error::with_invalid_data("not unicode"))
127        }
128    };
129
130    // Uses priority defined in
131    // <https://www.gnu.org/software/gettext/manual/html_node/Locale-Environment-Variables.html>
132    let lc_all = env_var("LC_ALL")?;
133    let lang = env_var("LANG")?;
134
135    if lang.is_none() && lc_all.is_none() {
136        return Err(Error::empty_record());
137    }
138
139    // Standard locales that have a higher global precedence than their specific
140    // counterparts, indicating that one should not perform any localization.
141    // https://www.gnu.org/software/libc/manual/html_node/Standard-Locales.html
142    if let Some(l) = &lang {
143        if l == "C" || l == "POSIX" {
144            return Ok(LanguagePreferences {
145                fallbacks: Vec::new(),
146                ..Default::default()
147            });
148        }
149    }
150
151    // The LANGUAGE environment variable takes precedence if and only if
152    // localization is enabled, i.e., LC_ALL / LANG is not "C" or "POSIX".
153    // <https://www.gnu.org/software/gettext/manual/html_node/The-LANGUAGE-variable.html>
154    if let Some(language) = env_var("LANGUAGE")? {
155        return Ok(LanguagePreferences {
156            fallbacks: language
157                .split(":")
158                .map(Language::from_str)
159                .collect::<Result<_>>()?,
160            ..Default::default()
161        });
162    }
163
164    // All fields other than LANGUAGE can only contain a single value, so we
165    // don't need to perform any splitting at this point.
166    let lang_from_var = |var| -> Result<Option<Language>> {
167        env_var(var)?.as_deref().map(Language::from_str).transpose()
168    };
169
170    Ok(LanguagePreferences {
171        fallbacks: lang
172            .as_ref()
173            .map(|l| -> Result<_> { Ok([Language::from_str(l)?].to_vec()) })
174            .transpose()?
175            .unwrap_or(Vec::new()),
176        collation: lang_from_var("LC_COLLATE")?,
177        char_classes: lang_from_var("LC_CTYPE")?,
178        monetary: lang_from_var("LC_MONETARY")?,
179        messages: lang_from_var("LC_MESSAGES")?,
180        numeric: lang_from_var("LC_NUMERIC")?,
181        time: lang_from_var("LC_TIME")?,
182    })
183}
