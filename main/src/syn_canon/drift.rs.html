12/// `group` with its stream mapped through `f`, delimiter and span kept.
13pub(crate) fn map_group(group: &Group, f: impl FnOnce(TokenStream) -> TokenStream) -> TokenTree {
14    let mut rebuilt = Group::new(group.delimiter(), f(group.stream()));
15    rebuilt.set_span(group.span());
16    TokenTree::Group(rebuilt)
17}

19/// `lit` in its canonical decimal spelling.
20pub(crate) fn canonical_literal(lit: proc_macro2::Literal) -> TokenTree {
21    let span = lit.span();
22    let rebuilt = match syn::Lit::new(lit) {
23        syn::Lit::Int(i) => {
24            let decimal = ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{0}{1}", i.base10_digits(),
                i.suffix()))
    })format!("{}{}", i.base10_digits(), i.suffix());
25            match syn::parse_str::<syn::LitInt>(&decimal) {
26                Ok(mut read)
27                    if read.base10_digits() == i.base10_digits() && read.suffix() == i.suffix() =>
28                {
29                    read.set_span(span);
30                    syn::Lit::Int(read)
31                }
32                // The suffix reads as a radix prefix or an exponent after the digits, `0b0buu`.
33                _ => syn::Lit::Int(i),
34            }
35        }
36        syn::Lit::Float(f) => {
37            let digits = canon_float_str(f.base10_digits());
38            syn::Lit::Float(syn::LitFloat::new(&::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{1}{0}", f.suffix(), digits))
    })format!("{digits}{}", f.suffix()), span))
39        }
40        syn::Lit::Str(s) => syn::Lit::Str(syn::LitStr::new(&s.value(), span)),
41        syn::Lit::ByteStr(b) => syn::Lit::ByteStr(syn::LitByteStr::new(&b.value(), span)),
42        syn::Lit::CStr(c) => syn::Lit::CStr(syn::LitCStr::new(c.value().as_c_str(), span)),
43        syn::Lit::Char(c) => syn::Lit::Char(syn::LitChar::new(c.value(), span)),
44        syn::Lit::Byte(b) => syn::Lit::Byte(syn::LitByte::new(b.value(), span)),
45        other => other,
46    };
47    rebuilt
48        .to_token_stream()
49        .into_iter()
50        .next()
51        .expect("rebuilt literal produces at least one token")
52}

54/// Canonical decimal string for a float's digit part, text only so the
55/// canonical form never depends on the feature set. `syn` has already
56/// dropped the underscores, a `+` exponent sign and an upper case `E`.
57fn canon_float_str(s: &str) -> String {
58    let (mantissa, exponent) = match s.split_once('e') {
59        Some((mantissa, exponent)) => (mantissa, Some(exponent)),
60        None => (s, None),
61    };
62    let (whole, fraction) = mantissa.split_once('.').unwrap_or((mantissa, ""));
63    let fraction = fraction.trim_end_matches('0');
64    let mut out = String::with_capacity(s.len());
65    out.push_str(whole);
66    out.push('.');
67    out.push_str(if fraction.is_empty() { "0" } else { fraction });
68    if let Some(exponent) = exponent {
69        out.push('e');
70        let (sign, digits) = exponent
71            .strip_prefix('-')
72            .map_or(("", exponent), |digits| ("-", digits));
73        out.push_str(sign);
74        let digits = digits.trim_start_matches('0');
75        out.push_str(if digits.is_empty() { "0" } else { digits });
76    }
77    out
78}

391    /// Fold a function body's tail `return`, and its tail semicolon when the
392    /// signature makes the value `()`, before and after the return fold since
393    /// each can expose the other.
394    fn fold_fn_body(&self, sig: &syn::Signature, block: &mut syn::Block) {
395        let unit = self.compiles && #[allow(non_exhaustive_omitted_patterns)] match sig.output {
    syn::ReturnType::Default => true,
    _ => false,
}matches!(sig.output, syn::ReturnType::Default);
396        if unit {
397            drop_tail_semicolon(&mut block.stmts);
398        }
399        fold_tail_return(&mut block.stmts);
400        if unit {
401            drop_tail_semicolon(&mut block.stmts);
402        }
403    }

406/// The block of a loop, or the then block of an `if` chain without a final
407/// `else`, whose value rustc requires to be `()`.
408fn unit_block(expr: &mut syn::Expr) -> Option<&mut syn::Block> {
409    match expr {
410        syn::Expr::ForLoop(e) => Some(&mut e.body),
411        syn::Expr::While(e) => Some(&mut e.body),
412        syn::Expr::Loop(e) => Some(&mut e.body),
413        syn::Expr::If(e) if discards_value(e) => Some(&mut e.then_branch),
414        _ => None,
415    }
416}

418/// Whether the `if` chain from `expr_if` ends without an `else` block.
419fn discards_value(mut expr_if: &syn::ExprIf) -> bool {
420    loop {
421        match expr_if.else_branch.as_ref().map(|(_, e)| e.as_ref()) {
422            None => return true,
423            Some(syn::Expr::If(inner)) => expr_if = inner,
424            Some(_) => return false,
425        }
426    }
427}

429/// Drop the `mut` of a pattern that is one binding, by-value whatever the
430/// scrutinee. Nested under a reference, `mut` resets a borrowing binding mode
431/// before edition 2024, so a nested one stays.
432fn drop_binding_mut(pat: &mut syn::Pat) {
433    let pat = match pat {
434        syn::Pat::Type(typed) => typed.pat.as_mut(),
435        syn::Pat::Guard(guarded) => guarded.pat.as_mut(),
436        other => other,
437    };
438    if let syn::Pat::Ident(ident) = pat
439        && ident.by_ref.is_none()
440    {
441        ident.mutability = None;
442    }
443}

445/// Drop the semicolon of a tail statement whose value is `()` either way, the
446/// rewrite clippy's `semicolon_if_nothing_returned` makes in reverse.
447fn drop_tail_semicolon(stmts: &mut [syn::Stmt]) {
448    match stmts.last_mut() {
449        Some(syn::Stmt::Expr(_, semi)) => *semi = None,
450        Some(syn::Stmt::Macro(mac)) => mac.semi_token = None,
451        _ => {}
452    }
453}

455/// Drop the semicolon after every non-tail statement ending in a block, an
456/// empty statement rustc's `redundant_semicolons` reports.
457fn drop_block_statement_semicolons(stmts: &mut [syn::Stmt]) {
458    let Some((_, rest)) = stmts.split_last_mut() else {
459        return;
460    };
461    for stmt in rest {
462        if let syn::Stmt::Expr(expr, semi) = stmt
463            && #[allow(non_exhaustive_omitted_patterns)] match expr {
    syn::Expr::If(_) | syn::Expr::Match(_) | syn::Expr::Block(_) |
        syn::Expr::Unsafe(_) | syn::Expr::While(_) | syn::Expr::Loop(_) |
        syn::Expr::ForLoop(_) | syn::Expr::TryBlock(_) | syn::Expr::Const(_)
        => true,
    _ => false,
}matches!(
464                expr,
465                syn::Expr::If(_)
466                    | syn::Expr::Match(_)
467                    | syn::Expr::Block(_)
468                    | syn::Expr::Unsafe(_)
469                    | syn::Expr::While(_)
470                    | syn::Expr::Loop(_)
471                    | syn::Expr::ForLoop(_)
472                    | syn::Expr::TryBlock(_)
473                    | syn::Expr::Const(_)
474            )
475        {
476            *semi = None;
477        }
478    }
479}

481/// An arm body `{ expr }` becomes `expr`, the form rustfmt writes.
482fn unwrap_arm_block(arm: &mut syn::Arm) {
483    unwrap_single_expr_block(&mut arm.body);

486/// A closure or arm body `{ expr }` becomes `expr` when the block holds
487/// exactly one tail expression and carries no label or attributes.
488fn unwrap_single_expr_block(body: &mut Box<syn::Expr>) {
489    let syn::Expr::Block(block) = body.as_mut() else {
490        return;
491    };
492    if block.label.is_some()
493        || !block.attrs.is_empty()
494        || !#[allow(non_exhaustive_omitted_patterns)] match block.block.stmts.as_slice() {
    [syn::Stmt::Expr(_, None)] => true,
    _ => false,
}matches!(block.block.stmts.as_slice(), [syn::Stmt::Expr(_, None)])
495    {
496        return;
497    }
498    if let Some(syn::Stmt::Expr(expr, None)) = block.block.stmts.pop() {
499        **body = expr;
500    }
501}

503/// An `else { if … }` whose block holds only that `if`, with no attribute
504/// on it, becomes `else if …`. An else block itself never carries a label
505/// or attributes.
506fn collapse_else_if(expr_if: &mut syn::ExprIf) {
507    let Some((_, otherwise)) = &mut expr_if.else_branch else {
508        return;
509    };
510    let syn::Expr::Block(block) = otherwise.as_mut() else {
511        return;
512    };
513    if !#[allow(non_exhaustive_omitted_patterns)] match block.block.stmts.as_slice() {
    [syn::Stmt::Expr(syn::Expr::If(inner), None)] if inner.attrs.is_empty() =>
        true,
    _ => false,
}matches!(block.block.stmts.as_slice(), [syn::Stmt::Expr(syn::Expr::If(inner), None)] if inner.attrs.is_empty())
514    {
515        return;
516    }
517    if let Some(syn::Stmt::Expr(inner, None)) = block.block.stmts.pop() {
518        **otherwise = inner;
519    }
520}

522/// Remove every bare empty statement from `stmts`.
523fn drop_empty_stmts(stmts: &mut Vec<syn::Stmt>) {
524    stmts.retain(|stmt| {
525        !#[allow(non_exhaustive_omitted_patterns)] match stmt {
    syn::Stmt::Expr(syn::Expr::Verbatim(v), Some(_)) if v.is_empty() => true,
    _ => false,
}matches!(
526            stmt,
527            syn::Stmt::Expr(syn::Expr::Verbatim(v), Some(_)) if v.is_empty()
528        )
529    });

532/// Replace a tail `return expr`, with or without the semicolon, with
533/// `expr`, then propagate through every tail position forwarding its value,
534/// an `if` only with an `else` clause because rustc rejects a valued return
535/// in a discarded then position (`E0317`). A tail unit `return` adds
536/// nothing and goes. A return keeping a live attribute, `#[cfg]` for
537/// instance, must not fold, the fold would drop the condition.
538fn fold_tail_return(stmts: &mut Vec<syn::Stmt>) {
539    let Some(last) = stmts.last_mut() else {
540        return;
541    };
542    match last {
543        syn::Stmt::Expr(syn::Expr::Return(ret), _) => {
544            strip_inert_attrs(&mut ret.attrs);
545            if !ret.attrs.is_empty() {
546                return;
547            }
548            if let Some(mut inner) = ret.expr.take() {
549                fold_tail_expr(&mut inner);
550                *last = syn::Stmt::Expr(*inner, None);
551            } else {
552                stmts.pop();
553                fold_tail_return(stmts);
554            }
555        }
556        syn::Stmt::Expr(expr, None) => fold_tail_expr(expr),
557        _ => {}
558    }
559}
560
561/// Continue the fold through an expression whose tail value is the enclosing
562/// tail's value.
563fn fold_tail_expr(expr: &mut syn::Expr) {
564    match expr {
565        syn::Expr::Block(block) => fold_tail_return(&mut block.block.stmts),
566        syn::Expr::Unsafe(block) => fold_tail_return(&mut block.block.stmts),
567        syn::Expr::If(expr_if) if expr_if.else_branch.is_some() => {
568            fold_tail_return(&mut expr_if.then_branch.stmts);
569            if let Some((_, otherwise)) = &mut expr_if.else_branch {
570                fold_tail_expr(otherwise);
571            }
572        }
573        syn::Expr::Match(expr_match) => {
574            for arm in &mut expr_match.arms {
575                fold_tail_expr(&mut arm.body);
576                unwrap_single_expr_block(&mut arm.body);
577            }
578        }
579        // An arm body `return value`, written bare or as a block the arm fold unwrapped.
580        syn::Expr::Return(ret) => {
581            strip_inert_attrs(&mut ret.attrs);
582            if ret.attrs.is_empty()
583                && let Some(inner) = ret.expr.take()
584            {
585                *expr = *inner;
586                fold_tail_expr(expr);
587            }
588        }
589        _ => {}
590    }
591}

648/// Move the bounds written on type and lifetime parameters into the `where`
649/// clause, ahead of its own predicates and in parameter order, the place
650/// they would take written there.
651fn bounds_to_where(generics: &mut syn::Generics) {
652    let mut moved = Vec::new();
653    for param in &mut generics.params {
654        match param {
655            syn::GenericParam::Type(ty) if !ty.bounds.is_empty() => {
656                ty.colon_token = None;
657                moved.push(syn::WherePredicate::Type(syn::PredicateType {
658                    attrs: Vec::new(),
659                    lifetimes: None,
660                    bounded_ty: syn::Type::Path(syn::TypePath {
661                        attrs: Vec::new(),
662                        qself: None,
663                        path: ty.ident.clone().into(),
664                    }),
665                    colon_token: syn::token::Colon::default(),
666                    bounds: core::mem::take(&mut ty.bounds),
667                }));
668            }
669            syn::GenericParam::Lifetime(lifetime) if !lifetime.bounds.is_empty() => {
670                lifetime.colon_token = None;
671                moved.push(syn::WherePredicate::Lifetime(syn::PredicateLifetime {
672                    attrs: Vec::new(),
673                    lifetime: lifetime.lifetime.clone(),
674                    colon_token: syn::token::Colon::default(),
675                    bounds: core::mem::take(&mut lifetime.bounds),
676                }));
677            }
678            _ => {}
679        }
680    }
681    if moved.is_empty() {
682        return;
683    }
684    let clause = generics.make_where_clause();
685    let written = core::mem::take(&mut clause.predicates);
686    clause.predicates.extend(moved);
687    clause.predicates.extend(written);
688}

710/// order and grouping carry no meaning.

712    "Clone",
713    "Copy",
714    "Debug",

718    "Ord",

723/// Whether `attrs` derive anything outside the std derives, a proc macro that reads the item.
724fn has_macro_derive(attrs: &[syn::Attribute]) -> bool {
725    use syn::parse::Parser;
726
727    let parser = syn::punctuated::Punctuated::<syn::Path, ::syn::token::Commasyn::Token![,]>::parse_terminated;
728    attrs
729        .iter()
730        .filter(|attr| attr.path().is_ident("derive"))
731        .any(|attr| {
732            let syn::Meta::List(list) = &attr.meta else {
733                return true;
734            };
735            parser.parse2(list.tokens.clone()).map_or(true, |paths| {
736                paths.iter().any(|path| {
737                    path.get_ident()
738                        .is_none_or(|name| !STD_DERIVES.iter().any(|std| name == std))
739                })
740            })
741        })
742}

744/// Merge an item's `derive` attributes into one sorted list at the place of
745/// the first, when every derive among them is a std one. A proc macro derive
746/// can depend on what ran before it, so any other name keeps them as written.
747fn fold_std_derives(attrs: &mut Vec<syn::Attribute>) {
748    use syn::parse::Parser;
749
750    let mut names: Vec<syn::Ident> = Vec::new();
751    let mut first = None;
752    for (index, attr) in attrs.iter().enumerate() {
753        if !attr.path().is_ident("derive") {
754            continue;
755        }
756        let syn::Meta::List(list) = &attr.meta else {
757            return;
758        };
759        let parser = syn::punctuated::Punctuated::<syn::Path, ::syn::token::Commasyn::Token![,]>::parse_terminated;
760        let Ok(paths) = parser.parse2(list.tokens.clone()) else {
761            return;
762        };
763        for path in paths {
764            match path.get_ident() {
765                Some(name) if STD_DERIVES.contains(&name.to_string().as_str()) => {
766                    names.push(name.clone());
767                }
768                _ => return,
769            }
770        }
771        first.get_or_insert(index);
772    }
773    let Some(first) = first else {
774        return;
775    };
776    names.sort_by_cached_key(ToString::to_string);
777    attrs.retain(|attr| !attr.path().is_ident("derive"));
778    attrs.insert(first, ::syn::__private::parse_quote({
        let mut _s = ::quote::__private::TokenStream::new();
        ::quote::__private::push_pound(&mut _s);
        ::quote::__private::push_group(&mut _s,
            ::quote::__private::Delimiter::Bracket,
            {
                let mut _s = ::quote::__private::TokenStream::new();
                ::quote::__private::push_ident(&mut _s, "derive");
                ::quote::__private::push_group(&mut _s,
                    ::quote::__private::Delimiter::Parenthesis,
                    {
                        let mut _s = ::quote::__private::TokenStream::new();
                        {
                            use ::quote::__private::ext::*;
                            let mut _first = true;
                            let has_iter = ::quote::__private::HasIterator::<false>;
                            #[allow(unused_mut)]
                            let (mut names, i) = names.quote_into_iter();
                            let has_iter = has_iter | i;
                            <_ as
                                    ::quote::__private::CheckHasIterator<true>>::check(has_iter);
                            while true {
                                let names =
                                    match names.next() {
                                        Some(_x) => ::quote::__private::RepInterp(_x),
                                        None => break,
                                    };
                                if !_first { ::quote::__private::push_comma(&mut _s); }
                                _first = false;
                                ::quote::ToTokens::to_tokens(&names, &mut _s);
                            }
                        }
                        _s
                    });
                _s
            });
        _s
    })syn::parse_quote!(#[derive(#(#names),*)]));
779}

872/// Replace the `use` entries of `list` by one sorted item per leaf path,
873/// placed at the front.
874fn hoist_uses<T>(
875    list: &mut Vec<T>,
876    into_use: impl Fn(T) -> Result<syn::ItemUse, T>,
877    wrap: impl Fn(syn::ItemUse) -> T,
878) {
879    let mut leaves = Vec::new();
880    let rest: Vec<T> = core::mem::take(list)
881        .into_iter()
882        .filter_map(|entry| {
883            let item = match into_use(entry) {
884                Ok(item) => item,
885                Err(entry) => return Some(entry),
886            };
887            let mut trees = Vec::new();
888            flatten_use_tree(&[], item.tree, &mut trees);
889            leaves.extend(trees.into_iter().map(|tree| syn::ItemUse {
890                attrs: item.attrs.clone(),
891                vis: item.vis.clone(),
892                use_token: item.use_token,
893                leading_colon: item.leading_colon,
894                tree,
895                semi_token: item.semi_token,
896            }));
897            None
898        })
899        .collect();
900    leaves.sort_by_cached_key(|leaf| {
901        ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{0} {1}",
                leaf.vis.to_token_stream(), leaf.tree.to_token_stream()))
    })format!(
902            "{} {}",
903            leaf.vis.to_token_stream(),
904            leaf.tree.to_token_stream()
905        )
906    });
907    list.extend(leaves.into_iter().map(wrap));
908    list.extend(rest);
909}

911/// Hoist non-`use`, non-`macro_rules!` item statements that appear before
912/// the first `macro_rules!` definition in a block to the front, right
913/// after the sorted `use` items placed there by `hoist_uses`.
914///
915/// Only items whose attributes `strip_inert_attrs` would all remove are
916/// hoisted. A live attribute may reference a local binding, and hoisting
917/// it before that binding changes the visit order and breaks alpha
918/// renaming.
919///
920/// Items in that region move to the front in their original relative order
921/// among items. Non-item statements follow them in their original relative
922/// order. Everything from the first `Stmt::Item(Item::Macro)` onward is
923/// untouched, because `macro_rules!` visibility is sequential.
924fn hoist_block_items(stmts: &mut Vec<syn::Stmt>) {
925    let use_count = stmts.partition_point(|s| #[allow(non_exhaustive_omitted_patterns)] match s {
    syn::Stmt::Item(syn::Item::Use(_)) => true,
    _ => false,
}matches!(s, syn::Stmt::Item(syn::Item::Use(_))));
926
927    let rel_macro = stmts[use_count..]
928        .iter()
929        .position(|s| #[allow(non_exhaustive_omitted_patterns)] match s {
    syn::Stmt::Item(syn::Item::Macro(_)) => true,
    _ => false,
}matches!(s, syn::Stmt::Item(syn::Item::Macro(_))));
930    let hoist_end = use_count + rel_macro.unwrap_or(stmts.len() - use_count);
931
932    if hoist_end == use_count {
933        return;
934    }
935
936    let tail = stmts.split_off(hoist_end);
937    let region = stmts.split_off(use_count);
938
939    let mut items = Vec::new();
940    let mut rest = Vec::new();
941    for mut stmt in region {
942        let hoistable = match &mut stmt {
943            syn::Stmt::Item(item) => {
944                item_attrs(item).is_none_or(|attrs| attrs.iter().all(is_inert_attr))
945            }
946            _ => false,
947        };
948        if hoistable {
949            items.push(stmt);
950        } else {
951            rest.push(stmt);
952        }
953    }
954
955    stmts.extend(items);
956    stmts.extend(rest);
957    stmts.extend(tail);
958}

960/// Split `tree` into its leaf paths, `a::{self}` becoming `a`.
961fn flatten_use_tree(prefix: &[syn::Ident], tree: syn::UseTree, out: &mut Vec<syn::UseTree>) {
962    match tree {
963        syn::UseTree::Path(path) => {
964            let mut prefix = prefix.to_vec();
965            prefix.push(path.ident);
966            flatten_use_tree(&prefix, *path.tree, out);
967        }
968        syn::UseTree::Group(group) => {
969            for item in group.items {
970                flatten_use_tree(prefix, item, out);
971            }
972        }
973        syn::UseTree::Name(name) if name.ident == "self" => {
974            if let [head @ .., last] = prefix {
975                let leaf = syn::UseTree::Name(syn::UseName {
976                    ident: last.clone(),
977                });
978                out.push(path_tree(head, leaf));
979            }
980        }
981        leaf => out.push(path_tree(prefix, leaf)),
982    }
983}
984
985fn path_tree(prefix: &[syn::Ident], leaf: syn::UseTree) -> syn::UseTree {
986    match prefix {
987        [] => leaf,
988        [head, rest @ ..] => syn::UseTree::Path(syn::UsePath {
989            ident: head.clone(),
990            colon2_token: syn::token::PathSep::default(),
991            tree: Box::new(path_tree(rest, leaf)),
992        }),
993    }
994}

1005/// The format string's index among a std formatting macro's arguments, and
1006/// whether the macro is a panic, whose lone literal before edition 2021 is
1007/// no format string.
1008pub(crate) fn format_operand(path: &syn::Path) -> Option<(usize, bool)> {
1009    let name = path.segments.last()?.ident.to_string();
1010    Some(match name.as_str() {
1011        "print" | "println" | "eprint" | "eprintln" | "format" | "format_args" => (0, false),
1012        "write" | "writeln" => (1, false),
1013        "panic" | "unreachable" | "todo" | "unimplemented" => (0, true),
1014        "assert" | "debug_assert" => (1, true),
1015        "assert_eq" | "assert_ne" | "debug_assert_eq" | "debug_assert_ne" => (2, true),
1016        _ => return None,
1017    })
1018}

1020/// Move identifier arguments of a formatting macro into its format string,
1021/// `println!("{}", x)` becoming `println!("{x}")`, in `mac` and in every
1022/// formatting call among its arguments. Returns whether `mac` changed.
1023fn inline_format_args(mac: &mut syn::Macro) -> bool {
1024    use syn::parse::Parser;
1025
1026    let parser = syn::punctuated::Punctuated::<syn::Expr, ::syn::token::Commasyn::Token![,]>::parse_terminated;
1027    let Ok(mut args) = parser.parse2(mac.tokens.clone()) else {
1028        return false;
1029    };
1030    let mut nested = NestedFormat(false);
1031    for arg in &mut args {
1032        nested.visit_expr_mut(arg);
1033    }
1034    let inlined = inline_positional(&mac.path, &mut args);
1035    if nested.0 || inlined {
1036        mac.tokens = args.to_token_stream();
1037    }
1038    nested.0 || inlined
1039}

1042/// drift visit never enters.
1043struct NestedFormat(bool);
1044
1045impl VisitMut for NestedFormat {
1046    fn visit_macro_mut(&mut self, mac: &mut syn::Macro) {
1047        self.0 |= inline_format_args(mac);
1048    }

1051/// Inline the identifier arguments of `args` when `path` names a non-panic
1052/// formatting macro. Returns whether anything moved.
1053fn inline_positional(
1054    path: &syn::Path,
1055    args: &mut syn::punctuated::Punctuated<syn::Expr, ::syn::token::Commasyn::Token![,]>,
1056) -> bool {
1057    let Some((at, false)) = format_operand(path) else {
1058        return false;
1059    };
1060    let positional: Vec<&syn::Expr> = args.iter().skip(at + 1).collect();
1061    if positional
1062        .iter()
1063        .any(|arg| #[allow(non_exhaustive_omitted_patterns)] match arg {
    syn::Expr::Assign(_) => true,
    _ => false,
}matches!(arg, syn::Expr::Assign(_)))
1064    {
1065        return false;
1066    }
1067    let names: Vec<Option<String>> = positional.into_iter().map(capturable).collect();
1068    let Some(syn::Expr::Lit(syn::ExprLit {
1069        lit: syn::Lit::Str(format),
1070        ..
1071    })) = args.iter_mut().nth(at)
1072    else {
1073        return false;
1074    };
1075    let Some(text) = inline_placeholders(&format.value(), &names) else {
1076        return false;
1077    };
1078    *format = syn::LitStr::new(&text, format.span());
1079    let kept = core::mem::take(args)
1080        .into_iter()
1081        .enumerate()
1082        .filter(|(index, _)| *index <= at || names[index - at - 1].is_none())
1083        .map(|(_, arg)| arg);
1084    args.extend(kept);
1085    true
1086}

1088/// The identifier a format string captures for `arg`, `None` for any other
1089/// expression and for the names a placeholder cannot spell.
1090fn capturable(arg: &syn::Expr) -> Option<String> {
1091    let syn::Expr::Path(path) = arg else {
1092        return None;
1093    };
1094    if !path.attrs.is_empty() {
1095        return None;
1096    }
1097    let name = path.path.get_ident()?.to_string();
1098    let spellable =
1099        !name.starts_with("r#") && !#[allow(non_exhaustive_omitted_patterns)] match name.as_str() {
    "self" | "Self" | "crate" | "super" => true,
    _ => false,
}matches!(name.as_str(), "self" | "Self" | "crate" | "super");
1100    spellable.then_some(name)
1101}

1103/// `format` with each implicit `{}` or `{:spec}` placeholder whose argument
1104/// in `names` is an identifier spelled inline. `None` when nothing moves, or
1105/// when an explicit index, a `$` or `*` spec or a count mismatch makes the
1106/// argument order matter.
1107fn inline_placeholders(format: &str, names: &[Option<String>]) -> Option<String> {
1108    let mut out = String::with_capacity(format.len());
1109    let mut next = 0;
1110    let mut chars = format.chars().peekable();
1111    while let Some(c) = chars.next() {
1112        match c {
1113            '{' | '}' if chars.peek() == Some(&c) => {
1114                chars.next();
1115                out.push(c);
1116                out.push(c);
1117            }
1118            '{' => {
1119                let mut body = String::new();
1120                loop {
1121                    match chars.next()? {
1122                        '}' => break,
1123                        c => body.push(c),
1124                    }
1125                }
1126                let (arg, spec) = body
1127                    .split_once(':')
1128                    .map_or((body.as_str(), None), |(a, s)| (a, Some(s)));
1129                let explicit_index = !arg.is_empty() && arg.bytes().all(|b| b.is_ascii_digit());
1130                if explicit_index || spec.is_some_and(|spec| spec.contains(['$', '*'])) {
1131                    return None;
1132                }
1133                out.push('{');
1134                if arg.is_empty() {
1135                    out.push_str(names.get(next)?.as_deref().unwrap_or_default());
1136                    next += 1;
1137                } else {
1138                    out.push_str(arg);
1139                }
1140                if let Some(spec) = spec {
1141                    out.push(':');
1142                    out.push_str(spec);
1143                }
1144                out.push('}');
1145            }
1146            '}' => return None,
1147            c => out.push(c),
1148        }
1149    }
1150    (next == names.len() && names.iter().any(Option::is_some)).then_some(out)
1151}
