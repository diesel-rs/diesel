1//! Alpha-renaming of local binders in a parsed file.
2//!
3//! Every identifier the file itself binds (`let` patterns, function
4//! names and parameters, closures, generics, lifetimes, local types,
5//! `const`/`static` items, local `macro_rules!` names, and `use`
6//! aliases) is rewritten to a positional name such as `_canon_0`, so
7//! two files that differ only in local names collapse to the same
8//! canonical form. Free identifiers, field names, method names,
9//! attribute paths, and string literals are never rewritten.
10

24/// A canonical binder name, positional in the deterministic traversal.
25fn canonical(counter: &mut usize) -> String {
26    let name = ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("_canon_{0}", counter))
    })format!("_canon_{counter}");
27    *counter += 1;
28    name
29}

33enum Ns {
34    /// Variables, function names, const items.
35    Value,
36    /// Types, traits, and the `Self` alias of an impl or trait.
37    Type,
38    /// Lifetimes.
39    Lifetime,
40    /// Loop and block labels, a namespace rustc keeps distinct from lifetimes.
41    Label,
42    /// Local macro names.
43    Macro,
44}

46/// A scope frame, one per block, function, generics list, impl, closure
47/// or `for<'a>` binder, with one map per namespace.
48#[derive(#[automatically_derived]
impl ::core::default::Default for Frame {
    #[inline]
    fn default() -> Self { Self(::core::default::Default::default()) }
}Default)]
49struct Frame([BTreeMap<String, String>; 5]);

51impl Frame {
52    fn bind(&mut self, ns: Ns, name: String, canon: String) {
53        // A fieldless enum's discriminant, 0 to 4.
54        self.0[ns as usize].insert(name, canon);
55    }
56
57    fn lookup(&self, ns: Ns, name: &str) -> Option<&String> {
58        // A fieldless enum's discriminant, 0 to 4.
59        self.0[ns as usize].get(name)
60    }
61
62    /// First binder of `name` in `ns`, in priority order.
63    fn lookup_any(&self, ns: &[Ns], name: &str) -> Option<&String> {
64        ns.iter().find_map(|ns| self.lookup(*ns, name))
65    }
66}

68/// Point `ident` at `canon`, keeping its span.
69fn rename(ident: &mut Ident, canon: &str) {
70    *ident = Ident::new(canon, ident.span());
71}

73/// The scope stack, the positional counter, the saved module frames,
74/// and the self type of the enclosing impl.
75struct Renamer {
76    frames: Vec<Frame>,
77    counter: usize,
78    mod_frames: BTreeMap<String, Frame>,
79    self_ty: Option<syn::Type>,
80    in_type: bool,
81    /// A bare `<T>` was just visited, so the next path names associated items of `T`.
82    after_bare_qself: bool,
83}
84
85impl Renamer {
86    fn new() -> Self {
87        Self {
88            frames: ::alloc::boxed::box_assume_init_into_vec_unsafe(::alloc::intrinsics::write_box_via_move(::alloc::boxed::Box::new_uninit(),
        [Frame::default()]))vec![Frame::default()],
89            counter: 0,
90            mod_frames: BTreeMap::new(),
91            self_ty: None,
92            in_type: false,
93            after_bare_qself: false,
94        }
95    }
96
97    fn push(&mut self) {
98        self.frames.push(Frame::default());
99    }
100
101    fn pop(&mut self) {
102        self.frames.pop();
103    }
104
105    /// Bind `name` in the current frame and return its canonical form.
106    fn bind(&mut self, ns: Ns, name: &str) -> String {
107        let canon = canonical(&mut self.counter);
108        self.top_bind(ns, name, canon.clone());
109        canon
110    }
111
112    /// Bind `name` to `canon` in the current frame.
113    fn top_bind(&mut self, ns: Ns, name: &str, canon: String) {
114        self.frames
115            .last_mut()
116            .expect("a scope frame")
117            .bind(ns, name.to_string(), canon);
118    }
119
120    /// Bind `ident` in the current frame and rename it.
121    fn bind_ident(&mut self, ns: Ns, ident: &mut Ident) -> String {
122        let canon = self.bind(ns, &ident.to_string());
123        rename(ident, &canon);
124        canon
125    }
126
127    /// Bind `name` in the value and type namespaces to one canon. A use
128    /// alias or a const parameter prints once, two canons would leave a
129    /// position of the other namespace pointing at an undeclared binder.
130    fn bind_value_and_type(&mut self, name: &str) -> String {
131        let canon = self.bind(Ns::Value, name);
132        self.top_bind(Ns::Type, name, canon.clone());
133        canon
134    }
135
136    /// Nearest-binder lookup, namespace priority first.
137    fn lookup(&self, ns: &[Ns], name: &str) -> Option<String> {
138        for ns in ns {
139            for frame in self.frames.iter().rev() {
140                if let Some(canon) = frame.lookup(*ns, name) {
141                    return Some(canon.clone());
142                }
143            }
144        }
145        None
146    }
147
148    /// Rename `ident` to the canon it already holds in the current frame,
149    /// else to a new one.
150    fn rename_binder(&mut self, ns: Ns, ident: &mut Ident) -> String {
151        let name = ident.to_string();
152        let canon = match self.frames.last().and_then(|frame| frame.lookup(ns, &name)) {
153            Some(canon) => canon.clone(),
154            None => self.bind(ns, &name),
155        };
156        rename(ident, &canon);
157        canon
158    }
159
160    /// Rename `ident` to its nearest binder in `ns`, if any.
161    fn resolve(&self, ns: &[Ns], ident: &mut Ident) {
162        if let Some(canon) = self.lookup(ns, &ident.to_string()) {
163            rename(ident, &canon);
164        }
165    }
166
167    fn visit_attrs(&mut self, attrs: &mut [syn::Attribute]) {
168        for attr in attrs {
169            self.visit_attribute_mut(attr);
170        }
171    }
172
173    /// Bind a loop or block label in the current frame.
174    fn bind_label(&mut self, label: Option<&mut syn::Label>) {
175        if let Some(label) = label {
176            self.bind_ident(Ns::Label, &mut label.name.ident);
177        }
178    }
179
180    /// Resolve a label reference through the label namespace only, the
181    /// lifetime map must not answer, rustc keeps the two namespaces apart.
182    fn rename_label(&self, lifetime: &mut syn::Lifetime) {
183        self.resolve(&[Ns::Label], &mut lifetime.ident);
184    }

186    /// Drop a leading `::` when no local item binds the first segment `first`
187    /// of its path, `::std` and `std` then naming the same crate.
188    fn unroot(&self, leading_colon: &mut Option<::syn::token::PathSepsyn::Token![::]>, first: Option<&Ident>) {
189        if first.is_some_and(|first| self.lookup(&[Ns::Type], &first.to_string()).is_none()) {
190            *leading_colon = None;
191        }
192    }
193
194    /// Unroot `path` and rename the local binder its first segment names,
195    /// with the later segments a local module holds.
196    fn resolve_path(&self, path: &mut syn::Path) {
197        self.unroot(
198            &mut path.leading_colon,
199            path.segments.first().map(|s| &s.ident),
200        );
201        let ns: &[Ns] = if self.in_type {
202            &[Ns::Type]
203        } else {
204            &[Ns::Value, Ns::Type]
205        };
206        if path.leading_colon.is_none()
207            && let Some(first) = path.segments.first_mut()
208            && let Some(mut canon) = self.lookup(ns, &first.ident.to_string())
209        {
210            rename(&mut first.ident, &canon);
211            for segment in path.segments.iter_mut().skip(1) {
212                let Some(next) = self
213                    .mod_frames
214                    .get(&canon)
215                    .and_then(|frame| frame.lookup_any(ns, &segment.ident.to_string()))
216                else {
217                    break;
218                };
219                canon = next.clone();
220                rename(&mut segment.ident, &canon);
221            }
222        }
223    }

225    /// Rewrite macro tokens the parser cannot read, an identifier before
226    /// `!` resolving through the macro namespace too.
227    fn rename_tokens(&mut self, tokens: proc_macro2::TokenStream) -> proc_macro2::TokenStream {
228        let mut out = Vec::new();
229        let mut iter = tokens.into_iter().peekable();
230        while let Some(tree) = iter.next() {
231            out.push(match tree {
232                proc_macro2::TokenTree::Ident(mut ident) => {
233                    let bang = #[allow(non_exhaustive_omitted_patterns)] match iter.peek() {
    Some(proc_macro2::TokenTree::Punct(p)) if p.as_char() == '!' => true,
    _ => false,
}matches!(iter.peek(), Some(proc_macro2::TokenTree::Punct(p)) if p.as_char() == '!');
234                    let ns: &[Ns] = if bang {
235                        &[Ns::Value, Ns::Type, Ns::Macro]
236                    } else {
237                        &[Ns::Value, Ns::Type]
238                    };
239                    self.resolve(ns, &mut ident);
240                    proc_macro2::TokenTree::Ident(ident)
241                }
242                proc_macro2::TokenTree::Group(group) => {
243                    crate::drift::map_group(&group, |inner| self.rename_tokens(inner))
244                }
245                other => other,
246            });
247        }
248        proc_macro2::TokenStream::from_iter(out)
249    }

251    /// Rewrite macro arguments as expressions when they parse, so binders
252    /// inside them are visited, else token by token. The literal at
253    /// `format_at` is a format string whose inline arguments rename too.
254    fn rewrite_macro_tokens(
255        &mut self,
256        tokens: proc_macro2::TokenStream,
257        format_at: Option<usize>,
258    ) -> proc_macro2::TokenStream {
259        let Ok(mut args) = syn::parse2::<MacroArgs>(tokens.clone()) else {
260            return self.rename_tokens(tokens);
261        };
262        for (index, expr) in args.iter_mut().enumerate() {
263            self.visit_expr_mut(expr);
264            if Some(index) == format_at
265                && let syn::Expr::Lit(syn::ExprLit {
266                    lit: syn::Lit::Str(lit),
267                    ..
268                }) = expr
269            {
270                *lit = rewrite_format_str(self, lit);
271            }
272        }
273        args.to_token_stream()
274    }
275}

277/// Macro arguments as a comma list or the `elem; count` of `vec!`.
278enum MacroArgs {
279    List(syn::punctuated::Punctuated<syn::Expr, ::syn::token::Commasyn::Token![,]>),
280    Repeat(syn::punctuated::Punctuated<syn::Expr, ::syn::token::Semisyn::Token![;]>),
281}

283impl MacroArgs {
284    fn iter_mut(&mut self) -> syn::punctuated::IterMut<'_, syn::Expr> {
285        match self {
286            Self::List(exprs) => exprs.iter_mut(),
287            Self::Repeat(exprs) => exprs.iter_mut(),
288        }
289    }
290}

292impl syn::parse::Parse for MacroArgs {
293    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
294        let list = input.fork();
295        if let Ok(exprs) = list.call(syn::punctuated::Punctuated::parse_terminated) {
296            input.advance_to(&list);
297            return Ok(Self::List(exprs));
298        }
299        let span = input.span();
300        let exprs: syn::punctuated::Punctuated<syn::Expr, ::syn::token::Semisyn::Token![;]> =
301            input.call(syn::punctuated::Punctuated::parse_terminated)?;
302        if exprs.len() == 2 && !exprs.trailing_punct() {
303            Ok(Self::Repeat(exprs))
304        } else {
305            Err(syn::Error::new(span, "not a repeat"))
306        }
307    }
308}

329/// Rewrite one `{…}` placeholder body, its name and any `name$` width or
330/// precision in the spec.
331fn rewrite_format_arg(renamer: &Renamer, arg: &str, out: &mut String) {
332    let (name, spec) = arg
333        .split_once(':')
334        .map_or((arg, None), |(n, s)| (n, Some(s)));
335    push_format_name(renamer, name, out);
336    let Some(spec) = spec else {
337        return;
338    };
339    out.push(':');
340    let mut rest = spec;
341    while let Some((before, after)) = rest.split_once('$') {
342        let start = before
343            .char_indices()
344            .rfind(|&(_, c)| !c.is_alphanumeric() && c != '_')
345            .map_or(0, |(p, c)| p + c.len_utf8());
346        out.push_str(&before[..start]);
347        push_format_name(renamer, &before[start..], out);
348        out.push('$');
349        rest = after;
350    }
351    out.push_str(rest);
352}

354/// Rename inline `{name}` arguments of a format string, `{{` and `}}`
355/// escapes untouched.
356fn rewrite_format_str(renamer: &Renamer, lit: &syn::LitStr) -> syn::LitStr {
357    let value = lit.value();
358    let mut out = String::with_capacity(value.len());
359    let mut chars = value.chars().peekable();
360    while let Some(c) = chars.next() {
361        match c {
362            '{' | '}' if chars.peek() == Some(&c) => {
363                chars.next();
364                out.push(c);
365                out.push(c);
366            }
367            '{' => {
368                let mut inner = String::new();
369                let mut closed = false;
370                for c in chars.by_ref() {
371                    if c == '}' {
372                        closed = true;
373                        break;
374                    }
375                    inner.push(c);
376                }
377                out.push('{');
378                rewrite_format_arg(renamer, &inner, &mut out);
379                if closed {
380                    out.push('}');
381                }
382            }
383            other => out.push(other),
384        }
385    }
386    syn::LitStr::new(&out, lit.span())
387}

395/// True for an ident pattern that names a unit struct, variant, or
396/// const rather than a binding, by the uppercase naming convention.
397fn is_unit_path(id: &syn::PatIdent) -> bool {
398    id.by_ref.is_none()
399        && id.mutability.is_none()
400        && id.subpat.is_none()
401        && id.ident.to_string().starts_with(|c: char| c.is_uppercase())
402}

404/// Call `f` on every identifier pattern of `pat`, in traversal order.
405/// A shorthand `S { x }` binds the member name and drops the member when
406/// rendered, so its field takes the explicit `x: x` form on the way.
407fn walk_pat_idents(pat: &mut syn::Pat, f: &mut impl FnMut(&mut syn::PatIdent)) {
408    match pat {
409        syn::Pat::Ident(id) => {
410            f(id);
411            if let Some((_, sub)) = &mut id.subpat {
412                walk_pat_idents(sub, f);
413            }
414        }
415        syn::Pat::Tuple(syn::PatTuple { elems, .. })
416        | syn::Pat::Slice(syn::PatSlice { elems, .. })
417        | syn::Pat::TupleStruct(syn::PatTupleStruct { elems, .. }) => {
418            for pat in elems {
419                walk_pat_idents(pat, f);
420            }
421        }
422        syn::Pat::Struct(r#struct) => {
423            for field in &mut r#struct.fields {
424                field.colon_token.get_or_insert_with(Default::default);
425                walk_pat_idents(&mut field.pat, f);
426            }
427        }
428        syn::Pat::Or(r#or) => {
429            for pat in &mut r#or.cases {
430                walk_pat_idents(pat, f);
431            }
432        }
433        syn::Pat::Reference(syn::PatReference { pat, .. })
434        | syn::Pat::Type(syn::PatType { pat, .. })
435        | syn::Pat::Guard(syn::PatGuard { pat, .. }) => walk_pat_idents(pat, f),
436        _ => {}
437    }
438}

440/// Rewrite the binder identifiers of `pat` to their canonical names.
441fn rewrite_pat_bindings(renamer: &Renamer, pat: &mut syn::Pat) {
442    walk_pat_idents(pat, &mut |id| {
443        let ns: &[Ns] = if is_unit_path(id) {
444            &[Ns::Value, Ns::Type]
445        } else {
446            &[Ns::Value]
447        };
448        renamer.resolve(ns, &mut id.ident);
449    });
450}

452/// Push the names bound by `pat` to `out`, in traversal order.
453fn pattern_names(pat: &mut syn::Pat, out: &mut Vec<String>) {
454    walk_pat_idents(pat, &mut |id| {
455        if !is_unit_path(id) {
456            out.push(id.ident.to_string());
457        }
458    });
459}

461/// The names bound by the function parameters of `inputs`.
462fn fn_param_names<'a>(inputs: impl Iterator<Item = &'a mut syn::FnArg>) -> Vec<String> {
463    let mut out = Vec::new();
464    for input in inputs {
465        if let syn::FnArg::Typed(pat_type) = input {
466            pattern_names(&mut pat_type.pat, &mut out);
467        }
468    }

472/// Pre-bind item names so a use may precede its definition, skipping
473/// `macro_rules!` and `use`, which are visible only after their line.
474/// A module's items go into its own frame, saved under its canon.
475fn prebind<'a>(renamer: &mut Renamer, items: impl Iterator<Item = &'a syn::Item>) {
476    for item in items {
477        let (ns, ident) = match item {
478            syn::Item::Fn(i) => (Ns::Value, &i.sig.ident),
479            syn::Item::Const(i) => (Ns::Value, &i.ident),
480            syn::Item::Static(i) => (Ns::Value, &i.ident),
481            syn::Item::Struct(i) => (Ns::Type, &i.ident),
482            syn::Item::Enum(i) => (Ns::Type, &i.ident),
483            syn::Item::Union(i) => (Ns::Type, &i.ident),
484            syn::Item::Type(i) => (Ns::Type, &i.ident),
485            syn::Item::Trait(i) => (Ns::Type, &i.ident),
486            syn::Item::TraitAlias(i) => (Ns::Type, &i.ident),
487            syn::Item::Mod(i) => (Ns::Type, &i.ident),
488            _ => continue,
489        };
490        let canon = renamer.bind(ns, &ident.to_string());
491        if let syn::Item::Mod(syn::ItemMod {
492            content: Some((_, items)),
493            ..
494        }) = item
495        {
496            renamer.push();
497            prebind(renamer, items.iter());
498            let frame = renamer.frames.pop().expect("a scope frame");
499            renamer.mod_frames.insert(canon, frame);
500        }
501    }
502}

504impl Renamer {
505    /// Bind every name a use tree introduces and rewrite the tree. A plain
506    /// name is both the imported item and its local alias, so it becomes
507    /// `item as canon` to keep the item and rename the alias.
508    fn bind_use_tree(&mut self, tree: syn::UseTree) -> syn::UseTree {
509        match tree {
510            syn::UseTree::Path(mut path) => {
511                path.tree = Box::new(self.bind_use_tree(*path.tree));
512                syn::UseTree::Path(path)
513            }
514            syn::UseTree::Name(syn::UseName { ident }) => {
515                let canon = self.bind_value_and_type(&ident.to_string());
516                syn::UseTree::Rename(syn::UseRename {
517                    rename: Ident::new(&canon, ident.span()),
518                    ident,
519                    as_token: <::syn::token::Assyn::Token![as]>::default(),
520                })
521            }
522            syn::UseTree::Rename(mut alias) => {
523                let canon = self.bind_value_and_type(&alias.rename.to_string());
524                rename(&mut alias.rename, &canon);
525                syn::UseTree::Rename(alias)
526            }
527            syn::UseTree::Group(mut group) => {
528                group.items = group
529                    .items
530                    .into_pairs()
531                    .map(|pair| {
532                        let (tree, comma) = pair.into_tuple();
533                        syn::punctuated::Pair::new(self.bind_use_tree(tree), comma)
534                    })
535                    .collect();
536                syn::UseTree::Group(group)
537            }
538            glob @ syn::UseTree::Glob(_) => glob,
539        }
540    }

542    /// Bind every name `pat` binds, then visit and rewrite it.
543    fn bind_pat(&mut self, pat: &mut syn::Pat) {
544        let mut names = Vec::new();
545        pattern_names(pat, &mut names);
546        for name in &names {
547            self.bind(Ns::Value, name);
548        }
549        syn::visit_mut::visit_pat_mut(self, pat);
550        rewrite_pat_bindings(self, pat);
551    }
552
553    /// A function's generics frame and parameter frame around its
554    /// signature and body, its own name already handled.
555    fn visit_fn(
556        &mut self,
557        attrs: &mut [syn::Attribute],
558        sig: &mut syn::Signature,
559        body: Option<&mut syn::Block>,
560    ) {
561        begin_generics(self, &mut sig.generics);
562        self.visit_attrs(attrs);
563        self.push();
564        for param in fn_param_names(sig.inputs.iter_mut()) {
565            self.bind(Ns::Value, &param);
566        }
567        visit_fn_inputs(self, sig.inputs.iter_mut());
568        syn::visit_mut::visit_return_type_mut(self, &mut sig.output);
569        if let Some(body) = body {
570            self.visit_block_mut(body);
571        }
572        self.pop();
573        self.pop();
574    }
575
576    /// Rename a type item, then open its generics frame and visit its
577    /// attributes. The caller visits the rest and pops the frame.
578    fn begin_type_item(
579        &mut self,
580        ident: &mut Ident,
581        generics: &mut syn::Generics,
582        attrs: &mut [syn::Attribute],
583    ) {
584        self.rename_binder(Ns::Type, ident);
585        begin_generics(self, generics);
586        self.visit_attrs(attrs);
587    }
588}

590/// Visit function parameters and rewrite their binder patterns.
591fn visit_fn_inputs<'a>(renamer: &mut Renamer, inputs: impl Iterator<Item = &'a mut syn::FnArg>) {
592    for input in inputs {
593        syn::visit_mut::visit_fn_arg_mut(renamer, input);
594        if let syn::FnArg::Typed(pat_type) = input {
595            rewrite_pat_bindings(renamer, &mut pat_type.pat);
596        }
597    }

600/// The identifier of a bare single-segment type, if any.
601fn single_segment_type_name(ty: &syn::Type) -> Option<String> {
602    let syn::Type::Path(path) = ty else {
603        return None;
604    };
605    if path.qself.is_some() || path.path.segments.len() != 1 {
606        return None;
607    }
608    let segment = path.path.segments.first()?;
609    if #[allow(non_exhaustive_omitted_patterns)] match segment.arguments {
    syn::PathArguments::None | syn::PathArguments::AngleBracketed(_) => true,
    _ => false,
}matches!(
610        segment.arguments,
611        syn::PathArguments::None | syn::PathArguments::AngleBracketed(_)
612    ) {
613        Some(segment.ident.to_string())
614    } else {
615        None
616    }
617}
618
619/// Push a generics frame. All parameters are bound and rewritten first
620/// so that a bound may name a later parameter, then the bounds are
621/// visited with every parameter in scope.
622fn begin_generics(renamer: &mut Renamer, generics: &mut syn::Generics) {
623    renamer.push();
624    for param in &mut generics.params {
625        match param {
626            syn::GenericParam::Lifetime(lifetime) => {
627                renamer.bind_ident(Ns::Lifetime, &mut lifetime.lifetime.ident);
628            }
629            syn::GenericParam::Type(ty_param) => {
630                renamer.bind_ident(Ns::Type, &mut ty_param.ident);
631            }
632            syn::GenericParam::Const(r#const) => {
633                // A bare reference in `Foo<N>` reads as a type argument.
634                let canon = renamer.bind_value_and_type(&r#const.ident.to_string());
635                rename(&mut r#const.ident, &canon);
636            }
637        }
638    }
639    for param in &mut generics.params {
640        syn::visit_mut::visit_generic_param_mut(renamer, param);
641    }
642    if let Some(where_clause) = &mut generics.where_clause {
643        syn::visit_mut::visit_where_clause_mut(renamer, where_clause);
644    }
645}
646
647/// Walk a condition, binding each `let` pattern before the operands to
648/// its right. Only `&&` may carry a `let`, so every binary splits.
649fn walk_let_cond(renamer: &mut Renamer, cond: &mut syn::Expr) {
650    match cond {
651        syn::Expr::Binary(binary) => {
652            walk_let_cond(renamer, &mut binary.left);
653            walk_let_cond(renamer, &mut binary.right);
654        }
655        syn::Expr::Let(let_expr) => bind_let(renamer, let_expr),
656        other => syn::visit_mut::visit_expr_mut(renamer, other),
657    }
658}
659
660/// Visit a `let` expression, binding its pattern names before the pattern
661/// is rewritten.
662fn bind_let(renamer: &mut Renamer, let_expr: &mut syn::ExprLet) {
663    syn::visit_mut::visit_expr_mut(renamer, &mut let_expr.expr);
664    renamer.bind_pat(&mut let_expr.pat);
665}

667impl VisitMut for Renamer {
668    fn visit_file_mut(&mut self, file: &mut syn::File) {
669        prebind(self, file.items.iter());
670        syn::visit_mut::visit_file_mut(self, file);
671    }

673    fn visit_block_mut(&mut self, block: &mut syn::Block) {
674        self.push();
675        prebind(
676            self,
677            block.stmts.iter().filter_map(|stmt| match stmt {
678                syn::Stmt::Item(item) => Some(item),
679                _ => None,
680            }),
681        );
682        syn::visit_mut::visit_block_mut(self, block);
683        self.pop();
684    }

686    fn visit_local_mut(&mut self, local: &mut syn::Local) {
687        // The initializer, and a `let .. else` branch, resolve under the
688        // outer bindings.
689        if let Some(init) = &mut local.init {
690            syn::visit_mut::visit_expr_mut(self, &mut init.expr);
691            if let Some(diverge) = &mut init.diverge {
692                syn::visit_mut::visit_expr_mut(self, &mut diverge.1);
693            }
694        }
695        // The pattern's names become visible from here on.
696        self.bind_pat(&mut local.pat);
697    }

701        self.visit_fn(&mut item.attrs, &mut item.sig, Some(&mut item.block));
702    }

714        syn::visit_mut::visit_fields_mut(self, &mut item.fields);

734        syn::visit_mut::visit_type_mut(self, &mut item.ty);
735        self.pop();
736    }
737
738    fn visit_item_trait_mut(&mut self, item: &mut syn::ItemTrait) {
739        let canon = self.rename_binder(Ns::Type, &mut item.ident);
740        self.top_bind(Ns::Type, "Self", canon);
741        begin_generics(self, &mut item.generics);
742        self.visit_attrs(&mut item.attrs);
743        for trait_item in &mut item.items {
744            syn::visit_mut::visit_trait_item_mut(self, trait_item);
745        }
746        self.pop();
747    }

749    fn visit_item_trait_alias_mut(&mut self, item: &mut syn::ItemTraitAlias) {
750        self.begin_type_item(&mut item.ident, &mut item.generics, &mut item.attrs);
751        for bound in &mut item.bounds {
752            syn::visit_mut::visit_type_param_bound_mut(self, bound);
753        }
754        self.pop();
755    }
756
757    fn visit_item_impl_mut(&mut self, item: &mut syn::ItemImpl) {
758        self.push();
759        if let Some(name) = single_segment_type_name(&item.self_ty)
760            && let Some(canon) = self.lookup(&[Ns::Type], &name)
761        {
762            self.top_bind(Ns::Type, "Self", canon);
763        }
764        let has_generics = item.generics.lt_token.is_some();
765        if has_generics {
766            begin_generics(self, &mut item.generics);
767        }
768        self.visit_attrs(&mut item.attrs);
769        if let Some((path, _)) = &mut item.trait_ {
770            // The trait path of `impl Trait for Type` is a type position
771            // given as a bare path, no visit_type_path_mut wraps it.
772            let outer = core::mem::replace(&mut self.in_type, true);
773            self.visit_path_mut(path);
774            self.in_type = outer;
775        }
776        syn::visit_mut::visit_type_mut(self, &mut item.self_ty);
777        let outer = self.self_ty.replace(item.self_ty.as_ref().clone());
778        for impl_item in &mut item.items {
779            syn::visit_mut::visit_impl_item_mut(self, impl_item);
780        }
781        self.self_ty = outer;
782        if has_generics {
783            self.pop();
784        }
785        self.pop();
786    }

788    fn visit_item_const_mut(&mut self, item: &mut syn::ItemConst) {
789        self.rename_binder(Ns::Value, &mut item.ident);
790        self.visit_attrs(&mut item.attrs);
791        syn::visit_mut::visit_type_mut(self, &mut item.ty);
792        syn::visit_mut::visit_expr_mut(self, &mut item.expr);
793    }

796        self.rename_binder(Ns::Value, &mut item.ident);
797        self.visit_attrs(&mut item.attrs);
798        syn::visit_mut::visit_type_mut(self, &mut item.ty);
799        syn::visit_mut::visit_expr_mut(self, &mut item.expr);
800    }

802    fn visit_item_macro_mut(&mut self, item: &mut syn::ItemMacro) {
803        // `macro_rules! n {}` carries the name in `ident` and a body of
804        // macro pattern syntax. The legacy `macro n {}` form does not occur
805        // in doctests. A call at item level is visited like any other.
806        if let Some(ident) = item.ident.as_mut() {
807            self.bind_ident(Ns::Macro, ident);
808            self.visit_attrs(&mut item.attrs);
809        } else {
810            self.visit_attrs(&mut item.attrs);
811            self.visit_macro_mut(&mut item.mac);
812        }
813    }
814
815    fn visit_item_use_mut(&mut self, item: &mut syn::ItemUse) {
816        self.visit_attrs(&mut item.attrs);
817        let first = match &item.tree {
818            syn::UseTree::Path(path) => Some(&path.ident),
819            syn::UseTree::Name(name) => Some(&name.ident),
820            syn::UseTree::Rename(rename) => Some(&rename.ident),
821            syn::UseTree::Glob(_) | syn::UseTree::Group(_) => None,
822        };
823        self.unroot(&mut item.leading_colon, first);
824        let glob = syn::UseTree::Glob(syn::UseGlob {
825            star_token: <::syn::token::Starsyn::Token![*]>::default(),
826        });
827        item.tree = self.bind_use_tree(core::mem::replace(&mut item.tree, glob));
828    }
829
830    fn visit_item_mod_mut(&mut self, item: &mut syn::ItemMod) {
831        let canon = self.rename_binder(Ns::Type, &mut item.ident);
832        self.visit_attrs(&mut item.attrs);
833        if let Some((_, items)) = &mut item.content {
834            let frame = self.mod_frames.remove(&canon).unwrap_or_default();
835            self.frames.push(frame);
836            for sub_item in items.iter_mut() {
837                syn::visit_mut::visit_item_mut(self, sub_item);
838            }
839            let frame = self.frames.pop().expect("a scope frame");
840            self.mod_frames.insert(canon, frame);
841        }
842    }
843
844    fn visit_expr_closure_mut(&mut self, closure: &mut syn::ExprClosure) {
845        self.push();
846        if let Some(bound) = &mut closure.lifetimes {
847            self.visit_bound_lifetimes_mut(bound);
848        }
849        let mut names = Vec::new();
850        for input in &mut closure.inputs {
851            pattern_names(input, &mut names);
852        }
853        for name in &names {
854            self.bind(Ns::Value, name);
855        }
856        for input in &mut closure.inputs {
857            syn::visit_mut::visit_pat_mut(self, input);
858            rewrite_pat_bindings(self, input);
859        }
860        syn::visit_mut::visit_return_type_mut(self, &mut closure.output);
861        syn::visit_mut::visit_expr_mut(self, &mut closure.body);
862        self.pop();
863    }

866        self.push();
867        self.bind_pat(&mut arm.pat);
868        syn::visit_mut::visit_expr_mut(self, &mut arm.body);
869        self.pop();
870    }

875        self.push();
876        self.bind_label(for_loop.label.as_mut());
877        self.bind_pat(&mut for_loop.pat);

879        self.pop();

882    fn visit_expr_if_mut(&mut self, expr: &mut syn::ExprIf) {

884        self.push();
885        walk_let_cond(self, &mut expr.cond);
886        self.visit_block_mut(&mut expr.then_branch);
887        self.pop();
888        if let Some((_, else_expr)) = &mut expr.else_branch {
889            syn::visit_mut::visit_expr_mut(self, else_expr);
890        }

895        self.bind_label(expr.label.as_mut());

897        self.visit_block_mut(&mut expr.body);
898        self.pop();
899    }

905        self.pop();

911        self.visit_block_mut(&mut expr.block);

915    fn visit_expr_struct_mut(&mut self, expr: &mut syn::ExprStruct) {
916        // A shorthand `S { x }` prints only the member, so a bound `x`
917        // takes the explicit form before the rewrite.
918        for field in &mut expr.fields {
919            if field.colon_token.is_none()
920                && let syn::Member::Named(ident) = &field.member
921                && self.lookup(&[Ns::Value], &ident.to_string()).is_some()
922            {
923                field.colon_token = Some(syn::token::Colon::default());
924            }
925        }
926        syn::visit_mut::visit_expr_struct_mut(self, expr);
927    }

929    fn visit_qself_mut(&mut self, qself: &mut syn::QSelf) {
930        // syn visits every qualified path's `<T>` right before its path.
931        syn::visit_mut::visit_qself_mut(self, qself);
932        self.after_bare_qself = qself.position == 0;
933    }
934
935    fn visit_path_mut(&mut self, path: &mut syn::Path) {
936        // Only the first segment may name a local binder, and a leading
937        // colon always names an extern crate. Each later segment resolves
938        // through the saved frame of the local module before it. A type
939        // position reads the type namespace only, rustc-valid programs
940        // bind every name they show there in it. After a bare `<T>` the
941        // `::` is the separator and every segment an associated item.
942        if !core::mem::take(&mut self.after_bare_qself) {
943            self.resolve_path(path);
944        }
945        for segment in &mut path.segments {
946            match &mut segment.arguments {
947                syn::PathArguments::AngleBracketed(args) => {
948                    for arg in &mut args.args {
949                        if #[allow(non_exhaustive_omitted_patterns)] match arg {
    syn::GenericArgument::Const(_) => true,
    _ => false,
}matches!(arg, syn::GenericArgument::Const(_)) {
950                            // A const argument expression resolves values
951                            // again, the type context ends at its braces.
952                            let outer = core::mem::replace(&mut self.in_type, false);
953                            syn::visit_mut::visit_generic_argument_mut(self, arg);
954                            self.in_type = outer;
955                        } else {
956                            syn::visit_mut::visit_generic_argument_mut(self, arg);
957                        }
958                    }
959                }
960                syn::PathArguments::Parenthesized(args) => {
961                    for input in &mut args.inputs {
962                        self.visit_named_arg_mut(input);
963                    }
964                    syn::visit_mut::visit_return_type_mut(self, &mut args.output);
965                }
966                syn::PathArguments::None => {}
967            }
968        }
969    }

971    fn visit_macro_mut(&mut self, mac: &mut syn::Macro) {
972        self.unroot(
973            &mut mac.path.leading_colon,
974            mac.path.segments.first().map(|s| &s.ident),
975        );
976        if let Some(first) = mac.path.segments.first_mut() {
977            self.resolve(&[Ns::Macro], &mut first.ident);
978        }
979        let format_at = crate::drift::format_operand(&mac.path).map(|(at, _)| at);
980        mac.tokens = self.rewrite_macro_tokens(core::mem::take(&mut mac.tokens), format_at);
981    }

984        // Type positions resolve through the type namespace, a local value
985        // binder of the same name must not capture the annotation.
986        let outer = core::mem::replace(&mut self.in_type, true);
987        syn::visit_mut::visit_type_path_mut(self, node);
988        self.in_type = outer;

991    fn visit_type_mut(&mut self, ty: &mut syn::Type) {
992        if let Some(self_ty) = &self.self_ty
993            && let syn::Type::Path(path) = &*ty
994            && path.qself.is_none()
995            && path.path.is_ident("Self")
996        {
997            *ty = self_ty.clone();
998            return;
999        }
1000        syn::visit_mut::visit_type_mut(self, ty);
1001    }

1049        let outer = core::mem::replace(&mut self.in_type, true);

1056        // Parameter names in a function pointer type bind nothing, rustc
