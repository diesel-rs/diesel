20    /// Create a new `InspectReader`, wrapping `reader` and calling `f` for the
21    /// new data supplied by each read call.
22    ///
23    /// The closure will only be called with an empty slice if the inner reader
24    /// returns without reading data into the buffer. This happens at EOF, or if
25    /// `poll_read` is called with a zero-size buffer.
26    pub fn new(reader: R, f: F) -> InspectReader<R, F>
27    where
28        R: AsyncRead,
29        F: FnMut(&[u8]),
30    {
31        InspectReader { reader, f }
32    }

36        &self.reader
37    }
38
39    /// Acquires a mutable reference to the underlying reader.
40    ///
41    /// I/O performed directly on the underlying reader is not inspected.
42    pub fn get_mut(&mut self) -> &mut R {
43        &mut self.reader
44    }

126    /// buffer supplied to `poll_write_vectored`.

128    where
129        W: AsyncWrite,

133    }

160impl<W: AsyncWrite, F: FnMut(&[u8])> AsyncWrite for InspectWriter<W, F> {
161    fn poll_write(self: Pin<&mut Self>, cx: &mut Context<'_>, buf: &[u8]) -> Poll<Result<usize>> {
162        let me = self.project();
163        let res = me.writer.poll_write(cx, buf);
164        if let Poll::Ready(Ok(count)) = res {
165            if count != 0 {
166                (me.f)(&buf[..count]);
167            }
168        }
169        res
170    }
171
172    fn poll_flush(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Result<()>> {
173        let me = self.project();
174        me.writer.poll_flush(cx)
175    }
176
177    fn poll_shutdown(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Result<()>> {
178        let me = self.project();
179        me.writer.poll_shutdown(cx)
180    }
181
182    fn poll_write_vectored(
183        self: Pin<&mut Self>,
184        cx: &mut Context<'_>,
185        bufs: &[IoSlice<'_>],
186    ) -> Poll<Result<usize>> {
187        let me = self.project();
188        let res = me.writer.poll_write_vectored(cx, bufs);
189        if let Poll::Ready(Ok(mut count)) = res {
190            for buf in bufs {
191                if count == 0 {
192                    break;
193                }
194                let size = count.min(buf.len());
195                if size != 0 {
196                    (me.f)(&buf[..size]);
197                    count -= size;
198                }
199            }
200        }
201        res
202    }
203
204    fn is_write_vectored(&self) -> bool {
205        self.writer.is_write_vectored()
206    }
207}
