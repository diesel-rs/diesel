6//! SHA-2, SHA-3 and the legacy SHA-1 digest algorithms.
7//!
8//! If all the data is available in a single contiguous slice then the `digest`
9//! function should be used. Otherwise, the digest can be calculated in
10//! multiple steps using `Context`.
11
12//! # Example
13//!
14//! ```
15//! use aws_lc_rs::digest;
16//!
17//! // Using `digest::digest`
18//! let one_shot = digest::digest(&digest::SHA384, b"hello, world");
19//!
20//! // Using `digest::Context`
21//! let mut ctx = digest::Context::new(&digest::SHA384);
22//! ctx.update(b"hello");
23//! ctx.update(b", ");
24//! ctx.update(b"world");
25//! let multi_part = ctx.finish();
26//!
27//! assert_eq!(&one_shot.as_ref(), &multi_part.as_ref());
28//! ```
29

70pub struct Context {
71    /// The context's algorithm.
72    pub(crate) algorithm: &'static Algorithm,
73    digest_ctx: DigestContext,
74    // The spec specifies that SHA-1 and SHA-256 support up to
75    // 2^64-1 bits of input. SHA-384 and SHA-512 support up to
76    // 2^128-1 bits.
77    // Implementations of `digest` only support up
78    // to 2^64-1 bits of input, which should be sufficient enough for
79    // practical use cases.
80    msg_len: u64,
81    max_input_reached: bool,
82}

84impl Context {
85    /// Constructs a new context.
86    ///
87    /// # Panics
88    ///
89    /// `new` panics if it fails to initialize an aws-lc digest context for the given
90    /// algorithm.
91    #[must_use]
92    pub fn new(algorithm: &'static Algorithm) -> Self {
93        Self {
94            algorithm,
95            digest_ctx: DigestContext::new(algorithm).unwrap(),
96            msg_len: 0u64,
97            max_input_reached: false,
98        }
99    }
100
101    /// Updates the message to digest with all the data in `data`.
102    ///
103    /// # Panics
104    /// Panics if update causes total input length to exceed maximum allowed (`u64::MAX`).
105    #[inline]
106    pub fn update(&mut self, data: &[u8]) {
107        Self::try_update(self, data).expect("digest update failed");
108    }
109
110    #[inline]
111    fn try_update(&mut self, data: &[u8]) -> Result<(), Unspecified> {
112        unsafe {
113            // Check if the message has reached the algorithm's maximum allowed input, or overflowed
114            // the msg_len counter.
115            let (msg_len, overflowed) = self.msg_len.overflowing_add(data.len() as u64);
116            if overflowed || msg_len > self.algorithm.max_input_len {
117                return Err(Unspecified);
118            }
119
120            self.msg_len = msg_len;
121            self.max_input_reached = self.msg_len == self.algorithm.max_input_len;
122
123            // Doesn't require boundary_check! guard
124            if 1 != EVP_DigestUpdate(
125                self.digest_ctx.as_mut_ptr(),
126                data.as_ptr().cast(),
127                data.len(),
128            ) {
129                return Err(Unspecified);
130            }
131            Ok(())
132        }
133    }
134
135    /// Finalizes the digest calculation and returns the digest value.
136    ///
137    /// `finish` consumes the context so it cannot be (mis-)used after `finish`
138    /// has been called.
139    ///
140    /// # Panics
141    /// Panics if the digest is unable to be finalized
142    #[inline]
143    #[must_use]
144    pub fn finish(self) -> Digest {
145        Self::try_finish(self).expect("EVP_DigestFinal failed")
146    }
147
148    #[inline]
149    fn try_finish(mut self) -> Result<Digest, Unspecified> {
150        let mut output = [0u8; MAX_OUTPUT_LEN];
151        let mut out_len = MaybeUninit::<c_uint>::uninit();
152        if 1 != {
    {
        unsafe {
            EVP_DigestFinal(self.digest_ctx.as_mut_ptr(), output.as_mut_ptr(),
                out_len.as_mut_ptr())
        }
    }
}indicator_check!(unsafe {

159            return Err(Unspecified);
160        }
161
162        Ok(Digest {
163            algorithm: self.algorithm,
164            message: output,

166        })
167    }

177/// Returns the digest of `data` using the given digest algorithm.
178///
179// # FIPS
180// This function must only be used with one of the following algorithms:
181// * `SHA1_FOR_LEGACY_USE_ONLY`
182// * `SHA224`
183// * `SHA256`
184// * `SHA384`
185// * `SHA512`
186// * `SHA512_256`
187//
188/// # Examples:
189///
190/// ```
191/// # {
192/// use aws_lc_rs::{digest, test};
193/// let expected_hex = "09ca7e4eaa6e8ae9c7d261167129184883644d07dfba7cbfbc4c8a2e08360d5b";
194/// let expected: Vec<u8> = test::from_hex(expected_hex).unwrap();
195/// let actual = digest::digest(&digest::SHA256, b"hello, world");
196///
197/// assert_eq!(&expected, &actual.as_ref());
198/// # }
199/// ```
200#[inline]
201#[must_use]
202pub fn digest(algorithm: &'static Algorithm, data: &[u8]) -> Digest {
203    let mut output = [0u8; MAX_OUTPUT_LEN];
204    (algorithm.one_shot_hash)(data, &mut output);
205
206    Digest {
207        algorithm,
208        message: output,
209        len: algorithm.output_len,
210    }
211}

226impl Digest {
227    /// Imports a digest value provide by an external source. This allows for the signing of
228    /// content that might not be directly accessible.
229    ///
230    /// WARNING: Ensure that the digest is provided by a trusted source.
231    /// When possible, prefer to directly compute the digest of content.
232    ///
233    /// # Errors
234    /// Returns `Unspecified` if the imported value is the wrong length for the specified algorithm.
235    pub fn import_less_safe(
236        digest: &[u8],
237        algorithm: &'static Algorithm,
238    ) -> Result<Self, Unspecified> {
239        if digest.len() != algorithm.output_len {
240            return Err(Unspecified);
241        }
242        let mut my_digest = [0u8; MAX_OUTPUT_LEN];
243        my_digest[0..digest.len()].copy_from_slice(&digest[0..digest.len()]);
244        Ok(Digest {
245            message: my_digest,
246            len: digest.len(),
247            algorithm,
248        })
249    }
250
251    /// The algorithm that was used to calculate the digest value.
252    #[inline]
253    #[must_use]
254    pub fn algorithm(&self) -> &'static Algorithm {
255        self.algorithm
256    }
257}

277
278    /// The size of the chaining value of the digest function, in bytes. For
279    /// non-truncated algorithms (SHA-1, SHA-256, SHA-512), this is equal to
280    /// `output_len`. For truncated algorithms (e.g. SHA-224, SHA-384, SHA-512/256),

290    )]
291    pub chaining_len: usize,
292
293    /// The internal block length.
294    pub block_len: usize,
295
296    // max_input_len is computed as u64 instead of usize to prevent overflowing on 32-bit machines.
297    max_input_len: u64,
298
299    one_shot_hash: fn(msg: &[u8], output: &mut [u8]),
300
301    pub(crate) id: AlgorithmID,
302}
