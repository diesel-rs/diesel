28fn sha1msg1(a: [u32; 4], b: [u32; 4]) -> [u32; 4] {
29    let [_, _, w2, w3] = a;
30    let [w4, w5, _, _] = b;
31    [a[0] ^ w2, a[1] ^ w3, a[2] ^ w4, a[3] ^ w5]
32}

34fn sha1msg2(a: [u32; 4], b: [u32; 4]) -> [u32; 4] {
35    let [x0, x1, x2, x3] = a;
36    let [_, w13, w14, w15] = b;
37
38    let w16 = (x0 ^ w13).rotate_left(1);
39    let w17 = (x1 ^ w14).rotate_left(1);
40    let w18 = (x2 ^ w15).rotate_left(1);
41    let w19 = (x3 ^ w16).rotate_left(1);
42
43    [w16, w17, w18, w19]
44}

51fn sha1_digest_round_x4(abcd: [u32; 4], work: [u32; 4], i: i8) -> [u32; 4] {
52    match i {
53        0 => sha1rnds4c(abcd, add(work, [K[0]; 4])),
54        1 => sha1rnds4p(abcd, add(work, [K[1]; 4])),
55        2 => sha1rnds4m(abcd, add(work, [K[2]; 4])),
56        3 => sha1rnds4p(abcd, add(work, [K[3]; 4])),
57        _ => { ::core::panicking::unreachable_display(&"unknown icosaround index"); }unreachable!("unknown icosaround index"),
58    }
59}

61fn sha1rnds4c(abcd: [u32; 4], msg: [u32; 4]) -> [u32; 4] {
62    let [mut a, mut b, mut c, mut d] = abcd;
63    let [t, u, v, w] = msg;
64    let mut e = 0u32;
65
66    macro_rules! bool3ary_202 {
67        ($a:expr, $b:expr, $c:expr) => {
68            $c ^ ($a & ($b ^ $c))
69        };
70    } // Choose, MD5F, SHA1C
71
72    e = e
73        .wrapping_add(a.rotate_left(5))
74        .wrapping_add(d ^ (b & (c ^ d))bool3ary_202!(b, c, d))
75        .wrapping_add(t);
76    b = b.rotate_left(30);
77
78    d = d
79        .wrapping_add(e.rotate_left(5))
80        .wrapping_add(c ^ (a & (b ^ c))bool3ary_202!(a, b, c))
81        .wrapping_add(u);
82    a = a.rotate_left(30);
83
84    c = c
85        .wrapping_add(d.rotate_left(5))
86        .wrapping_add(b ^ (e & (a ^ b))bool3ary_202!(e, a, b))
87        .wrapping_add(v);
88    e = e.rotate_left(30);
89
90    b = b
91        .wrapping_add(c.rotate_left(5))
92        .wrapping_add(a ^ (d & (e ^ a))bool3ary_202!(d, e, a))
93        .wrapping_add(w);
94    d = d.rotate_left(30);
95
96    [b, c, d, e]
97}

107        };

146    } // Majority, SHA1M

248pub fn compress(state: &mut [u32; 5], blocks: &[[u8; BLOCK_SIZE]]) {
249    let mut block_u32 = [0u32; BLOCK_SIZE / 4];
250    // since LLVM can't properly use aliasing yet it will make
251    // unnecessary state stores without this copy
252    let mut state_cpy = *state;
253    for block in blocks.iter() {
254        for (o, chunk) in block_u32.iter_mut().zip(block.chunks_exact(4)) {
255            *o = u32::from_be_bytes(chunk.try_into().unwrap());
256        }
257        sha1_digest_block_u32(&mut state_cpy, &block_u32);
258    }
259    *state = state_cpy;
