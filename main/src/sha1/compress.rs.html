21const BLOCK_SIZE: usize = <Sha1Core as BlockSizeUser>::BlockSize::USIZE;

23/// SHA-1 compression function
24#[cfg_attr(docsrs, doc(cfg(feature = "compress")))]
25pub fn compress(state: &mut [u32; 5], blocks: &[Block<Sha1Core>]) {
26    // SAFETY: GenericArray<u8, U64> and [u8; 64] have
27    // exactly the same memory layout
28    let blocks: &[[u8; BLOCK_SIZE]] =
29        unsafe { &*(blocks as *const _ as *const [[u8; BLOCK_SIZE]]) };
30    compress_inner(state, blocks);
