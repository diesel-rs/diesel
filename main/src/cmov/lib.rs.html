30/// Condition: the argument given to [`Cmov`] and [`CmovEq`] representing an effective boolean
31/// condition by virtue of being zero or non-zero.
32///
33/// Using a `u8` for this type helps prevent rustc optimizers from speculating about it as if it
34/// were a boolean value.
35pub type Condition = u8;
36
37/// Conditional move
38pub trait Cmov {
39    /// Move if non-zero.
40    ///
41    /// Moves `value` to `self` in constant-time if `condition` is non-zero.
42    fn cmovnz(&mut self, value: &Self, condition: Condition);
43
44    /// Move if zero.
45    ///
46    /// Moves `value` to `self` in constant-time if `condition` is equal to zero.
47    fn cmovz(&mut self, value: &Self, condition: Condition) {
48        let nz = {
    let mut value: Condition = condition;
    value |= value.wrapping_neg();
    core::hint::black_box(value >> (Condition::BITS - 1)).wrapping_neg()
}masknz!(condition: Condition);
49        self.cmovnz(value, !nz);
50    }
51}

53/// Conditional move with equality comparison
54pub trait CmovEq {
55    /// Move if both inputs are not equal.
56    ///
57    /// Moves `input` to `output` in constant-time if `self` and `rhs` are NOT equal.
58    fn cmovne(&self, rhs: &Self, input: Condition, output: &mut Condition) {
59        let mut tmp = 1u8;
60        self.cmoveq(rhs, 0u8, &mut tmp);
61        tmp.cmoveq(&1u8, input, output);
62    }
63
64    /// Move if both inputs are equal.
65    ///
66    /// Moves `input` to `output` in constant-time if `self` and `rhs` are equal.
67    fn cmoveq(&self, rhs: &Self, input: Condition, output: &mut Condition) {
68        let mut tmp = 1u8;
69        self.cmovne(rhs, 0u8, &mut tmp);
70        tmp.cmoveq(&1, input, output);
71    }
72}

92impl Cmov for u128 {
93    #[inline]
94    fn cmovnz(&mut self, value: &Self, condition: Condition) {
95        let mut lo = (*self & 0xFFFF_FFFF_FFFF_FFFF) as u64;
96        let mut hi = (*self >> 64) as u64;
97
98        lo.cmovnz(&((*value & 0xFFFF_FFFF_FFFF_FFFF) as u64), condition);
99        hi.cmovnz(&((*value >> 64) as u64), condition);
100
101        *self = u128::from(lo) | (u128::from(hi) << 64);
102    }

309impl Cmov for cmp::Ordering {
310    #[inline]
311    fn cmovnz(&mut self, src: &Self, condition: Condition) {
312        // `Ordering` is `#[repr(i8)]` where:
313        //
314        // - `Less` => -1
315        // - `Equal` => 0
316        // - `Greater` => 1
317        //
318        // Given this, it's possible to operate on orderings as if they're `i8`, which allows us to
319        // use the `CtSelect` impl on `i8` to select between them.
320        let mut n = *self as i8;
321        n.cmovnz(&(*src as i8), condition);
322
323        // SAFETY: `Ordering` is `#[repr(i8)]` and `ret` has been assigned to
324        // a value which was originally a valid `Ordering` then cast to `i8`
325        #[allow(trivial_casts, unsafe_code)]
326        unsafe {
327            *self = *(&raw const n).cast::<Self>();
328        }
329    }
330}
