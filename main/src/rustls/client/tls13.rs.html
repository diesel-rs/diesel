248enum KeyExchangeChoice {
249    Whole(Box<dyn ActiveKeyExchange>),
250    Component(Box<dyn ActiveKeyExchange>),
251}

253impl KeyExchangeChoice {
254    /// Decide between `our_key_share` or `our_key_share.hybrid_component()`
255    /// based on the selection of the server expressed in `their_key_share`.
256    fn new(
257        config: &Arc<ClientConfig>,
258        cx: &mut ClientContext<'_>,
259        our_key_share: Box<dyn ActiveKeyExchange>,
260        their_key_share: &KeyShareEntry,
261    ) -> Result<Self, ()> {
262        if our_key_share.group() == their_key_share.group {
263            return Ok(Self::Whole(our_key_share));
264        }
265
266        let (component_group, _) = our_key_share
267            .hybrid_component()
268            .ok_or(())?;
269
270        if component_group != their_key_share.group {
271            return Err(());
272        }
273
274        // correct the record for the benefit of accuracy of
275        // `negotiated_key_exchange_group()`
276        let actual_skxg = config
277            .find_kx_group(component_group, ProtocolVersion::TLSv1_3)
278            .ok_or(())?;
279        cx.common.kx_state = KxState::Start(actual_skxg);
280
281        Ok(Self::Component(our_key_share))
282    }
283
284    fn complete(self, peer_pub_key: &[u8]) -> Result<SharedSecret, Error> {
285        match self {
286            Self::Whole(akx) => akx.complete(peer_pub_key),
287            Self::Component(akx) => akx.complete_hybrid_component(peer_pub_key),
288        }
289    }
290}

370pub(super) fn prepare_resumption(
371    config: &ClientConfig,
372    cx: &mut ClientContext<'_>,
373    resuming_session: &Retrieved<&persist::Tls13ClientSessionValue>,
374    exts: &mut ClientExtensions<'_>,
375    doing_retry: bool,
376) {
377    let resuming_suite = resuming_session.suite();
378    cx.common.suite = Some(resuming_suite.into());
379    // The EarlyData extension MUST be supplied together with the
380    // PreSharedKey extension.
381    let max_early_data_size = resuming_session.max_early_data_size();
382    if config.enable_early_data && max_early_data_size > 0 && !doing_retry {
383        cx.data
384            .early_data
385            .enable(max_early_data_size as usize);
386        exts.early_data_request = Some(());
387    }
388
389    // Finally, and only for TLS1.3 with a ticket resumption, include a binder
390    // for our ticket.  This must go last.
391    //
392    // Include an empty binder. It gets filled in below because it depends on
393    // the message it's contained in (!!!).
394    let obfuscated_ticket_age = resuming_session.obfuscated_ticket_age();
395
396    let binder_len = resuming_suite
397        .common
398        .hash_provider
399        .output_len();
400    let binder = ::alloc::vec::from_elem_global(0u8, binder_len)vec![0u8; binder_len];
401
402    let psk_identity =
403        PresharedKeyIdentity::new(resuming_session.ticket().to_vec(), obfuscated_ticket_age);
404    let psk_offer = PresharedKeyOffer::new(psk_identity, binder);
405    exts.preshared_key_offer = Some(psk_offer);
406}

1201                    .send_cert_verify_error_alert(err)
1202            })?;
1203
1204        // 2. Verify their signature on the handshake.
1205        let handshake_hash = self.transcript.current_hash();
1206        let sig_verified = self
1207            .config
1208            .verifier
1209            .verify_tls13_signature(
1210                construct_server_verify_message(&handshake_hash).as_ref(),
1211                end_entity,
1212                cert_verify,
1213            )
1214            .map_err(|err| {
1215                cx.common
1216                    .send_cert_verify_error_alert(err)
1217            })?;
1218
1219        cx.common.peer_certificates = Some(self.server_cert.cert_chain.into_owned());
1220        self.transcript.add_message(&m);
1221
1222        Ok(Box::new(ExpectFinished {
1223            config: self.config,
1224            server_name: self.server_name,
1225            randoms: self.randoms,
1226            suite: self.suite,
1227            transcript: self.transcript,
1228            key_schedule: self.key_schedule,
1229            kx_group: self.kx_group,
1230            client_auth: self.client_auth,
1231            cert_verified,
1232            sig_verified,
1233            ech_retry_configs: self.ech_retry_configs,
1234        }))
1235    }

1253fn emit_compressed_certificate_tls13(
1254    flight: &mut HandshakeFlightTls13<'_>,
1255    certkey: &CertifiedKey,
1256    auth_context: Option<Vec<u8>>,
1257    compressor: &dyn compress::CertCompressor,
1258    config: &ClientConfig,
1259) {
1260    let mut cert_payload = CertificatePayloadTls13::new(certkey.cert.iter(), None);
1261    cert_payload.context = PayloadU8::new(auth_context.clone().unwrap_or_default());
1262
1263    let Ok(compressed) = config
1264        .cert_compression_cache
1265        .compression_for(compressor, &cert_payload)
1266    else {
1267        return emit_certificate_tls13(flight, Some(certkey), auth_context);
1268    };
1269
1270    flight.add(HandshakeMessagePayload(
1271        HandshakePayload::CompressedCertificate(compressed.compressed_cert_payload()),
1272    ));
1273}

1275fn emit_certificate_tls13(
1276    flight: &mut HandshakeFlightTls13<'_>,
1277    certkey: Option<&CertifiedKey>,
1278    auth_context: Option<Vec<u8>>,
1279) {
1280    let certs = certkey
1281        .map(|ck| ck.cert.as_ref())
1282        .unwrap_or(&[][..]);
1283    let mut cert_payload = CertificatePayloadTls13::new(certs.iter(), None);
1284    cert_payload.context = PayloadU8::new(auth_context.unwrap_or_default());
1285
1286    flight.add(HandshakeMessagePayload(HandshakePayload::CertificateTls13(
1287        cert_payload,
1288    )));
1289}

1291fn emit_certverify_tls13(
1292    flight: &mut HandshakeFlightTls13<'_>,
1293    signer: &dyn Signer,
1294) -> Result<(), Error> {
1295    let message = construct_client_verify_message(&flight.transcript.current_hash());
1296
1297    let scheme = signer.scheme();
1298    let sig = signer.sign(message.as_ref())?;
1299    let dss = DigitallySignedStruct::new(scheme, sig);
1300
1301    flight.add(HandshakeMessagePayload(
1302        HandshakePayload::CertificateVerify(dss),
1303    ));
1304    Ok(())
1305}

1307fn emit_finished_tls13(flight: &mut HandshakeFlightTls13<'_>, verify_data: &crypto::hmac::Tag) {
1308    let verify_data_payload = Payload::new(verify_data.as_ref());
1309
1310    flight.add(HandshakeMessagePayload(HandshakePayload::Finished(
1311        verify_data_payload,
1312    )));
1313}

1315fn emit_end_of_early_data_tls13(transcript: &mut HandshakeHash, common: &mut CommonState) {
1316    if common.is_quic() {
1317        return;
1318    }
1319
1320    let m = Message {
1321        version: ProtocolVersion::TLSv1_3,
1322        payload: MessagePayload::handshake(HandshakeMessagePayload(
1323            HandshakePayload::EndOfEarlyData,
1324        )),
1325    };
1326
1327    transcript.add_message(&m);
1328    common.send_msg(m, true);
1329}

1331struct ExpectFinished {
1332    config: Arc<ClientConfig>,
1333    server_name: ServerName<'static>,
1334    randoms: ConnectionRandoms,
1335    suite: &'static Tls13CipherSuite,
1336    transcript: HandshakeHash,
1337    key_schedule: KeyScheduleHandshake,
1338    kx_group: NamedGroup,
1339    client_auth: Option<ClientAuthDetails>,
1340    cert_verified: verify::ServerCertVerified,
1341    sig_verified: verify::HandshakeSignatureValid,
1342    ech_retry_configs: Option<Vec<EchConfigPayload>>,
1343}
1344
1345impl State<ClientConnectionData> for ExpectFinished {
1346    fn handle<'m>(
1347        self: Box<Self>,
1348        cx: &mut ClientContext<'_>,
1349        m: Message<'m>,
1350    ) -> hs::NextStateOrError<'m>
1351    where
1352        Self: 'm,
1353    {
1354        let mut st = *self;
1355        let finished =
1356            match &m.payload {
    MessagePayload::Handshake {
        parsed: crate::msgs::handshake::HandshakeMessagePayload(HandshakePayload::Finished(hm)),
        .. } => Ok(hm),
    payload =>
        Err(crate::check::inappropriate_handshake_message(payload,
                &[crate::ContentType::Handshake],
                &[HandshakeType::Finished])),
}require_handshake_msg!(m, HandshakeType::Finished, HandshakePayload::Finished)?;
1357
1358        let handshake_hash = st.transcript.current_hash();
1359        let expect_verify_data = st
1360            .key_schedule
1361            .sign_server_finish(&handshake_hash);
1362
1363        let fin = match ConstantTimeEq::ct_eq(expect_verify_data.as_ref(), finished.bytes()).into()
1364        {
1365            true => verify::FinishedMessageVerified::assertion(),
1366            false => {
1367                return Err(cx
1368                    .common
1369                    .send_fatal_alert(AlertDescription::DecryptError, Error::DecryptError));
1370            }
1371        };
1372
1373        st.transcript.add_message(&m);
1374
1375        let hash_after_handshake = st.transcript.current_hash();
1376        /* The EndOfEarlyData message to server is still encrypted with early data keys,
1377         * but appears in the transcript after the server Finished. */
1378        if cx.common.early_traffic {
1379            emit_end_of_early_data_tls13(&mut st.transcript, cx.common);
1380            cx.common.early_traffic = false;
1381            cx.data.early_data.finished();
1382            st.key_schedule
1383                .set_handshake_encrypter(cx.common);
1384        }
1385
1386        let mut flight = HandshakeFlightTls13::new(&mut st.transcript);
1387
1388        /* Send our authentication/finished messages.  These are still encrypted
1389         * with our handshake keys. */
1390        if let Some(client_auth) = st.client_auth {
1391            match client_auth {
1392                ClientAuthDetails::Empty {
1393                    auth_context_tls13: auth_context,
1394                } => {
1395                    emit_certificate_tls13(&mut flight, None, auth_context);
1396                }
1397                ClientAuthDetails::Verify {
1398                    auth_context_tls13: auth_context,
1399                    ..
1400                } if cx.data.ech_status == EchStatus::Rejected => {
1401                    // If ECH was offered, and rejected, we MUST respond with
1402                    // an empty certificate message.
1403                    emit_certificate_tls13(&mut flight, None, auth_context);
1404                }
1405                ClientAuthDetails::Verify {
1406                    certkey,
1407                    signer,
1408                    auth_context_tls13: auth_context,
1409                    compressor,
1410                } => {
1411                    if let Some(compressor) = compressor {
1412                        emit_compressed_certificate_tls13(
1413                            &mut flight,
1414                            &certkey,
1415                            auth_context,
1416                            compressor,
1417                            &st.config,
1418                        );
1419                    } else {
1420                        emit_certificate_tls13(&mut flight, Some(&certkey), auth_context);
1421                    }
1422                    emit_certverify_tls13(&mut flight, signer.as_ref())?;
1423                }
1424            }
1425        }
1426
1427        let (key_schedule_pre_finished, verify_data) = st
1428            .key_schedule
1429            .into_pre_finished_client_traffic(
1430                hash_after_handshake,
1431                flight.transcript.current_hash(),
1432                &*st.config.key_log,
1433                &st.randoms.client,
1434            );
1435
1436        emit_finished_tls13(&mut flight, &verify_data);
1437        flight.finish(cx.common);
1438
1439        // Remember what KX group the server liked for next time.
1440        st.config
1441            .resumption
1442            .store
1443            .set_kx_hint(st.server_name.clone(), st.kx_group);
1444
1445        /* We're now sure this server supports TLS1.3.  But if we run out of TLS1.3 tickets
1446         * when connecting to it again, we definitely don't want to attempt a TLS1.2 resumption. */
1447        st.config
1448            .resumption
1449            .store
1450            .remove_tls12_session(&st.server_name);
1451
1452        /* Now move to our application traffic keys. */
1453        cx.common.check_aligned_handshake()?;
1454        let (key_schedule, resumption) =
1455            key_schedule_pre_finished.into_traffic(cx.common, st.transcript.current_hash());
1456
1457        // Now that we've reached the end of the normal handshake we must enforce ECH acceptance by
1458        // sending an alert and returning an error (potentially with retry configs) if the server
1459        // did not accept our ECH offer. This must happen before the connection is made available
1460        // for application data, since the server has only been authenticated for the ECH config's
1461        // `public_name`.
1462        if cx.data.ech_status == EchStatus::Rejected {
1463            return Err(ech::fatal_alert_required(st.ech_retry_configs, cx.common));
1464        }
1465
1466        cx.common
1467            .start_traffic(&mut cx.sendable_plaintext);
1468
1469        let st = ExpectTraffic {
1470            config: st.config.clone(),
1471            session_storage: st.config.resumption.store.clone(),
1472            server_name: st.server_name,
1473            suite: st.suite,
1474            key_schedule,
1475            resumption,
1476            _cert_verified: st.cert_verified,
1477            _sig_verified: st.sig_verified,
1478            _fin_verified: fin,
1479        };
1480
1481        Ok(match cx.common.is_quic() {
1482            true => Box::new(ExpectQuicTraffic(st)),
1483            false => Box::new(st),
1484        })
1485    }
1486
1487    fn into_owned(self: Box<Self>) -> hs::NextState<'static> {
1488        self
1489    }
1490}

1507impl ExpectTraffic {
1508    fn handle_new_ticket_impl(
1509        &mut self,
1510        cx: &mut KernelContext<'_>,
1511        nst: &NewSessionTicketPayloadTls13,
1512    ) -> Result<(), Error> {
1513        let secret = self
1514            .resumption
1515            .derive_ticket_psk(&nst.nonce.0);
1516
1517        let now = self.config.current_time()?;
1518
1519        #[allow(unused_mut)]
1520        let mut value = persist::Tls13ClientSessionValue::new(
1521            self.suite,
1522            nst.ticket.clone(),
1523            secret.as_ref(),
1524            cx.peer_certificates
1525                .cloned()
1526                .unwrap_or_default(),
1527            &self.config.verifier,
1528            &self.config.client_auth_cert_resolver,
1529            now,
1530            nst.lifetime,
1531            nst.age_add,
1532            nst.extensions
1533                .max_early_data_size
1534                .unwrap_or_default(),
1535        );
1536
1537        if cx.is_quic() {
1538            if let Some(sz) = nst.extensions.max_early_data_size {
1539                if sz != 0 && sz != 0xffff_ffff {
1540                    return Err(PeerMisbehaved::InvalidMaxEarlyDataSize.into());
1541                }
1542            }
1543
1544            if let Some(quic_params) = &cx.quic.params {
1545                value.set_quic_params(quic_params);
1546            }
1547        }
1548
1549        self.session_storage
1550            .insert_tls13_ticket(self.server_name.clone(), value);
1551        Ok(())
1552    }

1554    fn handle_new_ticket_tls13(
1555        &mut self,
1556        cx: &mut ClientContext<'_>,
1557        nst: &NewSessionTicketPayloadTls13,
1558    ) -> Result<(), Error> {
1559        let mut kcx = KernelContext {
1560            peer_certificates: cx.common.peer_certificates.as_ref(),
1561            protocol: cx.common.protocol,
1562            quic: &cx.common.quic,
1563        };
1564        cx.common.tls13_tickets_received = cx
1565            .common
1566            .tls13_tickets_received
1567            .saturating_add(1);
1568        self.handle_new_ticket_impl(&mut kcx, nst)
1569    }
1570
1571    fn handle_key_update(
1572        &mut self,
1573        common: &mut CommonState,
1574        key_update_request: &KeyUpdateRequest,
1575    ) -> Result<(), Error> {
1576        if let Protocol::Quic = common.protocol {
1577            return Err(common.send_fatal_alert(
1578                AlertDescription::UnexpectedMessage,
1579                PeerMisbehaved::KeyUpdateReceivedInQuicConnection,
1580            ));
1581        }
1582
1583        // Mustn't be interleaved with other handshake messages.
1584        common.check_aligned_handshake()?;
1585
1586        if common.should_update_key(key_update_request)? {
1587            self.key_schedule
1588                .update_encrypter_and_notify(common);
1589        }
1590
1591        // Update our read-side keys.
1592        self.key_schedule
1593            .update_decrypter(common);
1594        Ok(())
1595    }
1596}
