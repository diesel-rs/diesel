293    /// Returns [`Choice::TRUE`] if `x < y`, and [`Choice::FALSE`] otherwise.

300        }
301        #[cfg(not(target_pointer_width = "32"))]
302        {
303            Self::from_u128_lsb((((!x) & y) | (((!x) | y) & (x).wrapping_sub(y))) >> (u128::BITS - 1)crate::primitives::bitlt!(x, y, u128::BITS))

307    #[cfg(any(target_pointer_width = "32", test))]
308    #[inline]
309    #[allow(clippy::cast_possible_truncation)]
310    const fn from_u128_lt_halves(x: u128, y: u128) -> Self {
311        // Full-width subtraction can introduce conditional borrow propagation.
312        // Keep the half-width predicates opaque so LLVM cannot combine them into a short-circuiting
313        // multiword comparison.
314        let x_hi = (x >> 64) as u64;
315        let y_hi = (y >> 64) as u64;
316        let hi_lt = Self::from_u64_lt(x_hi, y_hi).to_u8();
317        let hi_eq = Self::from_u64_eq(x_hi, y_hi).to_u8();
318        let lo_lt = Self::from_u64_lt(x as u64, y as u64).to_u8();
319        Self::from_u8_lsb(hi_lt | (hi_eq & lo_lt))
320    }

420    /// Convert [`Choice`] to a `u8`, attempting to apply a "best effort" optimization barrier.

423        // `black_box` is documented as working on a "best effort" basis. That's fine, this type is

620impl From<subtle::Choice> for Choice {
621    #[inline]
622    fn from(choice: subtle::Choice) -> Choice {
623        Choice(choice.unwrap_u8())
624    }
625}
626
627#[cfg(feature = "subtle")]
628impl From<Choice> for subtle::Choice {
629    #[inline]
630    fn from(choice: Choice) -> subtle::Choice {
631        subtle::Choice::from(choice.0)
632    }
633}

635#[cfg(feature = "subtle")]
636impl subtle::ConditionallySelectable for Choice {
637    #[inline]
638    fn conditional_select(a: &Self, b: &Self, choice: subtle::Choice) -> Self {
639        CtSelect::ct_select(a, b, choice.into())
640    }
641}

643#[cfg(feature = "subtle")]
644impl subtle::ConstantTimeEq for Choice {
645    #[inline]
646    fn ct_eq(&self, other: &Self) -> subtle::Choice {
647        CtEq::ct_eq(self, other).into()
648    }
649}

889    #[test]
890    fn from_u128_comparison_boundaries() {
891        const LT: Choice = Choice::from_u128_lt_halves((1 << 96) - 1, 1 << 96);
892        const LE: Choice = Choice::from_u128_le(1 << 64, 1 << 64);
893        assert!(LT.to_bool());
894        assert!(LE.to_bool());
895
896        let values = [
897            0,
898            1,
899            (1 << 32) - 1,
900            1 << 32,
901            (1 << 64) - 1,
902            1 << 64,
903            (1 << 64) + 1,
904            (1 << 96) - 1,
905            1 << 96,
906            u128::MAX - 1,
907            u128::MAX,
908        ];
909
910        for x in values {
911            for y in values {
912                assert_eq!(x < y, Choice::from_u128_lt_halves(x, y).to_bool());
913                assert_eq!(x < y, Choice::from_u128_lt(x, y).to_bool());
914                assert_eq!(x <= y, Choice::from_u128_le(x, y).to_bool());
915                assert_eq!(x < y, x.ct_lt(&y).to_bool());
916                assert_eq!(x > y, x.ct_gt(&y).to_bool());
917            }
918        }
919    }
