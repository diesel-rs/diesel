18pub(crate) struct SharedPool<M: ManageConnection + Send> {
19    pub(crate) statics: Builder<M>,
20    pub(crate) manager: M,
21    pub(crate) internals: Mutex<PoolInternals<M>>,
22    pub(crate) notify: Arc<Notify>,
23    pub(crate) statistics: AtomicStatistics,
24}
25
26impl<M: ManageConnection + Send> SharedPool<M> {
27    pub(crate) fn new(statics: Builder<M>, manager: M) -> Self {
28        Self {
29            statics,
30            manager,
31            internals: Mutex::new(PoolInternals::default()),
32            notify: Arc::new(Notify::new()),
33            statistics: AtomicStatistics::default(),
34        }
35    }
36
37    pub(crate) fn try_put(self: &Arc<Self>, conn: M::Connection) -> Result<(), M::Connection> {
38        let mut locked = self.internals.lock();
39        let mut approvals = locked.approvals(&self.statics, 1);
40        let Some(approval) = approvals.next() else {
41            return Err(conn);
42        };
43        let conn = Conn::new(conn);
44        locked.put(conn, Some(approval), self.clone());
45        Ok(())
46    }
47
48    pub(crate) fn reap(&self) -> ApprovalIter {
49        let mut locked = self.internals.lock();
50        let (iter, closed_idle_timeout, closed_max_lifetime) = locked.reap(&self.statics);
51        drop(locked);
52        self.statistics
53            .record_connections_reaped(closed_idle_timeout, closed_max_lifetime);
54        iter
55    }
56
57    pub(crate) fn start_get(self: &Arc<Self>) -> Getting<M> {
58        Getting::new(self.clone())
59    }
60
61    pub(crate) fn forward_error(&self, err: M::Error) {
62        self.statics.error_sink.sink(err);
63    }
64}

68pub(crate) struct PoolInternals<M: ManageConnection> {
69    conns: VecDeque<IdleConn<M::Connection>>,
70    num_conns: u32,
71    pending_conns: u32,
72    in_flight: u32,
73}

75impl<M: ManageConnection> PoolInternals<M> {
76    pub(crate) fn put(
77        &mut self,
78        conn: Conn<M::Connection>,
79        approval: Option<Approval>,
80        pool: Arc<SharedPool<M>>,
81    ) {
82        if approval.is_some() {
83            #[cfg(debug_assertions)]
84            {
85                self.pending_conns -= 1;
86                self.num_conns += 1;
87            }
88            #[cfg(not(debug_assertions))]
89            {
90                self.pending_conns = self.pending_conns.saturating_sub(1);
91                self.num_conns = self.num_conns.saturating_add(1);
92            }
93        }
94
95        // Queue it in the idle queue
96        let conn = IdleConn::from(conn);
97        match pool.statics.queue_strategy {
98            QueueStrategy::Fifo => self.conns.push_back(conn),
99            QueueStrategy::Lifo => self.conns.push_front(conn),
100        }
101
102        pool.notify.notify_one();
103    }
104
105    pub(crate) fn connect_failed(&mut self, _: Approval) {
106        #[cfg(debug_assertions)]
107        {
108            self.pending_conns -= 1;
109        }
110        #[cfg(not(debug_assertions))]
111        {
112            self.pending_conns = self.pending_conns.saturating_sub(1);
113        }
114    }
115
116    pub(crate) fn dropped(&mut self, num: u32, config: &Builder<M>) -> ApprovalIter {
117        #[cfg(debug_assertions)]
118        {
119            self.num_conns -= num;
120        }
121        #[cfg(not(debug_assertions))]
122        {
123            self.num_conns = self.num_conns.saturating_sub(num);
124        }
125
126        self.wanted(config)
127    }
128
129    pub(crate) fn wanted(&mut self, config: &Builder<M>) -> ApprovalIter {
130        let available = self.conns.len() as u32 + self.pending_conns;
131        let min_idle = config.min_idle.unwrap_or(0);
132        let wanted = min_idle.saturating_sub(available);
133        self.approvals(config, wanted)
134    }
135
136    fn approvals(&mut self, config: &Builder<M>, num: u32) -> ApprovalIter {
137        let current = self.num_conns + self.pending_conns;
138        let num = min(num, config.max_size.saturating_sub(current));
139        self.pending_conns += num;
140        ApprovalIter { num: num as usize }
141    }
142
143    pub(crate) fn reap(&mut self, config: &Builder<M>) -> (ApprovalIter, u64, u64) {
144        let mut closed_max_lifetime = 0;
145        let mut closed_idle_timeout = 0;
146        let now = Instant::now();
147        let before = self.conns.len();
148
149        self.conns.retain(|conn| {
150            let mut keep = true;
151            if let Some(timeout) = config.idle_timeout {
152                if now - conn.idle_start >= timeout {
153                    closed_idle_timeout += 1;
154                    keep &= false;
155                }
156            }
157            if let Some(lifetime) = config.max_lifetime {
158                if conn.conn.is_expired(now, lifetime) {
159                    closed_max_lifetime += 1;
160                    keep &= false;
161                }
162            }
163            keep
164        });
165
166        (
167            self.dropped((before - self.conns.len()) as u32, config),
168            closed_idle_timeout,
169            closed_max_lifetime,
170        )
171    }
172
173    pub(crate) fn state(&self, statistics: Statistics) -> State {
174        State {
175            connections: self.num_conns,
176            idle_connections: self.conns.len() as u32,
177            statistics,
178        }
179    }
180}

201    fn next(&mut self) -> Option<Self::Item> {
202        match self.num {
203            0 => None,
204            _ => {
205                self.num -= 1;
206                Some(Approval { _priv: () })
207            }
208        }
209    }

223pub(crate) struct Getting<M: ManageConnection + Send> {
224    inner: Arc<SharedPool<M>>,
225}
226
227impl<M: ManageConnection + Send> Getting<M> {
228    pub(crate) fn get(&self) -> (Option<Conn<M::Connection>>, ApprovalIter) {
229        let mut locked = self.inner.internals.lock();
230        if let Some(IdleConn { conn, .. }) = locked.conns.pop_front() {
231            return (Some(conn), locked.wanted(&self.inner.statics));
232        }
233
234        let approvals = match locked.in_flight > locked.pending_conns {
235            true => 1,
236            false => 0,
237        };
238
239        (None, locked.approvals(&self.inner.statics, approvals))
240    }
241}
242
243impl<M: ManageConnection + Send> Getting<M> {
244    fn new(inner: Arc<SharedPool<M>>) -> Self {
245        {
246            let mut locked = inner.internals.lock();
247            locked.in_flight += 1;
248        }
249        Getting { inner }
250    }
251}
252
253impl<M: ManageConnection + Send> Drop for Getting<M> {
254    fn drop(&mut self) {
255        let mut locked = self.inner.internals.lock();
256        locked.in_flight -= 1;
257    }
258}

304    pub(crate) get_waited: AtomicU64,

307    pub(crate) connections_created: AtomicU64,
308    pub(crate) connections_closed_broken: AtomicU64,
309    pub(crate) connections_closed_invalid: AtomicU64,
310    pub(crate) connections_closed_max_lifetime: AtomicU64,
311    pub(crate) connections_closed_idle_timeout: AtomicU64,
312}
313
314impl AtomicStatistics {
315    pub(crate) fn record_get_started(&self) {
316        self.get_started.fetch_add(1, Ordering::Relaxed);
317    }
318
319    pub(crate) fn record_get(&self, kind: StatsGetKind, wait_time_start: Option<Instant>) {
320        match kind {
321            StatsGetKind::Direct => self.get_direct.fetch_add(1, Ordering::Relaxed),
322            StatsGetKind::Waited => self.get_waited.fetch_add(1, Ordering::Relaxed),
323            StatsGetKind::TimedOut => self.get_timed_out.fetch_add(1, Ordering::Relaxed),
324        };
325
326        if let Some(wait_time_start) = wait_time_start {
327            let wait_time = Instant::now() - wait_time_start;
328            self.get_wait_time_micros
329                .fetch_add(wait_time.as_micros() as u64, Ordering::Relaxed);
330        }
331    }
332
333    pub(crate) fn record(&self, kind: StatsKind) {
334        match kind {
335            StatsKind::Created => &self.connections_created,
336            StatsKind::ClosedBroken => &self.connections_closed_broken,
337            StatsKind::ClosedInvalid => &self.connections_closed_invalid,
338        }
339        .fetch_add(1, Ordering::Relaxed);
340    }
341
342    pub(crate) fn record_connections_reaped(
343        &self,
344        closed_idle_timeout: u64,
345        closed_max_lifetime: u64,
346    ) {
347        self.connections_closed_idle_timeout
348            .fetch_add(closed_idle_timeout, Ordering::Relaxed);
349        self.connections_closed_max_lifetime
350            .fetch_add(closed_max_lifetime, Ordering::Relaxed);
351    }
352}
