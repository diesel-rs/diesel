5pub fn read_be_i32(buf: &mut &[u8]) -> Result<i32, Box<dyn Error + Sync + Send>> {
6    let val = buf
7        .get(..4)
8        .ok_or("invalid buffer size")?
9        .try_into()
10        .unwrap();
11    *buf = &buf[4..];
12    Ok(i32::from_be_bytes(val))
13}
