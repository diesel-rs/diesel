40/// A trait defining how to update a record and fetch the updated entry
41/// on a certain backend.
42///
43/// The only case where it is required to work with this trait is while
44/// implementing a new connection type.
45/// Otherwise use [`SaveChangesDsl`]
46///
47/// For implementing this trait for a custom backend:
48/// * The `Changes` generic parameter represents the changeset that should be stored
49/// * The `Output` generic parameter represents the type of the response.
50pub trait UpdateAndFetchResults<Changes, Output>: Connection {
51    /// See the traits documentation.
52    fn update_and_fetch(&mut self, changeset: Changes) -> QueryResult<Output>;
53}

58#[cfg(feature = "postgres")]
59impl<'b, Changes, Output> UpdateAndFetchResults<Changes, Output> for PgConnection
60where
61    Changes: Copy + AsChangeset<Target = <Changes as HasTable>::Table> + IntoUpdateTarget,
62    Update<Changes, Changes>: LoadQuery<'b, PgConnection, Output>,
63    <Changes::Table as Table>::AllColumns: ValidGrouping<()>
64        + crate::expression::SelectableExpression<
65            crate::query_builder::returning::ReturningQuerySource<
66                crate::query_builder::returning::UpdateStmt,
67                Changes::Table,
68            >,
69        >,
70    <<Changes::Table as Table>::AllColumns as ValidGrouping<()>>::IsAggregate:
71        MixedAggregates<is_aggregate::No, Output = is_aggregate::No>,
72{
73    fn update_and_fetch(&mut self, changeset: Changes) -> QueryResult<Output> {
74        crate::update(changeset).set(changeset).get_result(self)
75    }
76}

159/// Sugar for types which implement both `AsChangeset` and `Identifiable`
160///
161/// On backends which support the `RETURNING` keyword,
162/// `foo.save_changes(&conn)` is equivalent to
163/// `update(&foo).set(&foo).get_result(&conn)`.
164/// On other backends, two queries will be executed.
165///
166/// # Example
167///
168/// ```rust
169/// # include!("../doctest_setup.rs");
170/// # use schema::animals;
171/// #
172/// #[derive(Queryable, Debug, PartialEq)]
173/// struct Animal {
174///     id: i32,
175///     species: String,
176///     legs: i32,
177///     name: Option<String>,
178/// }
179///
180/// #[derive(AsChangeset, Identifiable)]
181/// #[diesel(table_name = animals)]
182/// struct AnimalForm<'a> {
183///     id: i32,
184///     name: &'a str,
185/// }
186///
187/// # fn main() {
188/// #     run_test();
189/// # }
190/// #
191/// # fn run_test() -> QueryResult<()> {
192/// #     use self::animals::dsl::*;
193/// #     let connection = &mut establish_connection();
194/// let form = AnimalForm {
195///     id: 2,
196///     name: "Super scary",
197/// };
198/// let changed_animal = form.save_changes(connection)?;
199/// let expected_animal = Animal {
200///     id: 2,
201///     species: String::from("spider"),
202///     legs: 8,
203///     name: Some(String::from("Super scary")),
204/// };
205/// assert_eq!(expected_animal, changed_animal);
206/// #     Ok(())
207/// # }
208/// ```
209pub trait SaveChangesDsl<Conn> {
210    /// See the trait documentation.
211    fn save_changes<T>(self, connection: &mut Conn) -> QueryResult<T>
212    where
213        Self: Sized,
214        Conn: UpdateAndFetchResults<Self, T>,
215    {
216        connection.update_and_fetch(self)
217    }
218}
