11    ///
12    /// Note that this function follows SQL semantics around `None`/`null` values,
13    /// so `eq(None)` will never match. Use [`is_null`](ExpressionMethods::is_null()) instead.
14    ///
15    #[cfg_attr(
16        any(feature = "__sqlite-shared", feature = "postgres"),
17        doc = "To get behavior that is more like the Rust `=` operator you can also use the"
18    )]
19    #[cfg_attr(
20        feature = "__sqlite-shared",
21        doc = "sqlite-specific [`is`](crate::SqliteExpressionMethods::is())"
22    )]
23    #[cfg_attr(all(feature = "__sqlite-shared", feature = "postgres"), doc = "or the")]
24    #[cfg_attr(
25        feature = "postgres",
26        doc = "postgres-specific [`is_not_distinct_from`](crate::PgExpressionMethods::is_not_distinct_from())"
27    )]

29    ///
30    /// # Example
31    ///
32    /// ```rust
33    /// # include!("../doctest_setup.rs");
34    /// #
35    /// # fn main() {
36    /// #     use schema::users::dsl::*;
37    /// #     let connection = &mut establish_connection();
38    /// let data = users.select(id).filter(name.eq("Sean"));

40    /// # }

44    /// ```rust

48    /// # // native sqlite library, which miri cannot track provenance for.
49    /// # #[cfg(miri)]
50    /// # fn main() {}
51    /// # #[cfg(not(miri))]

106    /// Creates a SQL `IN` statement.
107    ///
108    /// Queries using this method will not typically be
109    /// placed in the prepared statement cache. However,
110    /// in cases when a subquery is passed to the method, that
111    /// query will use the cache (assuming the subquery
112    /// itself is safe to cache).
113    /// On PostgreSQL, this method automatically performs a `= ANY()`
114    /// query if this is possible. For cases where this is not possible
115    /// like for example if values is a vector of arrays we
116    /// generate an ordinary `IN` expression instead.
117    ///
118    /// # Example
119    ///
120    /// ```rust
121    /// # include!("../doctest_setup.rs");
122    /// #
123    /// # fn main() {
124    /// #     use schema::users;
125    /// #     use schema::posts;
126    /// #     let connection = &mut establish_connection();
127    /// #     diesel::sql_query("INSERT INTO users (name) VALUES
128    /// #         ('Jim')").execute(connection).unwrap();
129    /// let data = users::table
130    ///     .select(users::id)
131    ///     .filter(users::name.eq_any(vec!["Sean", "Jim"]));
132    /// assert_eq!(Ok(vec![1, 3]), data.load(connection));
133    ///
134    /// // Calling `eq_any` with an empty array is the same as doing `WHERE 1=0`
135    /// let data = users::table
136    ///     .select(users::id)
137    ///     .filter(users::name.eq_any(Vec::<String>::new()));
138    /// assert_eq!(Ok(vec![]), data.load::<i32>(connection));
139    ///
140    /// // Calling `eq_any` with a subquery is the same as using
141    /// // `WHERE {column} IN {subquery}`.
142    ///
143    /// let subquery = users::table
144    ///     .filter(users::name.eq("Sean"))
145    ///     .select(users::id)
146    ///     .into_boxed();
147    /// let data = posts::table
148    ///     .select(posts::id)
149    ///     .filter(posts::user_id.eq_any(subquery));
150    /// assert_eq!(Ok(vec![1, 2]), data.load::<i32>(connection));
151    ///
152    /// # }
153    /// ```
154    #[doc(alias = "in")]
155    fn eq_any<T>(self, values: T) -> dsl::EqAny<Self, T>
156    where
157        Self::SqlType: SqlType,
158        T: AsInExpression<Self::SqlType>,
159    {
160        Grouped(In::new(self, values.as_in_expression()))
161    }

163    /// Creates a SQL `NOT IN` statement.
164    ///
165    /// Queries using this method will not be
166    /// placed in the prepared statement cache. On PostgreSQL, this
167    /// method automatically performs a `!= ALL()` query if this is possible.
168    /// For cases where this is not possible
169    /// like for example if values is a vector of arrays we
170    /// generate an ordinary `NOT IN` expression instead.
171    ///
172    /// # Example
173    ///
174    /// ```rust
175    /// # include!("../doctest_setup.rs");
176    /// #
177    /// # fn main() {
178    /// #     use schema::users::dsl::*;
179    /// #     let connection = &mut establish_connection();
180    /// #     diesel::sql_query("INSERT INTO users (name) VALUES
181    /// #         ('Jim')").execute(connection).unwrap();
182    /// let data = users.select(id).filter(name.ne_all(vec!["Sean", "Jim"]));
183    /// assert_eq!(Ok(vec![2]), data.load(connection));
184    ///
185    /// let data = users.select(id).filter(name.ne_all(vec!["Tess"]));
186    /// assert_eq!(Ok(vec![1, 3]), data.load(connection));
187    ///
188    /// // Calling `ne_any` with an empty array is the same as doing `WHERE 1=1`
189    /// let data = users.select(id).filter(name.ne_all(Vec::<String>::new()));
190    /// assert_eq!(Ok(vec![1, 2, 3]), data.load(connection));
191    /// # }
192    /// ```
193    #[doc(alias = "in")]
194    fn ne_all<T>(self, values: T) -> dsl::NeAny<Self, T>
195    where
196        Self::SqlType: SqlType,
197        T: AsInExpression<Self::SqlType>,
198    {
199        Grouped(NotIn::new(self, values.as_in_expression()))
200    }

804/// Methods present on all expressions
805pub trait NullableExpressionMethods: Expression + Sized {
806    /// Converts this potentially non-null expression into one which is treated
807    /// as nullable. This method has no impact on the generated SQL, and is only
808    /// used to allow certain comparisons that would otherwise fail to compile.
809    ///
810    /// # Example
811    /// ```no_run
812    /// # #![allow(dead_code)]
813    /// # include!("../doctest_setup.rs");
814    /// # use diesel::sql_types::*;
815    /// # use schema::users;
816    /// #
817    /// table! {
818    ///     posts {
819    ///         id -> Integer,
820    ///         user_id -> Integer,
821    ///         author_name -> Nullable<VarChar>,
822    ///     }
823    /// }
824    /// #
825    /// # joinable!(posts -> users (user_id));
826    /// # allow_tables_to_appear_in_same_query!(posts, users);
827    ///
828    /// fn main() {
829    ///     use self::posts::dsl::{author_name, posts};
830    ///     use self::users::dsl::*;
831    ///     let connection = &mut establish_connection();
832    ///
833    ///     let data = users
834    ///         .inner_join(posts)
835    ///         .filter(name.nullable().eq(author_name))
836    ///         .select(name)
837    ///         .load::<String>(connection);
838    ///     println!("{:?}", data);
839    /// }
840    /// ```
841    fn nullable(self) -> dsl::Nullable<Self> {
842        nullable::Nullable::new(self)
843    }
844
845    /// Converts this potentially nullable expression into one which will be **assumed**
846    /// to be not-null. This method has no impact on the generated SQL, however it will
847    /// enable you to attempt deserialization of the returned value in a non-`Option`.
848    ///
849    /// This is meant to cover for cases where you know that given the `WHERE` clause
850    /// the field returned by the database will never be `NULL`.
851    ///
852    /// This **will cause runtime errors** on `load()` if the "assume" turns out to be incorrect.
853    ///
854    /// # Examples
855    /// ## Normal usage
856    /// ```rust
857    /// # #![allow(dead_code)]
858    /// # include!("../doctest_setup.rs");
859    /// # use diesel::sql_types::*;
860    /// #
861    /// table! {
862    ///     animals {
863    ///         id -> Integer,
864    ///         species -> VarChar,
865    ///         legs -> Integer,
866    ///         name -> Nullable<VarChar>,
867    ///     }
868    /// }
869    ///
870    /// # // Reading the query result requires reading memory allocated by the
871    /// # // native sqlite library, which miri cannot track provenance for.
872    /// #[cfg(miri)]
873    /// fn main() {}
874    /// #[cfg(not(miri))]
875    /// fn main() {
876    ///     use self::animals::dsl::*;
877    ///     let connection = &mut establish_connection();
878    ///
879    ///     let result = animals
880    ///         .filter(name.is_not_null())
881    ///         .select(name.assume_not_null())
882    ///         .load::<String>(connection);
883    ///     assert!(result.is_ok());
884    /// }
885    /// ```
886    ///
887    /// ## Incorrect usage
888    /// ```rust
889    /// # #![allow(dead_code)]
890    /// # include!("../doctest_setup.rs");
891    /// # use diesel::sql_types::*;
892    /// #
893    /// table! {
894    ///     animals {
895    ///         id -> Integer,
896    ///         species -> VarChar,
897    ///         legs -> Integer,
898    ///         name -> Nullable<VarChar>,
899    ///     }
900    /// }
901    ///
902    /// # // Reading the query result requires reading memory allocated by the
903    /// # // native sqlite library, which miri cannot track provenance for.
904    /// #[cfg(miri)]
905    /// fn main() {}
906    /// #[cfg(not(miri))]
907    /// fn main() {
908    ///     use diesel::result::{Error, UnexpectedNullError};
909    ///     use self::animals::dsl::*;
910    ///     let connection = &mut establish_connection();
911    ///
912    ///     let result = animals
913    ///         .select(name.assume_not_null())
914    ///         .load::<String>(connection);
915    ///     assert!(matches!(
916    ///         result,
917    ///         Err(Error::DeserializationError(err)) if err.is::<UnexpectedNullError>()
918    ///     ));
919    /// }
920    /// ```
921    ///
922    /// ## Advanced usage - use only if you're sure you know what you're doing!
923    ///
924    /// This will cause the `Option` to be `None` where the `left_join` succeeded but the
925    /// `author_name` turned out to be `NULL`, due to how `Option` deserialization works.
926    /// (see [`Queryable` documentation](crate::deserialize::Queryable))
927    ///
928    /// ```rust
929    /// # #![allow(dead_code)]
930    /// # include!("../doctest_setup.rs");
931    /// # use diesel::sql_types::*;
932    /// # use schema::users;
933    /// #
934    /// table! {
935    ///     posts {
936    ///         id -> Integer,
937    ///         user_id -> Integer,
938    ///         author_name -> Nullable<Text>,
939    ///     }
940    /// }
941    /// #
942    /// # joinable!(posts -> users (user_id));
943    /// # allow_tables_to_appear_in_same_query!(posts, users);
944    ///
945    /// # // Reading the query result requires reading memory allocated by the
946    /// # // native sqlite library, which miri cannot track provenance for.
947    /// #[cfg(miri)]
948    /// fn main() {}
949    /// #[cfg(not(miri))]
950    /// fn main() {
951    ///     use self::posts;
952    ///     use self::users;
953    ///     let connection = &mut establish_connection();
954    ///
955    /// #   diesel::sql_query("ALTER TABLE posts ADD COLUMN author_name Text")
956    /// #       .execute(connection)
957    /// #       .unwrap();
958    /// #   diesel::update(posts::table.filter(posts::user_id.eq(1)))
959    /// #       .set(posts::author_name.eq("Sean"))
960    /// #       .execute(connection);
961    ///
962    ///     let result = posts::table
963    ///         .left_join(users::table)
964    ///         .select((
965    ///             posts::id,
966    ///             (users::id, posts::author_name.assume_not_null()).nullable(),
967    ///         ))
968    ///         .order_by(posts::id)
969    ///         .load::<(i32, Option<(i32, String)>)>(connection);
970    ///     let expected = Ok(vec![
971    ///         (1, Some((1, "Sean".to_owned()))),
972    ///         (2, Some((1, "Sean".to_owned()))),
973    ///         (3, None),
974    ///     ]);
975    ///     assert_eq!(expected, result);
976    /// }
977    /// ```
978    fn assume_not_null(self) -> dsl::AssumeNotNull<Self> {
979        assume_not_null::AssumeNotNull::new(self)
980    }
981}
