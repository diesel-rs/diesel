285impl<T> CopyToQuery<T, NotSet>
286where
287    T: CopyTarget,
288{
289    /// Copy data from the database by returning an iterator of deserialized data
290    ///
291    /// This function allows to easily load data from the database via a `COPY TO` statement.
292    /// It does **not** allow to configure any settings via the `with_*` method, as it internally
293    /// sets the required options itself. It will use the binary format to deserialize the result
294    /// into the specified type `U`. Column selection is performed via [`Selectable`].
295    pub fn load<U, C>(self, conn: &mut C) -> QueryResult<impl Iterator<Item = QueryResult<U>> + '_>
296    where
297        U: FromSqlRow<<U::SelectExpression as Expression>::SqlType, Pg> + Selectable<Pg>,
298        U::SelectExpression: AppearsOnTable<T::Table> + CopyTarget<Table = T::Table>,
299        C: ExecuteCopyToConnection,
300    {
301        let command = CopyToCommand {
302            p: PhantomData::<U::SelectExpression>,
303            options: CopyToOptions {
304                header: None,
305                common: CommonOptions {
306                    format: Some(CopyFormat::Binary),
307                    ..Default::default()
308                },
309            },
310        };
311        let out = ExecuteCopyToConnection::execute(conn, command)?;
312        decode_data::<U, C>(out)
313    }
314}
315
316fn decode_data<'a, U, C>(
317    mut out: C::CopyToBuffer<'a>,
318) -> QueryResult<impl Iterator<Item = QueryResult<U>> + 'a>
319where
320    U: FromSqlRow<<U::SelectExpression as Expression>::SqlType, Pg> + Selectable<Pg>,
321    C: ExecuteCopyToConnection + 'a,
322{
323    let io_result_mapper = |e| crate::result::Error::DeserializationError(Box::new(e));
324    // see https://www.postgresql.org/docs/current/sql-copy.html for
325    // a description of the binary format
326    //
327    // We don't write oids
328
329    let buffer = out.fill_buf().map_err(io_result_mapper)?;
330    expect_buffer_len(buffer, super::COPY_MAGIC_HEADER.len() + 8)?;
331    if buffer[..super::COPY_MAGIC_HEADER.len()] != super::COPY_MAGIC_HEADER {
332        return Err(crate::result::Error::DeserializationError(
333            "Unexpected protocol header".into(),
334        ));
335    }
336    // we care only about bit 16-31 here, so we can just skip the bytes in between
337    let flags_backward_incompatible = i16::from_be_bytes(
338        (&buffer[super::COPY_MAGIC_HEADER.len() + 2..super::COPY_MAGIC_HEADER.len() + 4])
339            .try_into()
340            .expect("Exactly 2 byte"),
341    );
342    if flags_backward_incompatible != 0 {
343        return Err(crate::result::Error::DeserializationError(
344            ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Unexpected flag value: {0:x}",
                flags_backward_incompatible))
    })format!("Unexpected flag value: {flags_backward_incompatible:x}").into(),
345        ));
346    }
347    let header_size = usize::try_from(i32::from_be_bytes(
348        (&buffer[super::COPY_MAGIC_HEADER.len() + 4..super::COPY_MAGIC_HEADER.len() + 8])
349            .try_into()
350            .expect("Exactly 4 byte"),
351    ))
352    .map_err(|e| crate::result::Error::DeserializationError(Box::new(e)))?;
353    out.consume(super::COPY_MAGIC_HEADER.len() + 8 + header_size);
354    let mut len = None;
355    let mut finished = false;
356    Ok(core::iter::from_fn(move || {
357        if finished {
358            return None;
359        }
360        if let Some(len) = len {
361            out.consume(len);
362            if let Err(e) = out.fill_buf().map_err(io_result_mapper) {
363                finished = true;
364                return Some(Err(e));
365            }
366        }
367        let buffer = C::get_buffer(&out);
368        len = Some(buffer.len());
369        if let Err(e) = expect_buffer_len(buffer, 2) {
370            finished = true;
371            return Some(Err(e));
372        }
373        let tuple_count = i16::from_be_bytes((&buffer[..2]).try_into().expect("Exactly 2 bytes"));
374        if tuple_count > 0 {
375            let tuple_count = match usize::try_from(tuple_count) {
376                Ok(o) => o,
377                Err(e) => {
378                    return Some(Err(crate::result::Error::DeserializationError(Box::new(e))));
379                }
380            };
381            let mut buffers = Vec::with_capacity(tuple_count);
382            let mut offset = 2;
383            for _t in 0..tuple_count {
384                if let Err(e) = expect_buffer_len(buffer, offset + 4) {
385                    finished = true;
386                    return Some(Err(e));
387                }
388                let data_size = i32::from_be_bytes(
389                    (&buffer[offset..offset + 4])
390                        .try_into()
391                        .expect("Exactly 4 bytes"),
392                );
393                if data_size < 0 {
394                    offset += 4;
395                    buffers.push(None);
396                } else {
397                    match usize::try_from(data_size) {
398                        Ok(data_size) => {
399                            if let Err(e) = expect_buffer_len(buffer, offset + 4 + data_size) {
400                                finished = true;
401                                return Some(Err(e));
402                            }
403                            buffers.push(Some(&buffer[offset + 4..offset + 4 + data_size]));
404                            offset += data_size + 4;
405                        }
406                        Err(e) => {
407                            return Some(Err(crate::result::Error::DeserializationError(
408                                Box::new(e),
409                            )));
410                        }
411                    }
412                }
413            }
414
415            let row = C::make_row(&out, buffers);
416            Some(U::build_from_row(&row).map_err(crate::result::Error::DeserializationError))
417        } else {
418            out.consume(len.expect("It's always set here"));
419            finished = true;
420            if let Err(e) = out.fill_buf().map_err(io_result_mapper) {
421                return Some(Err(e));
422            }
423            None
424        }
425    }))
426}

428fn expect_buffer_len(buffer: &[u8], expected: usize) -> QueryResult<()> {
429    if buffer.len() < expected {
430        Err(crate::result::Error::DeserializationError(
431            ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Expected at least {1} bytes but got {0} bytes",
                buffer.len(), expected))
    })format!(
432                "Expected at least {expected} bytes but got {} bytes",
433                buffer.len()
434            )
435            .into(),
436        ))
437    } else {
438        Ok(())
439    }
440}

552/// Creates a `COPY TO` statement
553///
554/// This function constructs a `COPY TO` statement which copies data
555/// from the database **to** a client side target. It's designed to move
556/// larger amounts of data out of the database.
557///
558/// This function accepts a target selection (table name or list of columns) as argument.
559///
560/// There are two ways to use a `COPY TO` statement with diesel:
561///
562/// * By using [`CopyToQuery::load`] directly to load the deserialized result
563///   directly into a specified type
564/// * By using the `with_*` methods to configure the format sent by the database
565///   and then by calling [`CopyToQuery::load_raw`] to receive the raw data
566///   sent by the database.
567///
568/// The first variant uses the `BINARY` format internally to receive
569/// the selected data efficiently. It automatically sets the right options
570/// and does not allow to change them via `with_*` methods.
571///
572/// The second variant allows you to control the behaviour of the
573/// generated `COPY TO` statement in detail. You can use the various
574/// `with_*` methods for that before issuing the statement via [`CopyToQuery::load_raw`].
575/// That method will return an type that implements [`std::io::BufRead`], which
576/// allows you to directly read the response from the database in the configured
577/// format.
578/// See [the postgresql documentation](https://www.postgresql.org/docs/current/sql-copy.html)
579/// for more details about the supported formats.
580///
581/// If you don't have any specific needs you should prefer using the more
582/// convenient first variant.
583///
584/// This functionality is postgresql specific.
585///
586/// # Examples
587///
588/// ## Via [`CopyToQuery::load()`]
589///
590/// ```rust
591/// # include!("../../../doctest_setup.rs");
592/// # use crate::schema::users;
593///
594/// #[derive(Queryable, Selectable, PartialEq, Debug)]
595/// #[diesel(table_name = users)]
596/// #[diesel(check_for_backend(diesel::pg::Pg))]
597/// struct User {
598///     name: String,
599/// }
600///
601/// # fn run_test() -> QueryResult<()> {
602/// # let connection = &mut establish_connection();
603/// let out = diesel::copy_to(users::table)
604///     .load::<User, _>(connection)?
605///     .collect::<Result<Vec<_>, _>>()?;
606///
607/// assert_eq!(
608///     out,
609///     vec![
610///         User {
611///             name: "Sean".into()
612///         },
613///         User {
614///             name: "Tess".into()
615///         }
616///     ]
617/// );
618/// # Ok(())
619/// # }
620/// # fn main() {
621/// #    run_test().unwrap();
622/// # }
623/// ```
624///
625/// ## Via [`CopyToQuery::load_raw()`]
626///
627/// ```rust
628/// # include!("../../../doctest_setup.rs");

630/// # use crate::schema::users;
631/// use diesel::pg::CopyFormat;
632/// use std::io::Read;
633/// # let connection = &mut establish_connection();
634///
635/// let mut copy = diesel::copy_to(users::table)
636///     .with_format(CopyFormat::Csv)
637///     .load_raw(connection)?;
638///
639/// let mut out = String::new();
640/// copy.read_to_string(&mut out).unwrap();
641/// assert_eq!(out, "1,Sean\n2,Tess\n");
642/// # Ok(())
643/// # }
644/// # fn main() {
645/// #    run_test().unwrap();
646/// # }
647/// ```
648#[cfg(feature = "postgres_backend")]
649pub fn copy_to<T>(target: T) -> CopyToQuery<T, NotSet>
650where
651    T: CopyTarget,
652{
653    CopyToQuery {
654        target,
655        options: NotSet,
656    }

671    /// Mimics `CopyToBuffer`: hands out one `CopyData` message at a time
672    /// and only moves on to the next one once the current one is consumed
673    struct Messages {
674        messages: VecDeque<Vec<u8>>,
675        offset: usize,
676    }
677
678    impl Messages {
679        fn new(messages: Vec<Vec<u8>>) -> Self {
680            Self {
681                messages: messages.into(),
682                offset: 0,
683            }
684        }
685
686        fn data_slice(&self) -> &[u8] {
687            self.messages
688                .front()
689                .map(|m| &m[self.offset..])
690                .unwrap_or_default()
691        }
692    }
693
694    impl Read for Messages {
695        fn read(&mut self, buf: &mut [u8]) -> std::io::Result<usize> {
696            let data = self.fill_buf()?;
697            let len = usize::min(buf.len(), data.len());
698            buf[..len].copy_from_slice(&data[..len]);
699            self.consume(len);
700            Ok(len)
701        }
702    }
703
704    impl BufRead for Messages {
705        fn fill_buf(&mut self) -> std::io::Result<&[u8]> {
706            if self.data_slice().is_empty() && self.messages.pop_front().is_some() {
707                self.offset = 0;
708            }
709            Ok(self.data_slice())
710        }
711
712        fn consume(&mut self, amt: usize) {
713            let len = self.messages.front().map_or(0, Vec::len);
714            self.offset = usize::min(len, self.offset + amt);
715        }
716    }

819    #[track_caller]
820    fn first_row_error(first_message: &[&[u8]]) -> String {
821        let first_message = [
822            COPY_MAGIC_HEADER.as_slice(),
823            [0; 8].as_slice(),
824            &first_message.concat(),
825        ]
826        .concat();
827        let mut rows = decode(vec![first_message]).unwrap();
828        let error = match rows.next() {
829            Some(Err(e)) => e.to_string(),
830            Some(Ok(row)) => panic!("expected an error, got {row:?}"),
831            None => panic!("expected an error, got the end of the data"),
832        };
833        assert!(
834            rows.next().is_none(),
835            "the iterator must end after an error"
836        );
837        error
838    }
