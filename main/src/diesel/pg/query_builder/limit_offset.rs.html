24impl QueryFragment<Pg> for BoxedCloneLimitOffsetClause<'_, Pg> {
25    fn walk_ast<'b>(&'b self, mut out: AstPass<'_, 'b, Pg>) -> QueryResult<()> {
26        if let Some(ref limit) = self.limit {
