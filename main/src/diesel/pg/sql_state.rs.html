1//! The SQLSTATE codes PostgreSQL defines, as listed in
2//! [its documentation](https://www.postgresql.org/docs/current/errcodes-appendix.html)
3//! and taken from
4//! [`errcodes.txt`](https://github.com/postgres/postgres/blob/REL_18_0/src/backend/utils/errcodes.txt).
5//!
6//! Each constant is named after PostgreSQL's `ERRCODE_` macro and documented
7//! with its PL/pgSQL condition name.
8//!
9//! ```rust
10//! use diesel::pg::sql_state;
11//! use diesel::result::SqlState;
12//!
13//! fn retryable(code: SqlState) -> bool {
14//!     matches!(
15//!         code,
16//!         sql_state::T_R_SERIALIZATION_FAILURE | sql_state::T_R_DEADLOCK_DETECTED
17//!     )
18//! }
19//!
20//! assert!(retryable("40001".parse().unwrap()));
21//! assert!(!retryable("23505".parse().unwrap()));
22//! ```
23
24use crate::result::SqlState;
25
26macro_rules! codes {
27    ($($(#[$doc:meta])* $name:ident = $code:literal;)*) => {
28        $(
29            $(#[$doc])*
30            pub const $name: SqlState = SqlState::new(*$code).expect("invalid SQLSTATE");
31        )*
32    };
33}
