8#[cfg(all(feature = "postgres_backend", feature = "uuid"))]
9impl FromSql<Uuid, Pg> for uuid::Uuid {
10    fn from_sql(value: PgValue<'_>) -> deserialize::Result<Self> {
11        uuid::Uuid::from_slice(value.as_bytes()).map_err(Into::into)
12    }
13}
14
15#[cfg(all(feature = "postgres_backend", feature = "uuid"))]
16impl ToSql<Uuid, Pg> for uuid::Uuid {
17    fn to_sql<'b>(&'b self, out: &mut Output<'b, '_, Pg>) -> serialize::Result {
18        out.write_all(self.as_bytes())
19            .map(|_| IsNull::No)
20            .map_err(Into::into)
21    }
22}

24#[cfg(test)]
25mod tests {
26    use super::*;
27    use crate::test_helpers::format_error;

29    #[diesel_test_helper::test]
30    fn uuid_to_sql() {
31        use crate::query_builder::bind_collector::ByteWrapper;
32
33        let mut buffer = Vec::new();
34        let bytes = [
35            0xFF_u8, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0x61, 0x62, 0x63, 0x64, 0x65, 0x66,
36            0x31, 0x32,
37        ];
38
39        let test_uuid = uuid::Uuid::from_slice(&bytes).unwrap();
40        let mut bytes = Output::test(ByteWrapper(&mut buffer));
41        ToSql::<Uuid, Pg>::to_sql(&test_uuid, &mut bytes).unwrap();
42        assert_eq!(&buffer, test_uuid.as_bytes());
43    }
44
45    #[diesel_test_helper::test]
46    fn some_uuid_from_sql() {
47        let bytes = [
48            0xFF_u8, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0x61, 0x62, 0x63, 0x64, 0x65, 0x66,
49            0x31, 0x32,
50        ];
51        let input_uuid = uuid::Uuid::from_slice(&bytes).unwrap();
52        let output_uuid =
53            <uuid::Uuid as FromSql<Uuid, Pg>>::from_sql(PgValue::for_test(input_uuid.as_bytes()))
54                .unwrap();
55        assert_eq!(input_uuid, output_uuid);
56    }
57
58    #[diesel_test_helper::test]
59    fn bad_uuid_from_sql() {
60        let uuid = uuid::Uuid::from_sql(PgValue::for_test(b"boom"));
61        assert!(uuid.is_err());
62        // The error message changes slightly between different
63        // uuid versions, so we just check on the relevant parts
64        // The exact error messages are either:
65        // "invalid bytes length: expected 16, found 4"
66        // or
67        // "invalid length: expected 16 bytes, found 4"
68        let error_message = format_error(&*uuid.unwrap_err());
69        assert!(error_message.starts_with("invalid"));
70        assert!(error_message.contains("length"));
71        assert!(error_message.contains("expected 16"));
72        assert!(error_message.ends_with("found 4"));
73    }
74
75    #[diesel_test_helper::test]
76    fn no_uuid_from_sql() {
77        let uuid = uuid::Uuid::from_nullable_sql(None);
78        assert_eq!(
79            format_error(&*uuid.unwrap_err()),
80            "Unexpected null for non-null column"
81        );
82    }
