27/// Refer to the pre-modification value of `col` in a Mariadb `RETURNING`
28/// clause.
29///
30/// This corresponds to the SQL `RETURNING OLD_VALUE(col)` syntax introduced in
31/// Mariadb 13.0.
32///
33/// # Requires Mariadb 13.0 or newer
34///
35/// Diesel emits `OLD_VALUE(col)` in the SQL it sends to the database. Earlier
36/// versions of Mariadb will reject the query at execution time.
37///
38/// # Statement compatibility
39///
40/// `old_value(col)` is valid inside the `RETURNING` clause of:
41///
42/// * an `UPDATE` statement, where it has the same Rust SQL type as `col`
43///   (since every returned row necessarily came from a pre-existing row).
44///
45/// Use of `old_value(col)` in `INSERT` or `DELETE` `RETURNING`
46/// is rejected at compile time, because it is invalid
47/// there. (Note that `ON CONFLICT DO NOTHING` never returns untouched rows.)
48///
49/// # Example
50///
51/// ```rust
52/// # include!("../../doctest_setup.rs");
53/// #
54/// # #[cfg(feature = "mariadb")]
55/// # fn main() {
56/// #     use schema::users::dsl::*;
57/// #     use diesel::mariadb::returning::old_value;
58/// #     let connection = &mut establish_connection();
59/// #     // `RETURNING OLD_VALUE(col)` requires Mariadb 13.0+
60/// #     if !mariadb_server_supports_update_returning(connection) { return; }
61/// let was_and_now = diesel::update(users.find(1))
62///     .set(name.eq("Updated"))
63///     .returning((old_value(name), name))
64///     .get_result::<(String, String)>(connection);
65/// assert_eq!(Ok(("Sean".to_string(), "Updated".to_string())), was_and_now);
66/// # }
67/// # #[cfg(not(feature = "mariadb"))]
68/// # fn main() {}
69/// ```
70pub fn old_value<C: Column>(col: C) -> old_value<C> {
71    OldValue::new(col)
72}
