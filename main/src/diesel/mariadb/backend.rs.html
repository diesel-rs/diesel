83impl MapErrorNumber for Mariadb {
84    fn map_error_number(error_number: u32) -> crate::result::DatabaseErrorKind {
85        // These values are not exposed by the C API, but are documented
86        // at https://mariadb.com/docs/server/reference/error-codes/mariadb-error-code-reference
87        match error_number {
88            1022 | 1062 | 1586 | 1859 => DatabaseErrorKind::UniqueViolation,
89            1216 | 1217 | 1451 | 1557 | 1452 | 1830 | 1834 => {
90                DatabaseErrorKind::ForeignKeyViolation
91            }
92            1792 => DatabaseErrorKind::ReadOnlyTransaction,
93            1048 | 1364 => DatabaseErrorKind::NotNullViolation,
94            4025 => DatabaseErrorKind::CheckViolation,
95            1213 => DatabaseErrorKind::SerializationFailure,
96            _ => DatabaseErrorKind::Unknown,
97        }
98    }
