14/// Represents that a structure can be used to insert a new row into the
15/// database. This is automatically implemented for `&[T]` and `&Vec<T>` for
16/// inserting more than one record.
17///
18/// This trait can be [derived](derive@Insertable)
19pub trait Insertable<T> {
20    /// The `VALUES` clause to insert these records
21    ///
22    /// The types used here are generally internal to Diesel.
23    /// Implementations of this trait should use the `Values`
24    /// type of other `Insertable` types.
25    /// For example `<diesel::dsl::Eq<column, &str> as Insertable<table>>::Values`.
26    type Values;
27
28    /// Construct `Self::Values`
29    ///
30    /// Implementations of this trait typically call `.values`
31    /// on other `Insertable` types.
32    fn values(self) -> Self::Values;
33
34    /// Insert `self` into a given table.
35    ///
36    /// `foo.insert_into(table)` is identical to `insert_into(table).values(foo)`.
37    /// However, when inserting from a select statement,
38    /// this form is generally preferred.
39    ///
40    /// # Example
41    ///
42    /// ```rust
43    /// # include!("doctest_setup.rs");
44    /// #
45    /// # // Reading the query result requires reading memory allocated by the
46    /// # // native sqlite library, which miri cannot track provenance for.
47    /// # #[cfg(miri)]
48    /// # fn main() {}
49    /// # #[cfg(not(miri))]
50    /// # fn main() {
51    /// #     run_test().unwrap();
52    /// # }
53    /// #
54    /// # fn run_test() -> QueryResult<()> {
55    /// #     use schema::{posts, users};
56    /// #     let conn = &mut establish_connection();
57    /// #     diesel::delete(posts::table).execute(conn)?;
58    /// users::table
59    ///     .select((users::name.concat("'s First Post"), users::id))
60    ///     .insert_into(posts::table)
61    ///     .into_columns((posts::title, posts::user_id))
62    ///     .execute(conn)?;
63    ///
64    /// let inserted_posts = posts::table.select(posts::title).load::<String>(conn)?;
65    /// let expected = vec!["Sean's First Post", "Tess's First Post"];
66    /// assert_eq!(expected, inserted_posts);
67    /// #     Ok(())
68    /// # }
69    /// ```
70    fn insert_into(self, table: T) -> InsertStatement<T, Self::Values>
71    where
72        T: Table,
73        Self: Sized,
74    {
75        crate::insert_into(table).values(self)
76    }
77}

82pub trait CanInsertInSingleQuery<DB: Backend> {
83    /// How many rows will this query insert?
84    ///
85    /// This function should only return `None` when the query is valid on all
86    /// backends, regardless of how many rows get inserted.
87    fn rows_to_insert(&self) -> Option<usize>;
88}

90impl<T, DB> CanInsertInSingleQuery<DB> for &T
91where
92    T: ?Sized + CanInsertInSingleQuery<DB>,
93    DB: Backend,
94{
95    fn rows_to_insert(&self) -> Option<usize> {
96        (*self).rows_to_insert()
97    }
98}

325impl<T, Tab, const N: usize> Insertable<Tab> for alloc::rc::Rc<[T; N]>
326where
327    T: Insertable<Tab> + core::clone::Clone,
328{
329    // We can reuse the query id for [T; N] here as this compiles down to the
330    // same query. `T: Clone` is required because shared ownership of `Rc`
331    // prevents moving the array elements out; we clone via `<[T]>::to_vec()`.
332    // The `Box<[T; N]>` impl above does not need this bound because it owns
333    // the array exclusively and can move elements out via `Vec::from`.
334    type Values = BatchInsert<Vec<T::Values>, Tab, [T::Values; N], true>;
335
336    fn values(self) -> Self::Values {
337        let v = self.to_vec();
338        let values = v.into_iter().map(Insertable::values).collect::<Vec<_>>();
339        BatchInsert::new(values)
340    }
341}
