34    /// Bind a value for use with this SQL query. The given query should have
35    /// placeholders that vary based on the database type,
36    /// like [SQLite Parameter](https://sqlite.org/lang_expr.html#varparam) syntax,
37    /// [PostgreSQL PREPARE syntax](https://www.postgresql.org/docs/current/sql-prepare.html),
38    /// or [MySQL bind syntax](https://dev.mysql.com/doc/refman/8.0/en/mysql-stmt-bind-param.html).
39    ///
40    /// For binding a variable number of values in a loop, use `into_boxed` first.
41    ///
42    /// # Safety
43    ///
44    /// This function should be used with care, as Diesel cannot validate that
45    /// the value is of the right type nor can it validate that you have passed
46    /// the correct number of parameters.
47    ///
48    /// # Example
49    ///
50    /// ```
51    /// # include!("../doctest_setup.rs");
52    /// #
53    /// # use schema::users;
54    /// #
55    /// # #[derive(QueryableByName, Debug, PartialEq)]
56    /// # struct User {
57    /// #     id: i32,
58    /// #     name: String,
59    /// # }
60    /// #
61    /// # // Reading the query result requires reading memory allocated by the
62    /// # // native sqlite library, which miri cannot track provenance for.
63    /// # #[cfg(miri)]
64    /// # fn main() {}
65    /// # #[cfg(not(miri))]
66    /// # fn main() {
67    /// #     use diesel::sql_query;
68    /// #     use diesel::sql_types::{Integer, Text};
69    /// #
70    /// #     let connection = &mut establish_connection();
71    /// #     diesel::insert_into(users::table)
72    /// #         .values(users::name.eq("Jim"))
73    /// #         .execute(connection).unwrap();
74    /// # #[cfg(feature = "postgres")]
75    /// # let users = sql_query("SELECT * FROM users WHERE id > $1 AND name != $2");
76    /// # #[cfg(not(feature = "postgres"))]
77    /// // sqlite/mysql bind syntax
78    /// let users = sql_query("SELECT * FROM users WHERE id > ? AND name <> ?")
79    /// # ;
80    /// # let users = users
81    ///     .bind::<Integer, _>(1)
82    ///     .bind::<Text, _>("Tess")
83    ///     .get_results(connection);
84    /// let expected_users = vec![User {
85    ///     id: 3,
86    ///     name: "Jim".into(),
87    /// }];
88    /// assert_eq!(Ok(expected_users), users);
89    /// # }
90    /// ```
91    pub fn bind<ST, Value>(self, value: Value) -> UncheckedBind<Self, Value, ST> {
92        UncheckedBind::new(self, value)
93    }

95    /// Internally boxes the query, which allows to  calls `bind` and `sql` so that they don't
96    /// change the type nor the instance. This allows to call `bind` or `sql`
97    /// in a loop, e.g.:
98    ///
99    /// ```
100    /// # include!("../doctest_setup.rs");
101    /// #
102    /// # use schema::users;
103    /// #
104    /// # #[derive(QueryableByName, Debug, PartialEq)]
105    /// # struct User {
106    /// #     id: i32,
107    /// #     name: String,
108    /// # }
109    /// #
110    /// # // Reading the query result requires reading memory allocated by the
111    /// # // native sqlite library, which miri cannot track provenance for.
112    /// # #[cfg(miri)]
113    /// # fn main() {}
114    /// # #[cfg(not(miri))]
115    /// # fn main() {
116    /// #     use diesel::sql_query;
117    /// #     use diesel::sql_types::{Integer};
118    /// #
119    /// #     let connection = &mut establish_connection();
120    /// #     diesel::insert_into(users::table)
121    /// #         .values(users::name.eq("Jim"))
122    /// #         .execute(connection).unwrap();
123    /// let mut q = diesel::sql_query("SELECT * FROM users WHERE id IN(").into_boxed();
124    /// for (idx, user_id) in [3, 4, 5].into_iter().enumerate() {
125    ///     if idx != 0 {
126    ///         q = q.sql(", ");
127    ///     }
128    /// # #[cfg(feature = "postgres")]
129    /// # {
130    ///     q = q
131    ///         // postgresql bind syntax
132    ///         .sql(format!("${}", idx + 1))
133    ///         .bind::<Integer, _>(user_id);
134    /// # }
135    /// # #[cfg(not(feature = "postgres"))]
136    /// # {
137    /// #   q = q
138    /// #       .sql(format!("?"))
139    /// #       .bind::<Integer, _>(user_id);
140    /// # }
141    /// }
142    /// let users = q.sql(");").get_results(connection);
143    /// let expected_users = vec![User {
144    ///     id: 3,
145    ///     name: "Jim".into(),
146    /// }];
147    /// assert_eq!(Ok(expected_users), users);
148    /// # }
149    /// ```
150    ///
151    /// This allows doing things you otherwise couldn't do, e.g. `bind`ing in a
152    /// loop.
153    ///
154    /// Boxed queries are not cloneable.
155    /// If you need cloning, use [`into_boxed_clone`] to create a [`BoxedCloneSqlQuery`].
156    ///
157    /// [`into_boxed_clone`]: SqlQuery::into_boxed_clone()
158    pub fn into_boxed<'f, DB: Backend>(self) -> BoxedSqlQuery<'f, DB, Self> {
159        BoxedSqlQuery::new(self)

310    /// Append raw SQL after this query and the values bound to it.
311    ///
312    /// The wrapped query renders first, then the given SQL text. This allows
313    /// interleaving raw SQL fragments with [`bind`] calls when the query
314    /// builder cannot express the statement as a whole.
315    ///
316    /// # Safety
317    ///
318    /// Diesel passes the given string to the database as written. It must
319    /// therefore never contain values that come from outside your own code,
320    /// because anything interpolated into the SQL text can carry an SQL
321    /// injection. Pass such values with [`bind`] instead.
322    ///
323    /// # Examples
324    ///
325    /// ```rust
326    /// # include!("../doctest_setup.rs");
327    /// #
328    /// # use schema::users;
329    /// #
330    /// # #[derive(QueryableByName, Debug, PartialEq)]
331    /// # struct User {
332    /// #     id: i32,
333    /// #     name: String,
334    /// # }
335    /// #
336    /// # // Reading the query result requires reading memory allocated by the
337    /// # // native sqlite library, which miri cannot track provenance for.
338    /// # #[cfg(miri)]
339    /// # fn main() {}
340    /// # #[cfg(not(miri))]
341    /// # fn main() {
342    /// #     use diesel::sql_query;
343    /// #     use diesel::sql_types::Integer;
344    /// #
345    /// #     let connection = &mut establish_connection();
346    /// # #[cfg(feature = "postgres")]
347    /// # let base = sql_query("SELECT id, name FROM users WHERE id >= $1");
348    /// # #[cfg(not(feature = "postgres"))]
349    /// // Checkout the documentation of your database for the correct
350    /// // bind placeholder
351    /// let base = sql_query("SELECT id, name FROM users WHERE id >= ?");
352    /// // The appended fragment decides the order of the returned rows.
353    /// let users = base
354    ///     .bind::<Integer, _>(1)
355    ///     .sql(" ORDER BY id DESC")
356    ///     .load::<User>(connection);
357    /// assert_eq!(
358    ///     Ok(vec![
359    ///         User {
360    ///             id: 2,
361    ///             name: "Tess".into()
362    ///         },
363    ///         User {
364    ///             id: 1,
365    ///             name: "Sean".into()
366    ///         }
367    ///     ]),
368    ///     users
369    /// );
370    /// # }
371    /// ```
372    ///
373    /// [`bind`]: Self::bind()
374    pub fn sql<T: Into<String>>(self, sql: T) -> SqlQuery<Self> {
375        SqlQuery::new(self, sql.into())
376    }

480/// Walk a boxed SQL query, whichever pointer holds its binds.
481fn walk_boxed_sql_query<'b, DB, Query, P>(
482    query: &'b Query,
483    sql: &'b str,
484    binds: &'b [P],
485    mut out: AstPass<'_, 'b, DB>,
486) -> QueryResult<()>
487where
488    DB: Backend,
489    Query: QueryFragment<DB>,
490    P: QueryFragment<DB>,
491{
492    out.unsafe_to_cache_prepared();
493    query.walk_ast(out.reborrow())?;
494    out.push_sql(sql);
495
496    for b in binds {
497        b.walk_ast(out.reborrow())?;
498    }
499    Ok(())
500}
