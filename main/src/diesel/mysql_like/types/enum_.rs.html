6#[diagnostic::do_not_recommend]
7impl<DB: MysqlLikeBackend> EnumMapping<DB> for EnumTypeMapping {
8    fn map_to_database_value<'b>(
9        output: &mut crate::serialize::Output<'b, '_, DB>,
10        variant: &'static EnumVariant,
11    ) -> crate::serialize::Result {
12        output.write_all(variant.sql_name.as_bytes())?;
13        Ok(crate::serialize::IsNull::No)
14    }
15
16    fn map_from_database_value(
17        raw: <DB as crate::backend::Backend>::RawValue<'_>,
18        type_name: &'static str,
19        variants: &'static [EnumVariant],
20    ) -> crate::deserialize::Result<usize> {
21        let name = raw.as_bytes();
22        variants
23            .iter()
24            .position(|v| v.sql_name.as_bytes() == name)
25            .ok_or_else(|| {
26                ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Invalid enum variant `{0}` for `{2}`. Allowed variants are {1}",
                String::from_utf8_lossy(name),
                variants.iter().map(|v|
                                ::alloc::__export::must_use({
                                        ::alloc::fmt::format(format_args!("`{0}`", v.sql_name))
                                    })).collect::<Vec<_>>().join(", "), type_name))
    })format!(
27                    "Invalid enum variant `{}` for `{type_name}`. Allowed variants are {}",
28                    String::from_utf8_lossy(name),
29                    variants
30                        .iter()
31                        .map(|v| format!("`{}`", v.sql_name))
32                        .collect::<Vec<_>>()
33                        .join(", ")
34                )
35                .into()
36            })
37    }
38}
