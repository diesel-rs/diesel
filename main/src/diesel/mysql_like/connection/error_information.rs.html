11/// Reads the result of `mysql_sqlstate` or `mysql_stmt_sqlstate`.
12///
13/// # Safety
14///
15/// `sqlstate` must be null or point to a NUL terminated buffer that stays
16/// unchanged for the duration of the call.
17pub(super) unsafe fn sqlstate_from_ptr(sqlstate: *const c_char) -> Option<SqlState> {
18    if sqlstate.is_null() {
19        return None;
20    }
21    // SAFETY: non-null, and the caller guarantees NUL termination and stability.
22    let bytes = unsafe { CStr::from_ptr(sqlstate) }.to_bytes();
23    <[u8; 5]>::try_from(bytes).ok().and_then(SqlState::new)
24}
