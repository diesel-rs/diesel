26impl SqlState {
27    /// Returns `None` unless every byte is an ASCII digit or uppercase letter.
28    pub const fn new(code: [u8; 5]) -> Option<Self> {
29        let mut i = 0;
30        while i < code.len() {
31            if !#[allow(non_exhaustive_omitted_patterns)] match code[i] {
    b'0'..=b'9' | b'A'..=b'Z' => true,
    _ => false,
}matches!(code[i], b'0'..=b'9' | b'A'..=b'Z') {
32                return None;
33            }
34            i += 1;
35        }
36        Some(Self(code))
37    }
38
39    /// The five character code.
40    pub fn as_str(&self) -> &str {
41        core::str::from_utf8(&self.0).expect("a SQLSTATE consists of ASCII bytes")
42    }
43
44    /// The [`SqlStateClass`], made of the first two characters.
45    pub fn class(&self) -> SqlStateClass {
46        SqlStateClass([self.0[0], self.0[1]])
47    }
48
49    /// The subclass, made of the last three characters, where `000` means none.
50    pub fn subclass(&self) -> &str {
51        &self.as_str()[2..]
52    }
53}

96impl SqlStateClass {

99        core::str::from_utf8(&self.0).expect("a SQLSTATE consists of ASCII bytes")

102    /// Whether the class starts with `0`-`4` or `A`-`H`, which the SQL standard reserves.

104        #[allow(non_exhaustive_omitted_patterns)] match self.0[0] {
    b'0'..=b'4' | b'A'..=b'H' => true,
    _ => false,
}matches!(self.0[0], b'0'..=b'4' | b'A'..=b'H')
105    }

202    #[test]
203    fn new_accepts_only_digits_and_uppercase_letters() {
204        assert!(SqlState::new(*b"09AZ0").is_some());
205        for invalid in [b'/', b':', b'@', b'[', b'a', b'z', b' ', 0, 0xC3] {
206            let mut code = *b"23505";
207            code[4] = invalid;
208            assert_eq!(SqlState::new(code), None, "byte {invalid:#04x}");
209        }
210    }
211
212    #[test]
213    fn from_str_requires_exactly_five_bytes() {
214        assert_eq!("23505".parse::<SqlState>().unwrap().as_str(), "23505");
215        for invalid in ["", "2350", "235050", "2350\u{e9}", "2350\u{0}", "23505 "] {
216            assert_eq!(
217                invalid.parse::<SqlState>(),
218                Err(InvalidSqlState),
219                "{invalid:?}"
220            );
221        }
222    }
223
224    #[test]
225    fn class_and_subclass_split_after_two_characters() {
226        let code: SqlState = "40P01".parse().unwrap();
227        assert_eq!(code.class(), SqlStateClass::TRANSACTION_ROLLBACK);
228        assert_eq!(code.class().as_str(), "40");
229        assert_eq!(code.subclass(), "P01");
230    }
231
232    #[test]
233    fn standard_classes_start_with_0_to_4_or_a_to_h() {
234        for code in [
235            "00000", "40001", "0A000", "B0000", "G0000", "HV000", "H0000",
236        ] {
237            let code: SqlState = code.parse().unwrap();
238            assert!(code.class().is_standard(), "{code}");
239        }
240        for code in ["50000", "9Z000", "I0000", "P0001", "XX000"] {
241            let code: SqlState = code.parse().unwrap();
242            assert!(!code.class().is_standard(), "{code}");
243        }
244    }

246    #[test]
247    fn constants_match_as_patterns() {
248        const UNIQUE_VIOLATION: SqlState = SqlState::new(*b"23505").unwrap();
249        let code: SqlState = "23505".parse().unwrap();
250        assert!(matches!(code, UNIQUE_VIOLATION));
251        assert!(matches!(
252            code.class(),
253            SqlStateClass::INTEGRITY_CONSTRAINT_VIOLATION
254        ));
255        let other: SqlState = "22012".parse().unwrap();
256        assert!(!matches!(other, UNIQUE_VIOLATION));
257        assert!(!matches!(
258            other.class(),
259            SqlStateClass::INTEGRITY_CONSTRAINT_VIOLATION
260        ));
