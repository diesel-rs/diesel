7/// Manages the internal transaction state for a connection.
8///
9/// You will not need to interact with this trait, unless you are writing an
10/// implementation of [`Connection`].
11pub trait TransactionManager<Conn: Connection> {
12    /// Data stored as part of the connection implementation
13    /// to track the current transaction state of a connection
14    type TransactionStateData;
15
16    /// Begin a new transaction or savepoint
17    ///
18    /// If the transaction depth is greater than 0,
19    /// this should create a savepoint instead.
20    /// This function is expected to increment the transaction depth by 1.
21    fn begin_transaction(conn: &mut Conn) -> QueryResult<()>;
22
23    /// Rollback the inner-most transaction or savepoint
24    ///
25    /// If the transaction depth is greater than 1,
26    /// this should rollback to the most recent savepoint.
27    /// This function is expected to decrement the transaction depth by 1.
28    fn rollback_transaction(conn: &mut Conn) -> QueryResult<()>;
29
30    /// Commit the inner-most transaction or savepoint
31    ///
32    /// If the transaction depth is greater than 1,
33    /// this should release the most recent savepoint.
34    /// This function is expected to decrement the transaction depth by 1.
35    fn commit_transaction(conn: &mut Conn) -> QueryResult<()>;
36
37    /// Fetch the current transaction status as mutable
38    ///
39    /// Used to ensure that `begin_test_transaction` is not called when already
40    /// inside of a transaction, and that operations are not run in a `InError`
41    /// transaction manager.
42    #[diesel_derives::__diesel_public_if(
43        feature = "i-implement-a-third-party-backend-and-opt-into-breaking-changes"
44    )]
45    fn transaction_manager_status_mut(conn: &mut Conn) -> &mut TransactionManagerStatus;
46
47    /// Executes the given function inside of a database transaction
48    ///
49    /// Each implementation of this function needs to fulfill the documented
50    /// behaviour of [`Connection::transaction`]
51    fn transaction<F, R, E>(conn: &mut Conn, callback: F) -> Result<R, E>
52    where
53        F: FnOnce(&mut Conn) -> Result<R, E>,
54        E: From<Error>,
55    {
56        Self::begin_transaction(conn)?;
57        match callback(&mut *conn) {
58            Ok(value) => {
59                Self::commit_transaction(conn)?;
60                Ok(value)
61            }
62            Err(user_error) => match Self::rollback_transaction(conn) {
63                Ok(()) => Err(user_error),
64                Err(Error::BrokenTransactionManager) => {
65                    // In this case we are probably more interested by the
66                    // original error, which likely caused this
67                    Err(user_error)
68                }
69                Err(rollback_error) => Err(rollback_error.into()),
70            },
71        }
72    }
73
74    /// This methods checks if the connection manager is considered to be broken
75    /// by connection pool implementations
76    ///
77    /// A connection manager is considered to be broken by default if it either
78    /// contains an open transaction (because you don't want to have connections
79    /// with open transactions in your pool) or when the transaction manager is
80    /// in an error state.
81    #[diesel_derives::__diesel_public_if(
82        feature = "i-implement-a-third-party-backend-and-opt-into-breaking-changes"
83    )]
84    fn is_broken_transaction_manager(conn: &mut Conn) -> bool {
85        match Self::transaction_manager_status_mut(conn).transaction_state() {
86            // all transactions are closed
87            // so we don't consider this connection broken
88            Ok(ValidTransactionManagerStatus {
89                in_transaction: None,
90            }) => false,
91            // The transaction manager is in an error state
92            // Therefore we consider this connection broken
93            Err(_) => true,
94            // The transaction manager contains a open transaction
95            // we do consider this connection broken
96            // if that transaction was not opened by `begin_test_transaction`
97            Ok(ValidTransactionManagerStatus {
98                in_transaction: Some(s),
99            }) => !s.test_transaction,
100        }
101    }
102}

301    /// Update the transaction depth by adding the value of the `transaction_depth_change` parameter if the `query` is
302    /// `Ok(())`
303    pub fn change_transaction_depth(
304        &mut self,
305        transaction_depth_change: TransactionDepthChange,
306    ) -> QueryResult<()> {
307        match (&mut self.in_transaction, transaction_depth_change) {
308            (Some(in_transaction), TransactionDepthChange::IncreaseDepth) => {
309                // Can be replaced with saturating_add directly on NonZeroU32 once
310                // <https://github.com/rust-lang/rust/issues/84186> is stable
311                in_transaction.transaction_depth =
312                    NonZeroU32::new(in_transaction.transaction_depth.get().saturating_add(1))
313                        .expect("nz + nz is always non-zero");
314                Ok(())
315            }
316            (Some(in_transaction), TransactionDepthChange::DecreaseDepth) => {
317                // This sets `transaction_depth` to `None` as soon as we reach zero
318                match NonZeroU32::new(in_transaction.transaction_depth.get() - 1) {
319                    Some(depth) => in_transaction.transaction_depth = depth,
320                    None => self.in_transaction = None,
321                }
322                Ok(())
323            }
324            (None, TransactionDepthChange::IncreaseDepth) => {
325                self.in_transaction = Some(InTransactionStatus {
326                    transaction_depth: NonZeroU32::new(1).expect("1 is non-zero"),
327                    requires_rollback_maybe_up_to_top_level: false,
328                    test_transaction: false,
329                    rolled_back_by_database: false,
330                });
331                Ok(())
332            }
333            (None, TransactionDepthChange::DecreaseDepth) => {
334                // We screwed up something somewhere
335                // we cannot decrease the transaction count if
336                // we are not inside a transaction
337                Err(Error::NotInTransaction)
338            }
339        }
340    }

455
456        if rolled_back_by_database {
457            // The database has already rolled back every level of this transaction.
458            return Self::get_transaction_state(conn)?
459                .change_transaction_depth(TransactionDepthChange::DecreaseDepth);
460        }
461
462        match conn.batch_execute(&rollback_sql) {
463            Ok(()) => {
464                match Self::get_transaction_state(conn)?
465                    .change_transaction_depth(TransactionDepthChange::DecreaseDepth)
466                {
467                    Ok(()) => {}
468                    Err(Error::NotInTransaction) if rolling_back_top_level => {
469                        // Transaction exit may have already been detected by connection
470                        // implementation. It's fine.
471                    }
472                    Err(e) => return Err(e),
473                }
474                Ok(())
475            }
476            Err(rollback_error) => {
477                let tm_status = Self::transaction_manager_status_mut(conn);
478                match tm_status {
479                    TransactionManagerStatus::Valid(ValidTransactionManagerStatus {
480                        in_transaction:
481                            Some(InTransactionStatus {
482                                transaction_depth,
483                                requires_rollback_maybe_up_to_top_level,
484                                ..
485                            }),
486                    }) if transaction_depth.get() > 1 => {
487                        // A savepoint failed to rollback - we may still attempt to repair
488                        // the connection by rolling back higher levels.
489
490                        // To make it easier on the user (that they don't have to really
491                        // look at actual transaction depth and can just rely on the number
492                        // of times they have called begin/commit/rollback) we still
493                        // decrement here:
494                        *transaction_depth = NonZeroU32::new(transaction_depth.get() - 1)
495                            .expect("Depth was checked to be > 1");
496                        *requires_rollback_maybe_up_to_top_level = true;
497                        if requires_rollback_maybe_up_to_top_level_before_execute {
498                            // In that case, we tolerate that savepoint releases fail
499                            // -> we should ignore errors
500                            return Ok(());
501                        }
502                    }
503                    TransactionManagerStatus::Valid(ValidTransactionManagerStatus {
504                        in_transaction: None,
505                    }) => {
506                        // we would have returned `NotInTransaction` if that was already the state
507                        // before we made our call
508                        // => Transaction manager status has been fixed by the underlying connection
509                        // so we don't need to set_in_error
510                    }
511                    TransactionManagerStatus::Valid(ValidTransactionManagerStatus {
512                        in_transaction:
513                            Some(InTransactionStatus {
514                                rolled_back_by_database: true,
515                                ..
516                            }),
517                    }) => {
518                        // The connection detected that the failed `ROLLBACK` still ended
519                        // the transaction, so no transaction is left to roll back.
520                        *tm_status = TransactionManagerStatus::default();
521                    }
522                    _ => tm_status.set_in_error(),
523                }
524                Err(rollback_error)
525            }
526        }
527    }

529    /// If a top-level `COMMIT` fails and leaves the transaction open, a rollback
530    /// will be attempted. If the rollback succeeds, the original error will be returned,
531    /// otherwise [`Error::RollbackErrorOnCommit`] carrying both errors will be returned.
532    /// In the second case the connection will be considered broken
533    /// as it contains a uncommitted unabortable open transaction.
534    fn commit_transaction(conn: &mut Conn) -> QueryResult<()> {
535        let transaction_state = Self::get_transaction_state(conn)?;
536        let transaction_depth = transaction_state.transaction_depth();
537        let (commit_sql, committing_top_level) = match transaction_depth {
538            None => return Err(Error::NotInTransaction),
539            Some(transaction_depth) if transaction_depth.get() == 1 => {
540                (Cow::Borrowed("COMMIT"), true)
541            }
542            Some(transaction_depth) => (
543                Cow::Owned(::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("RELEASE SAVEPOINT diesel_savepoint_{0}",
                transaction_depth.get() - 1))
    })alloc::format!(
544                    "RELEASE SAVEPOINT diesel_savepoint_{}",
545                    transaction_depth.get() - 1
546                )),
547                false,
548            ),
549        };
550        let depth = transaction_state
551            .transaction_depth()
552            .expect("We know that we are in a transaction here");
553        let rolled_back_by_database = transaction_state
554            .in_transaction
555            .as_ref()
556            .is_some_and(|in_transaction| in_transaction.rolled_back_by_database);
557        if rolled_back_by_database {
558            // Nothing of this transaction can be committed anymore.
559            Self::rollback_transaction(conn)?;
560            return Err(Error::TransactionRolledBackByDatabase);
561        }
562        conn.instrumentation().on_connection_event(
563            super::instrumentation::InstrumentationEvent::CommitTransaction { depth },
564        );
565        match conn.batch_execute(&commit_sql) {
566            Ok(()) => {
567                match Self::get_transaction_state(conn)?
568                    .change_transaction_depth(TransactionDepthChange::DecreaseDepth)
569                {
570                    Ok(()) => {}
571                    Err(Error::NotInTransaction) if committing_top_level => {
572                        // Transaction exit may have already been detected by connection.
573                        // It's fine
574                    }
575                    Err(e) => return Err(e),
576                }
577                Ok(())
578            }
579            Err(commit_error) => {
580                if let TransactionManagerStatus::Valid(ValidTransactionManagerStatus {
581                    in_transaction:
582                        Some(InTransactionStatus {
583                            requires_rollback_maybe_up_to_top_level,
584                            ..
585                        }),
586                }) = conn.transaction_state().status
587                    && (committing_top_level || requires_rollback_maybe_up_to_top_level)
588                {
589                    match Self::rollback_transaction(conn) {
590                        Ok(()) => {}
591                        Err(rollback_error) => {
592                            conn.transaction_state().status.set_in_error();
593                            return Err(Error::RollbackErrorOnCommit {
594                                rollback_error: Box::new(rollback_error),
595                                commit_error: Box::new(commit_error),
596                            });
597                        }
598                    }
599                }
600                Err(commit_error)
601            }
602        }
603    }

606        &mut conn.transaction_state().status
607    }
