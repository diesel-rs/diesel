91///
92/// This trait represents a database connection. It can be used to query the database through
93/// the query dsl provided by diesel, custom extensions or raw sql queries.
94///
95/// # Implementing a custom connection
96///
97/// There are several reasons why you would want to implement a custom connection implementation:
98///
99/// * To wrap an existing connection for instrumentation purposes
100/// * To use a different underlying library to provide a connection implementation
101/// for already existing backends.
102/// * To add support for an unsupported database system
103///
104/// Implementing a `Connection` in a third party crate requires
105/// enabling the
106/// `i-implement-a-third-party-backend-and-opt-into-breaking-changes`
107/// crate feature which grants access to some of diesel's implementation details.
108///
109///
110/// ## Wrapping an existing connection impl
111///
112/// Wrapping an existing connection allows you to customize the implementation to
113/// add additional functionality, like for example instrumentation. For this use case
114/// you only need to implement `Connection`, [`LoadConnection`] and all super traits.
115/// You should forward any method call to the wrapped connection type.
116/// It is **important** to also forward any method where diesel provides a
117/// default implementation, as the wrapped connection implementation may
118/// contain a customized implementation.
119///
120/// To allow the integration of your new connection type with other diesel features
121#[cfg_attr(
122    feature = "r2d2",
123    doc = "it may be useful to also implement [`R2D2Connection`](crate::r2d2::R2D2Connection)"
124)]

126    not(feature = "r2d2"),
127    doc = "it may be useful to also implement `R2D2Connection`"
128)]
129/// and [`MigrationConnection`](crate::migration::MigrationConnection).
130///
131/// ## Provide a new connection implementation for an existing backend
132///
133/// Implementing a new connection based on an existing backend can enable the usage of
134/// other methods to connect to the database. One example here would be to replace
135/// the official diesel provided connection implementations with an implementation
136/// based on a pure rust connection crate.
137///
138/// **It's important to use prepared statements to implement the following methods:**
139/// * [`LoadConnection::load`]
140/// * [`Connection::execute_returning_count`]

260    /// Executes the given function inside of a database transaction
261    ///
262    /// This function executes the provided closure `f` inside a database
263    /// transaction. If there is already an open transaction for the current
264    /// connection savepoints will be used instead. The connection is committed if
265    /// the closure returns `Ok(_)`, it will be rolled back if it returns `Err(_)`.
266    /// For both cases the original result value will be returned from this function.
267    ///
268    /// If the transaction fails to commit due to a `SerializationFailure` or a
269    /// `ReadOnlyTransaction` a rollback will be attempted.
270    /// If the rollback fails, the error will be returned in a
271    /// [`Error::RollbackErrorOnCommit`],
272    /// from which you will be able to extract both the original commit error and
273    /// the rollback error.
274    /// In addition, the connection will be considered broken
275    /// as it contains a uncommitted unabortable open transaction. Any further
276    /// interaction with the transaction system will result in an returned error
277    /// in this case.
278    ///
279    /// If the closure returns an `Err(_)` and the rollback fails the function
280    /// will return that rollback error directly, and the transaction manager will
281    /// be marked as broken as it contains a uncommitted unabortable open transaction.
282    ///
283    /// If a nested transaction fails to release the corresponding savepoint
284    /// the error will be returned directly.
285    ///
286    /// # Example
287    ///
288    /// ```rust
289    /// # include!("../doctest_setup.rs");
290    /// use diesel::result::Error;
291    ///
292    /// # // Reading the query result requires reading memory allocated by the
293    /// # // native sqlite library, which miri cannot track provenance for.
294    /// # #[cfg(miri)]
295    /// # fn main() {}
296    /// # #[cfg(not(miri))]
297    /// # fn main() {
298    /// #     run_test().unwrap();
299    /// # }
300    /// #
301    /// # fn run_test() -> QueryResult<()> {
302    /// #     use schema::users::dsl::*;
303    /// #     let conn = &mut establish_connection();
304    /// conn.transaction::<_, Error, _>(|conn| {
305    ///     diesel::insert_into(users)
306    ///         .values(name.eq("Ruby"))
307    ///         .execute(conn)?;
308    ///
309    ///     let all_names = users.select(name).load::<String>(conn)?;
310    ///     assert_eq!(vec!["Sean", "Tess", "Ruby"], all_names);
311    ///
312    ///     Ok(())
313    /// })?;
314    ///
315    /// conn.transaction::<(), _, _>(|conn| {
316    ///     diesel::insert_into(users)
317    ///         .values(name.eq("Pascal"))
318    ///         .execute(conn)?;
319    ///
320    ///     let all_names = users.select(name).load::<String>(conn)?;
321    ///     assert_eq!(vec!["Sean", "Tess", "Ruby", "Pascal"], all_names);
322    ///
323    ///     // If we want to roll back the transaction, but don't have an
324    ///     // actual error to return, we can return `RollbackTransaction`.
325    ///     Err(Error::RollbackTransaction)
326    /// });
327    ///
328    /// let all_names = users.select(name).load::<String>(conn)?;
329    /// assert_eq!(vec!["Sean", "Tess", "Ruby"], all_names);
330    /// #     Ok(())
331    /// # }
332    /// ```
333    fn transaction<T, E, F>(&mut self, f: F) -> Result<T, E>
334    where
335        F: FnOnce(&mut Self) -> Result<T, E>,
336        E: From<Error>,
337    {
338        Self::TransactionManager::transaction(self, f)
339    }
340
341    /// Creates a transaction that will never be committed. This is useful for
342    /// tests. Panics if called while inside of a transaction or
343    /// if called with a connection containing a broken transaction
344    fn begin_test_transaction(&mut self) -> QueryResult<()> {
345        match Self::TransactionManager::transaction_manager_status_mut(self) {
346            TransactionManagerStatus::Valid(valid_status) => {
347                {
    match (&None, &valid_status.transaction_depth()) {
        (left_val, right_val) => {
            if !(*left_val == *right_val) {
                let kind = ::core::panicking::AssertKind::Eq;
                ::core::panicking::assert_failed(kind, &*left_val,
                    &*right_val, ::core::option::Option::None);
            }
        }
    }
}assert_eq!(None, valid_status.transaction_depth())
348            }
349            TransactionManagerStatus::InError => {
    ::core::panicking::panic_fmt(format_args!("Transaction manager in error"));
}panic!("Transaction manager in error"),
350        };
351        Self::TransactionManager::begin_transaction(self)?;
352        // set the test transaction flag
353        // to prevent that this connection gets dropped in connection pools
354        // Tests commonly set the poolsize to 1 and use `begin_test_transaction`
355        // to prevent modifications to the schema
356        Self::TransactionManager::transaction_manager_status_mut(self).set_test_transaction_flag();
357        Ok(())
358    }

360    /// Executes the given function inside a transaction, but does not commit
361    /// it. Panics if the given function returns an error.
362    ///
363    /// # Example
364    ///
365    /// ```rust
366    /// # include!("../doctest_setup.rs");
367    /// use diesel::result::Error;
368    ///
369    /// # // Reading the query result requires reading memory allocated by the
370    /// # // native sqlite library, which miri cannot track provenance for.
371    /// # #[cfg(miri)]
372    /// # fn main() {}
373    /// # #[cfg(not(miri))]
374    /// # fn main() {
375    /// #     run_test().unwrap();
376    /// # }
377    /// #
378    /// # fn run_test() -> QueryResult<()> {
379    /// #     use schema::users::dsl::*;
380    /// #     let conn = &mut establish_connection();
381    /// conn.test_transaction::<_, Error, _>(|conn| {
382    ///     diesel::insert_into(users)
383    ///         .values(name.eq("Ruby"))
384    ///         .execute(conn)?;
385    ///
386    ///     let all_names = users.select(name).load::<String>(conn)?;
387    ///     assert_eq!(vec!["Sean", "Tess", "Ruby"], all_names);
388    ///
389    ///     Ok(())
390    /// });
391    ///
392    /// // Even though we returned `Ok`, the transaction wasn't committed.
393    /// let all_names = users.select(name).load::<String>(conn)?;
394    /// assert_eq!(vec!["Sean", "Tess"], all_names);
395    /// #     Ok(())
396    /// # }
397    /// ```
398    fn test_transaction<T, E, F>(&mut self, f: F) -> T
399    where
400        F: FnOnce(&mut Self) -> Result<T, E>,
401        E: Debug,
402    {
403        let mut user_result = None;
404        let _ = self.transaction::<(), _, _>(|conn| {
405            user_result = Some(f(conn));
406            Err(Error::RollbackTransaction)
407        });
408        user_result
409            .expect("Transaction never executed")
410            .unwrap_or_else(|e| {
    ::core::panicking::panic_fmt(format_args!("Transaction did not succeed: {0:?}",
            e));
}panic!("Transaction did not succeed: {:?}", e))
411    }

557    /// This will return `None` if the underlying
558    /// connection does not match the corresponding
559    /// type, otherwise a mutable reference to the underlying connection is returned
560    pub fn downcast_mut<T>(&mut self) -> Option<&mut T>
561    where
562        T: Connection<Backend = DB> + 'static,
563    {
564        self.as_any_mut().downcast_mut::<T>()
565    }
566
567    /// Check if the current connection is
568    /// a specific connection type
569    pub fn is<T>(&self) -> bool
570    where
571        T: Connection<Backend = DB> + 'static,
572    {
573        self.as_any().is::<T>()
574    }
575}

577// These traits are considered private for different reasons:
578//
579// `ConnectionSealed` to control who can implement `Connection`,
580// so that we can later change the `Connection` trait
581//
582// `MultiConnectionHelper` is a workaround needed for the
583// `MultiConnection` derive. We might stabilize this trait with
584// the corresponding derive
585//
586// `ConnectionHelperType` as a workaround for the `LoadRowIter`
587// type def. That trait should not be used by any user outside of diesel,
588// it purely exists for backward compatibility reasons.
589pub(crate) mod private {
590
591    /// This trait restricts who can implement `Connection`
592    #[cfg_attr(
593        diesel_docsrs,
594        doc(cfg(feature = "i-implement-a-third-party-backend-and-opt-into-breaking-changes"))
595    )]
596    pub trait ConnectionSealed {}
597
598    /// This trait provides helper methods to convert a database lookup type
599    /// to/from an `std::any::Any` reference. This is used internally by the `#[derive(MultiConnection)]`
600    /// implementation
601    #[cfg_attr(
602        diesel_docsrs,
603        doc(cfg(feature = "i-implement-a-third-party-backend-and-opt-into-breaking-changes"))
604    )]
605    pub trait MultiConnectionHelper: super::Connection {
606        /// Convert the lookup type to any
607        fn to_any<'a>(
608            lookup: &mut <Self::Backend as crate::sql_types::TypeMetadata>::MetadataLookup,
609        ) -> &mut (dyn core::any::Any + 'a);
610
611        /// Get the lookup type from any
612        fn from_any(
613            lookup: &mut dyn core::any::Any,
614        ) -> Option<&mut <Self::Backend as crate::sql_types::TypeMetadata>::MetadataLookup>;
615    }
616
617    // These impls are only there for backward compatibility reasons
618    // Remove them on the next breaking release
619    #[allow(unreachable_pub)] // must be pub for the type def using this trait
620    #[cfg(all(feature = "with-deprecated", not(feature = "without-deprecated")))]
621    pub trait ConnectionHelperType<DB, B>: super::LoadConnection<B, Backend = DB> {
622        type Cursor<'conn, 'query>
623        where
624            Self: 'conn;
625    }
626    #[cfg(all(feature = "with-deprecated", not(feature = "without-deprecated")))]
627    impl<T, B> ConnectionHelperType<T::Backend, B> for T
628    where
629        T: super::LoadConnection<B>,
630    {
631        type Cursor<'conn, 'query>
632            = <T as super::LoadConnection<B>>::Cursor<'conn, 'query>
633        where
634            T: 'conn;
635    }
