26#[cfg(all(feature = "__sqlite-shared", feature = "serde_json"))]
27impl FromSql<sql_types::Jsonb, Sqlite> for serde_json::Value {
28    fn from_sql(mut value: SqliteValue<'_, '_, '_>) -> deserialize::Result<Self> {
29        use self::jsonb::*;
30
31        let bytes = value.read_blob();
32
33        if bytes.is_empty() {
34            return Err("Empty blob cannot be decoded as JSONB".into());
35        }
36
37        // Read the JSONB value from the byte stream
38        let (jsonb, size) = read_jsonb_value(bytes)?;
39        if size == bytes.len() {
40            Ok(jsonb)
41        } else {
42            Err("Payload contained more bytes than the encoded JSONB".into())
43        }
44    }
45}

50        use self::jsonb::*;
51
52        // Create a buffer to hold the binary JSONB encoding
53        let mut buffer = Vec::new();
54
55        // Write the JSON value into the buffer in JSONB format
56        write_jsonb_value(self, &mut buffer)?;
57
58        // Set the serialized binary data to the output
59        out.set_value(buffer);
60
61        Ok(IsNull::No)
62    }

96    #[allow(unsafe_code)]
97    pub(super) fn read_jsonb_value(
98        bytes: &[u8],
99    ) -> deserialize::Result<(serde_json::Value, usize)> {
100        if bytes.is_empty() {
101            return Err("Empty JSONB data".into());
102        }
103        let (global_header, mut global_value) = read_header_and_value(bytes)?;
104
105        if global_value.is_array() || global_value.is_object() {
106            // we need to use pointers here, as the borrow checker does not understand
107            // that we only modify the last value in this stack. Given that we cannot
108            // invalidate any pointer that's up in the stack
109            let mut stack = ::alloc::boxed::box_assume_init_into_vec_unsafe(::alloc::intrinsics::write_box_via_move(::alloc::boxed::Box::new_uninit(),
        [(&mut global_value as *mut serde_json::Value,
                    global_header.payload_size)]))vec![(
110                &mut global_value as *mut serde_json::Value,
111                global_header.payload_size,
112            )];
113            let mut payload = &bytes[global_header.header_size..];
114            let mut total_read = 0;
115            // we must use a loop based decoding approach here instead of the much simpler recursive implementation
116            // as we otherwise run into stack overflows for deeply nested objects/arrays
117
118            while total_read < global_header.payload_size {
119                let Some((value, _size)) = stack.last().copied() else {
120                    break;
121                };
122                let value = unsafe {
123                    // SAFETY: The pointer was initialized before
124                    // We we cannot invalidate the underlying object
125                    &mut *value
126                };
127
128                if let serde_json::Value::Array(array) = value {
129                    let (header, value) = read_header_and_value(payload)?;
130
131                    array.push(value);
132                    let last_ref = array.last_mut().expect("Pushed above");
133                    let payload_size = if last_ref.is_object() || last_ref.is_array() {
134                        stack.push((last_ref as *mut _, total_read + header.total_size));
135                        header.header_size
136                    } else {
137                        header.total_size
138                    };
139                    total_read += payload_size;
140                    if payload.len() > payload_size {
141                        payload = &payload[payload_size..];
142                    } else {
143                        for (_, v) in stack {
144                            if v != total_read {
145                                return Err("Invalid size of payload declared".into());
146                            }
147                        }
148                        break;
149                    }
150                } else if let serde_json::Value::Object(object) = value {
151                    //       while total_read < payload_size {
152                    let (key_header, key) = read_header_and_value(payload)?;
153                    total_read += key_header.total_size;
154                    let serde_json::Value::String(key) = key else {
155                        return Err("Expected a string as object key".into());
156                    };
157                    if payload.len() > key_header.total_size {
158                        payload = &payload[key_header.total_size..];
159                    } else {
160                        return Err("No value found for object".into());
161                    }
162                    let (value_header, value) = read_header_and_value(payload)?;
163                    object.insert(key.clone(), value);
164                    let last_ref = object.get_mut(&key).expect("We inserted it above");
165                    let payload_size = if last_ref.is_object() || last_ref.is_array() {
166                        stack.push((last_ref as *mut _, total_read + value_header.total_size));
167                        value_header.header_size
168                    } else {
169                        value_header.total_size
170                    };
171                    total_read += payload_size;
172                    if payload.len() > payload_size {
173                        payload = &payload[payload_size..];
174                    } else {
175                        for (_, v) in stack {
176                            if v != total_read {
177                                return Err("Invalid size of payload declared".into());
178                            }
179                        }
180                        break;
181                    }
182                } else {
183                    ::core::panicking::panic("internal error: entered unreachable code")unreachable!()
184                }
185
186                while let Some(v) = stack.last().map(|(_, v)| *v) {
187                    if v > total_read {
188                        break;
189                    } else if v == total_read {
190                        stack.pop();
191                    } else {
192                        return Err("Invalid size of payload declared".into());
193                    }
194                }
195            }
196        }
197        Ok((global_value, global_header.total_size))
198    }

200    // This function decodes the jsonb header
201    // and the value for non-composite values. For composite values like array and object
202    // we only decode the "value header" and leave decoding
203    // the actual child values to future calls
204    fn read_header_and_value(
205        bytes: &[u8],
206    ) -> deserialize::Result<(JsonbHeader, serde_json::Value)> {
207        let header = read_jsonb_value_header(bytes)?;
208        let payload_bytes = &bytes[header.header_size..header.total_size];
209        let value = match header.element_type {
210            JSONB_NULL => Ok(serde_json::Value::Null),
211            JSONB_TRUE => Ok(serde_json::Value::Bool(true)),
212            JSONB_FALSE => Ok(serde_json::Value::Bool(false)),
213            JSONB_INT => read_jsonb_int(payload_bytes, header.payload_size),
214            JSONB_INT5 => Err("INT5 is not supported".into()),
215            JSONB_FLOAT => read_jsonb_float(payload_bytes, header.payload_size),
216            JSONB_FLOAT5 => Err("FLOAT5 is not supported".into()),
217            JSONB_TEXT => read_jsonb_text(payload_bytes, header.payload_size),
218            JSONB_TEXTJ => read_jsonb_textj(payload_bytes, header.payload_size),
219            JSONB_TEXTRAW => read_jsonb_text(payload_bytes, header.payload_size),
220            JSONB_TEXT5 => Err("TEXT5 is not supported".into()),
221            JSONB_ARRAY => Ok(serde_json::Value::Array(alloc::vec::Vec::new())),
222            JSONB_OBJECT => Ok(serde_json::Value::Object(serde_json::Map::new())),
223            _ => Err(::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Unsupported or reserved JSONB type: {0}",
                header.element_type))
    })alloc::format!(
224                "Unsupported or reserved JSONB type: {}",
225                header.element_type
226            )
227            .into()),
228        }?;
229        Ok((header, value))
230    }

232    fn read_jsonb_value_header(bytes: &[u8]) -> deserialize::Result<JsonbHeader> {
233        let first_byte = bytes[0];
234        let element_type = first_byte & 0x0F;
235        let size_hint = (first_byte & 0xF0) >> 4;
236        let (payload_size, header_size): (usize, usize) = match size_hint {
237            0x00..=0x0B => (size_hint as usize, 1), // Payload size is directly in the upper nibble
238            0x0C => {
239                if bytes.len() < 2 {
240                    return Err("Invalid JSONB data: insufficient bytes for payload size".into());
241                }
242                (bytes[1] as usize, 2) // 1 additional byte for payload size
243            }
244            0x0D => {
245                if bytes.len() < 3 {
246                    return Err("Invalid JSONB data: insufficient bytes for payload size".into());
247                }
248                (u16::from_be_bytes([bytes[1], bytes[2]]) as usize, 3) // 2 additional bytes
249            }
250            0x0E => {
251                if bytes.len() < 5 {
252                    return Err("Invalid JSONB data: insufficient bytes for payload size".into());
253                }
254                (
255                    u32::from_be_bytes([bytes[1], bytes[2], bytes[3], bytes[4]]) as usize,
256                    5,
257                ) // 4 additional bytes
258            }
259            0x0F => {
260                if bytes.len() < 9 {
261                    return Err("Invalid JSONB data: insufficient bytes for payload size".into());
262                }
263                (
264                    usize::try_from(u64::from_be_bytes([
265                        bytes[1], bytes[2], bytes[3], bytes[4], bytes[5], bytes[6], bytes[7],
266                        bytes[8],
267                    ]))
268                    .map_err(Box::new)?,
269                    9,
270                ) // 8 additional bytes
271            }
272            _ => return Err("Invalid payload size hint".into()),
273        };
274        let total_size = header_size
275            .checked_add(payload_size)
276            .ok_or("The provided payload size overflows usize")?;
277        if bytes.len() < total_size {
278            return Err(::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Invalid JSONB data: insufficient bytes for value of type {0}, expected {1} bytes, got {2}",
                element_type, total_size, bytes.len()))
    })alloc::format!(
279                "Invalid JSONB data: insufficient bytes for value of type {}, expected {} bytes, got {}",
280                element_type,
281                total_size,
282                bytes.len()
283            )
284            .into());
285        }
286
287        Ok(JsonbHeader {
288            element_type,
289            payload_size,
290            header_size,
291            total_size,
292        })
293    }

295    // Read a JSON integer in canonical format (INT)
296    pub(super) fn read_jsonb_int(
297        bytes: &[u8],
298        payload_size: usize,
299    ) -> deserialize::Result<serde_json::Value> {
300        // Ensure the bytes are at least as large as the payload size
301        if bytes.len() < payload_size {
302            return Err(::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Expected payload of size {0}, but got {1}",
                payload_size, bytes.len()))
    })alloc::format!(
303                "Expected payload of size {}, but got {}",
304                payload_size,
305                bytes.len()
306            )
307            .into());
308        }
309
310        // Read only the number of bytes specified by the payload size
311        let int_str = core::str::from_utf8(bytes).map_err(|_| "Invalid ASCII in JSONB integer")?;
312        let int_value = serde_json::from_str(int_str)
313            .map_err(|_| "Failed to parse JSONB")
314            .and_then(|v: serde_json::Value| {
315                v.is_i64()
316                    .then_some(v)
317                    .ok_or("Failed to parse JSONB integer")
318            })?;
319
320        Ok(int_value)
321    }

402    pub(super) fn write_jsonb_header(
403        buffer: &mut Vec<u8>,
404        element_type: u8,
405        payload_size: usize,
406    ) -> serialize::Result {
407        // Check if payload size exceeds the maximum allowed size
408        if payload_size > 2_147_483_647 {
409            return Err("Payload size exceeds the maximum allowed size of 2GB".into());
410        }
411
412        if payload_size <= 0x0B {
413            // Small payloads, 0 additional byte for size
414            buffer.push(
415                ((u8::try_from(payload_size).map_err(|e| e.to_string())?) << 4) | element_type,
416            );
417        } else if payload_size <= 0xFF {
418            // Medium payloads, 1 additional byte for size
419            buffer.extend_from_slice(&[
420                (0x0C << 4) | element_type,
421                u8::try_from(payload_size).map_err(|e| e.to_string())?,
422            ]);
423        } else if payload_size <= 0xFFFF {
424            // Larger payloads, 2 additional bytes for size
425            buffer.push((0x0D << 4) | element_type);
426            buffer.extend_from_slice(
427                &(u16::try_from(payload_size).map_err(|e| e.to_string())?).to_be_bytes(),
428            );
429        } else {
430            // Very large payloads, 4 additional bytes for size (up to 2 GiB)
431            buffer.push((0x0E << 4) | element_type);
432            buffer.extend_from_slice(
433                &(u32::try_from(payload_size).map_err(|e| e.to_string())?).to_be_bytes(),
434            );
435        };
436
437        Ok(IsNull::No)
438    }

440    fn place_composite_value(
441        buffer: &mut Vec<u8>,
442        stack: &mut [JsonValuePtr<'_>],
443        serialized_buffer: Vec<u8>,
444        tpe: u8,
445    ) -> serialize::Result {
446        let buffer = match stack.last_mut() {
447            None | Some(JsonValuePtr::Value(_)) => buffer,
448            Some(
449                JsonValuePtr::Array {
450                    serialized_buffer, ..
451                }
452                | JsonValuePtr::Object {
453                    serialized_buffer, ..
454                },
455            ) => serialized_buffer,
456        };
457        write_jsonb_header(buffer, tpe, serialized_buffer.len())?;
458        buffer.extend(serialized_buffer);
459
460        Ok(IsNull::No)
461    }

463    // Helper function to write a JSON value into a JSONB binary format
464    pub(super) fn write_jsonb_value(
465        value: &serde_json::Value,
466        buffer: &mut Vec<u8>,
467    ) -> serialize::Result {
468        let mut stack = ::alloc::boxed::box_assume_init_into_vec_unsafe(::alloc::intrinsics::write_box_via_move(::alloc::boxed::Box::new_uninit(),
        [JsonValuePtr::Value(value)]))vec![JsonValuePtr::Value(value)];
469
470        while let Some(value) = stack.pop() {
471            let add_to_stack = match value {
472                JsonValuePtr::Value(value) => write_plain_jsonb_value(buffer, value)?,
473                JsonValuePtr::Array {
474                    values,
475                    mut serialized_buffer,
476                } => {
477                    if let Some((el, tail)) = values.split_first() {
478                        let ret = write_plain_jsonb_value(&mut serialized_buffer, el)?;
479                        stack.push(JsonValuePtr::Array {
480                            values: tail,
481                            serialized_buffer,
482                        });
483                        ret
484                    } else {
485                        place_composite_value(buffer, &mut stack, serialized_buffer, JSONB_ARRAY)?;
486                        None
487                    }
488                }
489                JsonValuePtr::Object {
490                    object,
491                    mut keys,
492                    mut serialized_buffer,
493                } => {
494                    if let Some(next_key) = keys.next() {
495                        let value = object
496                            .get(next_key)
497                            .ok_or_else(|| ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Missing value for object key: `{0}`",
                next_key))
    })format!("Missing value for object key: `{next_key}`"))?;
498                        write_jsonb_string(next_key, &mut serialized_buffer)?;
499                        let ret = write_plain_jsonb_value(&mut serialized_buffer, value)?;
500                        stack.push(JsonValuePtr::Object {
501                            object,
502                            keys,
503                            serialized_buffer,
504                        });
505                        ret
506                    } else {
507                        place_composite_value(buffer, &mut stack, serialized_buffer, JSONB_OBJECT)?;
508                        None
509                    }
510                }
511            };
512            if let Some(next) = add_to_stack {
513                stack.push(next);
514            }
515        }
516
517        Ok(IsNull::No)
518    }

520    fn write_plain_jsonb_value<'a>(
521        buffer: &mut Vec<u8>,
522        value: &'a serde_json::Value,
523    ) -> Result<Option<JsonValuePtr<'a>>, Box<dyn Error + Send + Sync>> {
524        let ret = if value.is_null() {
525            write_jsonb_null(buffer)?;
526            None
527        } else if value.is_boolean() {
528            write_jsonb_bool(value.as_bool().ok_or("Failed to read JSONB value")?, buffer)?;
529            None
530        } else if value.is_number() {
531            write_jsonb_number(value, buffer)?;
532            None
533        } else if value.is_string() {
534            write_jsonb_string(value.as_str().ok_or("Failed to read JSONB value")?, buffer)?;
535            None
536        } else if value.is_array() {
537            let array = value.as_array().ok_or("Failed to read JSONB value")?;
538            Some(JsonValuePtr::Array {
539                values: array,
540                serialized_buffer: Vec::new(),
541            })
542        } else if value.is_object() {
543            let object = value.as_object().ok_or("Failed to read JSONB value")?;
544            Some(JsonValuePtr::Object {
545                object,
546                keys: Box::new(object.keys()),
547                serialized_buffer: Vec::new(),
548            })
549        } else {
550            return Err("Unsupported JSONB value type".into());
551        };
552        Ok(ret)
553    }

568    // Write a JSON number (integers and floats)
569    pub(super) fn write_jsonb_number(
570        n: &serde_json::Value,
571        buffer: &mut Vec<u8>,
572    ) -> serialize::Result {
573        let n = n.to_string();
574        let tpe = if n
575            .char_indices()
576            .any(|(idx, c)| !(c.is_ascii_digit() || (idx == 0 && (c == '-' || c == '+'))))
577        {
578            JSONB_FLOAT
579        } else {
580            JSONB_INT
581        };
582        write_jsonb_header(buffer, tpe, n.len())?;
583
584        // Write the ASCII text representation of the integer/float as the payload
585        buffer.extend_from_slice(n.as_bytes());
586
587        Ok(IsNull::No)
588    }
589
590    pub(super) fn write_jsonb_string(s: &str, buffer: &mut Vec<u8>) -> serialize::Result {
591        if s.chars().any(|c| c.is_control()) {
592            // If the string contains control characters, treat it as TEXTJ (escaped JSON)
593            write_jsonb_textj(s, buffer)
594        } else {
595            write_jsonb_header(buffer, JSONB_TEXT, s.len())?;
596            // Write the UTF-8 text of the string as the payload (no delimiters)
597            buffer.extend_from_slice(s.as_bytes());
598            Ok(IsNull::No)
599        }
600    }
601
602    pub(super) fn write_jsonb_textj(s: &str, buffer: &mut Vec<u8>) -> serialize::Result {
603        // Escaping the string for JSON (e.g., \n, \uXXXX)
604        let escaped_string = serde_json::to_string(&String::from(s))
605            .map_err(|_| "Failed to serialize string for TEXTJ")?;
606
607        // Remove the surrounding quotes from serde_json::to_string result
608        let escaped_string = &escaped_string[1..escaped_string.len() - 1];
609
610        // Write the header (JSONB_TEXTJ) and the length of the escaped string
611        write_jsonb_header(buffer, JSONB_TEXTJ, escaped_string.len())?;
612
613        // Write the escaped string as the payload
614        buffer.extend_from_slice(escaped_string.as_bytes());
615
616        Ok(IsNull::No)
617    }
618}
