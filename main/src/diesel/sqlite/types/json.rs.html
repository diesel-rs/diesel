27impl FromSql<sql_types::Jsonb, Sqlite> for serde_json::Value {
28    fn from_sql(mut value: SqliteValue<'_, '_, '_>) -> deserialize::Result<Self> {
29        use self::jsonb::*;
30
31        let bytes = value.read_blob();
32
33        if bytes.is_empty() {
34            return Err("Empty blob cannot be decoded as JSONB".into());
35        }
36
37        // Read the JSONB value from the byte stream
38        let (jsonb, size) = read_jsonb_value(bytes)?;
39        if size == bytes.len() {
40            Ok(jsonb)
41        } else {
42            Err("Payload contained more bytes than the encoded JSONB".into())
43        }
44    }

96    #[allow(unsafe_code)]
97    pub(super) fn read_jsonb_value(
98        bytes: &[u8],
99    ) -> deserialize::Result<(serde_json::Value, usize)> {
100        if bytes.is_empty() {
101            return Err("Empty JSONB data".into());
102        }
103        let (global_header, mut global_value) = read_header_and_value(bytes)?;
104
105        if global_value.is_array() || global_value.is_object() {
106            // we need to use pointers here, as the borrow checker does not understand
107            // that we only modify the last value in this stack. Given that we cannot
108            // invalidate any pointer that's up in the stack
109            let mut stack = ::alloc::boxed::box_assume_init_into_vec_unsafe(::alloc::intrinsics::write_box_via_move(::alloc::boxed::Box::new_uninit(),
        [(&mut global_value as *mut serde_json::Value,
                    global_header.payload_size)]))vec![(
110                &mut global_value as *mut serde_json::Value,
111                global_header.payload_size,
112            )];
113            let mut payload = &bytes[global_header.header_size..];
114            let mut total_read = 0;
115            // we must use a loop based decoding approach here instead of the much simpler recursive implementation
116            // as we otherwise run into stack overflows for deeply nested objects/arrays
117
118            while total_read < global_header.payload_size {
119                let Some((value, _size)) = stack.last().copied() else {
120                    break;
121                };
122                let value = unsafe {
123                    // SAFETY: The pointer was initialized before
124                    // We we cannot invalidate the underlying object
125                    &mut *value
126                };
127
128                if let serde_json::Value::Array(array) = value {
129                    let (header, value) = read_header_and_value(payload)?;
130
131                    array.push(value);
132                    let last_ref = array.last_mut().expect("Pushed above");
133                    let payload_size = if last_ref.is_object() || last_ref.is_array() {
134                        stack.push((last_ref as *mut _, total_read + header.total_size));
135                        header.header_size
136                    } else {
137                        header.total_size
138                    };
139                    total_read += payload_size;
140                    if payload.len() > payload_size {
141                        payload = &payload[payload_size..];
142                    } else {
143                        for (_, v) in stack {
144                            if v != total_read {
145                                return Err("Invalid size of payload declared".into());
146                            }
147                        }
148                        break;
149                    }
150                } else if let serde_json::Value::Object(object) = value {
151                    //       while total_read < payload_size {
152                    let (key_header, key) = read_header_and_value(payload)?;
153                    total_read += key_header.total_size;
154                    let serde_json::Value::String(key) = key else {
155                        return Err("Expected a string as object key".into());
156                    };
157                    if payload.len() > key_header.total_size {
158                        payload = &payload[key_header.total_size..];
159                    } else {
160                        return Err("No value found for object".into());
161                    }
162                    let (value_header, value) = read_header_and_value(payload)?;
163                    object.insert(key.clone(), value);
164                    let last_ref = object.get_mut(&key).expect("We inserted it above");
165                    let payload_size = if last_ref.is_object() || last_ref.is_array() {
166                        stack.push((last_ref as *mut _, total_read + value_header.total_size));
167                        value_header.header_size
168                    } else {
169                        value_header.total_size
170                    };
171                    total_read += payload_size;
172                    if payload.len() > payload_size {
173                        payload = &payload[payload_size..];
174                    } else {
175                        for (_, v) in stack {
176                            if v != total_read {
177                                return Err("Invalid size of payload declared".into());
178                            }
179                        }
180                        break;
181                    }
182                } else {
183                    ::core::panicking::panic("internal error: entered unreachable code")unreachable!()
184                }
185
186                while let Some(v) = stack.last().map(|(_, v)| *v) {
187                    if v > total_read {
188                        break;
189                    } else if v == total_read {
190                        stack.pop();
191                    } else {
192                        return Err("Invalid size of payload declared".into());
193                    }
194                }
195            }
196        }
197        Ok((global_value, global_header.total_size))
198    }

200    // This function decodes the jsonb header
201    // and the value for non-composite values. For composite values like array and object
202    // we only decode the "value header" and leave decoding
203    // the actual child values to future calls
204    fn read_header_and_value(
205        bytes: &[u8],
206    ) -> deserialize::Result<(JsonbHeader, serde_json::Value)> {
207        let header = read_jsonb_value_header(bytes)?;
208        let payload_bytes = &bytes[header.header_size..header.total_size];
209        let value = match header.element_type {
210            // sqlite writes a constant as a bare one byte header and refuses any other spelling
211            JSONB_NULL | JSONB_TRUE | JSONB_FALSE if header.total_size != 1 => {
212                Err("Invalid JSONB data: a constant must be a single byte".into())
213            }
214            JSONB_NULL => Ok(serde_json::Value::Null),
215            JSONB_TRUE => Ok(serde_json::Value::Bool(true)),
216            JSONB_FALSE => Ok(serde_json::Value::Bool(false)),
217            JSONB_INT => read_jsonb_int(payload_bytes, header.payload_size),
218            JSONB_INT5 => Err("INT5 is not supported".into()),
219            JSONB_FLOAT => read_jsonb_float(payload_bytes, header.payload_size),
220            JSONB_FLOAT5 => Err("FLOAT5 is not supported".into()),
221            JSONB_TEXT => read_jsonb_text(payload_bytes, header.payload_size),
222            JSONB_TEXTJ => read_jsonb_textj(payload_bytes, header.payload_size),
223            JSONB_TEXTRAW => read_jsonb_text(payload_bytes, header.payload_size),
224            JSONB_TEXT5 => Err("TEXT5 is not supported".into()),
225            JSONB_ARRAY => Ok(serde_json::Value::Array(alloc::vec::Vec::new())),
226            JSONB_OBJECT => Ok(serde_json::Value::Object(serde_json::Map::new())),
227            _ => Err(::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Unsupported or reserved JSONB type: {0}",
                header.element_type))
    })alloc::format!(
228                "Unsupported or reserved JSONB type: {}",
229                header.element_type
230            )
231            .into()),
232        }?;
233        Ok((header, value))
234    }

236    fn read_jsonb_value_header(bytes: &[u8]) -> deserialize::Result<JsonbHeader> {
237        let first_byte = bytes[0];
238        let element_type = first_byte & 0x0F;
239        let size_hint = (first_byte & 0xF0) >> 4;
240        let (payload_size, header_size): (usize, usize) = match size_hint {
241            0x00..=0x0B => (size_hint as usize, 1), // Payload size is directly in the upper nibble
242            0x0C => {
243                if bytes.len() < 2 {
244                    return Err("Invalid JSONB data: insufficient bytes for payload size".into());
245                }
246                (bytes[1] as usize, 2) // 1 additional byte for payload size
247            }
248            0x0D => {
249                if bytes.len() < 3 {
250                    return Err("Invalid JSONB data: insufficient bytes for payload size".into());
251                }
252                (u16::from_be_bytes([bytes[1], bytes[2]]) as usize, 3) // 2 additional bytes
253            }
254            0x0E => {
255                if bytes.len() < 5 {
256                    return Err("Invalid JSONB data: insufficient bytes for payload size".into());
257                }
258                (
259                    u32::from_be_bytes([bytes[1], bytes[2], bytes[3], bytes[4]]) as usize,
260                    5,
261                ) // 4 additional bytes
262            }
263            0x0F => {
264                if bytes.len() < 9 {
265                    return Err("Invalid JSONB data: insufficient bytes for payload size".into());
266                }
267                (
268                    usize::try_from(u64::from_be_bytes([
269                        bytes[1], bytes[2], bytes[3], bytes[4], bytes[5], bytes[6], bytes[7],
270                        bytes[8],
271                    ]))
272                    .map_err(Box::new)?,
273                    9,
274                ) // 8 additional bytes
275            }
276            _ => return Err("Invalid payload size hint".into()),
277        };
278        let total_size = header_size
279            .checked_add(payload_size)
280            .ok_or("The provided payload size overflows usize")?;
281        if bytes.len() < total_size {
282            return Err(::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Invalid JSONB data: insufficient bytes for value of type {0}, expected {1} bytes, got {2}",
                element_type, total_size, bytes.len()))
    })alloc::format!(
283                "Invalid JSONB data: insufficient bytes for value of type {}, expected {} bytes, got {}",
284                element_type,
285                total_size,
286                bytes.len()
287            )
288            .into());
289        }
290
291        Ok(JsonbHeader {
292            element_type,
293            payload_size,
294            header_size,
295            total_size,
296        })
297    }

299    // Read a JSON integer in canonical format (INT)
300    pub(super) fn read_jsonb_int(
301        bytes: &[u8],
302        payload_size: usize,
303    ) -> deserialize::Result<serde_json::Value> {
304        // Ensure the bytes are at least as large as the payload size
305        if bytes.len() < payload_size {
306            return Err(::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Expected payload of size {0}, but got {1}",
                payload_size, bytes.len()))
    })alloc::format!(
307                "Expected payload of size {}, but got {}",
308                payload_size,
309                bytes.len()
310            )
311            .into());
312        }
313
314        // Read only the number of bytes specified by the payload size
315        let int_str = core::str::from_utf8(bytes).map_err(|_| "Invalid ASCII in JSONB integer")?;
316        // An INT payload is integer text, so fractional and exponent forms are malformed.
317        // `serde_json` reports a non-finite `1e999` as neither float nor integer.
318        if int_str.contains(['.', 'e', 'E']) {
319            return Err("Failed to parse JSONB integer".into());
320        }
321        // `-0` is the one integer text `serde_json` turns into a float, and a signed
322        // zero only carries meaning for floats, so decode the integer it denotes.
323        if int_str == "-0" {
324            return Ok(serde_json::Value::Number(serde_json::Number::from(0)));
325        }
326        let int_value = serde_json::from_str(int_str)
327            .map_err(|_| "Failed to parse JSONB")
328            .and_then(|v: serde_json::Value| {
329                // Without `arbitrary_precision` an integer wider than 64 bits parses as
330                // a lossy float, which must not pass as the stored value.
331                (v.is_number() && !v.is_f64())
332                    .then_some(v)
333                    .ok_or("Failed to parse JSONB integer")
334            })?;
335
336        Ok(int_value)
337    }

418    pub(super) fn write_jsonb_header(
419        buffer: &mut Vec<u8>,
420        element_type: u8,
421        payload_size: usize,
422    ) -> serialize::Result {
423        // Check if payload size exceeds the maximum allowed size
424        if payload_size > 2_147_483_647 {
425            return Err("Payload size exceeds the maximum allowed size of 2GB".into());
426        }
427
428        if payload_size <= 0x0B {
429            // Small payloads, 0 additional byte for size
430            buffer.push(
431                ((u8::try_from(payload_size).map_err(|e| e.to_string())?) << 4) | element_type,
432            );
433        } else if payload_size <= 0xFF {
434            // Medium payloads, 1 additional byte for size
435            buffer.extend_from_slice(&[
436                (0x0C << 4) | element_type,
437                u8::try_from(payload_size).map_err(|e| e.to_string())?,
438            ]);
439        } else if payload_size <= 0xFFFF {
440            // Larger payloads, 2 additional bytes for size
441            buffer.push((0x0D << 4) | element_type);
442            buffer.extend_from_slice(
443                &(u16::try_from(payload_size).map_err(|e| e.to_string())?).to_be_bytes(),
444            );
445        } else {
446            // Very large payloads, 4 additional bytes for size (up to 2 GiB)
447            buffer.push((0x0E << 4) | element_type);
448            buffer.extend_from_slice(
449                &(u32::try_from(payload_size).map_err(|e| e.to_string())?).to_be_bytes(),
450            );
451        };
452
453        Ok(IsNull::No)
454    }

456    fn place_composite_value(
457        buffer: &mut Vec<u8>,
458        stack: &mut [JsonValuePtr<'_>],
459        serialized_buffer: Vec<u8>,
460        tpe: u8,
461    ) -> serialize::Result {
462        let buffer = match stack.last_mut() {
463            None | Some(JsonValuePtr::Value(_)) => buffer,
464            Some(
465                JsonValuePtr::Array {
466                    serialized_buffer, ..
467                }
468                | JsonValuePtr::Object {
469                    serialized_buffer, ..
470                },
471            ) => serialized_buffer,
472        };
473        write_jsonb_header(buffer, tpe, serialized_buffer.len())?;
474        buffer.extend(serialized_buffer);
475
476        Ok(IsNull::No)
477    }

487            let add_to_stack = match value {
488                JsonValuePtr::Value(value) => write_plain_jsonb_value(buffer, value)?,

490                    values,
491                    mut serialized_buffer,
492                } => {
493                    if let Some((el, tail)) = values.split_first() {
494                        let ret = write_plain_jsonb_value(&mut serialized_buffer, el)?;
495                        stack.push(JsonValuePtr::Array {
496                            values: tail,
497                            serialized_buffer,

501                        place_composite_value(buffer, &mut stack, serialized_buffer, JSONB_ARRAY)?;

504                }
505                JsonValuePtr::Object {
506                    object,
507                    mut keys,
508                    mut serialized_buffer,
509                } => {

511                        let value = object
512                            .get(next_key)
513                            .ok_or_else(|| ::alloc::__export::must_use({

523                        place_composite_value(buffer, &mut stack, serialized_buffer, JSONB_OBJECT)?;

526                }

536    fn write_plain_jsonb_value<'a>(
537        buffer: &mut Vec<u8>,
538        value: &'a serde_json::Value,
539    ) -> Result<Option<JsonValuePtr<'a>>, Box<dyn Error + Send + Sync>> {
540        let ret = if value.is_null() {
541            write_jsonb_null(buffer)?;
542            None
543        } else if value.is_boolean() {
544            write_jsonb_bool(value.as_bool().ok_or("Failed to read JSONB value")?, buffer)?;
545            None
546        } else if value.is_number() {
547            write_jsonb_number(value, buffer)?;
548            None
549        } else if value.is_string() {
550            write_jsonb_string(value.as_str().ok_or("Failed to read JSONB value")?, buffer)?;
551            None
552        } else if value.is_array() {
553            let array = value.as_array().ok_or("Failed to read JSONB value")?;
554            Some(JsonValuePtr::Array {
555                values: array,
556                serialized_buffer: Vec::new(),
557            })
558        } else if value.is_object() {
559            let object = value.as_object().ok_or("Failed to read JSONB value")?;
560            Some(JsonValuePtr::Object {
561                object,
562                keys: Box::new(object.keys()),
563                serialized_buffer: Vec::new(),
564            })
565        } else {
566            return Err("Unsupported JSONB value type".into());
567        };
568        Ok(ret)
569    }

578    pub(super) fn write_jsonb_bool(b: bool, buffer: &mut Vec<u8>) -> serialize::Result {
579        // Use the constants for true and false
580        write_jsonb_header(buffer, if b { JSONB_TRUE } else { JSONB_FALSE }, 0x0)?;

584    // Write a JSON number (integers and floats)
585    pub(super) fn write_jsonb_number(
586        n: &serde_json::Value,
587        buffer: &mut Vec<u8>,
588    ) -> serialize::Result {
589        let n = n.to_string();
590        let tpe = if n
591            .char_indices()
592            .any(|(idx, c)| !(c.is_ascii_digit() || (idx == 0 && (c == '-' || c == '+'))))
593        {
594            JSONB_FLOAT
595        } else {
596            JSONB_INT
597        };
598        write_jsonb_header(buffer, tpe, n.len())?;
599
600        // Write the ASCII text representation of the integer/float as the payload
601        buffer.extend_from_slice(n.as_bytes());
602
603        Ok(IsNull::No)
604    }
605
606    pub(super) fn write_jsonb_string(s: &str, buffer: &mut Vec<u8>) -> serialize::Result {
607        // strings needing a json escape go down TEXTJ, matching what sqlite's own jsonb() writes
608        // the scan is over bytes because json escapes nothing above 0x1F, unlike char::is_control
609        if s.bytes().any(|b| b < 0x20 || b == b'"' || b == b'\\') {
610            write_jsonb_textj(s, buffer)
611        } else {
612            write_jsonb_header(buffer, JSONB_TEXT, s.len())?;
613            buffer.extend_from_slice(s.as_bytes());
614            Ok(IsNull::No)
615        }
616    }
617
618    pub(super) fn write_jsonb_textj(s: &str, buffer: &mut Vec<u8>) -> serialize::Result {
619        // &s passes a sized &str, required by the serde_json 0.8.0 to_string bound
620        let escaped_string =
621            serde_json::to_string(&s).map_err(|_| "Failed to serialize string for TEXTJ")?;
622
623        let escaped_string = &escaped_string[1..escaped_string.len() - 1];
624
625        write_jsonb_header(buffer, JSONB_TEXTJ, escaped_string.len())?;
626
627        buffer.extend_from_slice(escaped_string.as_bytes());
628
629        Ok(IsNull::No)
630    }
