190    /// Shrink a database by releasing freelist pages, without the full rewrite
191    /// [`VACUUM`](https://www.sqlite.org/lang_vacuum.html) performs.
192    ///
193    /// `schema` selects an attached database by name, `None` targets `main`. `pages`
194    /// bounds how many pages are reclaimed. As SQLite specifies, `None` or a value
195    /// below one clears the whole freelist, as does a bound larger than it. Only
196    /// databases in [`AutoVacuumMode::Incremental`] have anything to reclaim, on any
197    /// other mode this succeeds and does nothing.
198    ///
199    /// ```rust
200    /// # include!("../../doctest_setup.rs");
201    /// # fn main() {
202    /// #     run_test().unwrap();
203    /// # }
204    /// # fn run_test() -> QueryResult<()> {
205    /// use diesel::sqlite::AutoVacuumMode;
206    /// let conn = &mut SqliteConnection::establish(":memory:").unwrap();
207    /// conn.set_auto_vacuum(None, AutoVacuumMode::Incremental)?;
208    /// // Reclaim at most 8 pages, then whatever is left.
209    /// conn.incremental_vacuum(None, Some(8))?;
210    /// conn.incremental_vacuum(None, None)?;
211    /// #     Ok(())
212    /// # }
213    /// ```
214    pub fn incremental_vacuum(
215        &mut self,
216        schema: Option<&str>,
217        pages: Option<u32>,
218    ) -> QueryResult<()> {
219        use crate::connection::SimpleConnection;
220        use crate::query_builder::QueryBuilder;
221        use crate::sqlite::SqliteQueryBuilder;
222
223        // SQLite frees one page per step of this statement, so it only empties the
224        // freelist when run to completion. `batch_execute` uses `sqlite3_exec`, which
225        // does that. A prepared statement would not: `StatementUse::run` steps once,
226        // which frees a single page and silently leaves the rest.
227        let mut query = SqliteQueryBuilder::new();
228        query.push_sql("PRAGMA ");
229        query.push_identifier(schema.unwrap_or("main"))?;
230        query.push_sql(".incremental_vacuum");
231        if let Some(pages) = pages {
232            query.push_sql("(");
233            query.push_sql(&pages.to_string());
234            query.push_sql(")");
235        }
236        self.batch_execute(&query.finish())
237    }

310    ///
311    /// A checkpoint stopped early by a reader or writer on another
312    /// connection is not an error: it sets
313    /// [`busy`](WalCheckpointOutcome::busy). On a database that is not in
314    /// WAL mode the call succeeds with both frame counts `None`, so it is
315    /// safe to issue unconditionally. Inside a transaction on its own
316    /// connection it fails with `SQLITE_LOCKED`.
317    ///
318    /// The mode argument requires SQLite 3.7.6 or later,
319    /// [`Truncate`](WalCheckpointMode::Truncate) requires 3.8.8 or later,
320    /// and [`Noop`](WalCheckpointMode::Noop) requires 3.51.0 or later.
321    /// Older versions do not report an error and treat an unrecognized
322    /// mode as [`Passive`](WalCheckpointMode::Passive).
323    ///
324    /// ```rust
325    /// # include!("../../doctest_setup.rs");
326    /// #
327    /// # fn main() {
328    /// #     run_test().unwrap();
329    /// # }
330    /// #
331    /// # fn run_test() -> QueryResult<()> {
332    /// use diesel::connection::SimpleConnection;
333    /// use diesel::sqlite::WalCheckpointMode;
334    /// # let dir = tempfile::tempdir().unwrap();
335    /// # let path = dir.path().join("app.db");
336    /// let conn = &mut SqliteConnection::establish(path.to_str().unwrap()).unwrap();
337    /// conn.batch_execute("PRAGMA journal_mode = WAL")?;
338    /// conn.batch_execute("CREATE TABLE logs (line TEXT NOT NULL)")?;
339    ///
340    /// let outcome = conn.wal_checkpoint(None, WalCheckpointMode::Truncate)?;
341    /// assert!(!outcome.busy);
342    /// // The whole WAL was moved into the database file and the log truncated.
343    /// assert_eq!(outcome.log_frames, Some(0));
344    /// assert_eq!(outcome.checkpointed_frames, Some(0));
345    /// #     Ok(())
346    /// # }
347    /// ```
348    pub fn wal_checkpoint(
349        &mut self,
350        schema: Option<&str>,
351        mode: WalCheckpointMode,
352    ) -> QueryResult<WalCheckpointOutcome> {
353        use crate::query_dsl::RunQueryDsl;
354
355        let (busy, log_frames, checkpointed_frames) =
356            WalCheckpoint { schema, mode }.get_result::<(i32, i64, i64)>(self)?;
357        Ok(WalCheckpointOutcome {
358            busy: busy != 0,
359            // On a database not in WAL mode both counts come back as -1.
360            log_frames: (log_frames >= 0).then_some(log_frames),
361            checkpointed_frames: (checkpointed_frames >= 0).then_some(checkpointed_frames),
362        })
363    }
364}

464// Like `Pragma`, no operand can be a bind parameter. Unlike `Pragma`, a
465// `None` schema stays unqualified on purpose: the unqualified pragma
466// checkpoints every attached database, while a qualified one targets a
467// single schema. The whole checkpoint runs on the first step of the
468// statement and yields exactly one row, so a prepared statement works here
469// (unlike `incremental_vacuum`).
470struct WalCheckpoint<'a> {
471    schema: Option<&'a str>,
472    mode: WalCheckpointMode,
473}
474
475impl QueryFragment<Sqlite> for WalCheckpoint<'_> {
476    fn walk_ast<'b>(&'b self, mut out: AstPass<'_, 'b, Sqlite>) -> QueryResult<()> {
477        out.push_sql("PRAGMA ");
478        if let Some(schema) = self.schema {
479            out.push_identifier(schema)?;
480            out.push_sql(".");
481        }
482        out.push_sql(match self.mode {
483            WalCheckpointMode::Passive => "wal_checkpoint(PASSIVE)",
484            WalCheckpointMode::Full => "wal_checkpoint(FULL)",
485            WalCheckpointMode::Restart => "wal_checkpoint(RESTART)",
486            WalCheckpointMode::Truncate => "wal_checkpoint(TRUNCATE)",
487            WalCheckpointMode::Noop => "wal_checkpoint(NOOP)",
488        });
489        Ok(())
490    }
491}
