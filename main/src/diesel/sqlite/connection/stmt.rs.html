45impl Statement {
46    pub(super) fn prepare(
47        raw_connection: &RawConnection,
48        sql: &str,
49        is_cached: PrepareForCache,
50        _: &[SqliteType],
51    ) -> QueryResult<Self> {
52        let mut stmt = ptr::null_mut();
53        let mut unused_portion = ptr::null();
54        let n_byte = sql
55            .len()
56            .try_into()
57            .map_err(|e| Error::SerializationError(Box::new(e)))?;
58        // the cast for `ffi::SQLITE_PREPARE_PERSISTENT` is required for old libsqlite3-sys versions
59        #[allow(clippy::unnecessary_cast)]
60        let prepare_result = unsafe {
61            ffi::sqlite3_prepare_v3(
62                raw_connection.internal_connection.as_ptr(),
63                CString::new(sql)?.as_ptr(),
64                n_byte,
65                if #[allow(non_exhaustive_omitted_patterns)] match is_cached {
    PrepareForCache::Yes { counter: _ } => true,
    _ => false,
}matches!(is_cached, PrepareForCache::Yes { counter: _ }) {
66                    ffi::SQLITE_PREPARE_PERSISTENT as u32
67                } else {
68                    0
69                },
70                &mut stmt,
71                &mut unused_portion,
72            )
73        };
74
75        ensure_sqlite_ok(prepare_result, raw_connection.internal_connection.as_ptr())?;
76
77        // sqlite3_prepare_v3 returns a null pointer for empty statements. This includes
78        // empty or only whitespace strings or any other non-op query string like a comment
79        let inner_statement = NonNull::new(stmt).ok_or_else(|| {
80            crate::result::Error::QueryBuilderError(Box::new(crate::result::EmptyQuery))
81        })?;
82        Ok(Statement { inner_statement })
83    }
84
85    // The caller of this function has to ensure that:
86    // * Any buffer provided as `SqliteBindValue::BorrowedBinary`, `SqliteBindValue::Binary`
87    // `SqliteBindValue::String` or `SqliteBindValue::BorrowedString` is valid
88    // till either a new value is bound to the same parameter or the underlying
89    // prepared statement is dropped.
90    unsafe fn bind(
91        &mut self,
92        tpe: SqliteType,
93        value: SqliteBindValueRef<'_>,
94        bind_index: i32,
95    ) -> QueryResult<Option<NonNull<[u8]>>> {
96        let mut ret_ptr = None;
97        let result = match (tpe, value) {
98            (_, SqliteBindValueRef::Null) => unsafe {
99                ffi::sqlite3_bind_null(self.inner_statement.as_ptr(), bind_index)
100            },
101            (SqliteType::Binary, SqliteBindValueRef::BorrowedBinary(bytes)) => {
102                let n = bytes
103                    .len()
104                    .try_into()
105                    .map_err(|e| Error::SerializationError(Box::new(e)))?;
106                unsafe {
107                    ffi::sqlite3_bind_blob(
108                        self.inner_statement.as_ptr(),
109                        bind_index,
110                        bytes.as_ptr() as *const libc::c_void,
111                        n,
112                        ffi::SQLITE_STATIC(),
113                    )
114                }
115            }
116            (SqliteType::Binary, SqliteBindValueRef::Binary(mut bytes)) => {
117                let len = bytes
118                    .len()
119                    .try_into()
120                    .map_err(|e| Error::SerializationError(Box::new(e)))?;
121                // We need a separate pointer here to pass it to sqlite
122                // as the returned pointer is a pointer to a dyn sized **slice**
123                // and not the pointer to the first element of the slice
124                let ptr = bytes.as_mut_ptr();
125                ret_ptr = NonNull::new(Box::into_raw(bytes));
126                unsafe {
127                    ffi::sqlite3_bind_blob(
128                        self.inner_statement.as_ptr(),
129                        bind_index,
130                        ptr as *const libc::c_void,
131                        len,
132                        ffi::SQLITE_STATIC(),
133                    )
134                }
135            }
136            (SqliteType::Text, SqliteBindValueRef::BorrowedString(bytes)) => {
137                let len = bytes
138                    .len()
139                    .try_into()
140                    .map_err(|e| Error::SerializationError(Box::new(e)))?;
141                unsafe {
142                    ffi::sqlite3_bind_text(
143                        self.inner_statement.as_ptr(),
144                        bind_index,
145                        bytes.as_ptr() as *const libc::c_char,
146                        len,
147                        ffi::SQLITE_STATIC(),
148                    )
149                }
150            }
151            (SqliteType::Text, SqliteBindValueRef::String(bytes)) => {
152                let mut bytes = Box::<[u8]>::from(bytes);
153                let len = bytes
154                    .len()
155                    .try_into()
156                    .map_err(|e| Error::SerializationError(Box::new(e)))?;
157                // We need a separate pointer here to pass it to sqlite
158                // as the returned pointer is a pointer to a dyn sized **slice**
159                // and not the pointer to the first element of the slice
160                let ptr = bytes.as_mut_ptr();
161                ret_ptr = NonNull::new(Box::into_raw(bytes));
162                unsafe {
163                    ffi::sqlite3_bind_text(
164                        self.inner_statement.as_ptr(),
165                        bind_index,
166                        ptr as *const libc::c_char,
167                        len,
168                        ffi::SQLITE_STATIC(),
169                    )
170                }
171            }
172            (SqliteType::Float, SqliteBindValueRef::F64(value))
173            | (SqliteType::Double, SqliteBindValueRef::F64(value)) => unsafe {
174                ffi::sqlite3_bind_double(
175                    self.inner_statement.as_ptr(),
176                    bind_index,
177                    value as libc::c_double,
178                )
179            },
180            (SqliteType::SmallInt, SqliteBindValueRef::I32(value))
181            | (SqliteType::Integer, SqliteBindValueRef::I32(value)) => unsafe {
182                ffi::sqlite3_bind_int(self.inner_statement.as_ptr(), bind_index, value)
183            },
184            (SqliteType::Long, SqliteBindValueRef::I64(value)) => unsafe {
185                ffi::sqlite3_bind_int64(self.inner_statement.as_ptr(), bind_index, value)
186            },
187            (t, b) => {
188                return Err(Error::SerializationError(
189                    ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("Type mismatch: Expected {0:?}, got {1}",
                t, b))
    })alloc::format!("Type mismatch: Expected {t:?}, got {b}").into(),
190                ));
191            }
192        };
193        match ensure_sqlite_ok(result, self.raw_connection()) {
194            Ok(()) => Ok(ret_ptr),
195            Err(e) => {
196                if let Some(ptr) = ret_ptr {
197                    // This is a `NonNul` ptr so it cannot be null
198                    // It points to a slice internally as we did not apply
199                    // any cast above.
200                    core::mem::drop(unsafe { Box::from_raw(ptr.as_ptr()) })
201                }
202                Err(e)
203            }
204        }
205    }
206
207    fn reset(&mut self) {
208        unsafe { ffi::sqlite3_reset(self.inner_statement.as_ptr()) };
209    }

211    /// Steps the statement once for its side effects.
212    pub(super) fn run(&mut self) -> QueryResult<()> {
213        // SAFETY: `inner_statement` stays a live prepared statement until `Drop`.
214        match unsafe { ffi::sqlite3_step(self.inner_statement.as_ptr()) } {
215            ffi::SQLITE_DONE | ffi::SQLITE_ROW => Ok(()),
216            _ => {
217                let error = last_error(self.raw_connection());
218                // Clears the step error so that `Drop` finalizes successfully.
219                self.reset();
220                Err(error)
221            }
222        }
223    }
224
225    fn clear_bindings(&mut self) {
226        unsafe { ffi::sqlite3_clear_bindings(self.inner_statement.as_ptr()) };
227    }
228
229    fn raw_connection(&self) -> *mut ffi::sqlite3 {
230        unsafe { ffi::sqlite3_db_handle(self.inner_statement.as_ptr()) }
231    }

234pub(super) fn ensure_sqlite_ok(
235    code: libc::c_int,
236    raw_connection: *mut ffi::sqlite3,
237) -> QueryResult<()> {
238    if code == ffi::SQLITE_OK {
239        Ok(())
240    } else {
241        Err(last_error(raw_connection))
242    }

245fn last_error(raw_connection: *mut ffi::sqlite3) -> Error {
246    let error_message = last_error_message(raw_connection);
247    let error_code = last_error_code(raw_connection);
248    let error_kind = match error_code {
249        ffi::SQLITE_CONSTRAINT_UNIQUE | ffi::SQLITE_CONSTRAINT_PRIMARYKEY => {
250            DatabaseErrorKind::UniqueViolation
251        }
252        ffi::SQLITE_CONSTRAINT_FOREIGNKEY => DatabaseErrorKind::ForeignKeyViolation,
253        // SQLITE_CONSTRAINT_TRIGGER is returned for ON DELETE RESTRICT violations,
254        // which are actually foreign key violations. We check the error message
255        // to distinguish from user-defined trigger failures.
256        ffi::SQLITE_CONSTRAINT_TRIGGER
257            if error_message.contains("FOREIGN KEY constraint failed") =>
258        {
259            DatabaseErrorKind::ForeignKeyViolation
260        }
261        ffi::SQLITE_CONSTRAINT_NOTNULL => DatabaseErrorKind::NotNullViolation,
262        ffi::SQLITE_CONSTRAINT_CHECK => DatabaseErrorKind::CheckViolation,
263        _ => DatabaseErrorKind::Unknown,
264    };
265    let error_information = Box::new(error_message);
266    DatabaseError(error_kind, error_information)
267}

269fn last_error_message(conn: *mut ffi::sqlite3) -> String {
270    let c_str = unsafe { CStr::from_ptr(ffi::sqlite3_errmsg(conn)) };
271    c_str.to_string_lossy().into_owned()
272}

274fn last_error_code(conn: *mut ffi::sqlite3) -> libc::c_int {
275    unsafe { ffi::sqlite3_extended_errcode(conn) }

294struct BoundStatement<'stmt, 'query> {
295    statement: MaybeCached<'stmt, Statement>,
296    // we need to store the query here to ensure no one does
297    // drop it till the end of the statement
298    // We use a boxed queryfragment here just to erase the
299    // generic type, we use NonNull to communicate
300    // that this is a shared buffer
301    query: Option<NonNull<dyn QueryFragment<Sqlite> + 'query>>,
302    // we need to store any owned bind values separately, as they are not
303    // contained in the query itself. We use NonNull to
304    // communicate that this is a shared buffer
305    binds_to_free: Vec<(i32, Option<NonNull<[u8]>>)>,
306    instrumentation: &'stmt mut dyn Instrumentation,
307    has_error: bool,
308}
309
310impl<'stmt, 'query> BoundStatement<'stmt, 'query> {
311    fn bind<T>(
312        mut statement: MaybeCached<'stmt, Statement>,
313        query: T,
314        instrumentation: &'stmt mut dyn Instrumentation,
315    ) -> QueryResult<BoundStatement<'stmt, 'query>>
316    where
317        T: QueryFragment<Sqlite> + QueryId + 'query,
318    {
319        // Don't use a trait object here to prevent using a virtual function call
320        // For sqlite this can introduce a measurable overhead
321        // Query is boxed here to make sure it won't move in memory anymore, so any bind
322        // it could output would stay valid.
323        let query = Box::new(query);
324        // make sure to reset any previously left bindings
325        statement.reset();
326        statement.clear_bindings();
327
328        let mut bind_collector = SqliteBindCollector::new();
329        query.collect_binds(&mut bind_collector, &mut (), &Sqlite)?;
330
331        let SqliteBindCollector { binds } = bind_collector;
332        if statement.parameter_count() != binds.len() as u64 {
333            return Err(
334                crate::connection::statement_cache::invalid_prepared_statement_parameter_count(
335                    statement.parameter_count().try_into().unwrap_or(usize::MAX),
336                    binds.len(),
337                ),
338            );
339        }
340
341        let mut ret = BoundStatement {
342            statement,
343            query: None,
344            binds_to_free: Vec::new(),
345            instrumentation,
346            has_error: false,
347        };
348
349        ret.bind_buffers(binds)?;
350
351        let query = query as Box<dyn QueryFragment<Sqlite> + 'query>;
352        ret.query = NonNull::new(Box::into_raw(query));
353
354        Ok(ret)
355    }
356
357    // This is a separated function so that
358    // not the whole constructor is generic over the query type T.
359    // This hopefully prevents binary bloat.
360    fn bind_buffers(
361        &mut self,
362        binds: Vec<(SqliteBindValueRef<'_>, SqliteType)>,
363    ) -> QueryResult<()> {
364        // It is useful to preallocate `binds_to_free` because it
365        // - Guarantees that pushing inside it cannot panic, which guarantees the `Drop`
366        //   impl of `BoundStatement` will always re-`bind` as needed
367        // - Avoids reallocations
368        self.binds_to_free.reserve(
369            binds
370                .iter()
371                .filter(|&(b, _)| {
372                    #[allow(non_exhaustive_omitted_patterns)] match b {
    SqliteBindValueRef::BorrowedBinary(_) |
        SqliteBindValueRef::BorrowedString(_) | SqliteBindValueRef::String(_)
        | SqliteBindValueRef::Binary(_) => true,
    _ => false,
}matches!(
373                        b,
374                        SqliteBindValueRef::BorrowedBinary(_)
375                            | SqliteBindValueRef::BorrowedString(_)
376                            | SqliteBindValueRef::String(_)
377                            | SqliteBindValueRef::Binary(_)
378                    )
379                })
380                .count(),
381        );
382        for (bind_idx, (bind, tpe)) in (1..).zip(binds) {
383            let is_borrowed_bind = #[allow(non_exhaustive_omitted_patterns)] match bind {
    SqliteBindValueRef::BorrowedString(_) |
        SqliteBindValueRef::BorrowedBinary(_) => true,
    _ => false,
}matches!(
384                bind,
385                SqliteBindValueRef::BorrowedString(_) | SqliteBindValueRef::BorrowedBinary(_)
386            );
387
388            // It's safe to call bind here as:
389            // * The type and value matches
390            // * We ensure that corresponding buffers lives long enough below
391            // * The statement is not used yet by `step` or anything else
392            let res = unsafe { self.statement.bind(tpe, bind, bind_idx) }?;
393
394            // it's important to push these only after
395            // the call to bind succeeded, otherwise we might attempt to
396            // call bind to an non-existing bind position in
397            // the destructor
398            if let Some(ptr) = res {
399                // Store the id + pointer for a owned bind
400                // as we must unbind and free them on drop
401                self.binds_to_free.push((bind_idx, Some(ptr)));
402            } else if is_borrowed_bind {
403                // Store the id's of borrowed binds to unbind them on drop
404                self.binds_to_free.push((bind_idx, None));
405            }
406        }
407        Ok(())
408    }
409
410    fn finish_query_with_error(mut self, e: &Error) {
411        self.has_error = true;
412        if let Some(q) = self.query {
413            // it's safe to get a reference from this ptr as it's guaranteed to not be null
414            let q = unsafe { q.as_ref() };
415            self.instrumentation.on_connection_event(
416                crate::connection::InstrumentationEvent::FinishQuery {
417                    query: &crate::debug_query(&q),
418                    error: Some(e),
419                },
420            );
421        }
422    }
423}

425impl Drop for BoundStatement<'_, '_> {
426    fn drop(&mut self) {
427        // First reset the statement, otherwise the bind calls
428        // below will fails
429        self.statement.reset();
430
431        for (idx, buffer) in core::mem::take(&mut self.binds_to_free) {
432            unsafe {
433                // It's always safe to bind null values, as there is no buffer that needs to outlife something
434                self.statement
435                    .bind(SqliteType::Text, SqliteBindValueRef::Null, idx)
436                    .expect(
437                        "Binding a null value should never fail. \
438                             If you ever see this error message please open \
439                             an issue at diesels issue tracker containing \
440                             code how to trigger this message.",
441                    );
442            }
443
444            if let Some(buffer) = buffer {
445                unsafe {
446                    // Constructing the `Box` here is safe as we
447                    // got the pointer from a box + it is guaranteed to be not null.
448                    core::mem::drop(Box::from_raw(buffer.as_ptr()));
449                }
450            }
451        }
452
453        if let Some(query) = self.query {
454            let query = unsafe {
455                // Constructing the `Box` here is safe as we
456                // got the pointer from a box + it is guaranteed to be not null.
457                Box::from_raw(query.as_ptr())
458            };
459            if !self.has_error {
460                self.instrumentation.on_connection_event(
461                    crate::connection::InstrumentationEvent::FinishQuery {
462                        query: &crate::debug_query(&query),
463                        error: None,
464                    },
465                );
466            }
467            core::mem::drop(query);
468            self.query = None;
469        }
470    }
471}

474pub struct StatementUse<'stmt, 'query> {
475    statement: BoundStatement<'stmt, 'query>,
476    // SAFETY:
477    // https://sqlite.org/c3ref/column_name.html
478    // This pointer is valid as long as:
479    // The returned string pointer is valid until either
480    // the prepared statement is destroyed by sqlite3_finalize()
481    // or until the statement is automatically reprepared by the first
482    // call to sqlite3_step() for a particular run or until the next call
483    // to sqlite3_column_name() or sqlite3_column_name16() on the same column.
484    //
485    // Step has an explicit `first` flag that resets this field
486    // in this case
487    // The `OnceCell` ensure that we never call
488    // `sqlite3_column_name` again after populating this field
489    //
490    // We also assume that we are the owner of `statement` and the
491    // inner pointer is not exposed anywhere else. This ensures
492    // no one else can call the functions mentioned above
493    //
494    // We use `*const str` instead of `&str`
495    // as this is essentially a self referential struct
496    // as we borrow into `statement`
497    column_names: OnceCell<Vec<*const str>>,
498}

500impl<'stmt, 'query> StatementUse<'stmt, 'query> {
501    pub(super) fn bind<T>(
502        statement: MaybeCached<'stmt, Statement>,
503        query: T,
504        instrumentation: &'stmt mut dyn Instrumentation,
505    ) -> QueryResult<StatementUse<'stmt, 'query>>
506    where
507        T: QueryFragment<Sqlite> + QueryId + 'query,
508    {
509        Ok(Self {
510            statement: BoundStatement::bind(statement, query, instrumentation)?,
511            column_names: OnceCell::new(),
512        })
513    }
514
515    pub(super) fn run(mut self) -> QueryResult<()> {
516        let r = unsafe {
517            // This is safe as we pass `first_step = true`
518            // and we consume the statement so nobody could
519            // access the columns later on anyway.
520            self.step(true).map(|_| ())
521        };
522        if let Err(ref e) = r {
523            self.statement.finish_query_with_error(e);
524        }
525        r
526    }
527
528    // This function is marked as unsafe incorrectly passing `false` to `first_step`
529    // for a first call to this function could cause access to freed memory via
530    // the cached column names.
531    //
532    // It's always safe to call this function with `first_step = true` as this removes
533    // the cached column names
534    pub(super) unsafe fn step(&mut self, first_step: bool) -> QueryResult<bool> {
535        let step_result =
536            unsafe { ffi::sqlite3_step(self.statement.statement.inner_statement.as_ptr()) };
537        let res = match step_result {
538            ffi::SQLITE_DONE => Ok(false),
539            ffi::SQLITE_ROW => Ok(true),
540            _ => Err(last_error(self.statement.statement.raw_connection())),
541        };
542        if first_step {
543            self.column_names = OnceCell::new();
544        }
545        res
546    }
547
548    // The returned string pointer is valid until either the prepared statement is
549    // destroyed by sqlite3_finalize() or until the statement is automatically
550    // reprepared by the first call to sqlite3_step() for a particular run or
551    // until the next call to sqlite3_column_name() or sqlite3_column_name16()
552    // on the same column.
553    //
554    // https://sqlite.org/c3ref/column_name.html
555    //
556    // Note: This function is marked as unsafe, as calling it can invalidate
557    // other existing column name pointers on the same column. To prevent that,
558    // it should maximally be called once per column at all.
559    unsafe fn column_name(&self, idx: i32) -> *const str {
560        let name = {
561            let column_name = unsafe {
562                ffi::sqlite3_column_name(self.statement.statement.inner_statement.as_ptr(), idx)
563            };
564            if !!column_name.is_null() {
    {
        ::core::panicking::panic_fmt(format_args!("The Sqlite documentation states that it only returns a null pointer here if we are in a OOM condition."));
    }
};assert!(
565                !column_name.is_null(),
566                "The Sqlite documentation states that it only returns a \
567                 null pointer here if we are in a OOM condition."
568            );
569            unsafe { CStr::from_ptr(column_name) }
570        };
571        name.to_str().expect(
572            "The Sqlite documentation states that this is UTF8. \
573             If you see this error message something has gone \
574             horribly wrong. Please open an issue at the \
575             diesel repository.",
576        ) as *const str
577    }

579    pub(super) fn column_count(&self) -> i32 {
580        unsafe { ffi::sqlite3_column_count(self.statement.statement.inner_statement.as_ptr()) }
581    }

583    pub(super) fn index_for_column_name(&mut self, field_name: &str) -> Option<usize> {
584        (0..self.column_count())
585            .find(|idx| self.field_name(*idx) == Some(field_name))
586            .map(|v| {
587                v.try_into()
588                    .expect("Diesel expects to run at least on a 32 bit platform")
589            })
590    }
591
592    pub(super) fn field_name(&self, idx: i32) -> Option<&str> {
593        let column_names = self.column_names.get_or_init(|| {
594            let count = self.column_count();
595            (0..count)
596                .map(|idx| unsafe {
597                    // By initializing the whole vec at once we ensure that
598                    // we really call this only once.
599                    self.column_name(idx)
600                })
601                .collect()
602        });
603
604        column_names
605            .get(usize::try_from(idx).expect("Diesel expects to run at least on a 32 bit platform"))
606            .and_then(|c| unsafe { c.as_ref() })
607    }

609    pub(super) fn copy_value(&self, idx: i32) -> QueryResult<Option<OwnedSqliteValue>> {
610        match self.column_value(idx) {
611            Some(ptr) => OwnedSqliteValue::copy_from_ptr(ptr),
612            None => Ok(None),
613        }
614    }
615
616    pub(super) fn column_value(&self, idx: i32) -> Option<NonNull<ffi::sqlite3_value>> {
617        let ptr = unsafe {
618            ffi::sqlite3_column_value(self.statement.statement.inner_statement.as_ptr(), idx)
619        };
620        NonNull::new(ptr)
621    }
622
623    pub(super) fn raw_connection(&self) -> *mut ffi::sqlite3 {
624        self.statement.statement.raw_connection()
625    }
