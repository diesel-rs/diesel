1//! Types for the SQLite authorizer callback.
2//!
3//! See [`sqlite3_set_authorizer`](https://sqlite.org/c3ref/set_authorizer.html)
4//! and the [action code constants](https://sqlite.org/c3ref/c_alter_table.html).
5//!
6//! The authorizer callback receives an [`AuthorizerContext`]. It is an enum
7//! with one variant per SQLite action code, and every variant carries the
8//! arguments for that action under descriptive field names, so there is no
9//! need to consult a table of what each positional argument means.
10//!
11//! Two fields recur across variants. `database` is the name of the database
12//! (`"main"`, `"temp"`, or an `ATTACH` alias) an object lives in, when SQLite
13//! reports it. `accessor` is the name of the inner-most trigger or view
14//! responsible for the access, or `None` when the access is directly from
15//! top-level SQL. A string argument is `None` when SQLite passes `NULL` or a
16//! value that is not valid UTF-8.
17

798impl SqliteConnection {
799    /// Registers an authorizer callback for SQL compilation access control.
800    /// Added in SQLite 3.0.0 (June 2004).
801    ///
802    /// The callback is invoked during `sqlite3_prepare()` (statement
803    /// compilation) to control access to database objects. It receives
804    /// an [`AuthorizerContext`] describing the operation and returns an
805    /// [`AuthorizerDecision`].
806    ///
807    /// The authorizer is consulted only at statement preparation, so treat it
808    /// as defense-in-depth rather than a complete sandbox. SQLite re-checks
809    /// already-prepared statements when an authorizer is installed. Removing the
810    /// authorizer clears diesel's statement cache so statements prepared while
811    /// it was active are re-prepared without it.
812    ///
813    /// The authorizer may be re-invoked during `sqlite3_step()` if a schema
814    /// change triggers statement recompilation.
815    ///
816    /// The callback must not modify the database connection. It is invoked
817    /// synchronously on the thread driving the connection, so it is never
818    /// called concurrently.
819    ///
820    /// Only one authorizer can be active at a time per connection.
821    /// Registering a new one replaces the previous. Panics in the callback
822    /// abort the process.
823    ///
824    /// See: [`sqlite3_set_authorizer`](https://sqlite.org/c3ref/set_authorizer.html)
825    ///
826    /// # Example
827    ///
828    /// ```rust
829    /// # use diesel::prelude::*;
830    /// use diesel::sqlite::{SqliteConnection, AuthorizerContext, AuthorizerDecision};
831    ///
832    /// # let conn = &mut SqliteConnection::establish(":memory:").unwrap();
833    /// conn.on_authorize(|ctx| match ctx {
834    ///     AuthorizerContext::Delete(_) => AuthorizerDecision::Deny,
835    ///     AuthorizerContext::DropTable(_) | AuthorizerContext::DropIndex(_) => {
836    ///         AuthorizerDecision::Deny
837    ///     }
838    ///     _ => AuthorizerDecision::Allow,
839    /// });
840    ///
841    /// // Later: remove the authorizer
842    /// conn.remove_authorizer();
843    /// ```
844    pub fn on_authorize<F>(&mut self, hook: F)
845    where
846        F: FnMut(AuthorizerContext<'_>) -> AuthorizerDecision + Send + 'static,
847    {
848        // No cache clear is needed here. Installing a non-null authorizer makes
849        // SQLite expire every prepared statement on the connection, and because
850        // diesel prepares with `sqlite3_prepare_v3` those statements are
851        // transparently re-prepared under the new authorizer on their next step.
852        self.raw_connection.set_authorizer(hook);
853        // The public documentation doesn't explicitly state what happens with existing
854        // prepared statements, so rather be safe and nuke them. We don't
855        // expect that people change the authorizer all the time, so this should be fine
856        self.statement_cache.clear();
857    }
858
859    /// Removes the authorizer callback.
860    ///
861    /// See [`on_authorize`](Self::on_authorize) for usage example.
862    pub fn remove_authorizer(&mut self) {
863        self.raw_connection.remove_authorizer();
864        // Removing an authorizer does not expire prepared statements: SQLite
865        // expires them only when a non-null authorizer is installed. A statement
866        // prepared while the authorizer was active can have its decisions
867        // compiled in (an `Ignore` on a column read, for example, bakes a `NULL`
868        // into the statement), and nothing invalidates that cached statement on
869        // removal, so it keeps returning the old result. Clear the cache to force
870        // the next query to re-prepare without the authorizer.
871        self.statement_cache.clear();
872    }
