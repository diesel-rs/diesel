8/// A read only SQLite Blob
9///
10/// This interface allows to incrementally read a blob from a SQLite database.
11/// Notably this type implements [`std::io::Read`] and [`std::io::Seek`] to integrate
12/// with standard Rust IO mechanisms.
13///
14/// You can use [`SqliteConnection::get_read_only_blob`](super::SqliteConnection::get_read_only_blob)
15/// to get a new instance of this type
16///
17/// See the [SQLite documentation](https://sqlite.org/c3ref/blob_open.html) for more details
18#[expect(missing_debug_implementations)]
19#[cfg_attr(not(feature = "std"), expect(dead_code))]
20pub struct SqliteReadOnlyBlob<'conn> {
21    // `None` once closed, so the handle is closed at most once
22    pub(crate) blob: Option<core::ptr::NonNull<ffi::sqlite3_blob>>,
23    pub(crate) read_index: usize,
24
25    pub(crate) blob_size: usize,
26    pub(crate) _pd: core::marker::PhantomData<&'conn mut ffi::sqlite3_blob>,
27}

49impl SqliteReadOnlyBlob<'_> {
50    /// Is the blob storage empty
51    pub fn is_empty(&self) -> bool {
52        self.len() == 0
53    }
54
55    /// The size of the underlying blob in bytes
56    pub fn len(&self) -> usize {
57        self.blob_size
58    }
59
60    /// Close the handle
61    ///
62    /// Even if an error is returned, the handle is still closed (from the sqlite documentation):
63    ///
64    /// > The BLOB handle is closed unconditionally. Even if this routine returns an error code,
65    /// > the handle is still closed.
66    pub fn close(mut self) -> Result<(), crate::result::Error> {
67        self.close_inner()
68    }
69
70    fn close_inner(&mut self) -> Result<(), crate::result::Error> {
71        let Some(blob) = self.blob.take() else {
72            return Err(crate::result::Error::ClosingHandle("handle already closed"));
73        };
74
75        // SAFETY: `blob` came from a successful `sqlite3_blob_open`, and taking it out of `self`
76        // guarantees that this open handle is closed at most once.
77        //
78        // From the sqlite3_blob_close documentation:
79        //
80        //     If an error occurs while committing the transaction, an error code is returned and
81        //     the transaction rolled back.
82        //
83        // As we are in read-only mode here, this is not an issue
84        let close_result = unsafe { ffi::sqlite3_blob_close(blob.as_ptr()) };
85
86        if close_result != ffi::SQLITE_OK {
87            let error_message = super::error_message(close_result);
88            return Err(crate::result::Error::ClosingHandle(error_message));
89        }
90
91        Ok(())
92    }
93}

103impl std::io::Read for SqliteReadOnlyBlob<'_> {
104    fn read(&mut self, buf: &mut [u8]) -> std::io::Result<usize> {
105        let Some(blob) = self.blob else {
106            return Err(std::io::Error::other("SQLite blob handle is closed"));
107        };
108        let buflen: i32 = buf.len().try_into().map_err(to_io_error)?;
109        let offset: i32 = self.read_index.try_into().map_err(to_io_error)?;
110
111        // From the sqlite docs:
112        //
113        // > If offset iOffset is less than N bytes from the end of the BLOB, SQLITE_ERROR is returned and no data is read.
114        //
115        // Thus we need to make sure to not provide a buffer that is too big for the remaining data
116        // from the blob.
117        let read_length: i32 = (i32::try_from(self.blob_size)
118            .map_err(to_io_error)?
119            .saturating_sub(offset))
120        .min(buflen);
121
122        // SAFETY: `blob` is `Some`, so the handle is open, and `'conn` keeps its connection alive.
123        // `read_length` is at most `buf.len()` and the bytes left after `offset`, so the write
124        // stays within `buf` and the read within the blob.
125        let ret = unsafe {
126            ffi::sqlite3_blob_read(
127                blob.as_ptr(),
128                buf.as_mut_ptr() as *mut core::ffi::c_void,
129                read_length,
130                offset,
131            )
132        };
133
134        if ret != ffi::SQLITE_OK {
135            let error_message = crate::sqlite::connection::error_message(ret);
136            return Err(std::io::Error::other(error_message.to_string()));
137        }
138
139        self.read_index += usize::try_from(read_length).map_err(to_io_error)?;
140        if true {
    if !(self.read_index <= self.blob_size) {
        ::core::panicking::panic("assertion failed: self.read_index <= self.blob_size")
    };
};debug_assert!(self.read_index <= self.blob_size);
141
142        usize::try_from(read_length).map_err(to_io_error)
143    }
144}

146#[cfg(feature = "std")]
147impl std::io::Seek for SqliteReadOnlyBlob<'_> {
148    fn seek(&mut self, pos: std::io::SeekFrom) -> std::io::Result<u64> {
149        match pos {
150            std::io::SeekFrom::Start(n) => {
151                self.read_index = usize::try_from(n).map_err(to_io_error)?.min(self.blob_size);
152            }
153            std::io::SeekFrom::End(n) => {
154                self.read_index = if n.is_positive() {
155                    self.blob_size
156                } else {
157                    self.blob_size
158                        .checked_sub(usize::try_from(n.unsigned_abs()).map_err(to_io_error)?)
159                        .ok_or(std::io::ErrorKind::InvalidInput)?
160                };
161            }
162            std::io::SeekFrom::Current(n) => {
163                let n = isize::try_from(n).map_err(to_io_error)?;
164
165                if n.is_negative() {
166                    self.read_index = self
167                        .read_index
168                        .checked_sub(n.unsigned_abs())
169                        .ok_or(std::io::ErrorKind::InvalidInput)?;
170                } else {
171                    self.read_index = (self.read_index + n.unsigned_abs()).min(self.blob_size);
172                }
173            }
174        }
175
176        u64::try_from(self.read_index).map_err(to_io_error)
177    }
