53impl SqliteConnection {
54    /// Registers a callback fired when SQLite encounters an unknown collation.
55    ///
56    /// The callback receives a borrowed `&mut SqliteConnection` and a
57    /// [`CollationNeededContext`] naming the missing collation. It should
58    /// install the missing collation via
59    /// [`register_collation`](Self::register_collation) and return, after
60    /// which SQLite retries the lookup. The callback body may also execute
61    /// arbitrary SQL through `conn`, provided it leaves no open transaction.
62    ///
63    /// Only one callback is active at a time, and re-registering replaces it.
64    /// Panics in the callback abort the process.
65    ///
66    /// If the callback registers a collation that is itself missing, SQLite
67    /// re-enters the callback. Guard against unbounded recursion in that case.
68    ///
69    /// Added in SQLite 3.0.0.
70    ///
71    /// See: [`sqlite3_collation_needed`](https://www.sqlite.org/c3ref/collation_needed.html)
72    ///
73    /// # Example
74    ///
75    /// ```rust
76    /// # use diesel::prelude::*;
77    /// # use diesel::sqlite::SqliteConnection;
78    /// # let conn = &mut SqliteConnection::establish(":memory:").unwrap();
79    /// conn.on_collation_needed(|conn, ctx| {
80    ///     if ctx.name.eq_ignore_ascii_case("RUSTNOCASE") {
81    ///         let _ = conn.register_collation("RUSTNOCASE", |a, b| {
82    ///             a.to_lowercase().cmp(&b.to_lowercase())
83    ///         });
84    ///     }
85    /// });
86    ///
87    /// // Later: remove the callback
88    /// conn.remove_collation_needed_hook();
89    /// ```
90    pub fn on_collation_needed<F>(&mut self, hook: F)
91    where
92        F: Fn(&mut SqliteConnection, CollationNeededContext<'_>) + Send + 'static,
93    {
94        self.raw_connection.set_collation_needed_hook(hook);
95    }
96
97    /// Removes the collation-needed callback.
98    ///
99    /// See [`on_collation_needed`](Self::on_collation_needed) for usage
100    /// example.
101    pub fn remove_collation_needed_hook(&mut self) {
102        self.raw_connection.remove_collation_needed_hook();
103    }
104}

152    #[diesel_test_helper::test]
153    fn replacing_collation_needed_hook_drops_old() {
154        use std::sync::atomic::AtomicBool;
155
156        let conn = &mut connection();
157
158        let first_fired = Arc::new(AtomicBool::new(false));
159        let first_fired2 = first_fired.clone();
160        conn.on_collation_needed(move |_conn, _ctx| {
161            first_fired2.store(true, Ordering::Relaxed);
162        });
163
164        let second_fired = Arc::new(AtomicBool::new(false));
165        let second_fired2 = second_fired.clone();
166        conn.on_collation_needed(move |conn, ctx| {
167            conn.register_collation(ctx.name, |a, b| a.cmp(b)).unwrap();
168            second_fired2.store(true, Ordering::Relaxed);
169        });
170
171        // `CREATE INDEX ... COLLATE FOO` forces SQLite to resolve FOO. A bare
172        // `SELECT ... COLLATE FOO` does not.
173        crate::sql_query("CREATE TABLE t_replace (x TEXT)")
174            .execute(conn)
175            .unwrap();
176        crate::sql_query("CREATE INDEX i_replace ON t_replace (x COLLATE REPLACE_ME_COLL)")
177            .execute(conn)
178            .unwrap();
179
180        assert!(
181            !first_fired.load(Ordering::Relaxed),
182            "the replaced hook must not have been invoked"
183        );
184        assert!(
185            second_fired.load(Ordering::Relaxed),
186            "the current hook should have fired"
187        );

191    fn collation_needed_fires_and_registers_collation() {
192        use crate::sqlite::SqliteTextRep;
193        use std::sync::atomic::AtomicBool;
194
195        let conn = &mut connection();
196
197        // The exact name is verified indirectly by `register_collation(ctx.name, ...)`.
198        // If the callback saw the wrong name, the retry for MYCOLL would still fail.
199        let fired = Arc::new(AtomicBool::new(false));
200        let saw_name = Arc::new(AtomicBool::new(false));
201        let saw_utf8 = Arc::new(AtomicBool::new(false));
202        let fired2 = fired.clone();
203        let saw_name2 = saw_name.clone();
204        let saw_utf8_2 = saw_utf8.clone();
205
206        conn.on_collation_needed(move |conn, ctx| {
207            fired2.store(true, Ordering::Relaxed);
208            if ctx.name == "MYCOLL" {
209                saw_name2.store(true, Ordering::Relaxed);
210            }
211            if ctx.text_rep == SqliteTextRep::Utf8 {
212                saw_utf8_2.store(true, Ordering::Relaxed);
213            }
214            conn.register_collation(ctx.name, |a, b| a.cmp(b)).unwrap();
215        });
216
217        // See `replacing_collation_needed_hook_drops_old` for why CREATE INDEX.
218        crate::sql_query("CREATE TABLE t_fires (x TEXT)")
219            .execute(conn)
220            .unwrap();
221        crate::sql_query("CREATE INDEX i_fires ON t_fires (x COLLATE MYCOLL)")
222            .execute(conn)
223            .unwrap();
224
225        assert!(
226            fired.load(Ordering::Relaxed),
227            "collation_needed callback should fire"
228        );
229        assert!(
230            saw_name.load(Ordering::Relaxed),
231            "callback should observe the exact missing collation name"

233        assert!(

247    #[diesel_test_helper::test]
248    fn callback_can_be_reentered_from_within_its_own_body() {
249        use std::sync::atomic::AtomicBool;
250
251        let conn = &mut connection();
252
253        // Both flips must land: the outer callback fires for OUTER_COLL, and
254        // then the SQL it executes internally triggers a second callback for
255        // INNER_COLL while the outer callback frame is still on the stack.
256        // This is the scenario the `Fn` (not `FnMut`) bound is designed to
257        // support.
258        let saw_outer = Arc::new(AtomicBool::new(false));
259        let saw_inner = Arc::new(AtomicBool::new(false));
260        let saw_outer2 = saw_outer.clone();
261        let saw_inner2 = saw_inner.clone();
262
263        conn.on_collation_needed(move |conn, ctx| {
264            if ctx.name.eq_ignore_ascii_case("OUTER_COLL") {
265                saw_outer2.store(true, Ordering::Relaxed);
266                conn.register_collation("OUTER_COLL", |a, b| a.cmp(b))
267                    .unwrap();
268                // From inside our own frame, drive SQL that needs INNER_COLL,
269                // which is also unregistered. SQLite must be able to call the
270                // trampoline re-entrantly to satisfy this.
271                crate::sql_query("CREATE TABLE t_reent_inner (x TEXT)")
272                    .execute(conn)
273                    .unwrap();
274                crate::sql_query(
275                    "CREATE INDEX i_reent_inner ON t_reent_inner (x COLLATE INNER_COLL)",
276                )
277                .execute(conn)
278                .unwrap();
279            } else if ctx.name.eq_ignore_ascii_case("INNER_COLL") {
280                saw_inner2.store(true, Ordering::Relaxed);
281                conn.register_collation("INNER_COLL", |a, b| a.cmp(b))
282                    .unwrap();
283            } else {
284                panic!("unexpected collation name: {}", ctx.name);
285            }
286        });
287
288        crate::sql_query("CREATE TABLE t_reent_outer (x TEXT)")
289            .execute(conn)
290            .unwrap();
291        crate::sql_query("CREATE INDEX i_reent_outer ON t_reent_outer (x COLLATE OUTER_COLL)")
292            .execute(conn)
293            .unwrap();
294
295        assert!(
296            saw_outer.load(Ordering::Relaxed),
297            "outer callback should fire for OUTER_COLL"
298        );
299        assert!(
300            saw_inner.load(Ordering::Relaxed),
301            "inner callback should fire re-entrantly from within the outer body"
302        );
303    }

305    #[diesel_test_helper::test]
306    fn remove_collation_needed_hook_stops_future_callbacks() {
307        let conn = &mut connection();
308        let calls: Arc<AtomicU32> = Arc::new(AtomicU32::new(0));
309        let calls2 = calls.clone();
310
311        conn.on_collation_needed(move |conn, ctx| {
312            calls2.fetch_add(1, Ordering::Relaxed);
313            conn.register_collation(ctx.name, |a, b| a.cmp(b)).unwrap();
314        });
315
316        // First trigger: callback fires and installs MYCOLL_STOP.
317        crate::sql_query("CREATE TABLE t_stop (x TEXT)")
318            .execute(conn)
319            .unwrap();
320        crate::sql_query("CREATE INDEX i_stop ON t_stop (x COLLATE MYCOLL_STOP)")
321            .execute(conn)
322            .unwrap();
323        let after_first = calls.load(Ordering::Relaxed);
324        assert!(after_first > 0, "callback should fire while registered");
325
326        conn.remove_collation_needed_hook();
327
328        // Second trigger with a fresh unregistered collation: SQL must fail
329        // (nothing left to install YOURCOLL_STOP) and the counter must not
330        // move. A regression that only drops the Rust box while leaving the
331        // C-side pointer registered would call into freed memory here.
332        let result = crate::sql_query("CREATE INDEX i_stop2 ON t_stop (x COLLATE YOURCOLL_STOP)")
333            .execute(conn);
334        assert!(
335            result.is_err(),
336            "SQL referencing an unregistered collation should fail after remove"
337        );
338        assert_eq!(
339            calls.load(Ordering::Relaxed),
340            after_first,
341            "callback must not fire after remove_collation_needed_hook"
342        );
343    }
