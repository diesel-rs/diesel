54impl SqliteConnection {
55    /// Registers a callback fired when SQLite encounters an unknown collation.
56    ///
57    /// The callback receives a [`CollationNeededContext`] naming the missing
58    /// collation and returns its comparator, which Diesel registers under that
59    /// name before SQLite retries the lookup. Returning `None` leaves the
60    /// collation missing, so the statement fails.
61    ///
62    /// Only one callback is active at a time, and re-registering replaces it.
63    /// Panics in the callback abort the process.
64    ///
65    /// Added in SQLite 3.0.0.
66    ///
67    /// See: [`sqlite3_collation_needed`](https://www.sqlite.org/c3ref/collation_needed.html)
68    ///
69    /// # Example
70    ///
71    /// ```rust
72    /// # use diesel::prelude::*;
73    /// # use diesel::sqlite::SqliteConnection;
74    /// # let conn = &mut SqliteConnection::establish(":memory:").unwrap();
75    /// conn.on_collation_needed(|ctx| {
76    ///     if ctx.name.eq_ignore_ascii_case("RUSTNOCASE") {
77    ///         Some(Box::new(|a: &str, b: &str| {
78    ///             a.to_lowercase().cmp(&b.to_lowercase())
79    ///         }))
80    ///     } else {
81    ///         None
82    ///     }
83    /// });
84    ///
85    /// // Later: remove the callback
86    /// conn.remove_collation_needed_hook();
87    /// ```
88    pub fn on_collation_needed<F>(&mut self, hook: F)
89    where
90        F: Fn(
91                CollationNeededContext<'_>,
92            ) -> Option<
93                Box<dyn Fn(&str, &str) -> core::cmp::Ordering + Send + core::panic::UnwindSafe>,
94            > + Send
95            + 'static,
96    {
97        self.raw_connection.set_collation_needed_hook(hook);
98    }
99
100    /// Removes the collation-needed callback.
101    ///
102    /// See [`on_collation_needed`](Self::on_collation_needed) for usage
103    /// example.
104    pub fn remove_collation_needed_hook(&mut self) {
105        self.raw_connection.remove_collation_needed_hook();
106    }
107}

161    fn replacing_collation_needed_hook_drops_old() {
162        use std::sync::atomic::AtomicBool;
163
164        let conn = &mut connection();
165
166        let first_fired = Arc::new(AtomicBool::new(false));
167        let first_fired2 = first_fired.clone();
168        conn.on_collation_needed(move |_ctx| {
169            first_fired2.store(true, Ordering::Relaxed);
170            None
171        });
172
173        let second_fired = Arc::new(AtomicBool::new(false));
174        let second_fired2 = second_fired.clone();
175        conn.on_collation_needed(move |_ctx| {
176            second_fired2.store(true, Ordering::Relaxed);
177            binary_order()
178        });
179
180        // `CREATE INDEX ... COLLATE FOO` forces SQLite to resolve FOO. A bare
181        // `SELECT ... COLLATE FOO` does not.
182        crate::sql_query("CREATE TABLE t_replace (x TEXT)")
183            .execute(conn)
184            .unwrap();
185        crate::sql_query("CREATE INDEX i_replace ON t_replace (x COLLATE REPLACE_ME_COLL)")
186            .execute(conn)
187            .unwrap();
188
189        assert!(
190            !first_fired.load(Ordering::Relaxed),
191            "the replaced hook must not have been invoked"
192        );
193        assert!(
194            second_fired.load(Ordering::Relaxed),
195            "the current hook should have fired"
196        );
197    }

199    #[diesel_test_helper::test]
200    fn collation_needed_fires_and_registers_collation() {
201        use crate::sqlite::SqliteTextRep;
202        use std::sync::atomic::AtomicBool;
203
204        let conn = &mut connection();
205
206        // Diesel registers the comparator under `ctx.name`, so a wrong name would still fail the retry for MYCOLL.
207        let fired = Arc::new(AtomicBool::new(false));
208        let saw_name = Arc::new(AtomicBool::new(false));
209        let saw_utf8 = Arc::new(AtomicBool::new(false));
210        let fired2 = fired.clone();
211        let saw_name2 = saw_name.clone();
212        let saw_utf8_2 = saw_utf8.clone();
213
214        conn.on_collation_needed(move |ctx| {
215            fired2.store(true, Ordering::Relaxed);
216            if ctx.name == "MYCOLL" {
217                saw_name2.store(true, Ordering::Relaxed);
218            }
219            if ctx.text_rep == SqliteTextRep::Utf8 {
220                saw_utf8_2.store(true, Ordering::Relaxed);
221            }
222            binary_order()
223        });
224
225        // See `replacing_collation_needed_hook_drops_old` for why CREATE INDEX.
226        crate::sql_query("CREATE TABLE t_fires (x TEXT)")
227            .execute(conn)
228            .unwrap();
229        crate::sql_query("CREATE INDEX i_fires ON t_fires (x COLLATE MYCOLL)")
230            .execute(conn)
231            .unwrap();
232
233        assert!(
234            fired.load(Ordering::Relaxed),
235            "collation_needed callback should fire"
236        );
237        assert!(
238            saw_name.load(Ordering::Relaxed),
239            "callback should observe the exact missing collation name"
240        );
241        assert!(
242            saw_utf8.load(Ordering::Relaxed),
243            "SQLite should request the UTF-8 encoding for a plain TEXT column"

255    #[diesel_test_helper::test]
256    fn returned_comparator_serves_the_requested_collation_and_none_leaves_it_missing() {
257        use crate::collation::Custom;
258        use crate::prelude::*;
259
260        crate::table! {
261            words (id) {
262                id -> Integer,
263                word -> Text,
264            }
265        }
266
267        let conn = &mut connection();
268        let calls = Arc::new(AtomicU32::new(0));
269        let calls2 = calls.clone();
270        conn.on_collation_needed(move |ctx| {
271            calls2.fetch_add(1, Ordering::Relaxed);
272            (ctx.name == "MYNOCASE").then(|| {
273                Box::new(|a: &str, b: &str| a.to_lowercase().cmp(&b.to_lowercase()))
274                    as Box<
275                        dyn Fn(&str, &str) -> core::cmp::Ordering + Send + core::panic::UnwindSafe,
276                    >
277            })
278        });
279        crate::sql_query("CREATE TABLE words (id INTEGER PRIMARY KEY, word TEXT NOT NULL)")
280            .execute(conn)
281            .unwrap();
282        crate::insert_into(words::table)
283            .values([words::word.eq("a"), words::word.eq("b")])
284            .execute(conn)
285            .unwrap();
286
287        let matching = |collation| {
288            words::table
289                .select(words::word)
290                .filter(words::word.collate(Custom(collation)).eq("A"))
291        };
292        assert_eq!(matching("MYNOCASE").load::<String>(conn).unwrap(), ["a"]);
293        assert_eq!(matching("MYNOCASE").load::<String>(conn).unwrap(), ["a"]);
294        assert_eq!(calls.load(Ordering::Relaxed), 1);
295        assert!(matching("UNKNOWN").load::<String>(conn).is_err());
296    }

298    #[diesel_test_helper::test]
299    fn remove_collation_needed_hook_stops_future_callbacks() {
300        let conn = &mut connection();
301        let calls: Arc<AtomicU32> = Arc::new(AtomicU32::new(0));
302        let calls2 = calls.clone();
303
304        conn.on_collation_needed(move |_ctx| {
305            calls2.fetch_add(1, Ordering::Relaxed);
306            binary_order()
307        });
308
309        // First trigger: callback fires and installs MYCOLL_STOP.
310        crate::sql_query("CREATE TABLE t_stop (x TEXT)")
311            .execute(conn)
312            .unwrap();
313        crate::sql_query("CREATE INDEX i_stop ON t_stop (x COLLATE MYCOLL_STOP)")
314            .execute(conn)
315            .unwrap();
316        let after_first = calls.load(Ordering::Relaxed);
317        assert!(after_first > 0, "callback should fire while registered");
318
319        conn.remove_collation_needed_hook();
320
321        // Second trigger with a fresh unregistered collation: SQL must fail
322        // (nothing left to install YOURCOLL_STOP) and the counter must not
323        // move. A regression that only drops the Rust box while leaving the
324        // C-side pointer registered would call into freed memory here.
325        let result = crate::sql_query("CREATE INDEX i_stop2 ON t_stop (x COLLATE YOURCOLL_STOP)")
326            .execute(conn);
327        assert!(
328            result.is_err(),
329            "SQL referencing an unregistered collation should fail after remove"
330        );
331        assert_eq!(
332            calls.load(Ordering::Relaxed),
333            after_first,
334            "callback must not fire after remove_collation_needed_hook"
335        );
336    }
