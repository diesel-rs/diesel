1//! Typed access to SQLite [pragmas](https://www.sqlite.org/pragma.html).
2//!
3//! A pragma is declared as a type implementing [`Pragma`], with [`ReadPragma`] when it returns rows and [`WritePragma`] when it can be assigned an argument. Its [`Scope`](Pragma::Scope) decides whether it is used per database or per connection, and the caller picks the Rust type each row decodes into.
4//!
5//! # Per-schema pragmas
6//!
7//! A [`PerSchema`] pragma applies to one database and is read and written through [`SchemaPragma`], where a `None` schema selects `main`.
8//!
9//! ```rust
10//! # include!("../doctest_setup.rs");
11//! use diesel::sqlite::pragma::{PerSchema, Pragma, ReadPragma, SchemaPragma, WritePragma};
12//!
13//! struct UserVersion;
14//!
15//! impl Pragma for UserVersion {
16//!     const NAME: &'static str = "user_version";
17//!     type Scope = PerSchema;
18//! }
19//!
20//! impl ReadPragma for UserVersion {
21//!     type SqlType = Integer;
22//! }
23//!
24//! impl WritePragma for UserVersion {
25//!     type Argument<'a> = i32;
26//! }
27//!
28//! # fn main() -> QueryResult<()> {
29//! let conn = &mut SqliteConnection::establish(":memory:").unwrap();
30//! UserVersion::set(7, None, conn)?;
31//! assert_eq!(7, UserVersion::get::<i32>(None, conn)?);
32//! #     Ok(())
33//! # }
34//! ```
35//!
36//! # Per-connection pragmas
37//!
38//! A [`PerConnection`] pragma applies to the whole connection and is read and written through [`ConnectionPragma`], which takes no schema.
39//!
40//! ```rust
41//! # include!("../doctest_setup.rs");
42//! use diesel::sqlite::pragma::{
43//!     ConnectionPragma, PerConnection, Pragma, ReadPragma, WritePragma,
44//! };
45//!
46//! struct ForeignKeys;
47//!
48//! impl Pragma for ForeignKeys {
49//!     const NAME: &'static str = "foreign_keys";
50//!     type Scope = PerConnection;
51//! }
52//!
53//! impl ReadPragma for ForeignKeys {
54//!     type SqlType = Bool;
55//! }
56//!
57//! impl WritePragma for ForeignKeys {
58//!     type Argument<'a> = bool;
59//! }
60//!
61//! # fn main() -> QueryResult<()> {
62//! let conn = &mut SqliteConnection::establish(":memory:").unwrap();
63//! ForeignKeys::set(true, conn)?;
64//! assert!(ForeignKeys::get::<bool>(conn)?);
65//! #     Ok(())
66//! # }
67//! ```

81/// A pragma, named by [`NAME`](Self::NAME) and scoped by [`Scope`](Self::Scope).
82pub trait Pragma: Sized {
83    /// The pragma name, written into the SQL unquoted.
84    ///
85    /// A name that is empty or holds anything but ASCII letters, digits and underscores fails with [`Error::QueryBuilderError`] before reaching SQLite. Naming a pragma SQLite knows is up to the implementer, since SQLite ignores unknown ones.
86    const NAME: &'static str;
87    /// [`PerSchema`] or [`PerConnection`].
88    type Scope: PragmaScope;
89}

284/// Reads and writes a [`PerSchema`] pragma, where a `None` schema selects `main`.
285pub trait SchemaPragma: Pragma<Scope = PerSchema> {
286    /// Read the first row, returning [`Error::NotFound`] if the pragma returns none.
287    fn get<Row>(schema: Option<&str>, conn: &mut SqliteConnection) -> QueryResult<Row>
288    where
289        Self: ReadPragma,
290        Sqlite: QueryMetadata<Self::SqlType>,
291        Row: FromSqlRow<Self::SqlType, Sqlite>,
292    {
293        first_row::<Self, Row>(conn, Some(schema.unwrap_or("main")))
294    }
295
296    /// Read every row.
297    fn get_all<Row>(schema: Option<&str>, conn: &mut SqliteConnection) -> QueryResult<Vec<Row>>
298    where
299        Self: ReadPragma,
300        Sqlite: QueryMetadata<Self::SqlType>,
301        Row: FromSqlRow<Self::SqlType, Sqlite>,
302    {
303        all_rows::<Self, Row>(conn, Some(schema.unwrap_or("main")))
304    }
305
306    /// Assign `argument`, running the statement to completion without confirming the value SQLite applied.
307    fn set<'a>(
308        argument: impl Into<Self::Argument<'a>>,
309        schema: Option<&str>,
310        conn: &mut SqliteConnection,
311    ) -> QueryResult<()>
312    where
313        Self: WritePragma,
314    {
315        assign::<Self>(conn, Some(schema.unwrap_or("main")), argument.into())
316    }
317}

532fn assign<P: WritePragma>(
533    conn: &mut SqliteConnection,
534    schema: Option<&str>,
535    argument: P::Argument<'_>,
536) -> QueryResult<()> {
537    use crate::connection::SimpleConnection;
538    use crate::query_builder::QueryBuilder;
539    use crate::sqlite::SqliteQueryBuilder;
540
541    // Some pragmas do one unit of work per step, `incremental_vacuum` frees one page per step.
542    // `sqlite3_exec` behind `batch_execute` runs to completion, while `StatementUse::run` steps once.
543    let mut query = SqliteQueryBuilder::new();

546        argument,

551}

600fn push_pragma<P: Pragma>(
601    out: &mut AstPass<'_, '_, Sqlite>,
602    schema: Option<&str>,
603) -> QueryResult<()> {
604    out.push_sql("PRAGMA ");
605    if let Some(schema) = schema {
606        out.push_identifier(schema)?;
607        out.push_sql(".");
608    }
609    push_bare_word(out, "name", P::NAME)
610}

612fn push_bare_word(
613    out: &mut AstPass<'_, '_, Sqlite>,
614    kind: &str,
615    word: &'static str,
616) -> QueryResult<()> {
617    if word.is_empty()
618        || !word
619            .bytes()
620            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'_')
621    {
622        return Err(Error::QueryBuilderError(
623            ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("`{0}` is not a valid pragma {1}",
                word, kind))
    })alloc::format!("`{word}` is not a valid pragma {kind}").into(),
624        ));
625    }
626    out.push_sql(word);
627    Ok(())
628}
