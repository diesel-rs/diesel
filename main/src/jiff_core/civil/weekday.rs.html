124    /// Add the given number of days to this weekday, using wrapping arithmetic,
125    /// and return the resulting weekday.
126    ///
127    /// Adding a multiple of `7` (including `0`) is guaranteed to produce the
128    /// same weekday as this one.
129    #[inline]
130    pub const fn wrapping_add(self, days: i64) -> Weekday {
131        let start = self.to_monday_zero_offset() as i64;
132        // We are careful to `rem_euclid` on `rhs` before doing
133        // wrapping arithmetic, otherwise the result is not
134        // correct. Namely, it would assume that, e.g., since
135        // `i64::MAX.rem_euclid(7)` is 0, then the next value would be
136        // `rem_euclid(7) == 1`. But `i64::MIN.rem_euclid(7)` is 6.
137        let end = (start.wrapping_add(days.rem_euclid(7)) % 7) as i8;
138        // Always valid because of the mod 7 above.
139        match Weekday::from_monday_zero_offset(end) {
    Ok(val) => val,
    Err(_) => {
        ::core::panicking::panic_fmt(format_args!("weekday is always 0..=6"));
    }
}unwrapr!(
140            Weekday::from_monday_zero_offset(end),
141            "weekday is always 0..=6",
142        )
143    }

145    /// Subtract the given number of days from this weekday, using wrapping
146    /// arithmetic, and return the resulting weekday.
147    ///
148    /// Subtracting a multiple of `7` (including `0`) is guaranteed to produce
149    /// the same weekday as this one.
150    #[inline]
151    pub const fn wrapping_sub(self, days: i64) -> Weekday {
152        // i64::MIN.rem_euclid(7) == 6
153        let days = match days.checked_neg() {
154            Some(days) => days,
155            None => -6,
156        };
157        self.wrapping_add(days)
158    }
