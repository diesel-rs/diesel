32#[repr(align(64))] // Align to cache lines
33struct Utf8Data {
34    table: [u8; 384],
35}
36
37impl Utf8Data {
38    /// The table lookup for backing prefix checks.
39    ///
40    /// # Safety-usable invariant
41    ///
42    /// The return value is 0 iff `first` and `second` form a prefix of a
43    /// well-formed multi-byte UTF-8 sequence (including a complete two-byte
44    /// sequence).
45    /// The lowest two bits are always 0.
46    #[inline(always)]
47    const fn lookup(&self, first: u8, second: u8) -> u8 {
48        self.table[second as usize] & self.table[first as usize + 0x80]
49    }
50}
51
52/// The first 256 entries are for looking up `second`.
53/// The rest are for looking up `first` on the assumption
54/// that `first` is not ASCII.
55///
56/// Lanes 0 and 1 are left unused to allow the tag
57/// bits of the third byte in a three-byte sequence
58/// to be shifted there for a single-branch comparison.
59static UTF8_DATA: Utf8Data = Utf8Data {
60    table: const {
61        let mut table = [0u8; 384];
62
63        // Trails
64        let mut i = 0u16;
65        while i < 0x100 {
66            let second = i as u8;
67            // If a `second` is valid given a lane, we leave a lane as zero.
68            // We put 1 on the lane if the `second` is invalid given a lane.
69            let mut combined = 1 << 2; // invalid lead
70            if second < 0x80 || second > 0xBF {
71                combined |= 1 << 3; // normal trail
72            }
73            if second < 0xA0 || second > 0xBF {
74                combined |= 1 << 4; // three-byte special lower bound
75            }
76            if second < 0x80 || second > 0x9F {
77                combined |= 1 << 5; // three-byte special upper bound
78            }
79            if second < 0x90 || second > 0xBF {
80                combined |= 1 << 6; // four-byte special lower bound
81            }
82            if second < 0x80 || second > 0x8F {
83                combined |= 1 << 7; // four-byte special upper bound
84            }
85            table[second as usize] = combined;
86            i += 1;
87        }
88
89        // Non-ASCII leads
90        i = 0x80;
91        while i < 0x100 {
92            let first = i as u8;
93            let lane = match first {
94                0xC2..=0xDF | 0xE1..=0xEC | 0xEE..=0xEF | 0xF1..=0xF3 => {
95                    1 << 3 // normal trail
96                }
97                0xE0 => {
98                    1 << 4 // three-byte special lower bound
99                }
100                0xED => {
101                    1 << 5 // three-byte special upper bound
102                }
103                0xF0 => {
104                    1 << 6 // four-byte special lower bound
105                }
106                0xF4 => {
107                    1 << 7 // four-byte special upper bound
108                }
109                _ => {
110                    // invalid lead
111                    1 << 2
112                }
113            };
114            table[0x80 + first as usize] = lane;
115            i += 1;
116        }
117
118        // We didn't explicitly generate table entries that are
119        // looked up when `first` is ASCII. However, from the construction
120        // of the entries for trails, the overlapping part ends up with
121        // suitable values. Let's verify that.
122        i = 0;
123        while i < 0x80 {
124            if table[0x80 + i as usize] & (1 << 2) == 0 {
125                ::core::panicking::panic_default_msg();panic!();
126            }
127            i += 1;
128        }
129
130        table
131    },
132};

140macro_rules! byte_methods {
141    ($type_specific_ident:ident,
142     $(#[$unchecked_meta:meta])*,
143     $(#[$checked_meta:meta])*,
144    ) => {
145        /// Whether `byte` is in range for this type.
146        ///
147        /// # Safety-usable invariant
148        ///
149        /// Iff this method returns `true`, the safety invariant of `new_unchecked()`
150        /// is satisfied.
151        #[inline(always)]
152        pub const fn is(byte: u8) -> bool {
153            byte >= Self::MIN.to_u8() && byte <= Self::MAX.to_u8()
154        }
155
156        $(#[$unchecked_meta])*
157        #[inline(always)]
158        pub const unsafe fn new_unchecked($type_specific_ident: u8) -> Self {
159            debug_assert!(Self::is($type_specific_ident));
160            // SAFETY: Our `repr(u8)` matches argument `u8` size and the
161            // value range is this function's safety invariant.
162            unsafe { core::mem::transmute($type_specific_ident) }
163        }
164
165        $(#[$checked_meta])*
166        #[inline(always)]
167        pub const fn try_new(byte: u8) -> Result<Self, Utf8ByteError> {
168            if Self::is(byte) {
169                // SAFETY: We checked the safety invariant immediately above.
170                Ok(unsafe { Self::new_unchecked(byte) })
171            } else {
172                Err(Utf8ByteError)
173            }
174        }
175
176        /// Convert to plain byte
177        #[inline(always)]
178        pub const fn to_u8(self) -> u8 {
179            self as u8
180        }
181    };

287macro_rules! sequence_methods {
288    ($(#[$as_bytes_meta:meta])*,
289     $(#[$as_str_meta:meta])*,
290    ) => {
291        $(#[$as_bytes_meta])*
292        #[inline(always)]
293        pub const fn as_bytes(&self) -> &[u8; Self::LEN] {
294            if core::mem::size_of::<Self>() != Self::LEN {
295                panic!();
296            }
297            let ptr: *const Self = core::ptr::from_ref(self);
298            // SAFETY: The `repr` of `Self` guarantees that it
299            // is safe to view `Self` as a sequence of `u8`, that
300            // the `u8`s are in the right order, and that the
301            // size of `Self` matches the length in the return type.
302            unsafe {
303                &*ptr.cast()
304            }
305        }
306
307        $(#[$as_str_meta])*
308        #[inline(always)]
309        pub const fn as_str(&self) -> &str {
310            // SAFETY: The safety-usable invariant of `as_bytes()` satisfies
311            // the safety invariant of `from_utf8_unchecked()`.
312            unsafe { core::str::from_utf8_unchecked(self.as_bytes()) }
313        }
314    };

1713        // The top five bits of `first` have to be 11110 for `first` to
1714        // be a valid four-byte lead byte.
1715        // From the field invariant, `check` must be 0b10 for `second`
1716        // and `third` to be contextually valid continuations assuming that
1717        // `first` is a four-byte lead byte.
1718        // We shift `check` left to use the structure of the full
1719        // four-byte check.
1720        (((self.check as u16) << 8) | ((self.first & 0b1111_1000) as u16)) == 0b10_1111_0000
