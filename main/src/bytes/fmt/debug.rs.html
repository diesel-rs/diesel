6/// Alternative implementation of `std::fmt::Debug` for byte slice.
7///
8/// Standard `Debug` implementation for `[u8]` is comma separated
9/// list of numbers. Since large amount of byte strings are in fact
10/// ASCII strings or contain a lot of ASCII strings (e. g. HTTP),
11/// it is convenient to print strings as ASCII when possible.
12impl Debug for BytesRef<'_> {
13    fn fmt(&self, f: &mut Formatter<'_>) -> Result {
14        f.write_fmt(format_args!("b\""))write!(f, "b\"")?;
15        for &b in self.0 {
16            // https://doc.rust-lang.org/reference/tokens.html#byte-escapes
17            if b == b'\n' {
18                f.write_fmt(format_args!("\\n"))write!(f, "\\n")?;
19            } else if b == b'\r' {
20                f.write_fmt(format_args!("\\r"))write!(f, "\\r")?;
21            } else if b == b'\t' {
22                f.write_fmt(format_args!("\\t"))write!(f, "\\t")?;
23            } else if b == b'\\' || b == b'"' {
24                f.write_fmt(format_args!("\\{0}", b as char))write!(f, "\\{}", b as char)?;
25            } else if b == b'\0' {
26                f.write_fmt(format_args!("\\0"))write!(f, "\\0")?;
27            // ASCII printable
28            } else if (0x20..0x7f).contains(&b) {
29                f.write_fmt(format_args!("{0}", b as char))write!(f, "{}", b as char)?;
30            } else {
31                f.write_fmt(format_args!("\\x{0:02x}", b))write!(f, "\\x{:02x}", b)?;
32            }
33        }
34        f.write_fmt(format_args!("\""))write!(f, "\"")?;
35        Ok(())
36    }
37}

39impl Debug for Bytes {
    fn fmt(&self, f: &mut Formatter<'_>) -> Result {
        Debug::fmt(&BytesRef(self.as_ref()), f)
    }
}fmt_impl!(Debug, Bytes);
