12fn copy(into: &mut [u8], from: &[u8], pos: &mut usize) -> usize {
13    let min = cmp::min(into.len(), from.len() - *pos);
14    into[..min].copy_from_slice(&from[*pos..*pos + min]);
15    *pos += min;
16    min
17}

67impl<R: BufRead> GzEncoder<R> {
68    /// Creates a new encoder which will use the given compression level.
69    ///
70    /// The encoder is not configured specially for the emitted header. For
71    /// header configuration, see the `GzBuilder` type.
72    ///
73    /// The data read from the stream `r` will be compressed and available
74    /// through the returned reader.
75    pub fn new(r: R, level: Compression) -> GzEncoder<R> {
76        GzBuilder::new().buf_read(r, level)
77    }
78
79    fn read_footer(&mut self, into: &mut [u8]) -> io::Result<usize> {
80        if self.pos == 8 {
81            return Ok(0);
82        }
83        let crc = self.inner.get_ref().crc();
84        let calced_crc_bytes = crc.sum().to_le_bytes();
85        let arr = [
86            calced_crc_bytes[0],
87            calced_crc_bytes[1],
88            calced_crc_bytes[2],
89            calced_crc_bytes[3],
90            crc.amount() as u8,
91            (crc.amount() >> 8) as u8,
92            (crc.amount() >> 16) as u8,
93            (crc.amount() >> 24) as u8,
94        ];
95        Ok(copy(into, &arr, &mut self.pos))
96    }
97}

122#[inline]
123fn finish(buf: &[u8; 8]) -> (u32, u32) {
124    let crc = (buf[0] as u32)
125        | ((buf[1] as u32) << 8)
126        | ((buf[2] as u32) << 16)
127        | ((buf[3] as u32) << 24);
128    let amt = (buf[4] as u32)
129        | ((buf[5] as u32) << 8)
130        | ((buf[6] as u32) << 16)
131        | ((buf[7] as u32) << 24);
132    (crc, amt)
133}

135impl<R: BufRead> Read for GzEncoder<R> {
136    fn read(&mut self, into: &mut [u8]) -> io::Result<usize> {
137        if into.is_empty() {
138            return Ok(0);
139        }
140        if self.eof {
141            return self.read_footer(into);
142        } else if self.pos < self.header.len() {
143            // Report header progress before attempting a fallible input read.
144            return Ok(copy(into, &self.header, &mut self.pos));
145        }
146        match self.inner.read(into) {
147            Ok(0) => {
148                self.eof = true;
149                self.pos = 0;
150                self.read_footer(into)
151            }
152            res => res,
153        }
154    }
155}

233impl<R: BufRead> GzDecoder<R> {
234    /// Creates a new decoder from the given reader, immediately parsing the
235    /// gzip header.
236    pub fn new(mut r: R) -> GzDecoder<R> {
237        let mut header_parser = GzHeaderParser::new();
238
239        let state = match header_parser.parse(&mut r) {
240            Ok(_) => GzState::Body(GzHeader::from(header_parser)),
241            Err(ref err) if io::ErrorKind::WouldBlock == err.kind() => {
242                GzState::Header(header_parser)
243            }
244            Err(err) => GzState::Err(err),
245        };
246
247        GzDecoder {
248            state,
249            reader: CrcReader::new(deflate::bufread::DeflateDecoder::new(r)),
250            multi: false,
251        }
252    }
253
254    fn multi(mut self, flag: bool) -> GzDecoder<R> {
255        self.multi = flag;
256        self
257    }
258}

273    }

276    ///

278    /// current position are preserved for subsequent reads by this decoder.
279    ///
280    /// To process a new stream, wait for this decoder to reach EOF and use
281    /// [`reset`](Self::reset); replacing the reader directly does not reset it.
282    pub fn get_mut(&mut self) -> &mut R {
283        self.reader.get_mut().get_mut()
284    }

288        self.reader.into_inner().into_inner()
289    }

291    /// Resets the state of this decoder entirely, swapping out the input
292    /// stream for another.
293    ///
294    /// This will reset the internal state of this decoder and replace the
295    /// input stream with the one provided, returning the previous input
296    /// stream. Future data read from this decoder will be the decompressed

304impl<R: BufRead> Read for GzDecoder<R> {
305    fn read(&mut self, into: &mut [u8]) -> io::Result<usize> {
306        loop {
307            match &mut self.state {
308                GzState::Header(parser) => {
309                    parser.parse(self.reader.get_mut().get_mut())?;
310                    self.state = GzState::Body(GzHeader::from(mem::take(parser)));
311                }
312                GzState::Body(header) => {
313                    if into.is_empty() {
314                        return Ok(0);
315                    }
316                    match self.reader.read(into)? {
317                        0 => {
318                            self.state = GzState::Finished(mem::take(header), 0, [0; 8]);
319                        }
320                        n => {
321                            return Ok(n);
322                        }
323                    }
324                }
325                GzState::Finished(header, pos, buf) => {
326                    if *pos < buf.len() {
327                        *pos += read_into(self.reader.get_mut().get_mut(), &mut buf[*pos..])?;
328                    } else {
329                        let (crc, amt) = finish(buf);
330
331                        if crc != self.reader.crc().sum() || amt != self.reader.crc().amount() {
332                            self.state = GzState::End(Some(mem::take(header)));
333                            return Err(corrupt());
334                        } else if self.multi {
335                            let is_eof = self
336                                .reader
337                                .get_mut()
338                                .get_mut()
339                                .fill_buf()
340                                .map(|buf| buf.is_empty())?;
341
342                            if is_eof {
343                                self.state = GzState::End(Some(mem::take(header)));
344                            } else {
345                                self.reader.reset();
346                                self.reader.get_mut().reset_data();
347                                self.state = GzState::Header(GzHeaderParser::new())
348                            }
349                        } else {
350                            self.state = GzState::End(Some(mem::take(header)));
351                        }
352                    }
353                }
354                GzState::Err(err) => {
355                    let result = Err(mem::replace(err, io::ErrorKind::Other.into()));
356                    self.state = GzState::End(None);
357                    return result;
358                }
359                GzState::End(_) => return Ok(0),
360            }
361        }
362    }
363}

475        let expected = "Hello World";

477        let compressed = {
478            let mut e = write::GzEncoder::new(Vec::new(), Compression::default());
479            e.write_all(expected.as_ref()).unwrap();
480            let mut b = e.finish().unwrap();
481            b.push(b'x');
482            b

485        let mut output = Vec::new();
486        let mut decoder = GzDecoder::new(compressed.as_slice());
487        let decoded_bytes = decoder.read_to_end(&mut output).unwrap();
488        assert_eq!(decoded_bytes, output.len());
489        let actual = core::str::from_utf8(&output).expect("String parsing error");
490        assert_eq!(
491            actual, expected,

496        assert_eq!(
497            decoder.read(&mut output).unwrap(),

502        assert_eq!(
503            reader.read_to_end(&mut output).unwrap(),
504            1,
505            "extra data is accessible in underlying buf-read"

507        assert_eq!(output, b"x");
