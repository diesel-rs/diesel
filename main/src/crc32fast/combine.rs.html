10// Calculates a(x) multiplied by b(x) modulo p(x), where p(x) is the CRC polynomial,
11// reflected. For speed, this requires that a not be zero.
12const fn multiply(a: u32, mut b: u32) -> u32 {
13    let mut p = 0u32;
14
15    let mut i = 0;
16    while i < 32 {
17        p ^= b & ((a >> (31 - i)) & 1).wrapping_neg();
18        b = (b >> 1) ^ ((b & 1).wrapping_neg() & POLY);
19        i += 1;
20    }
21
22    p
23}

25/// Returns x^(8 * len) modulo p(x), reflected. Multiplying a CRC state by it appends `len` zero
26/// bytes.
27const fn x8n(len: u64) -> u32 {
28    let mut p = 0x8000_0000; // x^0
29    let mut sq = 0x0080_0000; // x^8
30    let mut n = len;
31    while n != 0 {
32        if n & 1 != 0 {
33            p = multiply(sq, p);
34        }
35        sq = multiply(sq, sq);
36        n >>= 1;
37    }
38    p
39}

41/// The constants that fold a 128-bit accumulator forward by `bits` bits with two carry-less
42/// multiplies: `(x^(bits + 32), x^(bits - 32))` modulo p(x), reflected, as 33-bit values (the
43/// `K1`..`K4` of `specialized/pclmulqdq.rs`, which a test checks).
44#[cfg_attr(
45    not(all(stable_arm_crc32_intrinsics, target_arch = "aarch64")),
46    allow(dead_code)
47)]
48pub(crate) const fn fold_keys(bits: u64) -> [u64; 2] {
49    [
50        (x8n((bits + 32) / 8) as u64) << 1,
51        (x8n((bits - 32) / 8) as u64) << 1,
52    ]
53}

55/// Builds tables that append `len` zero bytes to a CRC state a byte at a time: the new state is
56/// the XOR of `table[k][(state >> (8 * k)) & 0xFF]` for `k` in `0..4`.
57#[cfg_attr(
58    not(all(stable_arm_crc32_intrinsics, target_arch = "aarch64")),
59    allow(dead_code)
60)]
61pub(crate) const fn shift_table(len: u64) -> [[u32; 256]; 4] {
62    let x = x8n(len);
63    let mut table = [[0u32; 256]; 4];
64    let mut k = 0;
65    while k < 4 {
66        let mut v = 0;
67        while v < 256 {
68            table[k][v] = multiply(x, (v as u32) << (8 * k));
69            v += 1;
70        }
71        k += 1;
72    }
73    table
74}

76/// Appends the zero bytes `table` was built for to the CRC state `state`.
77#[cfg_attr(
78    not(all(stable_arm_crc32_intrinsics, target_arch = "aarch64")),
79    allow(dead_code)
80)]
81#[inline(always)]
82pub(crate) fn shift(table: &[[u32; 256]; 4], state: u32) -> u32 {
83    table[0][(state & 0xFF) as usize]
84        ^ table[1][((state >> 8) & 0xFF) as usize]
85        ^ table[2][((state >> 16) & 0xFF) as usize]
86        ^ table[3][(state >> 24) as usize]
87}

89pub(crate) fn combine(crc1: u32, crc2: u32, len2: u64) -> u32 {
90    // Special case: If the length of the second chunk is zero, return the hash
91    // of the first chunk.
92    if len2 == 0 {
93        return crc1;
94    }
95
96    // We are padding the first checksum with len2-amount of zeroes. For efficiency,
97    // this is done in powers-of-two via a lookup table rather than one by one.
98    let mut p = crc1;
99    let n = 64 - len2.leading_zeros();
100    for i in 0..n {
101        if (len2 >> i & 1) != 0 {
102            p = multiply(X2N_TABLE[(i & 0x1F) as usize], p);
103        }
104    }
105
106    p ^ crc2
107}

110fn fold_keys_match_the_x86_constants() {
111    assert_eq!(fold_keys(128), [0x1751997d0, 0x0ccaa009e]);
112    assert_eq!(fold_keys(512), [0x154442bd4, 0x1c6e41596]);

117#[test]
118fn shift_table_appends_zeros() {
119    for &len in &[1u64, 7, 512, 32768] {
120        let table = shift_table(len);
121        let zeros = vec![0u8; len as usize];
122        for &state in &[0u32, 1, 0x8000_0000, 0xDEAD_BEEF, 0xFFFF_FFFF, 0x1234_5678] {
123            // `update_slow` takes and returns finalized CRCs, which are the inverted state.
124            let expected = !crate::baseline::update_slow(!state, &zeros);
125            assert_eq!(
126                shift(&table, state),
127                expected,
128                "len={len} state={state:#010x}"
129            );
130        }
131    }
132}
