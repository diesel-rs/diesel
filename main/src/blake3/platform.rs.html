314    pub fn xof_many(
315        &self,
316        cv: &CVWords,
317        block: &[u8; BLOCK_LEN],
318        block_len: u8,
319        mut counter: u64,
320        flags: u8,
321        out: &mut [u8],
322    ) {
323        if true {
    {
        match (&0, &(out.len() % BLOCK_LEN)) {
            (left_val, right_val) => {
                if !(*left_val == *right_val) {
                    let kind = ::core::panicking::AssertKind::Eq;
                    ::core::panicking::assert_failed(kind, &*left_val,
                        &*right_val,
                        ::core::option::Option::Some(format_args!("whole blocks only")));
                }
            }
        }
    };
};debug_assert_eq!(0, out.len() % BLOCK_LEN, "whole blocks only");
324        if out.is_empty() {
325            // The current assembly implementation always outputs at least 1 block.
326            return;
327        }
328        match self {
329            // Safe because detect() checked for platform support.
330            #[cfg(blake3_avx512_ffi)]
331            #[cfg(unix)]
332            #[cfg(any(target_arch = "x86", target_arch = "x86_64"))]
333            Platform::AVX512 => unsafe {
334                crate::avx512::xof_many(cv, block, block_len, counter, flags, out)
335            },
336            _ => {
337                // For platforms without an optimized xof_many, fall back to a loop over
338                // compress_xof. This is still faster than portable code.
339                for out_block in out.chunks_exact_mut(BLOCK_LEN) {
340                    // TODO: Use array_chunks_mut here once that's stable.
341                    let out_array: &mut [u8; BLOCK_LEN] = out_block.try_into().unwrap();
342                    *out_array = self.compress_xof(cv, block, block_len, counter, flags);
343                    counter += 1;
344                }
345            }
346        }
