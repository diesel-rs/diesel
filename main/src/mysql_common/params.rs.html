67impl Params {
68    /// Converts [`Params`] into a vector of values given the named parameters.
69    ///
70    /// `named_params` (if any) must follow the order they were given in the corresponding SQL
71    /// statement.
72    pub fn into_values(self, named_params: Option<&[Vec<u8>]>) -> Result<Vec<Value>, ParamsError> {
73        match self {
74            Params::Empty => match named_params {
75                Some(params) => {
76                    if let Some(first) = params.first() {
77                        Err(MissingNamedParameterError(first.clone()).into())
78                    } else {
79                        Ok(::alloc::vec::Vec::new()vec![])
80                    }
81                }
82                None => Ok(::alloc::vec::Vec::new()vec![]),
83            },
84            Params::Positional(values) => match named_params {
85                Some(named_params) if !named_params.is_empty() => {
86                    Err(ParamsConfusionError::PositionalParamsForNamedQuery.into())
87                }
88                _ => Ok(values),
89            },
90            Params::Named(map) => match named_params {
91                Some(named_params) if !named_params.is_empty() => {
92                    let mut values = ::alloc::vec::from_elem_global(Value::NULL, named_params.len())vec![Value::NULL; named_params.len()];
93                    let mut indexes = Vec::with_capacity(named_params.len());
94                    for (name, value) in map {
95                        let mut first = None;
96                        for (i, _) in named_params.iter().enumerate().filter(|(_, x)| **x == name) {
97                            indexes.push(i);
98                            if first.is_none() {
99                                first = Some(i);
100                            } else {
101                                values[i] = value.clone();
102                            }
103                        }
104                        if let Some(first) = first {
105                            values[first] = value;
106                        }
107                    }
108                    if indexes.len() != named_params.len() {
109                        indexes.sort_unstable();
110                        match indexes.into_iter().enumerate().find(|x| x.0 != x.1) {
111                            Some((missing, _)) => {
112                                Err(MissingNamedParameterError(named_params[missing].clone())
113                                    .into())
114                            }
115                            None => {
116                                match named_params.last() {
117                                    Some(last) => {
118                                        Err(MissingNamedParameterError(last.clone()).into())
119                                    }
120                                    None => {
121                                        // unreachable
122                                        Ok(values)
123                                    }
124                                }
125                            }
126                        }
127                    } else {
128                        Ok(values)
129                    }
130                }
131                _ => Err(ParamsConfusionError::NamedParamsForPositionalQuery.into()),
132            },
133        }
134    }
135
136    /// Will convert named parameters into positional assuming order passed in `named_params`
137    /// attribute.
138    #[deprecated = "use `into_values` instead"]
139    pub fn into_positional(
140        self,
141        named_params: &[Vec<u8>],
142    ) -> Result<Params, MissingNamedParameterError> {
143        match self {
144            Params::Named(mut map) => {
145                let mut params: Vec<Value> = Vec::new();
146                'params: for (i, name) in named_params.iter().enumerate() {
147                    match map.entry(name.clone()) {
148                        Occupied(entry) => {
149                            let mut x = named_params.len() - 1;
150                            while x > i {
151                                if *name == named_params[x] {
152                                    params.push(entry.get().clone());
153                                    continue 'params;
154                                }
155                                x -= 1;
156                            }
157                            params.push(entry.remove());
158                        }
159                        _ => return Err(MissingNamedParameterError(name.clone())),
160                    }
161                }
162                Ok(Params::Positional(params))
163            }
164            params => Ok(params),
165        }
166    }
167}
