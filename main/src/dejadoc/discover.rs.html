29/// Resolve the workspace containing `root` and the targets to scan.
30///
31/// # Errors
32///
33/// Fails when `cargo metadata` cannot be run for `root`.
34pub(crate) fn workspace(
35    root: &std::path::Path,
36    package: Option<&str>,
37    all_targets: bool,
38) -> crate::Result<Workspace> {
39    let metadata = cargo_metadata::MetadataCommand::new()
40        .current_dir(root)
41        .no_deps()
42        .exec()?;
43    let mut targets = Vec::new();
44    for package in metadata
45        .packages
46        .iter()
47        .filter(|p| package.is_none_or(|want| p.name == want))
48    {
49        for target in &package.targets {
50            if scan_target(&target.kind, all_targets) {
51                targets.push(Target {
52                    name: target.name.clone(),
53                    src: target.src_path.clone().into(),
54                });
55            }
56        }
57    }
58    Ok(Workspace {
59        root: metadata.workspace_root.clone().into(),
60        targets,
61    })
62}

64/// Whether a cargo target is scanned, from its kind list.
65fn scan_target(kinds: &[cargo_metadata::TargetKind], all_targets: bool) -> bool {
66    use cargo_metadata::TargetKind as CargoTargetKind;
67
68    if kinds
69        .iter()
70        .any(|k| #[allow(non_exhaustive_omitted_patterns)] match k {
    CargoTargetKind::Lib | CargoTargetKind::ProcMacro => true,
    _ => false,
}matches!(k, CargoTargetKind::Lib | CargoTargetKind::ProcMacro))
71    {
72        return true;
73    }
74    all_targets
75        && kinds
76            .iter()
77            .any(|k| #[allow(non_exhaustive_omitted_patterns)] match k {
    CargoTargetKind::Bin | CargoTargetKind::Example => true,
    _ => false,
}matches!(k, CargoTargetKind::Bin | CargoTargetKind::Example))
78}

80/// Parse the target's module tree, each file with its parsed contents and
81/// the module-path segments from the target root (empty for the root file).
82///
83/// # Errors
84///
85/// Fails when a module file cannot be canonicalized. Unreadable or
86/// unparseable files are skipped with a warning.
87pub(crate) fn module_tree(
88    target: &Target,
89) -> crate::Result<Vec<(PathBuf, syn::File, Vec<String>)>> {
90    let mut out = Vec::new();
91    let mut visited = BTreeSet::new();
92    collect(&target.src, &[], &mut out, &mut visited)?;
93    Ok(out)
94}

96fn collect(
97    file: &std::path::Path,
98    prefix: &[String],
99    out: &mut Vec<(PathBuf, syn::File, Vec<String>)>,
100    visited: &mut BTreeSet<PathBuf>,
101) -> crate::Result<()> {
102    let canonical = std::fs::canonicalize(file)?;
103    if !visited.insert(canonical) {
104        return Ok(());
105    }
106    let text = match std::fs::read_to_string(file) {
107        Ok(text) => text,
108        Err(err) => {
109            {
    ::std::io::_eprint(format_args!("dejadoc: cannot read {0}: {1}\n",
            file.display(), err));
};eprintln!("dejadoc: cannot read {}: {err}", file.display());
110            return Ok(());
111        }
112    };
113    let parsed = match syn::parse_file(&text) {
114        Ok(parsed) => parsed,
115        Err(err) => {
116            {
    ::std::io::_eprint(format_args!("dejadoc: cannot parse {0}: {1}\n",
            file.display(), err));
};eprintln!("dejadoc: cannot parse {}: {err}", file.display());
117            return Ok(());
118        }
119    };
120    let dir = file.parent().unwrap_or_else(|| std::path::Path::new("."));
121    // A file module with a companion directory keeps its child modules
122    // there; `#[path]` in that file is still relative to `dir`.
123    let base = match file.file_stem() {
124        Some(stem) if dir.join(stem).is_dir() => dir.join(stem),
125        _ => dir.to_path_buf(),
126    };
127    let mut children = Vec::new();
128    mod_decls(&parsed.items, dir, &base, &mut children);
129    out.push((file.to_path_buf(), parsed, prefix.to_vec()));
130    for (child, name) in children {
131        let mut sub = prefix.to_vec();
132        sub.push(name);
133        collect(&child, &sub, out, visited)?;
134    }
135    Ok(())
136}

138/// Collect the `mod name;` files and their module names, at every nesting
139/// depth, skipping modules whose `cfg` fails. `dir` is the directory
140/// containing the defining file (explicit `#[path]` resolves against it).
141/// `base` is the implicit child directory (the companion directory for a
142/// file module).
143fn mod_decls(
144    items: &[syn::Item],
145    dir: &std::path::Path,
146    base: &std::path::Path,
147    out: &mut Vec<(PathBuf, String)>,
148) {
149    use syn::Item;
150
151    for item in items {
152        let Item::Mod(moditem) = item else {
153            continue;
154        };
155        if !crate::cfg::allows(&moditem.attrs) {
156            continue;
157        }
158        if let Some((_, children)) = &moditem.content {
159            mod_decls(children, dir, base, out);
160            continue;
161        }
162        // Raw identifiers keep their name in the module path.
163        let name = moditem
164            .ident
165            .to_string()
166            .trim_start_matches("r#")
167            .to_string();
168        match resolve_mod_path(dir, base, &name, &moditem.attrs) {
169            Some(path) if path.exists() => out.push((path, name)),
170            Some(path) => {
    ::std::io::_eprint(format_args!("dejadoc: missing module file {0}\n",
            path.display()));
}eprintln!("dejadoc: missing module file {}", path.display()),
171            None => {}
172        }
173    }
174}

176/// File for `mod name;` in `dir`. A top-level `#[path = "…"]` wins, as it
177/// does for rustc. Otherwise the first existing `path` from a
178/// `#[cfg_attr(…, path = "…")]` is used, since cfg is not evaluated.
179/// Explicit paths resolve against `dir`. The plain fallback resolves against
180/// `base`.
181fn resolve_mod_path(
182    dir: &std::path::Path,
183    base: &std::path::Path,
184    name: &str,
185    attrs: &[syn::Attribute],
186) -> Option<PathBuf> {
187    use syn::{Expr, ExprLit, Lit, Meta};
188
189    if let Some(attr) = attrs.iter().find(|a| a.path().is_ident("path")) {
190        let Meta::NameValue(nv) = &attr.meta else {
191            return None;
192        };
193        let Expr::Lit(ExprLit {
194            lit: Lit::Str(s), ..
195        }) = &nv.value
196        else {
197            return None;
198        };
199        return Some(dir.join(s.value()));
200    }
201    let candidates: Vec<PathBuf> = attrs
202        .iter()
203        .filter(|a| a.path().is_ident("cfg_attr"))
204        .filter_map(|a| {
205            let Meta::List(list) = &a.meta else {
206                return None;
207            };
208            Some(cfg_attr_paths(&list.tokens))
209        })
210        .flatten()
211        .map(|p| dir.join(p))
212        .collect();
213    if let Some(path) = candidates.into_iter().find(|p| p.exists()) {
214        return Some(path);
215    }
216    let plain = base.join(::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{0}.rs", name))
    })format!("{name}.rs"));
217    if plain.exists() {
218        return Some(plain);
219    }
220    Some(base.join(name).join("mod.rs"))
221}

223/// The `path = "…"` values inside a `cfg_attr` attribute's tokens.
224fn cfg_attr_paths(tokens: &proc_macro2::TokenStream) -> Vec<String> {
225    let trees = tokens.clone().into_iter().collect::<Vec<_>>();
226    let mut out = Vec::new();
227    for window in trees.windows(3) {
228        if #[allow(non_exhaustive_omitted_patterns)] match (&window[0], &window[1]) {
    (proc_macro2::TokenTree::Ident(id), proc_macro2::TokenTree::Punct(eq)) if
        id == "path" && eq.as_char() == '=' => true,
    _ => false,
}matches!(
229            (&window[0], &window[1]),
230            (
231                proc_macro2::TokenTree::Ident(id),
232                proc_macro2::TokenTree::Punct(eq)
233            ) if id == "path" && eq.as_char() == '='
234        ) {
235            let text = window[2].to_string();
236            if let Ok(s) = syn::parse_str::<syn::LitStr>(&text) {
237                out.push(s.value());
238            }
239        }
240    }
241    out
242}
