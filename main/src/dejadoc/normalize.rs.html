36fn canonical(code: &str, compiles: bool) -> Canonical {
37    let unhidden: String = code.lines().map(map_line).collect::<Vec<_>>().join("\n");
38    let stripped = without_crate_attrs(&unhidden);
39    let body = stripped.as_ref();
40    let Some(mut file) = within_caps(body).then(|| parse_as_crate(body)).flatten() else {
41        let text = body.split_whitespace().collect::<Vec<_>>().join(" ");
42        return Canonical {
43            tokens: text.split_whitespace().count(),
44            text,
45            unparsed: true,
46        };
47    };
48    IncludeDepth.visit_file_mut(&mut file);
49    let stream = if compiles {
50        syn_canon::canonicalize(file)
51    } else {
52        syn_canon::canonicalize_failing(file)
53    };
54    from_stream(stream)
55}

57/// Deepest bracket and generic nesting a body may reach before it hashes as
58/// text, past it the recursive parse and visits risk the end of the stack.
59pub(crate) const MAX_NESTING: usize = 64;
60
61/// Most leaf tokens a body may hold before it hashes as text, bounding the
62/// length of operator, method and closure chains.
63pub(crate) const MAX_TOKENS: usize = 16_384;
64
65/// Whether `body` stays within `MAX_NESTING` and `MAX_TOKENS`, walked without
66/// recursion. An open `<` counts as nesting until its `>`, a `;`, a `{…}`
67/// block or the end of its group, so comparisons do not add up. A body that
68/// does not tokenize is within the caps and fails to parse later.
69fn within_caps(body: &str) -> bool {
70    use proc_macro2::{Delimiter, TokenTree};
71
72    let Ok(stream) = body.parse::<proc_macro2::TokenStream>() else {
73        return true;
74    };
75    // One entry per open group, its remaining trees and its open `<` count.
76    let mut stack = ::alloc::boxed::box_assume_init_into_vec_unsafe(::alloc::intrinsics::write_box_via_move(::alloc::boxed::Box::new_uninit(),
        [(stream.into_iter(), 0_usize)]))alloc::vec![(stream.into_iter(), 0_usize)];
77    let (mut depth, mut tokens) = (0_usize, 0_usize);
78    while let Some((trees, angles)) = stack.last_mut() {
79        let Some(tree) = trees.next() else {
80            depth -= *angles;
81            stack.pop();
82            depth = depth.saturating_sub(1);
83            continue;
84        };
85        match tree {
86            TokenTree::Group(group) => {
87                if group.delimiter() == Delimiter::Brace {
88                    depth -= *angles;
89                    *angles = 0;
90                }
91                let inner = group.stream();
92                drop(group);
93                stack.push((inner.into_iter(), 0));
94                depth += 1;
95            }
96            TokenTree::Punct(punct) => {
97                tokens += 1;
98                match punct.as_char() {
99                    '<' => {
100                        *angles += 1;
101                        depth += 1;
102                    }
103                    '>' if *angles > 0 => {
104                        *angles -= 1;
105                        depth -= 1;
106                    }
107                    ';' => {
108                        depth -= *angles;
109                        *angles = 0;
110                    }
111                    _ => {}
112                }
113            }
114            TokenTree::Ident(_) | TokenTree::Literal(_) => tokens += 1,
115        }
116        if depth > MAX_NESTING || tokens > MAX_TOKENS {
117            return false;
118        }
119    }
120    true
121}

139/// The body without the leading crate attributes that leave the test alone: lint levels,
140/// `feature` and literal docs. The others change the crate and stay.
141fn without_crate_attrs(body: &str) -> Cow<'_, str> {
142    let lifted = |attr: &syn::Attribute| {
143        let path = attr.path();
144        ["allow", "warn", "deny", "forbid", "expect", "feature"]
145            .iter()
146            .any(|name| path.is_ident(name))
147            || (path.is_ident("doc")
148                && #[allow(non_exhaustive_omitted_patterns)] match &attr.meta {
    syn::Meta::NameValue(nv) if
        #[allow(non_exhaustive_omitted_patterns)] match nv.value {
            syn::Expr::Lit(_) => true,
            _ => false,
        } => true,
    _ => false,
}matches!(&attr.meta, syn::Meta::NameValue(nv)
149                    if matches!(nv.value, syn::Expr::Lit(_))))
150    };
151    match syn::parse_str::<Lead>(body) {
152        Ok(lead) if lead.attrs.iter().any(lifted) => {
153            let mut kept: proc_macro2::TokenStream = lead
154                .attrs
155                .iter()
156                .filter(|attr| !lifted(attr))
157                .map(quote::ToTokens::to_token_stream)
158                .collect();
159            kept.extend(lead.rest);
160            Cow::Owned(kept.to_string())
161        }
162        _ => Cow::Borrowed(body),
163    }
164}

169fn parse_as_crate(body: &str) -> Option<syn::File> {
170    let is_main = |item: &syn::Item| #[allow(non_exhaustive_omitted_patterns)] match item {
    syn::Item::Fn(f) if f.sig.ident == "main" => true,
    _ => false,
}matches!(item, syn::Item::Fn(f) if f.sig.ident == "main");
171    let bare_extern = |item: &syn::Item| #[allow(non_exhaustive_omitted_patterns)] match item {
    syn::Item::ExternCrate(e) if e.rename.is_none() && e.attrs.is_empty() =>
        true,
    _ => false,
}matches!(item, syn::Item::ExternCrate(e) if e.rename.is_none() && e.attrs.is_empty());
172    let mut file = match syn::parse_str::<syn::File>(body) {
173        Ok(file) if file.items.iter().any(is_main) => file,
174        _ => {
175            let mut block: syn::Block = syn::parse_str(&::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{{\n{0}\n}}", body))
    })format!("{{\n{body}\n}}")).ok()?;
176            block
177                .stmts
178                .retain(|stmt| !#[allow(non_exhaustive_omitted_patterns)] match stmt {
    syn::Stmt::Item(item) if bare_extern(item) => true,
    _ => false,
}matches!(stmt, syn::Stmt::Item(item) if bare_extern(item)));
179            let output = take_result_tail(&mut block).map(|err| {
    let mut _s = ::quote::__private::TokenStream::new();
    ::quote::__private::push_rarrow(&mut _s);
    ::quote::__private::push_ident(&mut _s, "Result");
    ::quote::__private::push_lt(&mut _s);
    ::quote::__private::push_group(&mut _s,
        ::quote::__private::Delimiter::Parenthesis,
        ::quote::__private::TokenStream::new());
    ::quote::__private::push_comma(&mut _s);
    ::quote::ToTokens::to_tokens(&err, &mut _s);
    ::quote::__private::push_gt(&mut _s);
    _s
}quote::quote!(-> Result<(), #err>));
180            syn::parse2({
    let mut _s = ::quote::__private::TokenStream::new();
    ::quote::__private::push_ident(&mut _s, "fn");
    ::quote::__private::push_ident(&mut _s, "main");
    ::quote::__private::push_group(&mut _s,
        ::quote::__private::Delimiter::Parenthesis,
        ::quote::__private::TokenStream::new());
    ::quote::ToTokens::to_tokens(&output, &mut _s);
    ::quote::ToTokens::to_tokens(&block, &mut _s);
    _s
}quote::quote!(fn main() #output #block)).ok()?
181        }
182    };
183    file.items.retain(|item| !bare_extern(item));
184    Some(file)

187/// The error type of a trailing `Ok::<(), E>(())`, which becomes `Ok(())`.
188fn take_result_tail(block: &mut syn::Block) -> Option<syn::Type> {
189    let syn::Stmt::Expr(syn::Expr::Call(call), None) = block.stmts.last_mut()? else {
190        return None;
191    };
192    let unit_arg = call.args.len() == 1
193        && #[allow(non_exhaustive_omitted_patterns)] match call.args.first() {
    Some(syn::Expr::Tuple(t)) if t.elems.is_empty() => true,
    _ => false,
}matches!(call.args.first(), Some(syn::Expr::Tuple(t)) if t.elems.is_empty());
194    let syn::Expr::Path(func) = call.func.as_mut() else {
195        return None;
196    };
197    if !unit_arg
198        || func.qself.is_some()
199        || func.path.leading_colon.is_some()
200        || func.path.segments.len() != 1
201    {
202        return None;
203    }
204    let segment = func.path.segments.first_mut()?;
205    let syn::PathArguments::AngleBracketed(args) = &mut segment.arguments else {
206        return None;
207    };
208    let unit_ok = segment.ident == "Ok"
209        && args.args.len() == 2
210        && #[allow(non_exhaustive_omitted_patterns)] match &args.args[0] {
    syn::GenericArgument::Type(syn::Type::Tuple(t)) if t.elems.is_empty() =>
        true,
    _ => false,
}matches!(&args.args[0], syn::GenericArgument::Type(syn::Type::Tuple(t)) if t.elems.is_empty())
211        && #[allow(non_exhaustive_omitted_patterns)] match &args.args[1] {
    syn::GenericArgument::Type(_) => true,
    _ => false,
}matches!(&args.args[1], syn::GenericArgument::Type(_));
212    if !unit_ok {
213        return None;
214    }
215    let syn::GenericArgument::Type(err) = args.args.pop()? else {
216        return None;
217    };
218    segment.arguments = syn::PathArguments::None;
219    Some(err)
220}

222/// rustdoc's hidden-line rule per body line. A `##` line shows as a `#`
223/// line. A `# ` prefix (space required) hides the line but keeps its
224/// trimmed remainder. A bare `#` hides to an empty line. Everything
225/// else, including `#text` without the space, is kept verbatim.
226fn map_line(line: &str) -> String {
227    let trimmed = line.trim();
228    if trimmed.starts_with("##") {
229        line.replacen("##", "#", 1)
230    } else if let Some(stripped) = trimmed.strip_prefix("# ") {
231        stripped.to_string()
232    } else if trimmed == "#" {
233        String::new()
234    } else {
235        line.to_string()
236    }
237}

239/// Strip the leading `../` of `include`-style macro paths, which rustdoc resolves from the package root.
240struct IncludeDepth;
241
242impl VisitMut for IncludeDepth {
243    fn visit_macro_mut(&mut self, mac: &mut syn::Macro) {
244        let include = mac
245            .path
246            .segments
247            .last()
248            .is_some_and(|s| is_include(&s.ident));
249        mac.tokens = flatten_stream(core::mem::take(&mut mac.tokens), include);
250    }
251
252    fn visit_meta_list_mut(&mut self, list: &mut syn::MetaList) {
253        list.tokens = flatten_stream(core::mem::take(&mut list.tokens), false);
254    }
255}

257fn is_include(ident: &proc_macro2::Ident) -> bool {
258    let name = ident.unraw().to_string();
259    name == "include" || name.starts_with("include_")
260}

262/// Flatten include paths in one token list, `strip` set by an `include` macro name and kept only by its `!`.
263fn flatten_stream(stream: proc_macro2::TokenStream, strip: bool) -> proc_macro2::TokenStream {
264    use proc_macro2::{Group, TokenTree};
265
266    let mut strip = strip;
267    stream
268        .into_iter()
269        .map(|tree| match tree {
270            TokenTree::Ident(id) => {
271                strip = is_include(&id);
272                TokenTree::Ident(id)
273            }
274            TokenTree::Punct(p) => {
275                strip = strip && p.as_char() == '!';
276                TokenTree::Punct(p)
277            }
278            TokenTree::Group(group) => {
279                let mut rebuilt =
280                    Group::new(group.delimiter(), flatten_stream(group.stream(), strip));
281                rebuilt.set_span(group.span());
282                strip = false;
283                TokenTree::Group(rebuilt)
284            }
285            TokenTree::Literal(lit) => {
286                let lit = if strip {
287                    flat_literal(&lit).unwrap_or(lit)
288                } else {
289                    lit
290                };
291                strip = false;
292                TokenTree::Literal(lit)
293            }
294        })
295        .collect()
296}

298/// The literal with leading `../` components removed, `None` unless it is
299/// a string literal (plain, raw, or byte) whose text starts with depth.
300/// Prefix and hash runs are preserved, escape sequences are untouched.
301fn flat_literal(lit: &proc_macro2::Literal) -> Option<proc_macro2::Literal> {
302    let s = lit.to_string();
303    let open = s.find('"')?;
304    let prefix = &s[..open];
305    let hashes = prefix.trim_start_matches(|c: char| c != '#');
306    let close = s.len().checked_sub(1 + hashes.len())?;
307    let inner = s.get(open + 1..close)?;
308    if !inner.starts_with("../") {
309        return None;
310    }
311    let mut stripped = inner;
312    while let Some(rest) = stripped.strip_prefix("../") {
313        stripped = rest;
314    }
315    ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{0}\"{1}\"{2}", prefix, stripped,
                hashes))
    })format!("{prefix}\"{stripped}\"{hashes}").parse().ok()
316}

319    let text = stream.to_string();
320    Canonical {
321        text,
322        unparsed: false,
323        tokens: count_tokens(stream),
324    }

327/// Number of leaf tokens in a token stream.
328/// The multi-character operators, longest first, each counted as one token.
329const OPERATORS: [&str; 23] = [
330    "<<=", ">>=", "...", "..=", "=>", "==", "!=", "<=", ">=", "&&", "||", "+=", "-=", "*=", "/=",
331    "%=", "^=", "&=", "|=", "<<", ">>", "..", "->",
332];

334/// The size of a function's body in its canonical tokens, the signature left out. A path, a
335/// multi-character operator or a lifetime counts as one token, a comma or a bracket as none.
336/// A function canonicalized as a one-item file ends in its body.
337pub(crate) fn body_tokens(function: proc_macro2::TokenStream) -> usize {
338    match function.into_iter().last() {
339        Some(proc_macro2::TokenTree::Group(body)) => units(body.stream()),
340        _ => 0,
341    }
342}

344/// The tokens of `stream` as `body_tokens` counts them.
345fn units(stream: proc_macro2::TokenStream) -> usize {
346    use proc_macro2::{Spacing, TokenTree};
347
348    let mut trees = stream.into_iter().peekable();
349    let mut count = 0;
350    // The last token was an identifier, so a `::` after it continues one path.
351    let (mut ident, mut joined) = (false, false);
352    while let Some(tree) = trees.next() {
353        match tree {
354            TokenTree::Group(group) => {
355                count += units(group.stream());
356                (ident, joined) = (false, false);
357            }
358            TokenTree::Ident(_) => {
359                count += usize::from(!joined);
360                (ident, joined) = (true, false);
361            }
362            TokenTree::Literal(_) => {
363                count += 1;
364                (ident, joined) = (false, false);
365            }
366            TokenTree::Punct(first) => {
367                if first.as_char() == '\'' && #[allow(non_exhaustive_omitted_patterns)] match trees.peek() {
    Some(TokenTree::Ident(_)) => true,
    _ => false,
}matches!(trees.peek(), Some(TokenTree::Ident(_))) {
368                    trees.next();
369                    count += 1;
370                    (ident, joined) = (false, false);
371                    continue;
372                }
373                let mut run = alloc::string::String::from(first.as_char());
374                let mut spacing = first.spacing();
375                while spacing == Spacing::Joint
376                    && let Some(TokenTree::Punct(next)) = trees.peek()
377                {
378                    run.push(next.as_char());
379                    spacing = next.spacing();
380                    trees.next();
381                }
382                let mut rest = run.as_str();
383                while let Some(c) = rest.chars().next() {
384                    if let Some(tail) = rest.strip_prefix("::") {
385                        joined = ident;
386                        rest = tail;
387                        continue;
388                    }
389                    let op = OPERATORS.iter().find(|op| rest.starts_with(*op));
390                    count += usize::from(c != ',');
391                    (ident, joined) = (false, false);
392                    rest = &rest[op.map_or(c.len_utf8(), |op| op.len())..];
393                }
394            }
395        }
396    }
397    count
398}

400pub(crate) fn count_tokens(stream: proc_macro2::TokenStream) -> usize {
401    stream.into_iter().map(count_tree).sum()
402}
403
404fn count_tree(tree: proc_macro2::TokenTree) -> usize {
405    match tree {
406        proc_macro2::TokenTree::Group(group) => count_tokens(group.stream()),
407        _ => 1,
408    }
