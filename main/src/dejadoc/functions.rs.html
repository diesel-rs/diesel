45/// The remedy of a group's copies, from the first matching row of the table in the spec.
46pub(crate) fn remedy(sites: &[(DocTest, Context)]) -> Remedy {
47    let Some(((first, first_context), rest)) = sites.split_first() else {
48        return Remedy::Delete;
49    };
50    if rest
51        .iter()
52        .any(|(site, _)| site.self_type != first.self_type)
53    {
54        Remedy::GenericOrMacro
55    } else if sites
56        .iter()
57        .any(|(_, context)| context.owner == Owner::Trait || context.exported)
58    {
59        Remedy::HelperOrMacro
60    } else if rest
61        .iter()
62        .any(|(_, context)| context.cfg != first_context.cfg)
63    {
64        Remedy::MergeCfg
65    } else {
66        Remedy::Delete
67    }
68}

70/// The free functions, impl methods and trait default methods of `file`, nested functions excluded.
71pub(crate) fn functions(
72    prefix: &str,
73    file: &SourceFile,
74    root: &str,
75    crate_root: &str,
76    library: bool,
77) -> Vec<FnSite> {
78    let path = file
79        .path
80        .strip_prefix(root)
81        .and_then(|p| p.strip_prefix('/'))
82        .unwrap_or(&file.path);
83    let lines: Vec<&str> = file.text.lines().collect();
84    let mut walk = Walk {
85        path,
86        crate_root,
87        library,
88        lines: &lines,
89        cfg: Vec::new(),
90        out: Vec::new(),
91    };
92    walk.items(&file.parsed.items, prefix);
93    walk.out
94}

96/// The walk over one source file.
97struct Walk<'a> {
98    path: &'a str,
99    crate_root: &'a str,
100    /// Whether the file belongs to a library, whose `pub` functions are public API.
101    library: bool,
102    lines: &'a [&'a str],
103    /// The `cfg` attributes of the inline modules, impls and traits the walk is inside.
104    cfg: Vec<String>,
105    out: Vec<FnSite>,
106}
107
108impl Walk<'_> {
109    fn items(&mut self, items: &[syn::Item], module: &str) {
110        for item in items {
111            match item {
112                syn::Item::Fn(f) => {
113                    self.push(
114                        module,
115                        None,
116                        f,
117                        Parts {
118                            attrs: &f.attrs,
119                            vis: &f.vis,
120                            sig: &f.sig,
121                            block: &f.block,
122                        },
123                    );
124                }
125                syn::Item::Mod(m) => {
126                    if let Some((_, inner)) = &m.content {
127                        let depth = self.enter(&m.attrs);
128                        self.items(inner, &::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{1}::{0}", m.ident, module))
    })format!("{module}::{}", m.ident));
129                        self.cfg.truncate(depth);
130                    }
131                }
132                syn::Item::Impl(i) => {
133                    let self_type = type_text(&i.self_ty);
134                    let owner = if i.trait_.is_some() {
135                        Owner::Trait
136                    } else {
137                        Owner::Inherent
138                    };
139                    let depth = self.enter(&i.attrs);
140                    for member in &i.items {
141                        if let syn::ImplItem::Fn(f) = member {
142                            self.push(
143                                module,
144                                Some((&self_type, owner)),
145                                f,
146                                Parts {
147                                    attrs: &f.attrs,
148                                    vis: &f.vis,
149                                    sig: &f.sig,
150                                    block: &f.block,
151                                },
152                            );
153                        }
154                    }
155                    self.cfg.truncate(depth);
156                }
157                syn::Item::Trait(t) => {
158                    let self_type = t.ident.to_string();
159                    let inherited = syn::Visibility::Inherited;
160                    let depth = self.enter(&t.attrs);
161                    for member in &t.items {
162                        if let syn::TraitItem::Fn(f) = member
163                            && let Some(block) = &f.default
164                        {
165                            self.push(
166                                module,
167                                Some((&self_type, Owner::Trait)),
168                                f,
169                                Parts {
170                                    attrs: &f.attrs,
171                                    vis: &inherited,
172                                    sig: &f.sig,
173                                    block,
174                                },
175                            );
176                        }
177                    }
178                    self.cfg.truncate(depth);
179                }
180                _ => {}
181            }
182        }
183    }
184
185    /// Push the `cfg` attributes of an item the walk enters, returning the depth to restore.
186    fn enter(&mut self, attrs: &[syn::Attribute]) -> usize {
187        let depth = self.cfg.len();
188        self.cfg.extend(cfgs(attrs));
189        depth
190    }
191
192    /// Record the function `whole`, attributes and doc comments included, `method` holding
193    /// the self type and owner of a method.
194    fn push(
195        &mut self,
196        module: &str,
197        method: Option<(&str, Owner)>,
198        whole: &impl Spanned,
199        parts: Parts<'_>,
200    ) {
201        let Parts {
202            attrs,
203            vis,
204            sig,
205            block,
206        } = parts;
207        let (self_type, owner) = match method {
208            Some((self_type, owner)) => (Some(self_type), owner),
209            None => (None, Owner::Free),
210        };
211        let span = whole.span();
212        let (line, end) = (span.start().line, span.end().line);
213        let allow = line
214            .checked_sub(2)
215            .and_then(|above| self.lines.get(above))
216            .is_some_and(|above| above.trim() == "// dejadoc: allow");
217        let item = match self_type {
218            Some(ty) => ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{1}::{2}::{0}", sig.ident, module,
                ty))
    })format!("{module}::{ty}::{}", sig.ident),
219            None => ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{1}::{0}", sig.ident, module))
    })format!("{module}::{}", sig.ident),
220        };
221        let mut cfg = self.cfg.clone();
222        cfg.extend(cfgs(attrs));
223        cfg.sort_unstable();
224        let code = self
225            .lines
226            .get(line.saturating_sub(1)..end.min(self.lines.len()))
227            .unwrap_or_default()
228            .join("\n");
229        self.out.push(FnSite {
230            site: DocTest {
231                file: self.path.to_string(),
232                line: u32::try_from(line).unwrap_or(u32::MAX),
233                end: u32::try_from(end).ok(),
234                item,
235                info: Vec::new(),
236                code,
237                allow,
238                self_type: self_type.map(ToString::to_string),
239                public: self.library && #[allow(non_exhaustive_omitted_patterns)] match vis {
    syn::Visibility::Public(_) => true,
    _ => false,
}matches!(vis, syn::Visibility::Public(_)),
240            },
241            scope: ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{0}\n{1}", self.crate_root,
                module))
    })format!("{}\n{module}", self.crate_root),
242            func: syn::ItemFn {
243                attrs: attrs.iter().filter(|a| live(&a.meta)).cloned().collect(),
244                vis: syn::Visibility::Inherited,
245                modifiers: syn::FnModifiers::default(),
246                sig: compared_sig(sig, method.is_some()),
247                block: Box::new(block.clone()),
248            },
249            context: Context {
250                owner,
251                exported: attrs.iter().any(|a| exports(&a.meta)),
252                cfg,
253            },
254        });
255    }
256}

267/// The signature as compared. A method's name is not in scope in its body, where a bare
268/// path of that name reaches a free function, so the method's name becomes one no body names.
269fn compared_sig(sig: &syn::Signature, method: bool) -> syn::Signature {
270    let mut sig = sig.clone();
271    if method {
272        sig.ident = syn::Ident::new("__dejadoc_method", sig.ident.span());
273    }
274    sig
275}

285/// Whether `meta` takes part in the comparison. `cfg`, the hints that leave the body's
286/// meaning alone, the attributes that export a symbol, and a `cfg_attr` wrapping only those
287/// do not.
288fn live(meta: &syn::Meta) -> bool {
289    let path = meta.path();
290    if path.is_ident("cfg_attr") {
291        return wrapped(meta, 1).is_none_or(|metas| metas.iter().any(live));
292    }
293    if path.is_ident("unsafe") {
294        return wrapped(meta, 0).is_none_or(|metas| metas.iter().any(live));
295    }
296    // A `doc` computed by a macro runs at compile time and can fail.
297    if path.is_ident("doc")
298        && #[allow(non_exhaustive_omitted_patterns)] match meta {
    syn::Meta::NameValue(nv) if
        #[allow(non_exhaustive_omitted_patterns)] match nv.value {
            syn::Expr::Macro(_) => true,
            _ => false,
        } => true,
    _ => false,
}matches!(meta, syn::Meta::NameValue(nv) if matches!(nv.value, syn::Expr::Macro(_)))
299    {
300        return true;
301    }
302    let inert = [
303        "cfg",
304        "inline",
305        "cold",
306        "must_use",
307        "track_caller",
308        "doc",
309        "allow",
310        "warn",
311        "deny",
312        "forbid",
313        "expect",
314        "no_mangle",
315        "export_name",
316    ];
317    !inert.iter().any(|name| path.is_ident(name))
318        && path.segments.first().is_none_or(|s| s.ident != "rustfmt")
319}

321/// Whether `meta` exports a symbol, which the symbol's users then need.
322fn exports(meta: &syn::Meta) -> bool {
323    let path = meta.path();
324    if path.is_ident("unsafe") {
325        return wrapped(meta, 0).is_some_and(|metas| metas.iter().any(exports));
326    }
327    path.is_ident("no_mangle") || path.is_ident("export_name")
328}

330/// The metas `meta` wraps after its first `skip` arguments, `None` when they don't parse.
331fn wrapped(meta: &syn::Meta, skip: usize) -> Option<Vec<syn::Meta>> {
332    let syn::Meta::List(list) = meta else {
333        return None;
334    };
335    let parser = syn::punctuated::Punctuated::<syn::Meta, ::syn::token::Commasyn::Token![,]>::parse_terminated;
336    list.parse_args_with(parser)
337        .ok()
338        .map(|metas| metas.into_iter().skip(skip).collect())
339}

341/// The whole type as written, `Foo<u8>` and `a::Foo` telling apart types one last segment would merge.
342fn type_text(ty: &syn::Type) -> String {
343    let printed = quote::ToTokens::to_token_stream(ty).to_string();
344    let word = |c: Option<char>| c.is_some_and(|c| c.is_alphanumeric() || c == '_');
345    let mut out = String::with_capacity(printed.len());
346    let mut chars = printed.chars().peekable();
347    while let Some(c) = chars.next() {
348        if c != ' ' || (word(out.chars().last()) && word(chars.peek().copied())) {
349            out.push(c);
350        }
351    }
352    out
