12/// Drop the trailing comma of every token group. A one-tuple keeps its
13/// meaning because the fold of redundant parentheses already separates
14/// it from the parenthesised expression of the same element.
15pub(crate) fn strip_trailing_commas(stream: TokenStream) -> TokenStream {
16    stream
17        .into_iter()
18        .map(|tree| match tree {
19            TokenTree::Group(group) => {
20                let inner = strip_last_comma(strip_trailing_commas(group.stream()));
21                let mut rebuilt = Group::new(group.delimiter(), inner);
22                rebuilt.set_span(group.span());
23                TokenTree::Group(rebuilt)
24            }
25            other => other,
26        })
27        .collect()
28}

30/// `stream` without its last comma.
31fn strip_last_comma(stream: TokenStream) -> TokenStream {
32    let mut tokens: Vec<TokenTree> = stream.into_iter().collect();
33    if tokens.last().is_some_and(is_comma) {
34        tokens.pop();
35    }
36    tokens.into_iter().collect()
37}

39fn is_comma(tree: &TokenTree) -> bool {
40    #[allow(non_exhaustive_omitted_patterns)] match tree {
    TokenTree::Punct(p) if p.as_char() == ',' => true,
    _ => false,
}matches!(tree, TokenTree::Punct(p) if p.as_char() == ',')
41}

43/// Rewrite every literal token in `stream` to its canonical decimal spelling.
44///
45/// Skips the token tree of a macro call and of an attribute, both of
46/// which a procedural macro receives verbatim, so the spelling of a
47/// literal inside them can be observable.
48pub(crate) fn canonical_literals(stream: TokenStream) -> TokenStream {
49    let mut out = Vec::new();
50    let mut opaque = false;
51    for tree in stream {
52        let tree = match tree {
53            TokenTree::Literal(lit) => {
54                let span = lit.span();
55                match syn::parse_str::<syn::Lit>(&lit.to_string()) {
56                    Ok(parsed) => canon_one_lit(parsed, span),
57                    Err(_) => TokenTree::Literal(lit),
58                }
59            }
60            TokenTree::Group(group) if !opaque => {
61                let span = group.span();
62                let mut rebuilt = Group::new(group.delimiter(), canonical_literals(group.stream()));
63                rebuilt.set_span(span);
64                TokenTree::Group(rebuilt)
65            }
66            other => other,
67        };
68        opaque = #[allow(non_exhaustive_omitted_patterns)] match &tree {
    TokenTree::Punct(p) if
        #[allow(non_exhaustive_omitted_patterns)] match p.as_char() {
            '!' | '#' => true,
            _ => false,
        } => true,
    _ => false,
}matches!(&tree, TokenTree::Punct(p) if matches!(p.as_char(), '!' | '#'));
69        out.push(tree);
70    }
71    out.into_iter().collect()
72}

74fn canon_one_lit(lit: syn::Lit, span: proc_macro2::Span) -> TokenTree {
75    let rebuilt = match lit {
76        syn::Lit::Int(i) => {
77            let s = ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{0}{1}", i.base10_digits(),
                i.suffix()))
    })format!("{}{}", i.base10_digits(), i.suffix());
78            syn::Lit::Int(syn::LitInt::new(&s, span))
79        }
80        syn::Lit::Float(f) => {
81            let digits = canon_float_str(f.base10_digits());
82            syn::Lit::Float(syn::LitFloat::new(&::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{1}{0}", f.suffix(), digits))
    })format!("{digits}{}", f.suffix()), span))
83        }
84        syn::Lit::Str(s) => syn::Lit::Str(syn::LitStr::new(&s.value(), span)),
85        syn::Lit::ByteStr(b) => syn::Lit::ByteStr(syn::LitByteStr::new(&b.value(), span)),
86        syn::Lit::CStr(c) => syn::Lit::CStr(syn::LitCStr::new(c.value().as_c_str(), span)),
87        syn::Lit::Char(c) => syn::Lit::Char(syn::LitChar::new(c.value(), span)),
88        syn::Lit::Byte(b) => syn::Lit::Byte(syn::LitByte::new(b.value(), span)),
89        other => other,
90    };
91    rebuilt
92        .to_token_stream()
93        .into_iter()
94        .next()
95        .expect("rebuilt literal produces at least one token")
96}

98/// Canonical decimal string for a float's digit part, text only so the
99/// canonical form never depends on the feature set. `syn` has already
100/// dropped the underscores, a `+` exponent sign and an upper case `E`.
101fn canon_float_str(s: &str) -> String {
102    let (mantissa, exponent) = match s.split_once('e') {
103        Some((mantissa, exponent)) => (mantissa, Some(exponent)),
104        None => (s, None),
105    };
106    let (whole, fraction) = mantissa.split_once('.').unwrap_or((mantissa, ""));
107    let fraction = fraction.trim_end_matches('0');
108    let mut out = String::with_capacity(s.len());
109    out.push_str(whole);
110    out.push('.');
111    out.push_str(if fraction.is_empty() { "0" } else { fraction });
112    if let Some(exponent) = exponent {
113        out.push('e');
114        let (sign, digits) = exponent
115            .strip_prefix('-')
116            .map_or(("", exponent), |digits| ("-", digits));
117        out.push_str(sign);
118        let digits = digits.trim_start_matches('0');
119        out.push_str(if digits.is_empty() { "0" } else { digits });
120    }
121    out
122}

280/// An arm body `{ expr }` becomes `expr`, the form rustfmt writes.
281fn unwrap_arm_block(arm: &mut syn::Arm) {
282    unwrap_single_expr_block(&mut arm.body);
283}

285/// A closure or arm body `{ expr }` becomes `expr` when the block holds
286/// exactly one tail expression and carries no label or attributes.
287fn unwrap_single_expr_block(body: &mut Box<syn::Expr>) {
288    let syn::Expr::Block(block) = body.as_mut() else {
289        return;
290    };
291    if block.label.is_some()
292        || !block.attrs.is_empty()
293        || !#[allow(non_exhaustive_omitted_patterns)] match block.block.stmts.as_slice() {
    [syn::Stmt::Expr(_, None)] => true,
    _ => false,
}matches!(block.block.stmts.as_slice(), [syn::Stmt::Expr(_, None)])
294    {
295        return;
296    }
297    if let Some(syn::Stmt::Expr(expr, None)) = block.block.stmts.pop() {
298        **body = expr;
299    }
300}

302/// Remove every bare empty statement from `stmts`.
303fn drop_empty_stmts(stmts: &mut Vec<syn::Stmt>) {
304    stmts.retain(|stmt| {
305        !#[allow(non_exhaustive_omitted_patterns)] match stmt {
    syn::Stmt::Expr(syn::Expr::Verbatim(v), Some(_)) if v.is_empty() => true,
    _ => false,
}matches!(
306            stmt,
307            syn::Stmt::Expr(syn::Expr::Verbatim(v), Some(_)) if v.is_empty()
308        )
309    });
310}

312/// Replace a tail `return expr;` with `expr` as the tail expression.
313fn fold_tail_return(stmts: &mut Vec<syn::Stmt>) {
314    let n = stmts.len();
315    if n == 0 {
316        return;
317    }
318    let is_valued_tail_return =
319        #[allow(non_exhaustive_omitted_patterns)] match &stmts[n - 1] {
    syn::Stmt::Expr(syn::Expr::Return(r), Some(_)) if r.expr.is_some() =>
        true,
    _ => false,
}matches!(&stmts[n - 1], syn::Stmt::Expr(syn::Expr::Return(r), Some(_)) if r.expr.is_some());
320    if !is_valued_tail_return {
321        return;
322    }
323    let last = stmts.remove(n - 1);
324    if let syn::Stmt::Expr(syn::Expr::Return(mut ret), Some(_)) = last
325        && let Some(inner) = ret.expr.take()
326    {
327        stmts.push(syn::Stmt::Expr(*inner, None));
328    }
329}

331/// Give every statement macro but the last one a semicolon, which the
332/// brace form omits. A tail macro's value is the block's value.
333fn semicolon_non_tail_macros(stmts: &mut [syn::Stmt]) {
334    let Some((_, rest)) = stmts.split_last_mut() else {
335        return;
336    };
337    for stmt in rest {
338        if let syn::Stmt::Macro(mac) = stmt {
339            mac.semi_token.get_or_insert_with(Default::default);
340        }
341    }
342}

344/// Drop an explicit `-> ()`, which a signature means by omitting it.
345fn fold_unit_return(output: &mut syn::ReturnType) {
346    if let syn::ReturnType::Type(_, ty) = output
347        && #[allow(non_exhaustive_omitted_patterns)] match ty.as_ref() {
    syn::Type::Tuple(t) if t.elems.is_empty() => true,
    _ => false,
}matches!(ty.as_ref(), syn::Type::Tuple(t) if t.elems.is_empty())
348    {
349        *output = syn::ReturnType::Default;
350    }
351}

353/// Remove a no-attribute `Expr::Paren` wrapper, leaving the inner expression.
354fn fold_paren_expr(expr: &mut syn::Expr) {
355    if !#[allow(non_exhaustive_omitted_patterns)] match expr {
    syn::Expr::Paren(p) if p.attrs.is_empty() => true,
    _ => false,
}matches!(expr, syn::Expr::Paren(p) if p.attrs.is_empty()) {
356        return;
357    }
358    let dummy = syn::Expr::Verbatim(proc_macro2::TokenStream::new());
359    if let syn::Expr::Paren(paren) = core::mem::replace(expr, dummy) {
360        *expr = *paren.expr;
361    }
362}

    syn::Pat::Paren(p) if p.attrs.is_empty() => true,
    _ => false,
}matches!(pat, syn::Pat::Paren(p) if p.attrs.is_empty()) {

368    }
369    let dummy = syn::Pat::Verbatim(proc_macro2::TokenStream::new());
370    if let syn::Pat::Paren(paren) = core::mem::replace(pat, dummy) {

372    }

375/// Remove a `Type::Paren` wrapper, leaving the inner type.
376fn fold_paren_type(ty: &mut syn::Type) {
377    if !#[allow(non_exhaustive_omitted_patterns)] match ty {
    syn::Type::Paren(_) => true,
    _ => false,
}matches!(ty, syn::Type::Paren(_)) {
378        return;
379    }
380    let dummy = syn::Type::Verbatim(proc_macro2::TokenStream::new());
381    if let syn::Type::Paren(paren) = core::mem::replace(ty, dummy) {
382        *ty = *paren.elem;
383    }

386/// True for attributes that carry no program meaning for doctests: doc
387/// comments and lint-level directives.
388fn is_inert_attr(attr: &syn::Attribute) -> bool {
389    let p = attr.path();
390    p.is_ident("doc") || p.is_ident("allow") || p.is_ident("expect") || p.is_ident("warn")
391}

419/// Replace the `use` entries of `list` by one sorted item per leaf path,
420/// placed at the front.
421fn hoist_uses<T>(
422    list: &mut Vec<T>,
423    into_use: impl Fn(T) -> Result<syn::ItemUse, T>,
424    wrap: impl Fn(syn::ItemUse) -> T,
425) {
426    let mut leaves = Vec::new();
427    let rest: Vec<T> = core::mem::take(list)
428        .into_iter()
429        .filter_map(|entry| {
430            let item = match into_use(entry) {
431                Ok(item) => item,
432                Err(entry) => return Some(entry),
433            };
434            let mut trees = Vec::new();
435            flatten_use_tree(&[], item.tree, &mut trees);
436            leaves.extend(trees.into_iter().map(|tree| syn::ItemUse {
437                attrs: item.attrs.clone(),
438                vis: item.vis.clone(),
439                use_token: item.use_token,
440                leading_colon: item.leading_colon,
441                tree,
442                semi_token: item.semi_token,
443            }));
444            None
445        })
446        .collect();
447    leaves.sort_by_cached_key(|leaf| {
448        ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("{0} {1}",
                leaf.vis.to_token_stream(), leaf.tree.to_token_stream()))
    })format!(
449            "{} {}",
450            leaf.vis.to_token_stream(),
451            leaf.tree.to_token_stream()
452        )
453    });
454    list.extend(leaves.into_iter().map(wrap));
455    list.extend(rest);
456}

458/// True when every attribute on `item` would be removed by
459/// `strip_inert_attrs`, meaning hoisting it cannot move an attribute
460/// that references a local binding. `use` items never reach here,
461/// `hoist_uses` has already taken them.
462fn item_has_only_inert_attrs(item: &syn::Item) -> bool {
463    let attrs: &[syn::Attribute] = match item {
464        syn::Item::Const(v) => &v.attrs,
465        syn::Item::Enum(v) => &v.attrs,
466        syn::Item::ExternCrate(v) => &v.attrs,
467        syn::Item::Fn(v) => &v.attrs,
468        syn::Item::ForeignMod(v) => &v.attrs,
469        syn::Item::Impl(v) => &v.attrs,
470        syn::Item::Mod(v) => &v.attrs,
471        syn::Item::Static(v) => &v.attrs,
472        syn::Item::Struct(v) => &v.attrs,
473        syn::Item::Trait(v) => &v.attrs,
474        syn::Item::TraitAlias(v) => &v.attrs,
475        syn::Item::Type(v) => &v.attrs,
476        syn::Item::Union(v) => &v.attrs,
477        _ => return true,
478    };
479    attrs.iter().all(is_inert_attr)
480}

482/// Hoist non-`use`, non-`macro_rules!` item statements that appear before
483/// the first `macro_rules!` definition in a block to the front, right
484/// after the sorted `use` items placed there by `hoist_uses`.
485///
486/// Only items whose attributes are all inert are hoisted. An item with a
487/// non-inert attribute may have an argument that references a local
488/// binding; hoisting it before that binding changes the visit order and
489/// breaks alpha renaming.
490///
491/// Items in that region move to the front in their original relative order
492/// among items. Non-item statements follow them in their original relative
493/// order. Everything from the first `Stmt::Item(Item::Macro)` onward is
494/// untouched, because `macro_rules!` visibility is sequential.
495fn hoist_block_items(stmts: &mut Vec<syn::Stmt>) {
496    let use_count = stmts.partition_point(|s| #[allow(non_exhaustive_omitted_patterns)] match s {
    syn::Stmt::Item(syn::Item::Use(_)) => true,
    _ => false,
}matches!(s, syn::Stmt::Item(syn::Item::Use(_))));
497
498    let rel_macro = stmts[use_count..]
499        .iter()
500        .position(|s| #[allow(non_exhaustive_omitted_patterns)] match s {
    syn::Stmt::Item(syn::Item::Macro(_)) => true,
    _ => false,
}matches!(s, syn::Stmt::Item(syn::Item::Macro(_))));
501    let hoist_end = use_count + rel_macro.unwrap_or(stmts.len() - use_count);
502
503    if hoist_end == use_count {
504        return;
505    }
506
507    let tail = stmts.split_off(hoist_end);
508    let region = stmts.split_off(use_count);
509
510    let mut items = Vec::new();
511    let mut rest = Vec::new();
512    for stmt in region {
513        let hoistable = match &stmt {
514            syn::Stmt::Item(item) => item_has_only_inert_attrs(item),
515            _ => false,
516        };
517        if hoistable {
518            items.push(stmt);
519        } else {
520            rest.push(stmt);
521        }
522    }
523
524    stmts.extend(items);
525    stmts.extend(rest);
526    stmts.extend(tail);
527}

529/// Split `tree` into its leaf paths, `a::{self}` becoming `a`.
530fn flatten_use_tree(prefix: &[syn::Ident], tree: syn::UseTree, out: &mut Vec<syn::UseTree>) {
531    match tree {
532        syn::UseTree::Path(path) => {
533            let mut prefix = prefix.to_vec();
534            prefix.push(path.ident);
535            flatten_use_tree(&prefix, *path.tree, out);
536        }
537        syn::UseTree::Group(group) => {
538            for item in group.items {
539                flatten_use_tree(prefix, item, out);
540            }
541        }
542        syn::UseTree::Name(name) if name.ident == "self" => {
543            if let [head @ .., last] = prefix {
544                let leaf = syn::UseTree::Name(syn::UseName {
545                    ident: last.clone(),
546                });
547                out.push(path_tree(head, leaf));
548            }
549        }
550        leaf => out.push(path_tree(prefix, leaf)),
551    }
552}
553
554fn path_tree(prefix: &[syn::Ident], leaf: syn::UseTree) -> syn::UseTree {
555    match prefix {
556        [] => leaf,
557        [head, rest @ ..] => syn::UseTree::Path(syn::UsePath {
558            ident: head.clone(),
559            colon2_token: syn::token::PathSep::default(),
560            tree: Box::new(path_tree(rest, leaf)),
561        }),
562    }
563}

565/// Rewrite the outer delimiter of a macro call to parentheses.
566fn normalize_macro_delim(mac: &mut syn::Macro) {
567    if #[allow(non_exhaustive_omitted_patterns)] match mac.delimiter {
    syn::MacroDelimiter::Paren(_) => true,
    _ => false,
}matches!(mac.delimiter, syn::MacroDelimiter::Paren(_)) {
568        return;
569    }
570    let span = *mac.delimiter.span();
571    mac.delimiter = syn::MacroDelimiter::Paren(syn::token::Paren { span });
572}
