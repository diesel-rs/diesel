1//! `cfg` evaluation as rustdoc collects doctests on a 64-bit Linux host
2//! with every feature on and `test` off, so a scan gives the same answer
3//! on every machine.
4
5use alloc::string::ToString;

7use alloc::vec::Vec;
8use syn::{Expr, ExprLit, Lit, LitStr, Meta, Token, punctuated::Punctuated};

10/// Whether every `#[cfg(…)]` in `attrs` holds.
11pub(crate) fn allows(attrs: &[syn::Attribute]) -> bool {
12    attrs.iter().all(|attr| {
13        !attr.path().is_ident("cfg")
14            || match &attr.meta {
15                Meta::List(list) => list.parse_args::<Meta>().is_ok_and(|pred| holds(&pred)),
16                _ => true,
17            }
18    })
19}
20
21/// The metas a `#[cfg_attr(pred, …)]` applies, nested ones expanded in
22/// place, `None` unless it is one and its predicate holds.
23pub(crate) fn cfg_attr_metas(attr: &syn::Attribute) -> Option<Vec<Meta>> {
24    let mut metas = Vec::new();
25    expand(&attr.meta, &mut metas).then_some(metas)
26}
27
28/// Push what `meta` applies onto `out` when it is a `cfg_attr` whose
29/// predicate holds, and tell whether it was.
30fn expand(meta: &Meta, out: &mut Vec<Meta>) -> bool {
31    let Meta::List(list) = meta else {
32        return false;
33    };
34    if !list.path.is_ident("cfg_attr") {
35        return false;
36    }
37    let Ok(metas) = list.parse_args_with(Punctuated::<Meta, ::syn::token::CommaToken![,]>::parse_terminated) else {
38        return false;
39    };
40    let mut metas = metas.into_iter();
41    if !metas.next().is_some_and(|pred| holds(&pred)) {
42        return false;
43    }
44    for meta in metas {
45        if meta.path().is_ident("cfg_attr") {
46            expand(&meta, out);
47        } else {
48            out.push(meta);
49        }
50    }
51    true
52}

54/// The string literal `expr` is, if any.
55pub(crate) fn lit_str(expr: &Expr) -> Option<&LitStr> {
56    match expr {
57        Expr::Lit(ExprLit {
58            lit: Lit::Str(s), ..
59        }) => Some(s),
60        _ => None,
61    }
62}
63
64/// Whether a cfg predicate holds.
65pub(crate) fn holds(pred: &Meta) -> bool {
66    let list = match pred {
67        Meta::Path(path) => return path.get_ident().is_some_and(|i| flag(&i.to_string())),
68        Meta::NameValue(nv) => {
69            return lit_str(&nv.value).is_some_and(|value| {
70                nv.path
71                    .get_ident()
72                    .is_some_and(|i| pair(&i.to_string(), &value.value()))
73            });
74        }
75        Meta::List(list) => list,
76    };
77    if list.path.is_ident("not") {
78        return !list.parse_args::<Meta>().is_ok_and(|inner| holds(&inner));
79    }
80    let inner = list.parse_args_with(Punctuated::<Meta, ::syn::token::CommaToken![,]>::parse_terminated);
81    if list.path.is_ident("all") {
82        inner.is_ok_and(|inner| inner.iter().all(holds))
83    } else if list.path.is_ident("any") {
84        inner.is_ok_and(|inner| inner.iter().any(holds))
85    } else {
86        false
87    }
88}

90/// A bare cfg name that is set.
91fn flag(name: &str) -> bool {
92    #[allow(non_exhaustive_omitted_patterns)] match name {
    "doc" | "doctest" | "unix" | "debug_assertions" => true,
    _ => false,
}matches!(name, "doc" | "doctest" | "unix" | "debug_assertions")
93}
94
95/// A `name = "value"` cfg that is set.
96fn pair(name: &str, value: &str) -> bool {
97    match name {
98        "feature" => true,
99        "target_os" => value == "linux",
100        "target_family" => value == "unix",
101        "target_arch" => value == "x86_64",
102        "target_pointer_width" => value == "64",
103        "target_endian" => value == "little",
104        "target_env" => value == "gnu",
105        "target_vendor" => value == "unknown",
106        "target_abi" => value.is_empty(),
107        "target_has_atomic" => #[allow(non_exhaustive_omitted_patterns)] match value {
    "8" | "16" | "32" | "64" | "ptr" => true,
    _ => false,
}matches!(value, "8" | "16" | "32" | "64" | "ptr"),
108        "target_feature" => #[allow(non_exhaustive_omitted_patterns)] match value {
    "fxsr" | "sse" | "sse2" => true,
    _ => false,
}matches!(value, "fxsr" | "sse" | "sse2"),
109        "panic" => value == "unwind",
110        _ => false,
111    }
112}

119    fn attrs(src: &str) -> Vec<syn::Attribute> {
120        syn::parse_str::<syn::ItemFn>(&alloc::format!("{src}\nfn f() {{}}"))
121            .unwrap()
122            .attrs
123    }

125    #[test]
126    fn rustdoc_flags_hold_and_test_does_not() {
127        assert!(!allows(&attrs("#[cfg(test)]")));
128        assert!(allows(&attrs("#[cfg(not(test))]")));
129        assert!(allows(&attrs("#[cfg(doc)]")));
130        assert!(allows(&attrs("#[cfg(doctest)]")));
131        assert!(!allows(&attrs("#[cfg(not(doc))]")));
132        assert!(allows(&attrs("#[cfg(debug_assertions)]")));
133    }
134
135    #[test]
136    fn every_feature_is_on() {
137        assert!(allows(&attrs("#[cfg(feature = \"x\")]")));
138        assert!(!allows(&attrs("#[cfg(not(feature = \"x\"))]")));
139        assert!(allows(&attrs("#[cfg(any(test, feature = \"x\"))]")));
140        assert!(!allows(&attrs(
141            "#[cfg(all(feature = \"x\", not(doctest)))]"
142        )));
143    }

146    fn the_host_is_linux_x86_64() {

148            "unix",
149            "target_os = \"linux\"",
150            "target_family = \"unix\"",
151            "target_arch = \"x86_64\"",
152            "target_pointer_width = \"64\"",
153            "target_endian = \"little\"",
154            "target_env = \"gnu\"",
155            "target_vendor = \"unknown\"",

202    #[test]
203    fn nested_cfg_attr_expands_in_place_when_its_predicate_holds() {
204        let names = |src: &str| -> Vec<alloc::string::String> {
205            attrs(src)
206                .iter()
207                .filter_map(cfg_attr_metas)
208                .flatten()
209                .map(|meta| meta.path().get_ident().unwrap().to_string())
210                .collect()
211        };
212        assert_eq!(
213            names(
214                "#[cfg_attr(unix, must_use, cfg_attr(feature = \"a\", doc = \"x\", cfg_attr(windows, cold)), inline)]"
215            ),
216            ["must_use", "doc", "inline"]
217        );
218        assert_eq!(
219            names("#[cfg_attr(windows, cfg_attr(unix, doc = \"x\"))]"),
220            Vec::<alloc::string::String>::new()
221        );
222        // A malformed `cfg_attr`, outer or nested, applies nothing.
223        assert_eq!(
224            names("#[cfg_attr]\n#[cfg_attr(unix, = x)]\n#[cfg_attr(unix, cfg_attr, inline)]"),
225            ["inline"]
226        );
227        assert!(cfg_attr_metas(&attrs("#[cfg_attr]")[0]).is_none());
228    }
