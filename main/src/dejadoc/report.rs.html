9/// Render the report for humans.
10#[must_use]
11pub fn human(report: &Report, verbose: bool) -> String {
12    let mut out = String::new();
13    let _ = (&mut out).write_fmt(format_args!("{0} doctests, {1} unique, {2} duplicated groups\n",
        report.total, report.unique, report.groups.len()))writeln!(
14        &mut out,
15        "{} doctests, {} unique, {} duplicated groups",
16        report.total,
17        report.unique,
18        report.groups.len()
19    );
20    if report.groups.is_empty() {
21        return out;
22    }
23    out.push('\n');
24    for (i, group) in report.groups.iter().enumerate() {
25        let _ = (&mut out).write_fmt(format_args!("[{0}] {1} sites, {2} tokens\n", group.id,
        group.sites.len(), group.tokens))writeln!(
26            &mut out,
27            "[{}] {} sites, {} tokens",
28            group.id,
29            group.sites.len(),
30            group.tokens
31        );
32        for site in &group.sites {
33            let info = if site.info.is_empty() {
34                String::new()
35            } else {
36                ::alloc::__export::must_use({
        ::alloc::fmt::format(format_args!("   ({0})", site.info.join(", ")))
    })format!("   ({})", site.info.join(", "))
37            };
38            let _ = (&mut out).write_fmt(format_args!("  {0}:{1}  {2}{3}\n", site.file.as_str(),
        site.line, site.item, info))writeln!(
39                &mut out,
40                "  {}:{}  {}{}",
41                site.file.as_str(),
42                site.line,
43                site.item,
44                info
45            );
46            if verbose {
47                for line in site.code.lines() {
48                    let _ = (&mut out).write_fmt(format_args!("    {0}\n", line))writeln!(&mut out, "    {line}");
49                }
50            }
51        }
52        if i + 1 < report.groups.len() {
53            out.push('\n');
54        }
55    }
56    out.push('\n');
57    out.push_str(
58        "Remove the copies listed above, or keep one on purpose with a `rust,dejadoc` fence\n",
59    );
60    out
61}

63/// Render the report as JSON.
64///
65/// # Panics
66///
67/// `Report` is `Serialize`. Serialization failure would be a bug in this
68/// crate.
69#[must_use]
70pub fn json(report: &Report) -> String {
71    serde_json::to_string(report).expect("Report serializes to JSON")
72}

74/// Render the report as GitHub workflow annotations, one per copy to
75/// remove. They need no token, so a fork pull request still shows them
76/// anchored in the diff.
77#[must_use]
78pub fn annotations(report: &Report, no_fail: bool) -> String {
79    let level = if no_fail { "warning" } else { "error" };
80    let mut out = String::new();
81    for group in &report.groups {
82        let Some((kept, copies)) = group.sites.split_first() else {
83            continue;
84        };
85        for copy in copies {
86            let _ = (&mut out).write_fmt(format_args!("::{7} file={0},line={1},title=dejadoc::{2} duplicates the doctest of {3} at {4}:{5}, group {6}\n",
        copy.file.as_str(), copy.line, escape(&copy.item), escape(&kept.item),
        escape(kept.file.as_str()), kept.line, group.id, level))writeln!(
87                &mut out,
88                "::{level} file={},line={},title=dejadoc::{} duplicates the doctest of {} at {}:{}, group {}",
89                copy.file.as_str(),
90                copy.line,
91                escape(&copy.item),
92                escape(&kept.item),
93                escape(kept.file.as_str()),
94                kept.line,
95                group.id
96            );
97        }
98    }
99    out
100}

102/// A workflow command message, with the characters that would end it or
103/// its properties percent encoded.
104fn escape(text: &str) -> String {
105    text.replace('%', "%25")
106        .replace('\r', "%0D")
107        .replace('\n', "%0A")
108}
