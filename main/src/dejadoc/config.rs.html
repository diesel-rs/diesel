7/// Parameters from a `.dejadoc.toml` file.
8pub(crate) struct Config {
9    /// Report groups with at least this many sites.
10    pub(crate) threshold: Option<usize>,
11    /// Skip blocks with fewer tokens than this.
12    #[serde(rename = "min-tokens")]
13    pub(crate) min_tokens: Option<usize>,
14    /// Whether the duplicate function check runs.
15    pub(crate) functions: Option<bool>,
16    /// Skip functions whose body counts fewer tokens than this.
17    #[serde(rename = "fn-min-tokens")]
18    pub(crate) fn_min_tokens: Option<usize>,
19    /// Scan files marked generated as well.
20    #[serde(rename = "scan-generated")]
21    pub(crate) scan_generated: Option<bool>,
22    /// More phrases that mark a file generated in its first lines.
23    #[serde(rename = "generated-markers")]
24    pub(crate) generated_markers: Option<alloc::vec::Vec<alloc::string::String>>,
25}

27/// Load `path`, or the default config when the file is absent.
28///
29/// # Errors
30///
31/// Fails when the file exists but cannot be read or is not valid TOML.
32pub(crate) fn load(path: &Path) -> crate::Result<Config> {
33    if !path.exists() {
34        return Ok(Config::default());
35    }
36    let text = std::fs::read_to_string(path)?;
37    Ok(toml::from_str(&text)?)
38}

52    fn loads_parameters() {
53        let dir = tempfile::tempdir().unwrap();
54        let path = dir.path().join(".dejadoc.toml");
55        std::fs::write(
56            &path,
57            "threshold = 3\nmin-tokens = 4\nfunctions = false\nfn-min-tokens = 50\nscan-generated = true\ngenerated-markers = [\"by rust-bindgen\"]\n",
58        )
59        .unwrap();
60        let cfg = load(&path).unwrap();
61        assert_eq!(cfg.threshold, Some(3));
62        assert_eq!(cfg.min_tokens, Some(4));
63        assert_eq!(cfg.functions, Some(false));
64        assert_eq!(cfg.fn_min_tokens, Some(50));
65        assert_eq!(cfg.scan_generated, Some(true));
66        assert_eq!(
67            cfg.generated_markers,
68            Some(alloc::vec!["by rust-bindgen".to_string()])
69        );
70    }
