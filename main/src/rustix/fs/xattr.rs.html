30///
31/// Unlike the libc `getxattr` function, this function does not treat an
32/// buffer with zero length as a way to return the size of the attribute.
33/// To obtain the size, use [`getxattr_query`].
34///
35/// For a higher-level API to xattr functionality, see the [xattr] crate.
36///
37/// [xattr]: https://crates.io/crates/xattr
38///
39/// # References
40///  - [Linux]
41///  - [Apple]
42///
43/// [Linux]: https://man7.org/linux/man-pages/man2/getxattr.2.html
44/// [Apple]: https://developer.apple.com/library/archive/documentation/System/Conceptual/ManPages_iPhoneOS/man2/getxattr.2.html
45#[inline]
46pub fn getxattr<P: path::Arg, Name: path::Arg, Buf: Buffer<u8>>(
47    path: P,
48    name: Name,
49    mut value: Buf,
50) -> io::Result<Buf::Output> {
51    path.into_with_c_str(|path| {
52        name.into_with_c_str(|name| {
53            let parts = value.parts_mut();
54            // SAFETY: `getxattr` behaves.
55            let len = unsafe { backend::fs::syscalls::getxattr(path, name, parts)? };
56            if parts.1 == 0 {
57                // `getxattr` special-cases an input size of 0 as a size
58                // query, which we handle through a different API function.
59                unsafe { Ok(value.assume_init(0)) }
60            } else {
61                // SAFETY: `getxattr` behaves, and we've handled the special
62                // case of an empty buffer above.
63                unsafe { Ok(value.assume_init(len)) }
64            }
65        })
66    })
67}

408/// the list. This can be used to determine the size of a buffer to
409/// allocate for a subsequent call to [`flistxattr`], however keep in mind
410/// that other processes may modify attributes at any time.
411///
412/// # References
413///  - [Linux]
414///
415/// [Linux]: https://man7.org/linux/man-pages/man2/flistxattr.2.html
416#[inline]
417pub fn flistxattr_query<Fd: AsFd>(fd: Fd) -> io::Result<usize> {
418    // SAFETY: `flistxattr` behaves.
419    unsafe { backend::fs::syscalls::flistxattr(fd.as_fd(), (null_mut(), 0)) }
420}

422/// `removexattr(path, name)`—Remove an extended filesystem attribute.
423///
424/// # References
425///  - [Linux]
426///  - [Apple]
427///
428/// [Linux]: https://man7.org/linux/man-pages/man2/removexattr.2.html

431    path.into_with_c_str(|path| {
