18impl Routine<()> for ChangeUser {
19    fn call<'a>(self, conn: &'a mut Conn) -> BoxFuture<'a, crate::Result<()>>
20    where
21        Self: 'a,
22    {
23        #[cfg(feature = "tracing")]
24        let span = debug_span!(
25            "mysql_async::change_user",
26            mysql_async.connection.id = conn.id()
27        );
28
29        let com_change_user = ComChangeUser::new()
30            .with_user(conn.opts().user().map(|x| x.as_bytes()))
31            .with_database(conn.opts().db_name().map(|x| x.as_bytes()))
32            .with_auth_plugin_data(
33                conn.inner
34                    .auth_plugin
35                    .gen_data(conn.opts().pass(), &conn.inner.nonce)
36                    .as_deref(),
37            )
38            .with_more_data(Some(
39                ComChangeUserMoreData::new(if conn.inner.version >= (5, 5, 3) {
40                    UTF8MB4_GENERAL_CI
41                } else {
42                    UTF8_GENERAL_CI
43                })
44                .with_auth_plugin(Some(conn.inner.auth_plugin.clone()))
45                .with_connect_attributes(conn.opts().connect_attributes().cloned()),
46            ))
47            .into_owned();
48
49        let fut = async move {
50            conn.write_command(&com_change_user).await?;
51            conn.inner.auth_switched = false;
52            conn.continue_auth().await?;
53            Ok(())
54        };
55
56        #[cfg(feature = "tracing")]
57        let fut = instrument_result!(fut, span);
