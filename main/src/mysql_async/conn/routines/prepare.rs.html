27impl Routine<Arc<StmtInner>> for PrepareRoutine {
28    fn call<'a>(self, conn: &'a mut Conn) -> BoxFuture<'a, crate::Result<Arc<StmtInner>>>
29    where
30        Self: 'a,
31    {
32        #[cfg(feature = "tracing")]
33        let span = info_span!(
34            "mysql_async::prepare",
35            mysql_async.connection.id = conn.id(),
36            mysql_async.statement.id = field::Empty,
37            mysql_async.query.sql = field::Empty,
38        );
39        #[cfg(feature = "tracing")]
40        if tracing::span_enabled!(Level::DEBUG) {
41            // The statement may contain sensitive data. Restrict to DEBUG.
42            span.record(
43                "mysql_async.query.sql",
44                String::from_utf8_lossy(&self.query).as_ref(),
45            );
46        }
47
48        let fut = async move {
49            conn.write_command_data(Command::COM_STMT_PREPARE, &self.query)
50                .await?;
51
52            let packet = conn.read_packet().await?;
53            let mut inner_stmt = StmtInner::from_payload(&packet, conn.id(), self.query.clone())?;
54
55            #[cfg(feature = "tracing")]
56            Span::current().record("mysql_async.statement.id", inner_stmt.id());
57
58            if inner_stmt.num_params() > 0 {
59                let params = conn.read_column_defs(inner_stmt.num_params()).await?;
60                inner_stmt = inner_stmt.with_params(params);
61            }
62
63            if inner_stmt.num_columns() > 0 {
64                let columns = conn.read_column_defs(inner_stmt.num_columns()).await?;
65                inner_stmt = inner_stmt.with_columns(columns);
66            }
67
68            Ok(Arc::new(inner_stmt))
69        };
70
71        #[cfg(feature = "tracing")]
72        let fut = instrument_result!(fut, span);
73
74        fut.boxed()
75    }
76}
