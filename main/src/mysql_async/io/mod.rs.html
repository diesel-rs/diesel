53    ($e:expr) => {
54        loop {
55            match $e {

57                x => break x,
58            }
59        }

96impl Decoder for PacketCodec {
97    type Item = PooledBuf;
98    type Error = IoError;
99
100    fn decode(&mut self, src: &mut BytesMut) -> std::result::Result<Option<Self::Item>, IoError> {
101        if self.inner.decode(src, self.decode_buf.as_mut())? {
102            let new_buf = crate::buffer_pool().get();
103            Ok(Some(replace(&mut self.decode_buf, new_buf)))
104        } else {
105            Ok(None)
106        }
107    }
108}

114        Ok(self.inner.encode(&mut item.as_ref(), dst)?)
115    }

129/// This future will check that TcpStream is live.
130///
131/// This check is similar to a one, implemented by GitHub team for the go-sql-driver/mysql.
132#[derive(#[automatically_derived]impl<'a> ::core::fmt::Debug for CheckTcpStream<'a> {
    #[inline]
    #[rustc_trivial_field_reads]
    fn fmt(&self, f: &mut ::core::fmt::Formatter) -> ::core::fmt::Result {
        ::core::fmt::Formatter::debug_struct_field1_finish(f,
            "CheckTcpStream", &&self.0)
    }
}Debug)]
133struct CheckTcpStream<'a>(&'a mut TcpStream);
134
135impl Future for CheckTcpStream<'_> {
136    type Output = io::Result<()>;
137    fn poll(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
138        match self.0.poll_read_ready(cx) {
139            Poll::Ready(Ok(())) => {
140                // stream is readable
141                let mut buf = [0_u8; 1];
142                match self.0.try_read(&mut buf) {
143                    Ok(0) => Poll::Ready(Err(io::Error::new(BrokenPipe, "broken pipe"))),
144                    Ok(_) => Poll::Ready(Err(io::Error::new(Other, "stream should be empty"))),
145                    Err(err) if err.kind() == io::ErrorKind::WouldBlock => Poll::Ready(Ok(())),
146                    Err(err) => Poll::Ready(Err(err)),
147                }
148            }
149            Poll::Ready(Err(err)) => Poll::Ready(Err(err)),
150            Poll::Pending => Poll::Ready(Ok(())),
151        }
152    }
153}

155impl Endpoint {
156    #[cfg(unix)]
157    fn is_socket(&self) -> bool {
158        #[allow(non_exhaustive_omitted_patterns)] match self {
    Self::Socket(_) => true,
    _ => false,
}matches!(self, Self::Socket(_))
159    }
160
161    /// Checks, that connection is alive.
162    async fn check(&mut self) -> std::result::Result<(), IoError> {
163        //return Ok(());
164        match self {
165            Endpoint::Plain(Some(stream)) => {
166                CheckTcpStream(stream).await?;
167                Ok(())
168            }
169            #[cfg(feature = "native-tls-tls")]
170            Endpoint::Secure(tls_stream) => {
171                CheckTcpStream(tls_stream.get_mut().get_mut().get_mut()).await?;
172                Ok(())
173            }
174            #[cfg(feature = "rustls-tls")]
175            Endpoint::Secure(tls_stream) => {
176                let stream = tls_stream.get_mut().0;
177                CheckTcpStream(stream).await?;
178                Ok(())
179            }
180            #[cfg(unix)]
181            Endpoint::Socket(socket) => {
182                let _ = socket.write(&[]).await?;
183                Ok(())
184            }
185            Endpoint::Plain(None) => ::core::panicking::unreachable_default_msg()unreachable!(),
186        }
187    }
188
189    #[cfg(any(feature = "native-tls-tls", feature = "rustls-tls"))]
190    pub fn is_secure(&self) -> bool {
191        matches!(self, Endpoint::Secure(_))
192    }
193
194    pub fn set_tcp_nodelay(&self, val: bool) -> io::Result<()> {
195        match *self {
196            Endpoint::Plain(Some(ref stream)) => stream.set_nodelay(val)?,
197            Endpoint::Plain(None) => ::core::panicking::unreachable_default_msg()unreachable!(),
198            #[cfg(feature = "native-tls-tls")]
199            Endpoint::Secure(ref stream) => {
200                stream.get_ref().get_ref().get_ref().set_nodelay(val)?
201            }
202            #[cfg(feature = "rustls-tls")]
203            Endpoint::Secure(ref stream) => {
204                let stream = stream.get_ref().0;
205                stream.set_nodelay(val)?;
206            }
207            #[cfg(unix)]
208            Endpoint::Socket(_) => (/* inapplicable */),
209        }
210        Ok(())
211    }
212}

238impl AsyncRead for Endpoint {
239    fn poll_read(
240        self: Pin<&mut Self>,
241        cx: &mut Context<'_>,
242        buf: &mut ReadBuf<'_>,
243    ) -> Poll<std::result::Result<(), tokio::io::Error>> {
244        let this = self.get_mut();
245        loop {
    match match this {
            Self::Plain(stream) => {
                Pin::new(stream.as_mut().unwrap()).poll_read(cx, buf)
            }Self::Socket(stream) => Pin::new(stream).poll_read(cx, buf),
        } {
        Poll::Ready(Err(err)) if err.kind() == Interrupted => continue,
        x => break x,
    }
}with_interrupted!(match this {
246            Self::Plain(stream) => {
247                Pin::new(stream.as_mut().unwrap()).poll_read(cx, buf)
248            }
249            #[cfg(feature = "native-tls-tls")]
250            Self::Secure(stream) => Pin::new(stream).poll_read(cx, buf),
251            #[cfg(feature = "rustls-tls")]
252            Self::Secure(stream) => Pin::new(stream).poll_read(cx, buf),
253            #[cfg(unix)]
254            Self::Socket(stream) => Pin::new(stream).poll_read(cx, buf),
255        })
256    }
257}

343    pub(crate) async fn connect_tcp(
344        addr: &HostPortOrUrl,
345        keepalive: Option<Duration>,
346    ) -> io::Result<Stream> {
347        let tcp_stream = match addr {
348            HostPortOrUrl::HostPort {
349                host,
350                port,
351                resolved_ips,
352            } => match resolved_ips {
353                Some(ips) => {
354                    let addrs = ips
355                        .iter()
356                        .map(|ip| SocketAddr::new(*ip, *port))
357                        .collect::<Vec<_>>();
358                    TcpStream::connect(&*addrs).await?
359                }
360                None => TcpStream::connect((host.as_str(), *port)).await?,
361            },
362            HostPortOrUrl::Url(url) => {
363                let addrs = url.socket_addrs(|| Some(DEFAULT_PORT))?;
364                TcpStream::connect(&*addrs).await?
365            }
366        };
367
368        #[cfg(any(unix, windows))]
369        if let Some(duration) = keepalive {
370            socket2::SockRef::from(&tcp_stream)
371                .set_tcp_keepalive(&socket2::TcpKeepalive::new().with_time(duration))?;
372        }
373
374        Ok(Stream {
375            closed: false,
376            codec: Box::new(Framed::new(tcp_stream.into(), PacketCodec::default())).into(),
377        })
378    }

389    pub(crate) async fn make_secure(
390        &mut self,
391        domain: String,
392        tls_connector: &TlsConnector,
393    ) -> crate::error::Result<()> {
394        let codec = self.codec.take().unwrap();
395        let FramedParts { mut io, codec, .. } = codec.into_parts();
396        io.make_secure(domain, tls_connector).await?;
397        let codec = Framed::new(io, codec);
398        self.codec = Some(Box::new(codec));
399        Ok(())
400    }

452    pub(crate) async fn close(mut self) -> std::result::Result<(), IoError> {
453        self.closed = true;
454        if let Some(mut codec) = self.codec {
455            use futures_sink::Sink;
456            std::future::poll_fn(|cx| match Pin::new(&mut *codec).poll_close(cx) {
457                Poll::Ready(Err(IoError::Io(err))) if err.kind() == NotConnected => {
458                    Poll::Ready(Ok(()))
459                }
460                x => x,
461            })
462            .await?;
463        }
464        Ok(())
465    }

472        if !self.closed {
473            let item = match Pin::new(self.codec.as_mut().unwrap()).poll_next(cx) {
    ::futures_core::task::Poll::Ready(t) => t,
    ::futures_core::task::Poll::Pending =>
        return ::futures_core::task::Poll::Pending,
}ready!(Pin::new(self.codec.as_mut().unwrap()).poll_next(cx)).transpose()?;
474            Poll::Ready(Ok(item).transpose())
475        } else {
476            Poll::Ready(None)
477        }
478    }

485    #[cfg(unix)] // no sane way to retrieve current keepalive value on windows
486    #[tokio::test]
487    async fn should_connect_with_keepalive() {
488        use crate::{test_misc::get_opts, Conn};
489
490        let duration = Duration::from_millis(42_000);
491        let opts = get_opts()
492            .tcp_keepalive(Some(duration))
493            .prefer_socket(false);
494        let mut conn: Conn = Conn::new(opts).await.unwrap();
495        let stream = conn.stream_mut().unwrap();
496        let endpoint = stream.codec.as_mut().unwrap().get_ref();
497        let stream = match endpoint {
498            super::Endpoint::Plain(Some(stream)) => stream,
499            #[cfg(feature = "rustls-tls")]
500            super::Endpoint::Secure(tls_stream) => tls_stream.get_ref().0,
501            #[cfg(feature = "native-tls-tls")]
502            super::Endpoint::Secure(tls_stream) => tls_stream.get_ref().get_ref().get_ref(),
503            _ => unreachable!(),
504        };
505        let sock = unsafe {
506            use std::os::unix::prelude::*;
507            let raw = stream.as_raw_fd();
508            socket2::Socket::from_raw_fd(raw)
509        };
510
511        assert_eq!(sock.tcp_keepalive_time().unwrap(), duration);
512
513        std::mem::forget(sock);
514
515        conn.disconnect().await.unwrap();
516    }
