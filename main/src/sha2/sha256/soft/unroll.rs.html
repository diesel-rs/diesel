95#[inline(always)]
96fn load_round<const R: usize>(state: &mut [u32; 8], block: &[u8; 64], msg: &mut [u32; 16]) {
97    let chunk = block.as_chunks().0[R];
98    let w = u32::from_be_bytes(chunk);
99    msg[R] = w;
100    round::<R>(state, w);
101}

103#[inline(always)]
104fn schedule_round<const R: usize>(state: &mut [u32; 8], msg: &mut [u32; 16]) {
105    let w = msg[R % 16]
106        .wrapping_add(sig1(msg[(R + 14) % 16]))
107        .wrapping_add(msg[(R + 9) % 16])
108        .wrapping_add(sig0(msg[(R + 1) % 16]));
109    msg[R % 16] = w;
110    round::<R>(state, w);
111}

113#[inline(always)]
114fn round<const R: usize>(state: &mut [u32; 8], w: u32) {
115    let [a, b, c, d, e, f, g, h] = *state;
116
117    let t1 = sum1(e)
118        .wrapping_add(load_rk::<R>())
119        .wrapping_add(ch(e, f, g))
120        .wrapping_add(w)
121        .wrapping_add(h);
122    let t2 = sum0(a).wrapping_add(maj(a, b, c));
123
124    *state = [t1.wrapping_add(t2), a, b, c, d.wrapping_add(t1), e, f, g];
125}

127#[inline(always)]
128fn load_rk<const R: usize>() -> u32 {
129    use crate::consts::SHA256_RK;
130    const { if !(R < SHA256_RK.len()) {
    ::core::panicking::panic("assertion failed: R < SHA256_RK.len()")
}assert!(R < SHA256_RK.len()) };
131
132    // `read_volatile` forces the compiler to read round constants from the static
133    // instead of unnecessarily inlining values or materializaing them on the stack.
134    // It generally results in a smaller binary and improves performance.
135    #[cfg(not(any(target_arch = "x86", target_arch = "x86_64")))]
136    unsafe {
137        core::ptr::read_volatile(SHA256_RK.as_ptr().add(R))
138    }
139
140    // On x86 targets 32-bit constants can be encoded using immediate argument on the `add`
141    // instruction, so it's more efficient to inline them.
142    #[cfg(any(target_arch = "x86", target_arch = "x86_64"))]
143    unsafe {
144        core::ptr::read(SHA256_RK.as_ptr().add(R))
145    }
