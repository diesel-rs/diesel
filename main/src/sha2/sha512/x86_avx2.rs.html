11#[target_feature(enable = "avx2")]
12pub(super) fn compress(state: &mut [u64; 8], blocks: &[[u8; 128]]) {
13    let mut ms: MsgSchedule = [_mm_setzero_si128(); 8];
14    let mut t2: RoundStates = [_mm_setzero_si128(); 40];
15    let mut x = [_mm256_setzero_si256(); 8];
16
17    let (head, block_chunks) = blocks.as_rchunks::<2>();
18
19    if let [block] = head {
20        sha512_compress_x86_64_avx(state, block);
21    }
22
23    for block_chunk in block_chunks {
24        load_data_avx2(&mut x, &mut ms, &mut t2, block_chunk);
25
26        // First block
27        let mut current_state = *state;
28        rounds_0_63_avx2(&mut current_state, &mut x, &mut ms, &mut t2);
29        rounds_64_79(&mut current_state, &ms);
30        accumulate_state(state, &current_state);
31
32        // Second block
33        current_state = *state;
34        process_second_block(&mut current_state, &t2);
35        accumulate_state(state, &current_state);
36    }
37}

39#[inline]
40#[target_feature(enable = "avx")]
41fn sha512_compress_x86_64_avx(state: &mut [u64; 8], block: &[u8; 128]) {
42    let mut ms = [_mm_setzero_si128(); 8];
43    let mut x = [_mm_setzero_si128(); 8];
44
45    // Reduced to single iteration
46    let mut current_state = *state;
47    load_data_avx(&mut x, &mut ms, block.as_ptr().cast());
48    rounds_0_63_avx(&mut current_state, &mut x, &mut ms);
49    rounds_64_79(&mut current_state, &ms);
50    accumulate_state(state, &current_state);
51}

55fn load_data_avx(x: &mut [__m128i; 8], ms: &mut MsgSchedule, data: *const __m128i) {
56    #[allow(non_snake_case)]
57    let MASK = _mm_setr_epi32(0x04050607, 0x00010203, 0x0c0d0e0f, 0x08090a0b);
58
59    macro_rules! unrolled_iterations {
60        ($($i:literal),*) => {$(
61            x[$i] = unsafe { _mm_loadu_si128(data.add($i).cast()) };
62            x[$i] = _mm_shuffle_epi8(x[$i], MASK);
63
64            let y = _mm_add_epi64(x[$i], rk(2 * $i));
65
66            ms[$i] = y;
67        )*};
68    }
69
70    x[0] = unsafe { _mm_loadu_si128(data.add(0).cast()) };
x[0] = _mm_shuffle_epi8(x[0], MASK);
let y = _mm_add_epi64(x[0], rk(2 * 0));
ms[0] = y;
x[1] = unsafe { _mm_loadu_si128(data.add(1).cast()) };
x[1] = _mm_shuffle_epi8(x[1], MASK);
let y = _mm_add_epi64(x[1], rk(2 * 1));
ms[1] = y;
x[2] = unsafe { _mm_loadu_si128(data.add(2).cast()) };
x[2] = _mm_shuffle_epi8(x[2], MASK);
let y = _mm_add_epi64(x[2], rk(2 * 2));
ms[2] = y;
x[3] = unsafe { _mm_loadu_si128(data.add(3).cast()) };
x[3] = _mm_shuffle_epi8(x[3], MASK);
let y = _mm_add_epi64(x[3], rk(2 * 3));
ms[3] = y;
x[4] = unsafe { _mm_loadu_si128(data.add(4).cast()) };
x[4] = _mm_shuffle_epi8(x[4], MASK);
let y = _mm_add_epi64(x[4], rk(2 * 4));
ms[4] = y;
x[5] = unsafe { _mm_loadu_si128(data.add(5).cast()) };
x[5] = _mm_shuffle_epi8(x[5], MASK);
let y = _mm_add_epi64(x[5], rk(2 * 5));
ms[5] = y;
x[6] = unsafe { _mm_loadu_si128(data.add(6).cast()) };
x[6] = _mm_shuffle_epi8(x[6], MASK);
let y = _mm_add_epi64(x[6], rk(2 * 6));
ms[6] = y;
x[7] = unsafe { _mm_loadu_si128(data.add(7).cast()) };
x[7] = _mm_shuffle_epi8(x[7], MASK);
let y = _mm_add_epi64(x[7], rk(2 * 7));
ms[7] = y;unrolled_iterations!(0, 1, 2, 3, 4, 5, 6, 7);

112#[target_feature(enable = "avx")]
113fn rounds_0_63_avx(current_state: &mut State, x: &mut [__m128i; 8], ms: &mut MsgSchedule) {
114    let mut k64_idx: usize = SHA512_BLOCK_WORDS_NUM;
115
116    for _ in 0..4 {
117        for j in 0..8 {
118            let y = sha512_update_x_avx(x, rk(k64_idx));
119
120            {
121                let ms = cast_ms(ms);
122                sha_round(current_state, ms[2 * j]);
123                sha_round(current_state, ms[2 * j + 1]);
124            }
125
126            ms[j] = y;
127            k64_idx += 2;
128        }
129    }
130}

133#[target_feature(enable = "avx2")]
134fn rounds_0_63_avx2(
135    current_state: &mut State,
136    x: &mut [__m256i; 8],
137    ms: &mut MsgSchedule,
138    t2: &mut RoundStates,
139) {
140    let mut k64x4_idx: usize = SHA512_BLOCK_WORDS_NUM;
141
142    for i in 1..5 {
143        for j in 0..8 {
144            let t = rk(k64x4_idx);
145            let y = sha512_update_x_avx2(x, _mm256_set_m128i(t, t));
146
147            {
148                let ms = cast_ms(ms);
149                sha_round(current_state, ms[2 * j]);
150                sha_round(current_state, ms[2 * j + 1]);
151            }
152
153            ms[j] = _mm256_extracti128_si256(y, 0);
154            t2[8 * i + j] = _mm256_extracti128_si256(y, 1);
155
156            k64x4_idx += 2;
157        }
158    }

162fn rounds_64_79(current_state: &mut State, ms: &MsgSchedule) {
163    let ms = cast_ms(ms);
164    for i in 64..80 {
165        sha_round(current_state, ms[i & 0xf]);

176#[inline(always)]
177fn sha_round(s: &mut State, x: u64) {
178    macro_rules! big_sigma0 {
179        ($a:expr) => {
180            $a.rotate_right(28) ^ $a.rotate_right(34) ^ $a.rotate_right(39)
181        };
182    }
183    macro_rules! big_sigma1 {
184        ($a:expr) => {
185            $a.rotate_right(14) ^ $a.rotate_right(18) ^ $a.rotate_right(41)
186        };
187    }
188    macro_rules! bool3ary_202 {
189        ($a:expr, $b:expr, $c:expr) => {
190            $c ^ ($a & ($b ^ $c))
191        };
192    } // Choose, MD5F, SHA1C
193    macro_rules! bool3ary_232 {
194        ($a:expr, $b:expr, $c:expr) => {
195            ($a & $b) ^ ($a & $c) ^ ($b & $c)
196        };
197    } // Majority, SHA1M
198
199    macro_rules! rotate_state {
200        ($s:ident) => {{
201            let tmp = $s[7];
202            $s[7] = $s[6];
203            $s[6] = $s[5];
204            $s[5] = $s[4];
205            $s[4] = $s[3];
206            $s[3] = $s[2];
207            $s[2] = $s[1];
208            $s[1] = $s[0];
209            $s[0] = tmp;
210        }};
211    }
212
213    let t = x
214        .wrapping_add(s[7])
215        .wrapping_add(s[4].rotate_right(14) ^ s[4].rotate_right(18) ^ s[4].rotate_right(41)big_sigma1!(s[4]))
216        .wrapping_add(s[6] ^ (s[4] & (s[5] ^ s[6]))bool3ary_202!(s[4], s[5], s[6]));
217
218    s[7] = t
219        .wrapping_add(s[0].rotate_right(28) ^ s[0].rotate_right(34) ^ s[0].rotate_right(39)big_sigma0!(s[0]))
220        .wrapping_add((s[0] & s[1]) ^ (s[0] & s[2]) ^ (s[1] & s[2])bool3ary_232!(s[0], s[1], s[2]));
221    s[3] = s[3].wrapping_add(t);
222
223    {
    let tmp = s[7];
    s[7] = s[6];
    s[6] = s[5];
    s[5] = s[4];
    s[4] = s[3];
    s[3] = s[2];
    s[2] = s[1];
    s[1] = s[0];
    s[0] = tmp;
};rotate_state!(s);
224}

233macro_rules! fn_sha512_update_x {
234    ($name:ident, $tf:literal, $ty:ident, {
235        ADD64 = $ADD64:ident,
236        ALIGNR8 = $ALIGNR8:ident,
237        SRL64 = $SRL64:ident,
238        SLL64 = $SLL64:ident,
239        XOR = $XOR:ident,
240    }) => {
241        #[inline]
242        #[target_feature(enable = $tf)]
243        fn $name(x: &mut [$ty; 8], k64: $ty) -> $ty {
244            // q[2:1]
245            let mut t0 = $ALIGNR8(x[1], x[0], 8);
246            // q[10:9]
247            let mut t3 = $ALIGNR8(x[5], x[4], 8);
248            // q[2:1] >> s0[0]
249            let mut t2 = $SRL64(t0, 1);
250            // q[1:0] + q[10:9]
251            x[0] = $ADD64(x[0], t3);
252            // q[2:1] >> s0[2]
253            t3 = $SRL64(t0, 7);
254            // q[2:1] << (64 - s0[1])
255            let mut t1 = $SLL64(t0, 64 - 8);
256            // (q[2:1] >> s0[2]) ^
257            // (q[2:1] >> s0[0])
258            t0 = $XOR(t3, t2);
259            // q[2:1] >> s0[1]
260            t2 = $SRL64(t2, 8 - 1);
261            // (q[2:1] >> s0[2]) ^
262            // (q[2:1] >> s0[0]) ^
263            // q[2:1] << (64 - s0[1])
264            t0 = $XOR(t0, t1);
265            // q[2:1] << (64 - s0[0])
266            t1 = $SLL64(t1, 8 - 1);
267            // sigma1(q[2:1])
268            t0 = $XOR(t0, t2);
269            t0 = $XOR(t0, t1);
270            // q[15:14] >> s1[2]
271            t3 = $SRL64(x[7], 6);
272            // q[15:14] >> (64 - s1[1])
273            t2 = $SLL64(x[7], 64 - 61);
274            // q[1:0] + sigma0(q[2:1])
275            x[0] = $ADD64(x[0], t0);
276            // q[15:14] >> s1[0]
277            t1 = $SRL64(x[7], 19);
278            // q[15:14] >> s1[2] ^
279            // q[15:14] >> (64 - s1[1])
280            t3 = $XOR(t3, t2);
281            // q[15:14] >> (64 - s1[0])
282            t2 = $SLL64(t2, 61 - 19);
283            // q[15:14] >> s1[2] ^
284            // q[15:14] >> (64 - s1[1] ^
285            // q[15:14] >> s1[0]
286            t3 = $XOR(t3, t1);
287            // q[15:14] >> s1[1]
288            t1 = $SRL64(t1, 61 - 19);
289            // sigma1(q[15:14])
290            t3 = $XOR(t3, t2);
291            t3 = $XOR(t3, t1);
292
293            // q[1:0] + q[10:9] + sigma1(q[15:14]) + sigma0(q[2:1])
294            x[0] = $ADD64(x[0], t3);
295
296            // rotate
297            let temp = x[0];
298            x[0] = x[1];
299            x[1] = x[2];
300            x[2] = x[3];
301            x[3] = x[4];
302            x[4] = x[5];
303            x[5] = x[6];
304            x[6] = x[7];
305            x[7] = temp;
306
307            $ADD64(x[7], k64)
308        }
309    };
310}

    t3 = _mm_srli_epi64(t0, 7);

    t2 = _mm_srli_epi64(t2, 8 - 1);
    t0 = _mm_xor_si128(t0, t1);
    t1 = _mm_slli_epi64(t1, 8 - 1);
    t0 = _mm_xor_si128(t0, t2);
    t0 = _mm_xor_si128(t0, t1);
    t3 = _mm_srli_epi64(x[7], 6);
    t2 = _mm_slli_epi64(x[7], 64 - 61);
    x[0] = _mm_add_epi64(x[0], t0);
    t1 = _mm_srli_epi64(x[7], 19);
    t3 = _mm_xor_si128(t3, t2);
    t2 = _mm_slli_epi64(t2, 61 - 19);
    t3 = _mm_xor_si128(t3, t1);
    t1 = _mm_srli_epi64(t1, 61 - 19);
    t3 = _mm_xor_si128(t3, t2);
    t3 = _mm_xor_si128(t3, t1);
    x[0] = _mm_add_epi64(x[0], t3);
