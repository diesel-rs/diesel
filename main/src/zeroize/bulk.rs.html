19/// Fill a slice using an assembly optimization barrier or portable volatile writes.
20#[inline]
21pub(crate) fn fill_slice<T: Copy>(dst: &mut [T], src: T) {
22    if size_of::<T>() == 0 {
23        return;
24    }
25
26    if HAS_ASM {
27        dst.fill(src);
28        optimization_barrier(dst);
29    } else {
30        volatile_fill_slice(dst, src);
31    }
32}

34/// Zero bytes which may be uninitialized, including padding.
35///
36/// # Safety
37/// `dst` must be non-NULL and provide exclusive access to `len` writable bytes within a single
38/// allocation.
39///
40/// The size must not exceed `isize::MAX` and the address range must not wrap. If zeroes are not a
41/// valid bit pattern for the underlying type, the caller must restore a valid value before any
42/// typed access or drop.
43#[inline]
44pub(crate) unsafe fn zeroize_byte_ptr(dst: *mut u8, len: usize) {
45    if HAS_ASM {
46        // SAFETY: the caller supplies exclusive access to `len` writable bytes.
47        // `write_bytes` initializes every byte before constructing the slice.
48        unsafe {
49            core::ptr::write_bytes(dst, 0, len);
50            optimization_barrier(slice::from_raw_parts(dst, len));
51        }
52    } else {
53        use core::mem::MaybeUninit;
54
55        // SAFETY: the caller supplies exclusive access to this range. `MaybeUninit<u8>` permits
56        // uninitialized bytes and has byte alignment.
57        let bytes = unsafe { slice::from_raw_parts_mut(dst.cast::<MaybeUninit<u8>>(), len) };
58        volatile_fill_slice(bytes, MaybeUninit::new(0));
59    }
60}

62/// Fill a slice with a repeated value, preserving its type and alignment.
63#[inline(always)]
64fn volatile_fill_slice<T: Copy>(dst: &mut [T], src: T) {
65    // Target 16-byte chunks
66    match size_of::<T>() {
67        0 => {}
68        1 => volatile_fill_chunked::<_, 16>(dst, src),
69        2 => volatile_fill_chunked::<_, 8>(dst, src),
70        4 => volatile_fill_chunked::<_, 4>(dst, src),
71        8 => volatile_fill_chunked::<_, 2>(dst, src),
72        _ => volatile_fill_iterative(dst, src),
73    }
74}
75
76/// Prepare the repeated pattern and handle the tail independently of the bulk backend.
77#[inline(always)]
78fn volatile_fill_chunked<T: Copy, const N: usize>(dst: &mut [T], src: T) {
79    let (chunks, tail) = as_chunks_mut::<T, N>(dst);
80
81    if !chunks.is_empty() {
82        volatile_fill_iterative(chunks, [src; N]);
83    }
84
85    volatile_fill_iterative(tail, src);
86}
87
88/// Write individual elements or complete chunks with the same volatile primitive.
89#[inline(always)]
90fn volatile_fill_iterative<T: Copy>(dst: &mut [T], src: T) {
91    for item in dst {
92        volatile_write(item, src);
93    }
94}

96/// MSRV-compatible copy of core's `slice::as_chunks_mut` taken from Rust 1.88.0.

99#[inline]
100#[track_caller]
101#[must_use]

103    if !(N != 0) {

    }
};assert!(N != 0, "chunk size must be non-zero");
104    let len_rounded_down = slice.len() / N * N;
105    // SAFETY: The rounded-down value is always the same or smaller than the
106    // original length, and thus must be in-bounds of the slice.
107    let (multiple_of_n, remainder) = unsafe { slice.split_at_mut_unchecked(len_rounded_down) };
108    // SAFETY: We already panicked for zero, and ensured by construction
109    // that the length of the subslice is a multiple of N. Arrays have the same alignment as T.
110    let array_slice = unsafe {
111        slice::from_raw_parts_mut(multiple_of_n.as_mut_ptr().cast(), multiple_of_n.len() / N)
112    };
113    (array_slice, remainder)
