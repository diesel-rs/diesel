22/// Trait used by the deserializer for iterating over input. This is manually
23/// "specialized" for iterating over `&[u8]`. Once feature(specialization) is
24/// stable we can use actual specialization.
25///
26/// This trait is sealed and cannot be implemented for types outside of
27/// `serde_json`.
28pub trait Read<'de>: private::Sealed {
29    #[doc(hidden)]
30    fn next(&mut self) -> Result<Option<u8>>;
31    #[doc(hidden)]
32    fn peek(&mut self) -> Result<Option<u8>>;
33
34    /// Only valid after a call to peek(). Discards the peeked byte.
35    #[doc(hidden)]
36    fn discard(&mut self);
37
38    /// Position of the most recent call to next().
39    ///
40    /// The most recent call was probably next() and not peek(), but this method
41    /// should try to return a sensible result if the most recent call was
42    /// actually peek() because we don't always know.
43    ///
44    /// Only called in case of an error, so performance is not important.
45    #[doc(hidden)]
46    fn position(&self) -> Position;
47
48    /// Position of the most recent call to peek().
49    ///
50    /// The most recent call was probably peek() and not next(), but this method
51    /// should try to return a sensible result if the most recent call was
52    /// actually next() because we don't always know.
53    ///
54    /// Only called in case of an error, so performance is not important.
55    #[doc(hidden)]
56    fn peek_position(&self) -> Position;
57
58    /// Offset from the beginning of the input to the next byte that would be
59    /// returned by next() or peek().
60    #[doc(hidden)]
61    fn byte_offset(&self) -> usize;
62
63    /// Assumes the previous byte was a quotation mark. Parses a JSON-escaped
64    /// string until the next quotation mark using the given scratch space if
65    /// necessary. The scratch space is initially empty.
66    #[doc(hidden)]
67    fn parse_str<'s>(&'s mut self, scratch: &'s mut Vec<u8>) -> Result<Reference<'de, 's, str>>;
68
69    /// Assumes the previous byte was a quotation mark. Parses a JSON-escaped
70    /// string until the next quotation mark using the given scratch space if
71    /// necessary. The scratch space is initially empty.
72    ///
73    /// This function returns the raw bytes in the string with escape sequences
74    /// expanded but without performing unicode validation.
75    #[doc(hidden)]
76    fn parse_str_raw<'s>(
77        &'s mut self,
78        scratch: &'s mut Vec<u8>,
79    ) -> Result<Reference<'de, 's, [u8]>>;
80
81    /// Assumes the previous byte was a quotation mark. Parses a JSON-escaped
82    /// string until the next quotation mark but discards the data.
83    #[doc(hidden)]
84    fn ignore_str(&mut self) -> Result<()>;
85
86    /// Assumes the previous byte was a hex escape sequence ('\u') in a string.
87    /// Parses next hexadecimal sequence.
88    #[doc(hidden)]
89    fn decode_hex_escape(&mut self) -> Result<u16>;
90
91    /// Switch raw buffering mode on.
92    ///
93    /// This is used when deserializing `RawValue`.
94    #[cfg(feature = "raw_value")]
95    #[doc(hidden)]
96    fn begin_raw_buffering(&mut self);
97
98    /// Switch raw buffering mode off and provides the raw buffered data to the
99    /// given visitor.
100    #[cfg(feature = "raw_value")]
101    #[doc(hidden)]
102    fn end_raw_buffering<V>(&mut self, visitor: V) -> Result<V::Value>
103    where
104        V: Visitor<'de>;
105
106    /// Whether StreamDeserializer::next needs to check the failed flag. True
107    /// for IoRead, false for StrRead and SliceRead which can track failure by
108    /// truncating their input slice to avoid the extra check on every next
109    /// call.
110    #[doc(hidden)]
111    const should_early_return_if_failed: bool;
112
113    /// Mark a persistent failure of StreamDeserializer, either by setting the
114    /// flag or by truncating the input data.
115    #[doc(hidden)]
116    fn set_failed(&mut self, failed: &mut bool);
117}

124pub enum Reference<'b, 'c, T>
125where
126    T: ?Sized + 'static,
127{

132impl<'b, 'c, T> Deref for Reference<'b, 'c, T>
133where
134    T: ?Sized + 'static,
135{
136    type Target = T;

400            Ok(raw) => raw,
401            Err(err) => {
402                let prefix = &err.as_bytes()[..=err.utf8_error().valid_up_to()];
403                let line;
404                let column;

407                        + memchr::memchr_iter(b'\n', &prefix[..last_newline]).count()
408                        + 1;
409                    column = prefix.len() - last_newline - 1;

412                    column = self.raw_buffering_start_position.column + prefix.len();
413                }
414                return Err(Error::syntax(
415                    ErrorCode::InvalidUnicodeCodePoint,
416                    line,
417                    column,
418                ));
419            }
420        };
421        visitor.visit_map(OwnedRawDeserializer {
422            raw_value: Some(raw),
423        })

437impl<'a> SliceRead<'a> {
438    /// Create a JSON input source to read from a slice of bytes.
439    pub fn new(slice: &'a [u8]) -> Self {
440        SliceRead {
441            slice,
442            index: 0,
443            #[cfg(feature = "raw_value")]
444            raw_buffering_start_index: 0,
445        }
446    }
447
448    fn position_of_index(&self, i: usize) -> Position {
449        let start_of_line = match memchr::memrchr(b'\n', &self.slice[..i]) {
450            Some(position) => position + 1,
451            None => 0,
452        };
453        Position {
454            line: 1 + memchr::memchr_iter(b'\n', &self.slice[..start_of_line]).count(),
455            column: i - start_of_line,
456        }
457    }
458
459    fn skip_to_escape(&mut self, forbid_control_characters: bool) {
460        // Immediately bail-out on empty strings and consecutive escapes (e.g. \u041b\u0435)
461        if self.index == self.slice.len()
462            || is_escape(self.slice[self.index], forbid_control_characters)
463        {
464            return;
465        }
466        self.index += 1;
467
468        let rest = &self.slice[self.index..];
469
470        if !forbid_control_characters {
471            self.index += memchr::memchr2(b'"', b'\\', rest).unwrap_or(rest.len());
472            return;
473        }
474
475        // We wish to find the first byte in range 0x00..=0x1F or " or \. Ideally, we'd use
476        // something akin to memchr3, but the memchr crate does not support this at the moment.
477        // Therefore, we use a variation on Mycroft's algorithm [1] to provide performance better
478        // than a naive loop. It runs faster than equivalent two-pass memchr2+SWAR code on
479        // benchmarks and it's cross-platform, so probably the right fit.
480        // [1]: https://groups.google.com/forum/#!original/comp.lang.c/2HtQXvg7iKc/xOJeipH6KLMJ
481
482        #[cfg(fast_arithmetic = "64")]
483        type Chunk = u64;
484        #[cfg(fast_arithmetic = "32")]
485        type Chunk = u32;
486
487        const STEP: usize = mem::size_of::<Chunk>();
488        const ONE_BYTES: Chunk = Chunk::MAX / 255; // 0x0101...01
489
490        for chunk in rest.chunks_exact(STEP) {
491            let chars = Chunk::from_le_bytes(chunk.try_into().unwrap());
492            let contains_ctrl = chars.wrapping_sub(ONE_BYTES * 0x20) & !chars;
493            let chars_quote = chars ^ (ONE_BYTES * Chunk::from(b'"'));
494            let contains_quote = chars_quote.wrapping_sub(ONE_BYTES) & !chars_quote;
495            let chars_backslash = chars ^ (ONE_BYTES * Chunk::from(b'\\'));
496            let contains_backslash = chars_backslash.wrapping_sub(ONE_BYTES) & !chars_backslash;
497            let masked = (contains_ctrl | contains_quote | contains_backslash) & (ONE_BYTES << 7);
498            if masked != 0 {
499                // SAFETY: chunk is in-bounds for slice
500                self.index = unsafe { chunk.as_ptr().offset_from(self.slice.as_ptr()) } as usize
501                    + masked.trailing_zeros() as usize / 8;
502                return;
503            }
504        }
505
506        self.index += rest.len() / STEP * STEP;
507        self.skip_to_escape_slow();
508    }
509
510    #[cold]
511    #[inline(never)]
512    fn skip_to_escape_slow(&mut self) {
513        while self.index < self.slice.len() && !is_escape(self.slice[self.index], true) {
514            self.index += 1;
515        }
516    }
517
518    /// The big optimization here over IoRead is that if the string contains no
519    /// backslash escape sequences, the returned &str is a slice of the raw JSON
520    /// data so we avoid copying into the scratch space.
521    fn parse_str_bytes<'s, T, F>(
522        &'s mut self,
523        scratch: &'s mut Vec<u8>,
524        validate: bool,
525        result: F,
526    ) -> Result<Reference<'a, 's, T>>
527    where
528        T: ?Sized + 's,
529        F: for<'f> FnOnce(&'s Self, &'f [u8]) -> Result<&'f T>,
530    {
531        // Index of the first byte not yet copied into the scratch space.
532        let mut start = self.index;
533
534        loop {
535            self.skip_to_escape(validate);
536            if self.index == self.slice.len() {
537                return error(self, ErrorCode::EofWhileParsingString);
538            }
539            match self.slice[self.index] {
540                b'"' => {
541                    if scratch.is_empty() {
542                        // Fast path: return a slice of the raw JSON without any
543                        // copying.
544                        let borrowed = &self.slice[start..self.index];
545                        self.index += 1;
546                        return result(self, borrowed).map(Reference::Borrowed);
547                    } else {
548                        scratch.extend_from_slice(&self.slice[start..self.index]);
549                        self.index += 1;
550                        return result(self, scratch).map(Reference::Copied);
551                    }
552                }
553                b'\\' => {
554                    scratch.extend_from_slice(&self.slice[start..self.index]);
555                    self.index += 1;
556                    match parse_escape(self, validate, scratch) {
    core::result::Result::Ok(val) => val,
    core::result::Result::Err(err) => return core::result::Result::Err(err),
};tri!(parse_escape(self, validate, scratch));
557                    start = self.index;
558                }
559                _ => {
560                    self.index += 1;
561                    return error(self, ErrorCode::ControlCharacterWhileParsingString);
562                }
563            }
564        }
565    }
566}

638                    match ignore_escape(self) {
    core::result::Result::Ok(val) => val,
    core::result::Result::Err(err) => return core::result::Result::Err(err),
};tri!(ignore_escape(self));
639                }
640                _ => {
641                    return error(self, ErrorCode::ControlCharacterWhileParsingString);
642                }
643            }
644        }

647    #[inline]
648    fn decode_hex_escape(&mut self) -> Result<u16> {
649        match self.slice[self.index..] {
650            [a, b, c, d, ..] => {
651                self.index += 4;
652                match decode_four_hex_digits(a, b, c, d) {
653                    Some(val) => Ok(val),
654                    None => error(self, ErrorCode::InvalidEscape),
655                }
656            }
657            _ => {
658                self.index = self.slice.len();
659                error(self, ErrorCode::EofWhileParsingString)
660            }
661        }
662    }
663
664    #[cfg(feature = "raw_value")]
665    fn begin_raw_buffering(&mut self) {
666        self.raw_buffering_start_index = self.index;
667    }
668
669    #[cfg(feature = "raw_value")]
670    fn end_raw_buffering<V>(&mut self, visitor: V) -> Result<V::Value>
671    where
672        V: Visitor<'a>,
673    {
674        let raw = &self.slice[self.raw_buffering_start_index..self.index];
675        let raw = match str::from_utf8(raw) {
676            Ok(raw) => raw,
677            Err(err) => {
678                let at = self.raw_buffering_start_index + err.valid_up_to() + 1;
679                let position = self.position_of_index(at);
680                return Err(Error::syntax(
681                    ErrorCode::InvalidUnicodeCodePoint,
682                    position.line,
683                    position.column,
684                ));
685            }
686        };
687        visitor.visit_map(BorrowedRawDeserializer {
688            raw_value: Some(raw),
689        })
690    }
691
692    const should_early_return_if_failed: bool = false;
693
694    #[inline]
695    #[cold]
696    fn set_failed(&mut self, _failed: &mut bool) {
697        self.slice = &self.slice[..self.index];
698    }
699}

903fn as_str<'de, 's, R: Read<'de>>(read: &R, slice: &'s [u8]) -> Result<&'s str> {
904    match str::from_utf8(slice) {
905        Ok(s) => Ok(s),
906        Err(err) => {
907            // Point at the first invalid UTF-8 byte. `read` is positioned at
908            // the closing quote of the string whose content is `slice`. If the
909            // JSON string contained escape sequences, the column position is
910            // approximate.
911            let position = read.position();
912            Err(Error::syntax(
913                ErrorCode::InvalidUnicodeCodePoint,
914                position.line,
915                position
916                    .column
917                    .saturating_sub(slice.len() - err.valid_up_to()),
918            ))
919        }
920    }
921}

923/// Parses a JSON escape sequence and appends it into the scratch space. Assumes
924/// the previous byte read was a backslash.
925fn parse_escape<'de, R: Read<'de>>(
926    read: &mut R,
927    validate: bool,
928    scratch: &mut Vec<u8>,
929) -> Result<()> {
930    let ch = match next_or_eof(read) {
    core::result::Result::Ok(val) => val,
    core::result::Result::Err(err) => return core::result::Result::Err(err),
}tri!(next_or_eof(read));
931
932    match ch {
933        b'"' => scratch.push(b'"'),
934        b'\\' => scratch.push(b'\\'),
935        b'/' => scratch.push(b'/'),
936        b'b' => scratch.push(b'\x08'),
937        b'f' => scratch.push(b'\x0c'),
938        b'n' => scratch.push(b'\n'),
939        b'r' => scratch.push(b'\r'),
940        b't' => scratch.push(b'\t'),
941        b'u' => return parse_unicode_escape(read, validate, scratch),
942        _ => return error(read, ErrorCode::InvalidEscape),
943    }
944
945    Ok(())
946}

948/// Parses a JSON \u escape and appends it into the scratch space. Assumes `\u`
949/// has just been read.
950#[cold]
951fn parse_unicode_escape<'de, R: Read<'de>>(
952    read: &mut R,
953    validate: bool,
954    scratch: &mut Vec<u8>,
955) -> Result<()> {
956    let mut n = match read.decode_hex_escape() {
    core::result::Result::Ok(val) => val,
    core::result::Result::Err(err) => return core::result::Result::Err(err),
}tri!(read.decode_hex_escape());
957
958    // Non-BMP characters are encoded as a sequence of two hex escapes,
959    // representing UTF-16 surrogates. If deserializing a utf-8 string the
960    // surrogates are required to be paired, whereas deserializing a byte string
961    // accepts lone surrogates.
962    if validate && n >= 0xDC00 && n <= 0xDFFF {
963        // XXX: This is actually a trailing surrogate.
964        return error(read, ErrorCode::LoneLeadingSurrogateInHexEscape);
965    }
966
967    loop {
968        if n < 0xD800 || n > 0xDBFF {
969            // Every u16 outside of the surrogate ranges is guaranteed to be a
970            // legal char.
971            push_wtf8_codepoint(n as u32, scratch);
972            return Ok(());
973        }
974
975        // n is a leading surrogate, we now expect a trailing surrogate.
976        let n1 = n;
977
978        if match peek_or_eof(read) {
    core::result::Result::Ok(val) => val,
    core::result::Result::Err(err) => return core::result::Result::Err(err),
}tri!(peek_or_eof(read)) == b'\\' {
979            read.discard();
980        } else {
981            return if validate {
982                read.discard();
983                error(read, ErrorCode::UnexpectedEndOfHexEscape)
984            } else {
985                push_wtf8_codepoint(n1 as u32, scratch);
986                Ok(())
987            };
988        }
989
990        if match peek_or_eof(read) {
    core::result::Result::Ok(val) => val,
    core::result::Result::Err(err) => return core::result::Result::Err(err),
}tri!(peek_or_eof(read)) == b'u' {
991            read.discard();
992        } else {
993            return if validate {
994                read.discard();
995                error(read, ErrorCode::UnexpectedEndOfHexEscape)
996            } else {
997                push_wtf8_codepoint(n1 as u32, scratch);
998                // The \ prior to this byte started an escape sequence, so we
999                // need to parse that now. This recursive call does not blow the
1000                // stack on malicious input because the escape is not \u, so it
1001                // will be handled by one of the easy nonrecursive cases.
1002                parse_escape(read, validate, scratch)
1003            };
1004        }
1005
1006        let n2 = match read.decode_hex_escape() {
    core::result::Result::Ok(val) => val,
    core::result::Result::Err(err) => return core::result::Result::Err(err),
}tri!(read.decode_hex_escape());
1007
1008        if n2 < 0xDC00 || n2 > 0xDFFF {
1009            if validate {
1010                return error(read, ErrorCode::LoneLeadingSurrogateInHexEscape);
1011            }
1012            push_wtf8_codepoint(n1 as u32, scratch);
1013            // If n2 is a leading surrogate, we need to restart.
1014            n = n2;
1015            continue;
1016        }
1017
1018        // This value is in range U+10000..=U+10FFFF, which is always a valid
1019        // codepoint.
1020        let n = ((((n1 - 0xD800) as u32) << 10) | (n2 - 0xDC00) as u32) + 0x1_0000;
1021        push_wtf8_codepoint(n, scratch);
1022        return Ok(());
1023    }
1024}

1026/// Adds a WTF-8 codepoint to the end of the buffer. This is a more efficient
1027/// implementation of String::push. The codepoint may be a surrogate.
1028#[inline]
1029fn push_wtf8_codepoint(n: u32, scratch: &mut Vec<u8>) {
1030    if n < 0x80 {
1031        scratch.push(n as u8);
1032        return;
1033    }
1034
1035    scratch.reserve(4);
1036
1037    // SAFETY: After the `reserve` call, `scratch` has at least 4 bytes of
1038    // allocated but uninitialized memory after its last initialized byte, which
1039    // is where `ptr` points. All reachable match arms write `encoded_len` bytes
1040    // to that region and update the length accordingly, and `encoded_len` is
1041    // always <= 4.
1042    unsafe {
1043        let ptr = scratch.as_mut_ptr().add(scratch.len());
1044
1045        let encoded_len = match n {
1046            0..=0x7F => ::core::panicking::unreachable_default_msg()unreachable!(),
1047            0x80..=0x7FF => {
1048                ptr.write(((n >> 6) & 0b0001_1111) as u8 | 0b1100_0000);
1049                2
1050            }
1051            0x800..=0xFFFF => {
1052                ptr.write(((n >> 12) & 0b0000_1111) as u8 | 0b1110_0000);
1053                ptr.add(1)
1054                    .write(((n >> 6) & 0b0011_1111) as u8 | 0b1000_0000);
1055                3
1056            }
1057            0x1_0000..=0x10_FFFF => {
1058                ptr.write(((n >> 18) & 0b0000_0111) as u8 | 0b1111_0000);
1059                ptr.add(1)
1060                    .write(((n >> 12) & 0b0011_1111) as u8 | 0b1000_0000);
1061                ptr.add(2)
1062                    .write(((n >> 6) & 0b0011_1111) as u8 | 0b1000_0000);
1063                4
1064            }
1065            0x11_0000.. => ::core::panicking::unreachable_default_msg()unreachable!(),
1066        };
1067        ptr.add(encoded_len - 1)
1068            .write((n & 0b0011_1111) as u8 | 0b1000_0000);
1069
1070        scratch.set_len(scratch.len() + encoded_len);
1071    }
1072}

1074/// Parses a JSON escape sequence and discards the value. Assumes the previous
1075/// byte read was a backslash.
1076fn ignore_escape<'de, R>(read: &mut R) -> Result<()>
1077where
1078    R: ?Sized + Read<'de>,
1079{
1080    let ch = match next_or_eof(read) {
    core::result::Result::Ok(val) => val,
    core::result::Result::Err(err) => return core::result::Result::Err(err),
}tri!(next_or_eof(read));
1081
1082    match ch {
1083        b'"' | b'\\' | b'/' | b'b' | b'f' | b'n' | b'r' | b't' => {}
1084        b'u' => {
1085            // At this point we don't care if the codepoint is valid. We just
1086            // want to consume it. We don't actually know what is valid or not
1087            // at this point, because that depends on if this string will
1088            // ultimately be parsed into a string or a byte buffer in the "real"
1089            // parse.
1090
1091            match read.decode_hex_escape() {
    core::result::Result::Ok(val) => val,
    core::result::Result::Err(err) => return core::result::Result::Err(err),
};tri!(read.decode_hex_escape());
1092        }
1093        _ => {
1094            return error(read, ErrorCode::InvalidEscape);
1095        }
1096    }
1097
1098    Ok(())
1099}

1101const fn decode_hex_val_slow(val: u8) -> Option<u8> {
1102    match val {
1103        b'0'..=b'9' => Some(val - b'0'),
1104        b'A'..=b'F' => Some(val - b'A' + 10),
1105        b'a'..=b'f' => Some(val - b'a' + 10),
1106        _ => None,
1107    }
1108}

1110const fn build_hex_table(shift: usize) -> [i16; 256] {
1111    let mut table = [0; 256];
1112    let mut ch = 0;
1113    while ch < 256 {
1114        table[ch] = match decode_hex_val_slow(ch as u8) {
1115            Some(val) => (val as i16) << shift,
1116            None => -1,
1117        };
1118        ch += 1;
1119    }
1120    table
1121}

1123static HEX0: [i16; 256] = build_hex_table(0);
1124static HEX1: [i16; 256] = build_hex_table(4);

1126fn decode_four_hex_digits(a: u8, b: u8, c: u8, d: u8) -> Option<u16> {
1127    let a = HEX1[a as usize] as i32;
1128    let b = HEX0[b as usize] as i32;
1129    let c = HEX1[c as usize] as i32;
1130    let d = HEX0[d as usize] as i32;
1131
1132    let codepoint = ((a | b) << 8) | c | d;
1133
1134    // A single sign bit check.
1135    if codepoint >= 0 {
1136        Some(codepoint as u16)
1137    } else {
1138        None
1139    }
